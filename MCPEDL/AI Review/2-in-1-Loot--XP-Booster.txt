  const dx = a.x - b.x;
  const dy = a.y - b.y;
  const dz = a.z - b.z;

function addToTotals(totals, stack) {
  for (const entry of totals) {
    if (entry.stack.isStackableWith(stack)) {
      entry.amount += stack.amount;
      return;
    }
  }
  totals.push({ stack, amount: stack.amount });
}

function spawnTotals(dimension, location, totals) {
  for (const entry of totals) {
    const size = entry.stack.maxAmount;
    let left = entry.amount;
    while (left > 0) {
      const count = Math.min(size, left);
      const piece = entry.stack.clone();
      piece.amount = count;
      try {
        dimension.spawnItem(piece, location);
      } catch (error) {
        return;
      }
      left -= count;
    }
  }
}

function addExtraBlockLoot(dimension, center, permutation, tool, maxMultiplier, skipSelf) {
  const rounds = rollMultiplier(maxMultiplier) - 1;
  if (rounds <= 0) return;
  const typeId = permutation.type.id;
  const manager = world.getLootTableManager();
  const totals = [];
  for (let round = 0; round < rounds; round++) {
    let stacks;
    try {
      stacks = manager.generateLootFromBlockPermutation(permutation, tool);
    } catch (error) {
      break;
    }
    if (!stacks) break;
    for (const stack of stacks) {
      if (stack.maxAmount <= 1) continue;
      if (skipSelf && stack.typeId === typeId) continue;
      addToTotals(totals, stack);
    }
  }
  spawnTotals(dimension, center, totals);
}

function addExtraMobLoot(player, deadEntity, maxMultiplier) {
  const rounds = rollMultiplier(maxMultiplier) - 1;
  if (rounds <= 0) return;
  let dimension;
  let location;
  let tool;
  try {
    dimension = deadEntity.dimension;
    location = deadEntity.location;
    const equippable = player.getComponent(EntityComponentTypes.Equippable);
    tool = equippable ? equippable.getEquipment("Mainhand") : undefined;
  } catch (error) {
    return;
  }
  const manager = world.getLootTableManager();
  const totals = [];
  for (let round = 0; round < rounds; round++) {
    let stacks;
    try {
      stacks = manager.generateLootFromEntity(deadEntity, tool);
    } catch (error) {
      break;
    }
    if (!stacks) break;
    for (const stack of stacks) {
      if (stack.maxAmount <= 1) continue;
      addToTotals(totals, stack);
    }
  }
  spawnTotals(dimension, location, totals);
}

function blockKey(dimension, location) {
  return `${dimension.id}:${location.x},${location.y},${location.z}`;
}

world.afterEvents.playerInteractWithBlock.subscribe(({ block, itemStack }) => {
  if (!itemStack || itemStack.typeId !== "minecraft:bone_meal") return;

  fertilized.add(blockKey(block.dimension, block.location));

world.afterEvents.playerBreakBlock.subscribe((event) => {
  const { player, block, brokenBlockPermutation, dimension, itemStackBeforeBreak } = event;
  const origin = block.location;
  const wasFertilized = fertilized.delete(blockKey(dimension, origin));

  const maxMultiplier = lootMultiplier(player);
  if (maxMultiplier <= 1) return;
  if (isCreative(player)) return;
  const category = blockCategory(brokenBlockPermutation.type.id);
  if (!category || !getBoost(player)[category]) return;
  if (!isGrown(brokenBlockPermutation)) return;
  if (wasFertilized) return;
  if (hasSilkTouch(itemStackBeforeBreak)) return;

  const center = { x: origin.x + 0.5, y: origin.y + 0.5, z: origin.z + 0.5 };
  addExtraBlockLoot(dimension, center, brokenBlockPermutation, itemStackBeforeBreak, maxMultiplier, category === "ores");
});

world.afterEvents.entityDie.subscribe(({ deadEntity, damageSource }) => {
  const killer = damageSource ? damageSource.damagingEntity : undefined;
  if (!killer || killer.typeId !== "minecraft:player") return;
  if (!isMob(deadEntity)) return;

  const maxMultiplier = lootMultiplier(killer);
  if (maxMultiplier <= 1 || !getBoost(killer).mobs) return;

  addExtraMobLoot(killer, deadEntity, maxMultiplier);
});

function hasNearerPlayer(orbLocation, player, players) {
  const own = distanceSquared(orbLocation, player.location);
  for (const other of players) {
    if (other.id === player.id || !other.isValid) continue;
    if (other.dimension.id !== player.dimension.id) continue;
    if (distanceSquared(orbLocation, other.location) < own) return true;
  }
  return false;
}

system.runInterval(() => {
  const players = world.getAllPlayers();
  for (const player of players) {
    if (!player.isValid) continue;
    const multiplier = xpMultiplier(player);
    if (multiplier <= 1) continue;

    const dimension = player.dimension;
    const location = player.location;
    let orbs;
    try {
      orbs = dimension.getEntities({
        type: "minecraft:xp_orb",
        location,
        maxDistance: XP_RADIUS,
      });
    } catch (error) {
      continue;
    }

    for (const orb of orbs) {
      try {
        if (!orb.isValid) continue;
        const orbLocation = orb.location;
        if (hasNearerPlayer(orbLocation, player, players)) continue;
        if (!orb.hasTag(XP_TAG)) {
          const extra = multiplier - 1;
          orb.addTag(XP_TAG);
          for (let i = 0; i < extra; i++) {
            const bonus = dimension.spawnEntity("minecraft:xp_orb", location);
            bonus.addTag(XP_TAG);
          }
        }
        if (distanceSquared(orbLocation, location) > XP_PICKUP_SQUARED) orb.teleport(location);
      } catch (error) {
        continue;
      }
    }
  }
}, XP_INTERVAL_TICKS);

- **What Happens**: Drag the Loot slider across six levels, each giving a random bonus of up to 1x, 2x, 4x, 8x, 16x or 32x, for ores, crops and other Fortune-style blocks you break and for mobs you kill.

Ores, gilded blackstone, amethyst clusters, nether wart, sweet berries, wheat, carrots, potatoes, beetroot and mob loot are multiplied, so normal blocks such as dirt, stone and wood drop as usual, and unstackable items such as tools are left alone. Mining with Silk Touch drops the block once and is never multiplied, and blocks broken in Creative mode are not multiplied because they drop nothing. Blocks are boosted by rolling their loot again, so only items that block can really drop are multiplied, and mobs are boosted the same way, using the weapon you killed with, so rare drops stay rare but get more chances to appear. The Ores, Crops and Mobs switches let you choose which kinds of loot are boosted. Crops, nether wart and sweet berries are multiplied only when fully grown, and crops sped up with bone meal are not multiplied. Blocks that drop themselves or can be crafted back from their drops, such as ancient debris, melons, glowstone and sea lanterns, are never multiplied, so they cannot be farmed by placing and breaking. Fortune still works, and the booster multiplies on top of it. Extra drops are combined into full stacks, so large boosts create fewer items on the ground. Only real mobs count, so armor stands, boats, minecarts and mobs that carry items, such as donkeys, mules, llamas, allays, foxes and villagers, are not multiplied, and only mobs you kill yourself are boosted. Only drops that appear the instant you break a block or kill a mob are multiplied, so items thrown or dropped nearby are left alone.

### XP Gain Booster
- **What Happens**: Drag the XP slider across seven levels, 1x, 2x, 4x, 8x, 16x, 32x and 64x, and each XP orb near you is turned into that many orbs.
- **Good To Know**: The orbs are collected instantly, so Mending repairs your gear from the extra XP. Only the orbs closest to you are multiplied, so other players' orbs are never affected.

### About Screen
- **What Happens**: The About button shows the creator, the add-on version, a short note on both packs, the CurseForge and MCPEDL profiles and an unofficial-product notice.

### Redesigned UI
- **What Happens**: A small Resource Pack named Redesigned UI gives the Booster menus the ALNStudios look, with a dark panel, a blue border, a gold title and icon buttons.
- **Good To Know**: Enable the Resource Pack together with the Behavior Pack.

## How to Use

1. Rename a Book to Booster on an anvil.
2. Hold it and right-click, then choose XP & Loot Booster.
3. Drag the Loot and XP sliders, switch Boosters on or off, choose whether Ores, Crops and Mobs are boosted and press Apply. A higher sound means the boosters are on and a lower sound means they are off.

## Common Questions

- **Does the multiplier apply to other players?**: No. Each player chooses their own Loot and XP multipliers.
- **Can I turn it off?**: Yes. Open XP & Loot Booster, switch Boosters Off and press Apply. Your levels are kept.
