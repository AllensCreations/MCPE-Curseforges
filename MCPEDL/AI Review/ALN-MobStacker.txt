import { EntityComponentTypes, EquipmentSlot, ItemStack, system, world } from "@minecraft/server";
import {
  COOKED_BY_RAW,
  FARM_ANIMAL_SET,
  GEAR_DROP_CHANCE,
  GEAR_DROP_ITEMS,
  GEAR_LOOTING_BONUS,
  GRINDER_BATCH_SIZE,
  LOOT,
  PERSISTENT_TAG,
  STACKABLE_MOBS,
  STACK_COUNT_PROPERTY,
  WOOL_BY_COLOR,
  applyStack,
  mobName,
  readMeta,
  restoreAppearance,
  spawnMatching,
  stackCache,
} from "./stack_config.js";
import { clearTower, ensureTower } from "./stack_tower.js";

const SAMPLE_LIMIT = 200;
const MAX_ITEM_ENTITIES = 256;
const ITEMS_PER_TICK = 32;

function randomInt(min, max) {
  return min + Math.floor(Math.random() * (max - min + 1));
}

function lootMultiplier(looting) {
  return 1 + Math.floor(Math.random() * (looting + 1));
}

function addTotal(totals, id, amount) {
  if (id && amount > 0) totals.set(id, (totals.get(id) ?? 0) + amount);
}

function pickWeighted(entries) {
  let sum = 0;
  for (const entry of entries) sum += entry[3];
  let roll = Math.random() * sum;
  for (const entry of entries) {
    roll -= entry[3];
    if (roll < 0) return entry;
  }
  return entries[entries.length - 1];
}

function rollUnit(typeId, def, meta, looting, totals, playerKill) {
  for (const [id, min, max] of def.drops) {
    addTotal(totals, id, randomInt(min, max) * lootMultiplier(looting));
  }
  if (def.rolled) {
    const rolls = randomInt(def.rolled.rolls[0], def.rolled.rolls[1]);
    for (let r = 0; r < rolls; r++) {
      const entry = pickWeighted(def.rolled.entries);
      addTotal(totals, entry[0], randomInt(entry[1], entry[2]) * lootMultiplier(looting));
    }
  }
  if (playerKill && def.gear && Math.random() < GEAR_DROP_CHANCE + GEAR_LOOTING_BONUS * looting) {
    addTotal(totals, GEAR_DROP_ITEMS[randomInt(0, GEAR_DROP_ITEMS.length - 1)], 1);
  }
  if (playerKill && def.rare) {
    for (const rare of def.rare) {
      if (Math.random() < rare.chance + rare.bonus * looting) {
        addTotal(totals, rare.ids[randomInt(0, rare.ids.length - 1)], 1);
      }
    }
  }
  if (typeId === "minecraft:sheep" && !meta.sheared) {
    const wool = WOOL_BY_COLOR[meta.color] ?? WOOL_BY_COLOR[0];
    addTotal(totals, wool, 1);
  }
}

function rollExperience(def, meta) {
  if (FARM_ANIMAL_SET.has(def.typeId) && meta.isBaby) return 0;
  if (def.xp === "variant") return meta.variant > 0 ? meta.variant : 1;
  if (meta.isBaby && def.babyXp) return def.babyXp;
  return randomInt(def.xp[0], def.xp[1]);
}

function computeDrops(typeId, meta, units, looting, burning, playerKill) {
  const base = LOOT[typeId];
  const totals = new Map();
  let xp = 0;
  if (!base) return { totals, xp };
  const def = { ...base, typeId };
  const isBabyAnimal = FARM_ANIMAL_SET.has(typeId) && meta.isBaby;
  const sampled = Math.min(units, SAMPLE_LIMIT);
  const scale = units / sampled;
  for (let i = 0; i < sampled; i++) {
    if (playerKill) xp += rollExperience(def, meta);
    if (!isBabyAnimal) rollUnit(typeId, def, meta, looting, totals, playerKill);
  }
  if (scale !== 1) {
    xp = Math.round(xp * scale);
    for (const [id, amount] of totals) totals.set(id, Math.round(amount * scale));
  }
  if (burning) {
    for (const [raw, cooked] of Object.entries(COOKED_BY_RAW)) {
      const amount = totals.get(raw);
      if (amount) {
        totals.delete(raw);
        totals.set(cooked, amount);
      }
    }
  }
  return { totals, xp };
}

function spawnDrops(dimension, location, totals, xp) {
  const pending = [];
  for (const [id, amount] of totals) {
    let remaining = amount;
    let maxAmount = 64;
    try {
      maxAmount = new ItemStack(id, 1).maxAmount;
    } catch (error) {
      continue;
    }
    while (remaining > 0 && pending.length < MAX_ITEM_ENTITIES) {
      const size = Math.min(remaining, maxAmount);
      try {
        pending.push(new ItemStack(id, size));
      } catch (error) {
        break;
      }
      remaining -= size;
    }
  }
  let index = 0;
  const handle = system.runInterval(() => {
    for (let n = 0; n < ITEMS_PER_TICK && index < pending.length; n++) {
      try {
        dimension.spawnItem(pending[index], location);
      } catch (error) {
        index = pending.length;
      }
      index++;
    }
    if (index >= pending.length) system.clearRun(handle);
  }, 1);
  if (xp > 0) {
    try {
      dimension.spawnXp(location, xp);
    } catch (error) {
      return;
    }
  }
}

function isBurning(entity) {
  try {
    return entity.getComponent(EntityComponentTypes.OnFire) !== undefined;
  } catch (error) {
    return false;
  }
}

function lootingLevel(player) {
  try {
    const equippable = player.getComponent(EntityComponentTypes.Equippable);
    const item = equippable ? equippable.getEquipment(EquipmentSlot.Mainhand) : undefined;
    const enchantable = item ? item.getComponent("minecraft:enchantable") : undefined;
    const enchantment = enchantable ? enchantable.getEnchantment("looting") : undefined;
    return enchantment ? Math.max(0, Math.min(3, enchantment.level)) : 0;
  } catch (error) {
    return 0;
  }
}

function readDeadStack(dead) {
  const cached = stackCache.get(dead.id);
  stackCache.delete(dead.id);
  let count = 1;
  try {
    const value = dead.getDynamicProperty(STACK_COUNT_PROPERTY);
    if (typeof value === "number" && value >= 1) count = Math.floor(value);
  } catch (error) {
    count = cached ? cached.count : 1;
  }
  if (count <= 1 && cached) count = cached.count;
  let location = cached ? cached.location : undefined;
  let dimension;
  try {
    location = dead.location;
    dimension = dead.dimension;
  } catch (error) {
    dimension = undefined;
  }
  if (!dimension && cached) dimension = world.getDimension(cached.dimensionId);
  let meta = cached ? cached.meta : undefined;
  if (!meta) {
    try {
      meta = readMeta(dead);
    } catch (error) {
      meta = { isBaby: false, sheared: false, charged: false, color: -1, variant: -1, mark: -1, skin: -1 };
    }
  }
  return { count, location, dimension, meta };
}

function respawnStack(dimension, typeId, location, remaining, meta) {
  const entity = spawnMatching(dimension, typeId, location, meta, remaining > 1);
  restoreAppearance(entity, meta);
  if (remaining > 1) {
    entity.addTag(PERSISTENT_TAG);
    applyStack(entity, remaining, meta);
    ensureTower(entity, remaining, meta);
  }
}

function handleDeath(event) {
  const dead = event.deadEntity;
  const typeId = dead.typeId;
  if (!STACKABLE_MOBS.has(typeId)) return;
  clearTower(dead.id);
  const { count, location, dimension, meta } = readDeadStack(dead);
  if (count <= 1 || !location || !dimension) return;

  const killer = event.damageSource.damagingEntity;
  const isPlayerKill = killer !== undefined && killer.typeId === "minecraft:player";

  if (isPlayerKill && killer.isSneaking) {
    const looting = lootingLevel(killer);
    const { totals, xp } = computeDrops(typeId, meta, count - 1, looting, isBurning(dead), true);
    spawnDrops(dimension, location, totals, xp);
    killer.onScreenDisplay.setActionBar(
      "§6[MobStacker] §aCleared x" + count + " " + mobName(typeId, meta.isBaby) + " §7(Looting " + looting + ")"
    );
    return;
  }

  const batch = isPlayerKill ? 1 : Math.min(GRINDER_BATCH_SIZE, count);
  if (batch > 1) {
    const extra = computeDrops(typeId, meta, batch - 1, 0, isBurning(dead), false);
    spawnDrops(dimension, location, extra.totals, 0);
  }
  if (count - batch < 1) return;
  try {
    respawnStack(dimension, typeId, location, count - batch, meta);
  } catch (error) {
    return;
  }
}

export function startCombat() {
  world.afterEvents.entityDie.subscribe(handleDeath, { entityTypes: [...STACKABLE_MOBS] });
}

  }
  if (allowed === 0) towers.delete(host.id);
}

export function reconcileTowers(dimension, typeId) {
  if (!TOWER_MOB_HEIGHTS[typeId]) return;
  let visuals;
  try {
    visuals = dimension.getEntities({ type: typeId, tags: [VISUAL_TAG] });
  } catch (error) {
    return;
  }
  for (const visual of visuals) {
    try {
      const hostId = visual.getDynamicProperty(VISUAL_HOST_PROPERTY);
      let entry = towers.get(hostId);
      if (!entry || !entry.host.isValid) {
        const host = typeof hostId === "string" ? world.getEntity(hostId) : undefined;
        if (!host || !host.isValid) {
          visual.remove();
          continue;
        }
        entry = { host, meta: readMeta(host), visuals: [] };
        towers.set(hostId, entry);
      }
      if (!entry.visuals.some((known) => known.id === visual.id)) {
        entry.visuals.push(visual);
        makeInert(visual);
      }
    } catch (error) {
      continue;
    }
  }
}

function syncTowers() {
  for (const [hostId, entry] of towers) {
    const host = entry.host;
    if (!host.isValid) {
      clearTower(hostId);
      continue;
    }
    const location = host.location;
    const height = unitHeight(host.typeId, entry.meta);
    const rotation = host.getRotation();
    let level = 1;
    for (const visual of entry.visuals) {
      if (!visual.isValid) continue;
      const spot = { x: location.x, y: location.y + level * height, z: location.z };
      level++;
      try {
        const now = visual.location;
        const dx = now.x - spot.x;
        const dy = now.y - spot.y;
        const dz = now.z - spot.z;
        const turn = Math.abs(((visual.getRotation().y - rotation.y + 540) % 360) - 180);
        if (dx * dx + dy * dy + dz * dz <= TOWER_MOVE_TOLERANCE * TOWER_MOVE_TOLERANCE && turn <= TOWER_YAW_TOLERANCE) {
          continue;
        }
        visual.teleport(spot, { keepVelocity: false, rotation });
      } catch (error) {
        continue;
      }
    }
  }
}

export function startTower() {
  system.runInterval(syncTowers, TOWER_SYNC_TICKS);

  world.beforeEvents.playerInteractWithEntity.subscribe((event) => {
    try {
      if (event.target.hasTag(VISUAL_TAG)) event.cancel = true;
    } catch (error) {
      return;
    }
  });
}

# ALN:MobStacker — Lag-Free Farm and Grinder Condenser

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)
[![Format](https://img.shields.io/badge/Format-.MCPACK%20%7C%20.MCADDON%20%7C%20.MCWORLD-blue?style=flat-square)](https://github.com/AllensCreations/MCPE-Curseforges)

**Merges identical farm animals and grinder mobs within 4 blocks into one counted stack with no size limit.**

> ALN:MobStacker turns 25 supported mob types into single stacked mobs with a gold count label. Farms and AFK grinders keep fewer mobs in the world, and every kill still pays out.

## Overview

Build a farm or grinder as usual. Mobs of the same kind that stand close together merge into one mob labelled with how many it holds. Killing the stack removes one mob at a time, or you can sneak while you strike the final blow to clear the whole stack at once. Looting on your held weapon raises the drops from the cleared stack.

## Key Features

### Mob Merging
- **What Happens**: Identical mobs within 4 blocks of each other merge into one mob about once a second. The merged mob shows a label such as `[x12] Cow` in gold and white.
- **How It Works**: Twenty cows packed in a pen become one Cow labelled `[x20]`. The label appears when you look at the mob.
- **Good To Know**: There is no stack size limit. Mobs absorbed into a stack disappear without dropping anything. Merged stacks are set to stay in the world, so they should not despawn when you walk away. Baby farm animal stacks keep their original mob, so its growth is not reset, but killing one mob of a baby stack spawns a new baby that starts growing from the beginning.

### Stack Towers
- **What Happens**: A stack of farm animals shows up to 3 mobs standing on top of each other. Only the bottom mob is real, and the number label sits on it.
- **How It Works**: Hit the bottom mob to remove one from the stack. The mobs above cannot be hurt and copy the bottom mob's movement. You cannot feed, shear, milk, name or leash them.
- **Good To Know**: Towers appear only for Cow, Pig, Sheep, Chicken, Rabbit, Mooshroom, Goat and Frog stacks. Hostile stacks show the number only. A tower is shorter under a low ceiling.

### Supported Mobs
- **What Happens**: Farm animals: Cow, Pig, Sheep, Chicken, Rabbit, Mooshroom, Goat, Frog. Grinder mobs: Zombie, Skeleton, Spider, Cave Spider, Creeper, Drowned, Husk, Stray, Blaze, Magma Cube, Slime, Enderman, Witch, Zombified Piglin, Guardian, Bogged, Breeze.

### Merge Rules
- **What Happens**: Adults merge with adults and babies merge with babies, and the two never share a stack. Zombies, Husks, Drowned and Zombified Piglins each keep their own stacks. Sheep merge only with sheep of the same color and shear state, and slimes and magma cubes merge only with the same size.
- **Good To Know**: Tamed mobs, leashed mobs and mobs you named with a Name Tag never merge into a stack. Mobs wearing armor or holding items stay separate, including armored Zombies. Skeletons, Strays and Boggeds holding only their bow still merge.

### One-by-One Unpeeling
- **What Happens**: Killing a stacked mob normally removes one mob from the stack. The stack stays in place with its count lowered by one, and the mob you killed drops its normal loot.
- **How It Works**: A stack mob that dies to fall damage, lava, fire or anything else that is not a player takes up to 5 stack mobs with it at once, so a grinder clears a big stack faster. The extra mobs drop their normal items, without experience or rare drops.
- **Good To Know**: A baby stack is replaced by a baby and an adult stack by an adult, so a stack never turns into the other kind.

### Sneak Kill
- **What Happens**: Sneak while you kill a stacked mob and the whole remaining stack is cleared at once. The extra mobs drop their loot and experience together, and an action bar message shows the stack size, mob name and Looting level.
- **Good To Know**: Sneak Kill drops the regular items of each mob type, plus the rare drops of Rabbits, Zombies, Zombified Piglins and Drowned at their vanilla chances. Zombie and Husk stacks also have a small chance of dropping a piece of armor for each extra mob, a little higher with Looting. Cow, Pig, Sheep, Chicken, Rabbit and Mooshroom stacks that are burning when you kill them drop cooked meat. Big piles of drops appear over a few ticks, up to 256 item piles per kill.

### Looting Bonus
- **What Happens**: Looting on the weapon in your main hand, levels 1 to 3, raises the drops of each extra mob in a Sneak Kill. Each extra mob multiplies its drops by a random 1 up to the Looting level plus 1.
- **How It Works**: With Looting III, each extra Zombie in a cleared stack drops its Rotten Flesh multiplied by 1 to 4.

## How to Use

1. Enable the Behavior Pack on your world.
2. Gather several mobs of the same kind within 4 blocks of each other and wait a few seconds until one mob shows a count label.
3. Hit the stack to remove one mob at a time, or sneak while you kill it to clear the whole stack and collect the boosted drops.

## Common Questions

- **Is there a maximum stack size?**: No. A stack can hold any number of mobs.
- **Does it work on pets and named mobs?**: No. Tamed mobs and mobs with a custom name are never merged.
