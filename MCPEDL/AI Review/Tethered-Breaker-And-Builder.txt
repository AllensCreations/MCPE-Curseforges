import { world, system, ItemStack, EntityComponentTypes, GameMode } from "@minecraft/server";
import { ActionFormData, FormCancelationReason } from "@minecraft/server-ui";

const CONTROLLER_ID = "aln:tethered_breaker_builder_controller";
const STATE_KEY = "game_state";
const OPERATOR_LEVEL = 2;

let config = null;
let registered = false;
let stateCache;
const stateListeners = [];

const getState = () => {
  if (stateCache === undefined) {
    try {
      const v = world.getDynamicProperty(STATE_KEY);
      stateCache = typeof v === "string" ? v : "idle";
    } catch {
      return "idle";
    }
  }
  return stateCache;
};
const setState = (s) => {
  stateCache = s;
  try { world.setDynamicProperty(STATE_KEY, s); } catch {}
  for (const fn of stateListeners) {
    try { fn(s); } catch {}
  }
};
export const isRunning = () => getState() === "running";
export const onStateChange = (fn) => { stateListeners.push(fn); };

const isCreative = (player) => player.getGameMode() === GameMode.Creative;
const isHost = (player) => (typeof player.isOp === "function" ? player.isOp() : player.playerPermissionLevel === OPERATOR_LEVEL);
const containerOf = (player) => player.getComponent(EntityComponentTypes.Inventory)?.container;

function hasController(player) {
  const inv = containerOf(player);
  if (!inv) return false;
  for (let i = 0; i < inv.size; i++) {
    if (inv.getItem(i)?.typeId === CONTROLLER_ID) return true;
  }
  return false;
}

function giveController(player) {
  if (hasController(player)) return;
  containerOf(player)?.addItem(new ItemStack(CONTROLLER_ID, 1));
}

function consumeController(player) {
  if (isCreative(player)) return;
  const inv = containerOf(player);
  if (!inv) return;
  for (let i = 0; i < inv.size; i++) {
    if (inv.getItem(i)?.typeId === CONTROLLER_ID) {
      inv.setItem(i, undefined);
      player.playSound("random.break");
      player.sendMessage("The Controller broke.");
      return;
    }
  }
}

function runStart(player) {
  if (getState() === "running") {
    player.sendMessage("Already running.");
    return;
  }
  setState("running");
  consumeController(player);
  try {
    config.onStart(player);
  } catch (e) {
    setState("idle");
    giveController(player);
    player.sendMessage("The game could not start.");
  }
}

export function endGame() {
  if (getState() !== "running") return;
  setState("ended");
  try {
    config?.onEnd();
  } catch (e) {
    world.sendMessage("The game ended with an error.");
  }
  for (const p of world.getAllPlayers()) if (isHost(p)) giveController(p);
}

async function showHowToPlay(player) {
  const form = new ActionFormData().title("How to play").body(config.howToPlay).button("Back");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (!res.canceled && res.selection === 0) openController(player);
}

async function openController(player, attempt = 0) {
  const note = isCreative(player) ? "Creative: the Controller is kept." : "Start uses up the Controller.";
  const form = new ActionFormData()
    .title(config.title)
    .body(`${config.gameInfo}\n\nState: ${getState()}\n${note}`)
    .button("Start")
    .button("How to play");
  for (const x of config.extras) form.button(typeof x.label === "function" ? x.label() : x.label);
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (res.canceled) {
    if (res.cancelationReason === FormCancelationReason.UserBusy && attempt < 5) {
      system.runTimeout(() => openController(player, attempt + 1), 10);
    }
    return;
  }
  if (res.selection === 0) {
    runStart(player);
  } else if (res.selection === 1) {
    showHowToPlay(player);
  } else {
    try { config.extras[res.selection - 2]?.run(player); } catch {}
  }
}

export function registerController(options) {
  if (registered) return;
  registered = true;
  config = {
    title: options?.title ?? "Controller",
    gameInfo: options?.gameInfo ?? "",
    howToPlay: options?.howToPlay ?? "",
    onStart: options?.onStart ?? (() => {}),
    onEnd: options?.onEnd ?? (() => {}),
    extras: Array.isArray(options?.extras) ? options.extras : [],
  };

  world.afterEvents.itemUse.subscribe(({ source, itemStack }) => {
    if (itemStack?.typeId === CONTROLLER_ID && isHost(source)) openController(source);
  });

  system.afterEvents.scriptEventReceive.subscribe(({ id }) => {
    if (id === "allenscreations:end") endGame();
  });

  world.afterEvents.playerSpawn.subscribe(({ player }) => {
    if (isHost(player) && (isCreative(player) || getState() !== "running")) giveController(player);
  });
}

function computePair() {
  let breaker;
  let builder;
  for (const p of world.getAllPlayers()) {
    if (!p.isValid) continue;
    const r = getRole(p);
    if (r === BREAKER && !breaker) breaker = p;
    else if (r === BUILDER && !builder) builder = p;

  if (breaker && builder) return { breaker, builder };
  if (dummyValid()) {
    if (breaker) return { breaker, builder: dummy };
    if (builder) return { breaker: dummy, builder };
  }
  return null;
}

function renderBeam(dim, a, b, dist, particle) {
  const steps = Math.min(CONFIG.MAX_BEAM_PARTICLES, Math.max(2, Math.floor(dist / CONFIG.BEAM_SPACING)));
  const sx = (b.x - a.x) / steps;
  const sy = (b.y - a.y) / steps;
  const sz = (b.z - a.z) / steps;
  try {
    for (let i = 1; i < steps; i++) {
      beamPos.x = a.x + sx * i;
      beamPos.y = a.y + 1.1 + sy * i;
      beamPos.z = a.z + sz * i;
      dim.spawnParticle(particle, beamPos);
    }
  } catch {}
}

function pull(p, dx, dz, up, strength) {
  const m = mode(p);
  if (m === GameMode.Creative || m === GameMode.Spectator) return;
  kb.x = dx * strength;
  kb.z = dz * strength;
  const v = up > 0 ? Math.min(CONFIG.MAX_VPULL, up * strength) : 0;
  try { p.applyKnockback(kb, v); } catch {}
}

function tpTo(p, x, y, z) {
  const dest = { x, y, z };
  try {
    p.teleport(dest, { dimension: p.dimension, checkForBlocks: true });
  } catch {
    try { p.teleport(dest, { dimension: p.dimension }); } catch {}
  }
}

function clampRope(A, B, la, lb, dist) {
  const cA = mode(A) === GameMode.Creative;
  const cB = mode(B) === GameMode.Creative;
  if (cA && cB) return;
  const ancA = anchored(A);
  const ancB = anchored(B);
  let wa = 0.5;
  if (cA || (ancA && !ancB)) wa = 0;
  else if (cB || (ancB && !ancA)) wa = 1;
  const wb = 1 - wa;
  const f = (dist - CONFIG.ROPE_LIMIT) / dist;
  const mx = (lb.x - la.x) * f;
  const my = (lb.y - la.y) * f;
  const mz = (lb.z - la.z) * f;
  if (wa > 0) tpTo(A, la.x + mx * wa, la.y + my * wa, la.z + mz * wa);
  if (wb > 0) tpTo(B, lb.x - mx * wb, lb.y - my * wb, lb.z - mz * wb);
}

function anchored(p) {

}

function strain(p) {

}

  if (level === 1) return `§e[LEASH: TENSE] §fDistance: ${d}m`;
  if (level === 2) return `§6[LEASH: OVERSTRETCHED] §fDistance: ${d}m`;
  return `§c[LEASH: STRAINED] §fDistance: ${d}m! Damage incoming`;
}

  tether.waitCd += step;

  tether.waitCd = 0;
  for (const p of world.getAllPlayers()) if (getRole(p)) hud(p, WAIT_TEXT);

function tick() {
  if (!isRunning()) return;
  const step = CONFIG.TICK_STEP;
  const pr = getPair();
  if (!pr) { waitForPartner(step); return; }
  const { breaker: A, builder: B } = pr;

  if (!isAlive(A) || !isAlive(B)) { resetTether(); return; }
  if (mode(A) === GameMode.Spectator || mode(B) === GameMode.Spectator) {
    hud(A, "§7[LEASH: PAUSED] §fSpectator");
    hud(B, "§7[LEASH: PAUSED] §fSpectator");
    resetTether();
    return;
  }

  if (A.dimension.id !== B.dimension.id) {
    tether.dimTicks += step;
    const left = Math.max(0, Math.ceil((CONFIG.DIMENSION_RECALL_TICKS - tether.dimTicks) / 20));
    const text = `§c[LEASH: STRETCHED ACROSS DIMENSIONS] §fFollow your partner! Recall in ${left}s`;
    hud(A, text);
    hud(B, text);
    if (tether.dimTicks >= CONFIG.DIMENSION_RECALL_TICKS) {
      recall(A, B);
      tether.dimTicks = 0;
    }
    return;
  }
  tether.dimTicks = 0;

  const la = A.location;
  const lb = B.location;
  const dx = lb.x - la.x;
  const dy = lb.y - la.y;
  const dz = lb.z - la.z;
  const d2 = dx * dx + dy * dy + dz * dz;

  if (d2 > RECALL_SQ) {
    tether.farTicks += step;
    tether.strainTicks = 0;
    const left = Math.max(0, Math.ceil((CONFIG.RECALL_TICKS - tether.farTicks) / 20));
    const text = `§c[LEASH: SNAPPED] §f${Math.floor(Math.sqrt(d2))}m apart. Recall in ${left}s`;
    hud(A, text);
    hud(B, text);
    tether.key = -1;
    if (tether.farTicks >= CONFIG.RECALL_TICKS) {
      recall(A, B);
      tether.farTicks = 0;
    }
    return;
  }
  tether.farTicks = 0;

  let level = 0;
  if (d2 > DAMAGE_SQ) level = 3;
  else if (d2 > MAX_SQ) level = 2;
  else if (d2 >= TENSE_SQ) level = 1;

  const dist = Math.sqrt(d2);
  const half = Math.round(dist * 2);
  const key = level * 10000 + half;
  if (key !== tether.key) {
    tether.key = key;
    tether.text = leashText(level, (half / 2).toFixed(1));
  }

  tether.beamCd += step;
  if (tether.beamCd >= CONFIG.BEAM_INTERVAL_TICKS) {
    tether.beamCd = 0;
    if (d2 >= BEAM_MIN_SQ) renderBeam(A.dimension, la, lb, dist, PARTICLES[level]);
  }
  hud(A, tether.text);
  hud(B, tether.text);
  if (d2 > ROPE_SQ) clampRope(A, B, la, lb, dist);

  if (level < 2) {
    tether.strainTicks = 0;
    tether.soundCd = 0;
    return;
  }

  const hd = Math.sqrt(dx * dx + dz * dz);
  const ux = hd > 0.01 ? dx / hd : 0;
  const uz = hd > 0.01 ? dz / hd : 0;
  const vy = dy / dist;
  const s = Math.min(CONFIG.MAX_PULL, 0.2 + (dist - CONFIG.MAX_TETHER_DISTANCE) * 0.1);
  const anchorA = anchored(A);
  const anchorB = anchored(B);
  if (anchorA && !anchorB) pull(B, -ux, -uz, -vy, s);
  else if (anchorB && !anchorA) pull(A, ux, uz, vy, s);
  else {
    pull(A, ux, uz, vy, s * 0.6);
    pull(B, -ux, -uz, -vy, s * 0.6);
  }

  tether.soundCd += step;
  if (tether.soundCd >= 16) {
    tether.soundCd = 0;
    try { A.playSound("note.bass", { volume: 0.8, pitch: 0.5 }); } catch {}
    try { B.playSound("note.bass", { volume: 0.8, pitch: 0.5 }); } catch {}
  }

  if (level === 2) {
    tether.strainTicks = 0;
    return;
  }
  tether.strainTicks += step;
  const over = tether.strainTicks - CONFIG.STRAIN_GRACE_TICKS;
  if (over >= 0 && over % CONFIG.STRAIN_INTERVAL_TICKS < step) {
    strain(A);
    strain(B);
  }
}

function syncLoops() {
  if (isRunning()) {
    if (tickId === undefined) tickId = system.runInterval(tick, CONFIG.TICK_STEP);
    if (sweepId === undefined) sweepId = system.runInterval(sweep, CONFIG.PROMPT_SWEEP_TICKS);
    return;
  }
  if (tickId !== undefined) {
    system.clearRun(tickId);
    tickId = undefined;
  }
  if (sweepId !== undefined) {
    system.clearRun(sweepId);
    sweepId = undefined;
  }
  pair = null;
  hudState.clear();
  resetTether();
  tether.waitCd = 0;
}

onStateChange(() => {
  pairDirty = true;
  syncLoops();
});
system.run(syncLoops);

world.afterEvents.playerSpawn.subscribe(({ player, initialSpawn }) => {

      if (partner.typeId === "minecraft:player") partner.sendMessage(`§a[TETHERED] ${player.name} respawned at your position.`);
    } catch {}
  }, CONFIG.RESPAWN_DELAY_TICKS);

    if (!pr) return;
    setRole(pr.breaker, BUILDER);
    setRole(pr.builder, BREAKER);

  } else if (ev.id === "tether:breaker" || ev.id === "tether:builder") {
    const target = resolveTarget(ev);
    if (!target) return;

Work together as a mining-and-combat partner and a building specialist while an elastic tether keeps you close. Tethered: Breaker and Builder divides survival abilities between two players, making teamwork essential.

## Overview

Two players share a 10-meter tether. The Breaker handles mining and combat but cannot place blocks or fluids.

## Key Features

### Asymmetric Roles
- **What Happens**: Each player chooses Breaker or Builder when the challenge starts.

### Elastic Tether
- **What Happens**: Partners feel tension from 6 meters and are pulled toward each other beyond 8 meters, including up and down. They can never get more than 10 meters apart, and strain damage applies beyond 9 meters after a short grace period.

### Tension Feedback
- **What Happens**: A particle beam, sound cues and an on-screen display show four distance states: slack, tense, overstretched and strained.

### Recall
- **What Happens**: Partners who are very far apart or in different dimensions for too long are brought back together.

### Respawn Support
- **What Happens**: A partner who dies respawns beside the survivor with temporary protection.

### Controller
- **What Happens**: The host or an operator uses the ALNStudios Controller to start the challenge, read the rules and, for solo testing, spawn a Test Dummy partner to be leashed to.

## How to Use

1. Import and enable the pack in a Minecraft Bedrock world.
2. The host or an operator uses the Controller and chooses Start.
3. Each player picks a role from the menu.
4. The Breaker gathers and defends while the Builder constructs. Stay within the leash range.
