import { world, ItemStack, ItemTypes } from "@minecraft/server";

const LISTINGS_KEY = "aln_market_listings";
const PAYOUTS_KEY = "aln_market_payouts";
const MAX_JSON_LENGTH = 30000;
const INV_SLOTS = 36;

const UNSUPPORTED = /shulker_box|bundle|potion|tipped_arrow|banner|firework|filled_map|written_book|writable_book|goat_horn|decorated_pot|suspicious_/;

function readJson(key, fallback) {
    try {
        const raw = world.getDynamicProperty(key);
        if (typeof raw !== "string") return fallback;
        const parsed = JSON.parse(raw);
        return parsed && typeof parsed === "object" ? parsed : fallback;
    } catch {
        return fallback;
    }
}

function writeJson(key, value) {
    const json = JSON.stringify(value);
    if (json.length > MAX_JSON_LENGTH) return false;
    world.setDynamicProperty(key, json);
    return true;
}

export function loadListings() {
    const data = readJson(LISTINGS_KEY, []);
    return Array.isArray(data) ? data : [];
}
export function saveListings(listings) {
    return writeJson(LISTINGS_KEY, listings);
}
function loadPayouts() {
    const data = readJson(PAYOUTS_KEY, {});
    return Array.isArray(data) ? {} : data;
}
function savePayouts(map) {
    return writeJson(PAYOUTS_KEY, map);
}

export function newListingId() {
    return Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
}

export function normalizeItemId(raw) {
    if (typeof raw !== "string") return undefined;
    let id = raw.trim().toLowerCase().replace(/\s+/g, "_");
    if (!id) return undefined;
    if (!id.includes(":")) id = "minecraft:" + id;
    return id;
}

export function maxStack(typeId) {
    try {
        return new ItemStack(typeId, 1).maxAmount;
    } catch {
        return undefined;
    }
}

export function resolveItemId(raw) {
    const id = normalizeItemId(raw);
    if (!id) return undefined;
    try {
        const type = ItemTypes.get(id);
        if (!type || type.id === "minecraft:air") return undefined;
        return maxStack(type.id) ? type.id : undefined;
    } catch {
        return undefined;
    }
}

export function prettyName(typeId) {
    const [ns, name] = typeId.includes(":") ? typeId.split(":") : ["minecraft", typeId];
    const nice = name.split("_").map((w) => w.charAt(0).toUpperCase() + w.slice(1)).join(" ");
    return ns === "minecraft" ? nice : `${nice} (${ns})`;
}

export function getContainer(player) {
    return player.getComponent("minecraft:inventory").container;
}

export function isPlainItem(item) {
    if (!item || item.nameTag || UNSUPPORTED.test(item.typeId)) return false;
    if (item.getLore().length > 0) return false;
    const ench = item.getComponent("minecraft:enchantable");
    if (ench && ench.getEnchantments().length > 0) return false;
    const dur = item.getComponent("minecraft:durability");
    if (dur && dur.damage > 0) return false;
    return true;
}

export function getSellableSummary(player) {
    const container = getContainer(player);
    const summary = new Map();
    for (let i = 0; i < Math.min(container.size, INV_SLOTS); i++) {
        const item = container.getItem(i);
        if (!isPlainItem(item)) continue;
        summary.set(item.typeId, (summary.get(item.typeId) ?? 0) + item.amount);
    }
    return summary;
}

export function countItem(player, typeId) {
    return getSellableSummary(player).get(typeId) ?? 0;
}

export function removeItem(player, typeId, amount) {
    if (countItem(player, typeId) < amount) return false;
    const container = getContainer(player);
    let remaining = amount;
    for (let i = 0; i < Math.min(container.size, INV_SLOTS) && remaining > 0; i++) {
        const item = container.getItem(i);
        if (!item || item.typeId !== typeId || !isPlainItem(item)) continue;
        const take = Math.min(remaining, item.amount);
        if (take === item.amount) {
            container.setItem(i, undefined);
        } else {
            item.amount -= take;
            container.setItem(i, item);
        }
        remaining -= take;
    }
    return remaining === 0;
}

export function giveItem(player, typeId, amount) {
    const container = getContainer(player);
    const max = new ItemStack(typeId, 1).maxAmount;
    let remaining = amount;
    while (remaining > 0) {
        const n = Math.min(remaining, max);
        const overflow = container.addItem(new ItemStack(typeId, n));
        if (overflow) player.dimension.spawnItem(overflow, player.location);
        remaining -= n;
    }
}

export function findOnlinePlayer(playerId) {
    return world.getAllPlayers().find((p) => p.id === playerId);
}

function buildPayoutEntry(listing, buyerName) {
    return {
        typeId: listing.wantTypeId,
        amount: listing.wantAmount,
        name: listing.wantDisplayName,
        sold: listing.sellDisplayName,
        buyer: buyerName,
    };
}

export function tryQueuePayout(listing, buyerName, commit = true) {
    const map = loadPayouts();
    (map[listing.sellerId] ??= []).push(buildPayoutEntry(listing, buyerName));
    if (!commit) return JSON.stringify(map).length <= MAX_JSON_LENGTH;
    return savePayouts(map);
}

export function routePayment(listing, buyerName) {
    const seller = findOnlinePlayer(listing.sellerId);
    if (seller) {
        giveItem(seller, listing.wantTypeId, listing.wantAmount);
        seller.onScreenDisplay.setActionBar(
            `§a[PlayerTrade] §f${buyerName} bought your ${listing.sellDisplayName}! Received ${listing.wantAmount}x ${listing.wantDisplayName}!`
        );
        seller.playSound("random.levelup");
        return true;
    }
    return tryQueuePayout(listing, buyerName, true);
}

export function deliverPayouts(player) {
    const map = loadPayouts();
    const entries = map[player.id];
    if (!Array.isArray(entries) || entries.length === 0) return;

    delete map[player.id];
    if (!savePayouts(map)) return;

    for (const e of entries) {
        giveItem(player, e.typeId, e.amount);
        player.sendMessage(
            `§a[PlayerTrade] §f${e.buyer} bought your ${e.sold}! Received ${e.amount}x ${e.name}!`
        );
    }
    player.playSound("random.levelup");
}

function applyFilters(all, st, player, have) {
    const q = st.search.trim().toLowerCase();
    const out = all.filter((l) => {
        if (st.mine && l.sellerId !== player.id) return false;
        if (st.category > 0 && categoryOf(l.sellTypeId) !== st.category) return false;
        if (st.afford && (l.sellerId === player.id || (have.get(l.wantTypeId) ?? 0) < l.wantAmount)) return false;
        if (q) {
            const hit = l.sellDisplayName.toLowerCase().includes(q)
                || l.sellTypeId.includes(q.replace(/\s+/g, "_"))
                || l.sellerName.toLowerCase().includes(q);
            if (!hit) return false;
        }
        return true;
    });
    const sorters = [
        (a, b) => b.timestamp - a.timestamp,
        (a, b) => a.timestamp - b.timestamp,
        (a, b) => a.sellDisplayName.localeCompare(b.sellDisplayName),
        (a, b) => a.sellerName.localeCompare(b.sellerName),
    ];
    return out.sort(sorters[st.sort]);
}

async function showPurchase(player) {
    const st = { category: 0, sort: 0, search: "", mine: false, afford: false, page: 0 };

    while (true) {
        const all = loadListings();
        const have = getSellableSummary(player);
        const mineCount = all.filter((l) => l.sellerId === player.id).length;
        const list = applyFilters(all, st, player, have);
        const pages = Math.max(1, Math.ceil(list.length / PAGE_SIZE));
        st.page = Math.min(st.page, pages - 1);
        const slice = list.slice(st.page * PAGE_SIZE, (st.page + 1) * PAGE_SIZE);

        const active = [`§f${CATEGORIES[st.category]}`, `§f${SORTS[st.sort]}`];
        if (st.search.trim()) active.push(`§fSearch: "${st.search.trim()}"`);
        if (st.mine) active.push("§fMy listings");
        if (st.afford) active.push("§fAffordable");

        const form = new ActionFormData()
            .title("§b§lGlobal Market")
            .header("§b§lMARKETPLACE")
            .label(`§7Showing §e${list.length}§7 of §e${all.length}§7 listings  §8|  §7Page §e${st.page + 1}§7/§e${pages}\n§7${active.join("§8 • §7")}`);

        const actions = [];
        const add = (text, icon, fn) => { form.button(text, icon); actions.push(fn); };

        add("§l§9Filters & Search\n§r§7Category, sort, find items", "textures/items/paper", async () => {
            await showFilters(player, st);
        });
        add(`§l§6My Listings\n§r§7${mineCount} active`, "textures/items/name_tag", async () => {
            st.mine = !st.mine;
            st.page = 0;
        });

        form.divider();
        if (slice.length === 0) {
            form.label("§7No listings match your filters. Try changing the category or search.");
        }
        for (const l of slice) {
            const canPay = l.sellerId !== player.id && (have.get(l.wantTypeId) ?? 0) >= l.wantAmount;
            const priceColor = l.sellerId === player.id ? "§e" : canPay ? "§a" : "§c";
            add(
                `§f§l${l.sellAmount}x ${l.sellDisplayName}\n§r§7Wants ${priceColor}${l.wantAmount}x ${l.wantDisplayName} §8| §7${l.sellerName}`,
                iconFor(l.sellTypeId),
                async () => { await showListingDetail(player, l.id); }
            );
        }

        form.divider();
        if (st.page > 0) add("§7< Previous Page", undefined, async () => { st.page--; });
        if (st.page < pages - 1) add("§7Next Page >", undefined, async () => { st.page++; });
        add("§7Back to Menu", "textures/ui/cancel", async () => "back");

        const res = await showForm(form, player);
        if (res.canceled) return;
        const result = await actions[res.selection]?.();
        if (result === "back") return;
    }
}

async function showFilters(player, st) {
    const form = new ModalFormData()

async function showListingDetail(player, listingId) {
    const l = loadListings().find((x) => x.id === listingId);
    if (!l) {
        bar(player, "§c[PlayerTrade] §fThat listing is no longer available.");
        return;
    }
    const own = l.sellerId === player.id;
    const have = countItem(player, l.wantTypeId);
    const haveColor = have >= l.wantAmount ? "§a" : "§c";

    const form = new ActionFormData()
        .title(own ? "§6§lYour Listing" : "§b§lListing Details")
        .header(`§f§l${l.sellAmount}x ${l.sellDisplayName}`)
        .label(`§7Category: §f${CATEGORIES[categoryOf(l.sellTypeId)]}\n§7Seller: §e${l.sellerName}${own ? " §7(you)" : ""}\n§7Listed: §f${timeAgo(l.timestamp)}`)
        .divider()
        .label(`§7Price: §b${l.wantAmount}x ${l.wantDisplayName}\n§7You have: ${haveColor}${have}§7 / §f${l.wantAmount}`)
        .divider();

    if (own) form.button("§c§lCancel Listing\n§r§7Reclaim your escrow items", "textures/ui/cancel");
    else form.button(`§a§lBuy Now\n§r§7Pay ${l.wantAmount}x ${l.wantDisplayName}`, iconFor(l.sellTypeId));
    form.button("§7Back to Market");

    const res = await showForm(form, player);
    if (res.canceled || res.selection !== 0) return;

    if (own) cancelListing(player, l.id);
    else attemptPurchase(player, l.id);
}

function cancelListing(player, listingId) {
    const listings = loadListings();
    const idx = listings.findIndex((l) => l.id === listingId && l.sellerId === player.id);
    if (idx < 0) {
        bar(player, "§c[PlayerTrade] §fThat listing is no longer available.");
        return;
    }
    const [l] = listings.splice(idx, 1);
    if (!saveListings(listings)) {
        bar(player, "§c[PlayerTrade] §fCould not cancel the listing. Try again.");
        return;
    }
    giveItem(player, l.sellTypeId, l.sellAmount);
    player.playSound("random.orb");
    bar(player, `§a[PlayerTrade] §fListing cancelled. Reclaimed ${l.sellAmount}x ${l.sellDisplayName}.`);
}

function attemptPurchase(buyer, listingId) {
    const listings = loadListings();
    const idx = listings.findIndex((l) => l.id === listingId);
    if (idx < 0) {
        bar(buyer, "§c[PlayerTrade] §fThat listing is no longer available.");
        return;
    }
    const l = listings[idx];
    if (l.sellerId === buyer.id) return;

    if (countItem(buyer, l.wantTypeId) < l.wantAmount) {
        bar(buyer, `§c[PlayerTrade] §fYou need §b${l.wantAmount}x ${l.wantDisplayName} §fto buy this!`);
        return;
    }
    if (!findOnlinePlayer(l.sellerId) && !tryQueuePayout(l, buyer.name, false)) {
        bar(buyer, "§c[PlayerTrade] §fThe seller's payout vault is full. Try again later.");
        return;
    }

    if (!removeItem(buyer, l.wantTypeId, l.wantAmount)) {
        bar(buyer, "§c[PlayerTrade] §fPayment failed. No items were taken.");
        return;
    }
    listings.splice(idx, 1);
    if (!saveListings(listings)) {
        giveItem(buyer, l.wantTypeId, l.wantAmount);
        bar(buyer, "§c[PlayerTrade] §fTrade failed. Your payment was returned.");
        return;
    }

    giveItem(buyer, l.sellTypeId, l.sellAmount);
    routePayment(l, buyer.name);
    buyer.playSound("random.levelup");
    bar(buyer, `§a[PlayerTrade] §fBought ${l.sellAmount}x ${l.sellDisplayName} for ${l.wantAmount}x ${l.wantDisplayName}!`);
}

**Open a Global Market with a Book renamed Player Trade and barter items with any player, even offline ones.**

> Adds a player-to-player market opened with a renamed Book, with escrow-held listings and offline payouts. It suits worlds and servers where players want to trade items safely.

## Overview

ALN-GlobalMarket lets players list items for trade, name the item and amount they want in return, and let other players buy those listings. Listed items are held safely in escrow, and payments reach the seller even when they are offline.

## Key Features

### Safe Escrow
- **What Happens**: Items you list leave your inventory and are held until they sell or you cancel the listing.
- **Good To Know**: Enchanted, renamed, damaged and container items cannot be listed.

### Marketplace Browser
- **What Happens**: Browse listings with item icons, category filters, sorting and search, with prices shown green when you can afford them and red when you cannot.
- **How It Works**: Choose Purchase, then use Filters & Search or the My Listings button.

### Listing Details
- **What Happens**: A details screen shows the seller, how long ago the item was listed and how much of the payment item you hold before you buy.

### Offline Payouts
- **What Happens**: If the seller is offline when you buy, the payment is vaulted and delivered the next time they join.

### Easy Cancellation
- **What Happens**: Cancel your own listing at any time and get the held items back.
- **How It Works**: Choose Purchase, open your own listing, then choose Cancel Listing.

## How to Use

1. Rename a Book to "Player Trade" on an Anvil.
2. Use the Book to open the Global Market.
3. Choose Sell to list an item, or Purchase to browse listings.
