export function sessionKey(dimensionId, x, y, z) {
  return dimensionId + ":" + x + "," + y + "," + z;

export function discInfo(typeId) {
  const vanilla = typeId.startsWith("minecraft:");
  const shortId = typeId.replace(/^[^:]*:/, "").replace(/^(music_disc_|record_)/, "");
  const known = vanilla ? DISCS[shortId] : undefined;
  const soundId = "record." + shortId;
  if (known) {
    return { name: known[0], totalSeconds: known[1], soundId };
  }
  const pretty = shortId
    .split("_")
    .filter((word) => word.length > 0)
    .map((word) => word.charAt(0).toUpperCase() + word.slice(1))
    .join(" ");
  return { name: pretty || shortId, totalSeconds: undefined, soundId };
}

export function formatTime(totalSeconds) {
  const safe = Math.max(0, Math.floor(totalSeconds));
  const minutes = Math.floor(safe / 60);
  const seconds = safe % 60;
  return minutes + ":" + (seconds < 10 ? "0" : "") + seconds;
}

export function getRecordPlayer(block) {

export function readJukebox(block, dimensionId) {
  const player = getRecordPlayer(block);
  const key = sessionKey(dimensionId, block.location.x, block.location.y, block.location.z);
  if (!player) {
    sessions.delete(key);
    return { state: "invalid" };
  }
  let record;
  let playing = false;
  try {
    record = player.getRecord();
    playing = player.isPlaying();
  } catch (error) {
    sessions.delete(key);
    return { state: "invalid" };
  }
  if (!record) {
    sessions.delete(key);
    return { state: "empty" };
  }
  const info = discInfo(record.typeId);
  if (!playing) {
    sessions.delete(key);
    return { state: "stopped", info };
  }
  let session = sessions.get(key);
  if (!session || session.typeId !== record.typeId) {
    session = { typeId: record.typeId, startTick: system.currentTick };
    sessions.set(key, session);
  }
  let elapsedSeconds = (system.currentTick - session.startTick) / 20;
  if (info.totalSeconds !== undefined && elapsedSeconds >= info.totalSeconds + 2) {
    sessions.delete(key);
    return { state: "stopped", info };
  }
  if (info.totalSeconds !== undefined) {
    elapsedSeconds = Math.min(elapsedSeconds, info.totalSeconds);
  }
  return { state: "playing", info, elapsedSeconds, startTick: session.startTick };
}

function areaName(pos) {
  const part = (n) => String(n).replace("-", "n");
  return "alnhp_" + pos.dimension.replace("minecraft:", "") + "_" + part(pos.x) + "_" + part(pos.y) + "_" + part(pos.z);

function tickingManager() {
  try {
    const manager = world.tickingAreaManager;
    return manager && typeof manager.hasCapacity === "function" ? manager : undefined;
  } catch (error) {
    return undefined;
  }
}

function keepLoadedByManager(manager, pos) {
  const name = areaName(pos);
  if (manager.hasTickingArea(name)) {
    return true;
  }
  const options = {
    dimension: world.getDimension(pos.dimension),
    from: { x: pos.x, y: pos.y, z: pos.z },
    to: { x: pos.x, y: pos.y, z: pos.z }
  };
  if (!manager.hasCapacity(options)) {
    return false;
  }
  const pending = manager.createTickingArea(name, options);
  if (pending && typeof pending.catch === "function") {
    pending.catch(() => {});
  }
  return true;
}

export function keepLoaded(pos) {
  const manager = tickingManager();
  if (manager) {
    try {
      return keepLoadedByManager(manager, pos);
    } catch (error) {}
  }
  let dimension;
  try {
    dimension = world.getDimension(pos.dimension);
  } catch (error) {
    return false;
  }
  const command = "tickingarea add circle " + pos.x + " " + pos.y + " " + pos.z + " 1 " + areaName(pos);
  try {
    if (dimension.runCommand(command).successCount > 0) {
      return true;
    }
  } catch (error) {}
  releaseLoaded(pos);
  try {
    return dimension.runCommand(command).successCount > 0;
  } catch (error) {
    return false;
  }
}

export function releaseLoaded(pos) {
  const manager = tickingManager();
  if (manager) {
    try {
      if (manager.hasTickingArea(areaName(pos))) {
        manager.removeTickingArea(areaName(pos));
        return;
      }
    } catch (error) {}

const RETRY_TICKS = 100;

system.runInterval(() => {
  const list = loadRegistry();
  if (list.length === 0) {
    return;
  }
  const keep = [];
  let changed = false;
  for (const entry of list) {
    let dimension;
    try {
      dimension = world.getDimension(entry.dimension);
    } catch (error) {
      keep.push(entry);
      continue;
    }
    const location = { x: entry.x, y: entry.y, z: entry.z };
    try {
      if (entry.kept === false && system.currentTick % RETRY_TICKS < MUTE_INTERVAL_TICKS && keepLoaded(entry)) {
        entry.kept = true;
        changed = true;
      }
      if (!dimension.isChunkLoaded(location)) {
        keep.push(entry);
        continue;
      }
      const block = dimension.getBlock(location);
      if (!block) {
        keep.push(entry);
        continue;
      }
      if (block.typeId !== JUKEBOX_ID) {
        releaseLoaded(entry);
        continue;
      }
      keep.push(entry);
      const status = readJukebox(block, entry.dimension);
      if (status.state !== "playing") {
        continue;
      }
      const area = "x=" + entry.x + ",y=" + entry.y + ",z=" + entry.z + ",r=" + MUTE_RADIUS;
      const selector =
        system.currentTick - status.startTick <= MUTE_WINDOW_TICKS
          ? "@a[" + area + "]"
          : "@a[" + area + ",tag=!" + LISTEN_TAG + "]";
      try {
        dimension.runCommand("stopsound " + selector + " " + status.info.soundId);
      } catch (error) {}
    } catch (error) {
      keep.push(entry);
    }
  }
  if (changed || keep.length !== list.length) {
    saveRegistry(keep);
  }

import { world, system, EquipmentSlot } from "@minecraft/server";
import {
  HEADPHONES_ID,
  JUKEBOX_ID,
  LISTEN_TAG,
  MUTE_WINDOW_TICKS,
  PREFIX,
  formatTime,
  parsePairing,
  readJukebox
} from "./headphone_common.js";

const STREAM_INTERVAL_TICKS = 5;
const START_GRACE_TICKS = 100;
const MAX_UNKNOWN_HOLD_SECONDS = 600;
let canSeek = false;
const PAIRED = "\u00a7aPaired \u00a78| ";
const listening = new Set();
const streams = new Map();

function stopAudio(player) {
  if (streams.has(player.id)) {
    streams.delete(player.id);
    try {
      player.stopMusic();
    } catch (error) {}
  }
  try {
    if (player.hasTag(LISTEN_TAG)) {
      player.removeTag(LISTEN_TAG);
    }
  } catch (error) {}
}

function tagListening(player) {
  try {
    if (!player.hasTag(LISTEN_TAG)) {
      player.addTag(LISTEN_TAG);
    }
  } catch (error) {}
}

function ensureStream(player, status) {
  const current = streams.get(player.id);
  if (current && current.trackStart === status.startTick && current.soundId === status.info.soundId) {
    return current;
  }
  const sinceStart = system.currentTick - status.startTick;
  if (sinceStart < MUTE_WINDOW_TICKS + 2) {
    return "sync";
  }
  if (sinceStart > START_GRACE_TICKS && !canSeek) {
    if (current) {
      stopAudio(player);
    }
    tagListening(player);
    return "late";
  }
  if (current) {
    stopAudio(player);
  }
  try {
    player.addTag(LISTEN_TAG);
  } catch (error) {}
  let offset = 0;
  try {
    const instance = player.playMusic(status.info.soundId, { volume: 1.0, fade: 0.0, loop: false });
    if (instance && typeof instance.seekTo === "function") {
      canSeek = true;
      if (sinceStart > START_GRACE_TICKS) {
        instance.seekTo(sinceStart / 20);
        offset = sinceStart / 20;
      }
    }
  } catch (error) {}
  const stream = {
    soundId: status.info.soundId,
    name: status.info.name,
    total: status.info.totalSeconds,
    trackStart: status.startTick,
    startedTick: system.currentTick,
    offset
  };
  streams.set(player.id, stream);
  return stream;
}

function streamElapsed(stream) {
  const seconds = (system.currentTick - stream.startedTick) / 20 + (stream.offset || 0);
  return stream.total === undefined ? seconds : Math.min(seconds, stream.total);
}

function nowPlaying(stream) {
  const total = stream.total === undefined ? "--:--" : formatTime(stream.total);
  return (
    PREFIX +
    PAIRED +
    "\u00a7fNow Playing: \u00a7e" +
    stream.name +
    " \u00a78| \u00a7b[ " +
    formatTime(streamElapsed(stream)) +
    " / " +
    total +
    " ]"
  );
}

function holdOrStop(player, message) {
  const stream = streams.get(player.id);
  const limit = stream ? (stream.total === undefined ? MAX_UNKNOWN_HOLD_SECONDS : stream.total) : 0;
  if (stream && streamElapsed(stream) < limit) {
    return nowPlaying(stream);
  }
  stopAudio(player);
  return message;
}

function renderFeedAndStream(player, item) {
  const pairing = parsePairing(item);
  if (!pairing) {
    stopAudio(player);
    return PREFIX + "\u00a77Not paired. Sneak and use on a Jukebox.";
  }
  let dimension;
  try {
    dimension = world.getDimension(pairing.dimension);
  } catch (error) {
    return holdOrStop(player, PREFIX + PAIRED + "\u00a7cJukebox dimension is unavailable.");
  }
  const location = { x: pairing.x, y: pairing.y, z: pairing.z };
  if (!dimension.isChunkLoaded(location)) {
    return holdOrStop(player, PREFIX + PAIRED + "\u00a7cJukebox is not loaded.");
  }
  const block = dimension.getBlock(location);
  if (!block || block.typeId !== JUKEBOX_ID) {
    stopAudio(player);
    return PREFIX + PAIRED + "\u00a7cJukebox no longer exists.";
  }
  const status = readJukebox(block, pairing.dimension);
  if (status.state === "empty") {
    stopAudio(player);
    return PREFIX + PAIRED + "\u00a77Jukebox has no disc.";
  }
  if (status.state === "stopped") {
    const stream = streams.get(player.id);
    if (stream && stream.total !== undefined && streamElapsed(stream) < stream.total) {
      return nowPlaying(stream);
    }
    stopAudio(player);
    return PREFIX + PAIRED + "\u00a77Waiting for the next track. \u00a78| \u00a7e" + status.info.name;
  }
  if (status.state !== "playing") {
    stopAudio(player);
    return PREFIX + PAIRED + "\u00a7cJukebox data unavailable.";
  }
  const stream = ensureStream(player, status);
  if (stream === "late") {
    return PREFIX + PAIRED + "\u00a77Waiting for the next track. \u00a78| \u00a7e" + status.info.name;
  }
  if (stream === "sync") {
    return PREFIX + PAIRED + "\u00a77Syncing...";
  }
  return nowPlaying(stream);
}

system.runInterval(() => {
  const present = new Set();
  for (const player of world.getAllPlayers()) {
    present.add(player.id);
    try {
      const equippable = player.getComponent("minecraft:equippable");
      const head = equippable ? equippable.getEquipment(EquipmentSlot.Head) : undefined;
      if (head && head.typeId === HEADPHONES_ID) {
        listening.add(player.id);
        player.onScreenDisplay.setActionBar(renderFeedAndStream(player, head));
      } else if (listening.has(player.id)) {
        listening.delete(player.id);
        stopAudio(player);
        player.onScreenDisplay.setActionBar("");
      } else if (player.hasTag(LISTEN_TAG)) {
        player.removeTag(LISTEN_TAG);
      }
    } catch (error) {
      continue;
    }
  }
  for (const id of listening) {
    if (!present.has(id)) {
      listening.delete(id);
      streams.delete(id);
    }
  }
}, STREAM_INTERVAL_TICKS);
