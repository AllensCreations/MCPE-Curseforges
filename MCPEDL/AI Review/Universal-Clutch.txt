function trackPlayerFall(player, vel) {
  const loc = player.location;
  let tracker = fallTrackers.get(player.id);
  const isFallingDown = vel.y < -0.15 && !player.isOnGround;

  if (!isFallingDown) {
    fallTrackers.delete(player.id);
    return 0;
  }
  if (!tracker) {
    tracker = { startY: loc.y, fallDist: 0, mode: getPlayerMode(player), threshold: getPlayerThreshold(player) };
    fallTrackers.set(player.id, tracker);
  } else {
    if (loc.y > tracker.startY) tracker.startY = loc.y;
    tracker.fallDist = Math.max(0, tracker.startY - loc.y);
  }
  return tracker.fallDist;
}

function resolveClutchItem(itemStack) {

function findBestClutchItemInInventory(container, dimensionId) {
  if (!container) return null;
  for (const key of getClutchPriority(dimensionId)) {
    for (let slot = 0; slot < 9 && slot < container.size; slot++) {
      const item = container.getItem(slot);
      if (item && itemMatchesPriority(item, key)) return { slot, item, inHotbar: true };

    for (let slot = 9; slot < container.size; slot++) {
      const item = container.getItem(slot);
      if (item && itemMatchesPriority(item, key)) return { slot, item, inHotbar: false };
    }
  }
  return null;
}

function acquireClutchItem(player, mode) {
  const equippable = player.getComponent("minecraft:equippable");
  const container = player.getComponent("minecraft:inventory")?.container;
  const isNether = player.dimension?.id === "minecraft:nether";

  const mainhand = equippable?.getEquipment("Mainhand");
  if (usableInDimension(mainhand, isNether)) {
    const def = resolveClutchItem(mainhand);
    if (def) return { itemStack: mainhand, isOffhand: false, containerSlot: player.selectedSlotIndex, definition: def };
  }

  const offhand = equippable?.getEquipment("Offhand");
  if (usableInDimension(offhand, isNether)) {
    const def = resolveClutchItem(offhand);
    if (def) return { itemStack: offhand, isOffhand: true, containerSlot: null, definition: def };
  }

  if (mode === "simple" || !container) return null;

  const match = findBestClutchItemInInventory(container, player.dimension?.id);
  if (!match) return null;

  if (match.inHotbar) {
    player.selectedSlotIndex = match.slot;
    const switched = container.getItem(match.slot);
    const definition = resolveClutchItem(switched);
    if (!definition) return null;
    return { itemStack: switched, isOffhand: false, containerSlot: match.slot, definition };
  }

  try {
    const heldSlot = player.selectedSlotIndex;
    const hotbarItem = container.getItem(heldSlot);
    const invItem = container.getItem(match.slot);
    container.setItem(heldSlot, invItem);
    container.setItem(match.slot, hotbarItem);
    const switched = container.getItem(heldSlot);
    const definition = resolveClutchItem(switched);
    if (!definition) return null;
    return { itemStack: switched, isOffhand: false, containerSlot: heldSlot, definition };
  } catch {
    return null;
  }
}

function isPassableOrAir(block) {

function findGroundBelow(dimension, location) {
  const px = Math.floor(location.x);
  const pz = Math.floor(location.z);
  const startY = Math.floor(location.y);
  let worldMinY = -64;
  try {
    worldMinY = dimension.heightRange.min;
  } catch {}
  const minY = Math.max(worldMinY, startY - 9);

  for (let y = startY; y >= minY; y--) {
    let block;
    try {
      block = dimension.getBlock({ x: px, y, z: pz });
    } catch {
      return null;
    }
    if (!block) return null;
    if (isPassableOrAir(block)) continue;
    if (NO_CLUTCH_SURFACES.has(block.typeId)) return null;

    let airAbove;
    try {
      airAbove = dimension.getBlock({ x: px, y: y + 1, z: pz });
    } catch {
      return null;
    }
    if (!airAbove || !isPassableOrAir(airAbove)) return null;
    return { surfaceBlock: block, airBlockAbove: airAbove, surfaceY: y };
  }
  return null;
}

function swapInto(player, clutchData, fromType, toType) {
  try {
    if (isCreative(player)) return;
    const makeItem = () => new ItemStack(toType, 1);

    if (clutchData.isOffhand) {
      const equippable = player.getComponent("minecraft:equippable");
      const off = equippable?.getEquipment("Offhand");
      if (off && off.typeId === fromType) {
        equippable.setEquipment("Offhand", makeItem());
        return;
      }
    } else {
      const container = player.getComponent("minecraft:inventory")?.container;
      if (container && clutchData.containerSlot !== null) {
        const item = container.getItem(clutchData.containerSlot);
        if (item && item.typeId === fromType) {
          container.setItem(clutchData.containerSlot, makeItem());
          return;
        }
      }
    }

    const container = player.getComponent("minecraft:inventory")?.container;
    if (!container) return;
    for (let i = 0; i < container.size; i++) {
      const item = container.getItem(i);
      if (item && item.typeId === fromType) {
        if (item.amount === 1) {
          container.setItem(i, makeItem());
        } else {
          item.amount -= 1;
          container.setItem(i, item);
          container.addItem(makeItem());
        }
        return;
      }
    }
  } catch {}
}

function consumeClutchItem(player, clutchData, replacementTypeId) {
  try {
    if (isCreative(player)) return;

    if (clutchData.isOffhand) {
      const equippable = player.getComponent("minecraft:equippable");
      const offhand = equippable?.getEquipment("Offhand");
      if (!offhand) return;
      if (replacementTypeId) {
        equippable.setEquipment("Offhand", new ItemStack(replacementTypeId, 1));
      } else if (offhand.amount > 1) {
        offhand.amount -= 1;
        equippable.setEquipment("Offhand", offhand);
      } else {
        equippable.setEquipment("Offhand", undefined);
      }
      return;
    }

    const container = player.getComponent("minecraft:inventory")?.container;
    if (!container || clutchData.containerSlot === null) return;
    const current = container.getItem(clutchData.containerSlot);
    if (!current) return;
    if (replacementTypeId) {
      container.setItem(clutchData.containerSlot, new ItemStack(replacementTypeId, 1));
    } else if (current.amount > 1) {
      current.amount -= 1;
      container.setItem(clutchData.containerSlot, current);
    } else {
      container.setItem(clutchData.containerSlot, undefined);
    }
  } catch {}
}

function placeBoatClutch(dimension, airBlock, entityType, player) {
  try {
    const boat = dimension.spawnEntity(entityType, {
      x: airBlock.location.x + 0.5,
      y: airBlock.location.y,
      z: airBlock.location.z + 0.5
    });
    if (!boat) return false;

    clutchBoats.set(player.id, { boat, tick: system.currentTick });

    const seatPlayer = () => {
      try {
        if (!boat.isValid() || !player.isValid() || isDead(player)) return true;
        const rideable = boat.getComponent("minecraft:rideable");
        return rideable ? rideable.addRider(player) !== false : false;
      } catch {
        return false;
      }
    };
    if (!seatPlayer()) system.runTimeout(seatPlayer, 1);
    return true;
  } catch {
    return false;
  }
}

function executeAutoScoop(clutch) {
  const { player, playerId, targetPos, dimension, clutchData } = clutch;
  try {

    } catch {}

    if (replacementTypeId) {
      container.setItem(clutchData.containerSlot, new ItemStack(replacementTypeId, 1));
    } else if (current.amount > 1) {
      current.amount -= 1;
      container.setItem(clutchData.containerSlot, current);
    } else {
      container.setItem(clutchData.containerSlot, undefined);
    }
  } catch {}
}

function placeBoatClutch(dimension, airBlock, entityType, player) {
  try {
    const boat = dimension.spawnEntity(entityType, {
      x: airBlock.location.x + 0.5,
      y: airBlock.location.y,
      z: airBlock.location.z + 0.5
    });
    if (!boat) return false;

    clutchBoats.set(player.id, { boat, tick: system.currentTick });

    const seatPlayer = () => {
      try {
        if (!boat.isValid() || !player.isValid() || isDead(player)) return true;
        const rideable = boat.getComponent("minecraft:rideable");
        return rideable ? rideable.addRider(player) !== false : false;
      } catch {
        return false;
      }
    };
    if (!seatPlayer()) system.runTimeout(seatPlayer, 1);
    return true;
  } catch {
    return false;
  }
}

function isDead(player) {
  try {
    return (player.getComponent("minecraft:health")?.currentValue ?? 20) <= 0;
  } catch {
    return false;
  }
}

function executeAutoScoop(clutch) {
  const { player, playerId, targetPos, dimension, clutchData } = clutch;
  try {
    if (!player.isValid() || isDead(player)) return;

    try {
      const block = dimension.getBlock(targetPos);
      if (block && (block.typeId === "minecraft:water" || block.typeId === "minecraft:flowing_water")) {
        block.setType("minecraft:air");
      }
    } catch {}

    try { dimension.playSound("bucket.fill_water", player.location, { volume: 1.0, pitch: 1.0 }); } catch {}
    try { player.playAnimation(ANIM); } catch {}
    swapInto(player, clutchData, "minecraft:bucket", "minecraft:water_bucket");
  } catch {
  } finally {
    activeWaterClutches.delete(playerId);
  }
}

function executeAutoScoopPowderSnow(clutch) {
  const { player, playerId, targetPos, dimension, clutchData } = clutch;
  try {
    if (!player.isValid() || isDead(player)) return;

    let stillSnow = false;
    try {
      const block = dimension.getBlock(targetPos);
      if (block && block.typeId === "minecraft:powder_snow") {
        stillSnow = true;
        block.setType("minecraft:air");
      }
    } catch {}

    if (stillSnow) {
      try { dimension.playSound("bucket.fill_powder_snow", player.location, { volume: 1.0, pitch: 1.0 }); } catch {}
      try { player.playAnimation(ANIM); } catch {}
      swapInto(player, clutchData, "minecraft:bucket", "minecraft:powder_snow_bucket");
    }
  } catch {
  } finally {
    activePowderSnowClutches.delete(playerId);
  }
}

function executeClutch(player, velocity, clutchData, ground, distToSurface) {
  const { definition } = clutchData;
  const { airBlockAbove } = ground;
  const dimension = player.dimension;

  try { player.playAnimation(ANIM); } catch {}
  try { dimension.playSound(definition.placeSound, player.location, { volume: 1.0, pitch: 1.0 }); } catch {}

  let placed = false;

  switch (definition.category) {
    case "water": {
      try {
        airBlockAbove.setType("minecraft:water");
        placed = true;
        consumeClutchItem(player, clutchData, "minecraft:bucket");
        activeWaterClutches.set(player.id, {
          player,
          playerId: player.id,
          targetPos: airBlockAbove.location,
          dimension,
          deployedTick: system.currentTick,
          scheduled: false,
          clutchData
        });
      } catch {}
      break;
    }
    case "powder_snow": {
      try {
        airBlockAbove.setType("minecraft:powder_snow");
        placed = true;
        consumeClutchItem(player, clutchData, "minecraft:bucket");
        activePowderSnowClutches.set(player.id, {
          player,
          playerId: player.id,
          targetPos: airBlockAbove.location,
          dimension,
          deployedTick: system.currentTick,
          scheduled: false,
          clutchData
        });
      } catch {}
      break;
    }
    case "vines": {
      try {
        airBlockAbove.setType(definition.blockId);
        if (airBlockAbove.typeId === definition.blockId) {
          consumeClutchItem(player, clutchData);
          placed = true;
        }
      } catch {}
      break;
    }
    case "block":
    case "sweet_berries": {
      try {
        airBlockAbove.setType(definition.blockId);
        consumeClutchItem(player, clutchData);
        placed = true;
      } catch {}
      break;
    }
    case "boat": {
      if (placeBoatClutch(dimension, airBlockAbove, definition.entityType, player)) {
        consumeClutchItem(player, clutchData);
        placed = true;
      }
      break;
    }
  }

  if (placed) lastClutchTick.set(player.id, system.currentTick);
}

function processPending(map, executor, currentTick) {
  for (const [playerId, clutch] of map.entries()) {
    let valid = false;
    try { valid = clutch.player.isValid(); } catch {}
    if (!valid) {
      map.delete(playerId);
      continue;
    }
    if (clutch.scheduled) continue;

    let vel = { x: 0, y: 0, z: 0 };
    try { vel = clutch.player.getVelocity(); } catch {}
    const landed = clutch.player.isOnGround === true || Math.abs(vel.y) < 0.1 || currentTick - clutch.deployedTick > 15;
    if (landed) {
      clutch.scheduled = true;
      system.runTimeout(() => executor(clutch), 3);
    }
  }
}

system.runInterval(() => {
  const currentTick = system.currentTick;

  processPending(activeWaterClutches, executeAutoScoop, currentTick);
  processPending(activePowderSnowClutches, executeAutoScoopPowderSnow, currentTick);

  for (const player of world.getAllPlayers()) {
    try {
      if (!player.isValid()) continue;

      let vel;
      try {
        vel = player.getVelocity();
      } catch {
        continue;
      }

      let notFalling = true;
      try {
        notFalling = vel.y > -0.15 || player.isOnGround;
      } catch {}
      if (notFalling) {
        fallTrackers.delete(player.id);
        continue;
      }

      let blocked = false;
      try {
        blocked = player.isGliding || player.isInWater || player.isSwimming;
      } catch {}
      if (blocked || isInactiveGameMode(player) || !isClutchEnabled(player)) {
        fallTrackers.delete(player.id);
        continue;
      }

      const currentFallDist = trackPlayerFall(player, vel);
      if (vel.y > -0.5) continue;

      const threshold = fallTrackers.get(player.id)?.threshold ?? getPlayerThreshold(player);
      const ground = findGroundBelow(player.dimension, player.location);
      if (!ground) continue;

      const distToSurface = player.location.y - (ground.surfaceY + 1);
      if (distToSurface < 0.1) continue;

      const tracker = fallTrackers.get(player.id);
      const totalDrop = tracker ? tracker.startY - (ground.surfaceY + 1) : currentFallDist;
      if (currentFallDist < threshold && totalDrop < threshold) continue;

      const mode = tracker?.mode ?? getPlayerMode(player);
      const fallSpeed = Math.abs(vel.y);

      if (mode === "extreme") {
        if (tracker) {
          if (tracker.itemTick === undefined || currentTick - tracker.itemTick >= 10) {
            tracker.hasItems = hasAnyClutchItem(player) || hasTotemOfUndying(player);
            tracker.itemTick = currentTick;
          }
          if (!tracker.hasItems) continue;
        } else if (!hasAnyClutchItem(player) && !hasTotemOfUndying(player)) continue;
      }

      if (currentTick - (lastClutchTick.get(player.id) ?? -100) < 25) continue;

      const rot = player.getRotation();
      if (mode !== "extreme" && rot.x < 55) continue;

      const triggerDist = Math.min(7, Math.max(4, fallSpeed * 2.6 + 1.5));
      if (distToSurface > triggerDist) continue;

      const clutchData = acquireClutchItem(player, mode);

      if (clutchData) {
        executeClutch(player, vel, clutchData, ground, distToSurface);
      } else {
        tryTriggerTotemFailSafe(player, Math.max(currentFallDist, totalDrop));
      }
    } catch {}
  }
}, 1);

function clearClutchBoat(playerId) {
  const entry = clutchBoats.get(playerId);
  clutchBoats.delete(playerId);
  try {
    if (entry && entry.boat.isValid() && system.currentTick - entry.tick <= BOAT_CLEAR_TICKS) {
      entry.boat.remove();
    }
  } catch {}
}

world.afterEvents.entityDie.subscribe((event) => {
  try {
    const dead = event.deadEntity;
    if (dead.typeId !== "minecraft:player") return;
    clearClutchBoat(dead.id);
  } catch {}
});

world.afterEvents.playerSpawn.subscribe((event) => {
  try {
    if (event.initialSpawn) return;
    clearClutchBoat(event.player.id);
  } catch {}
});

world.afterEvents.playerLeave.subscribe((event) => {
  clutchBoats.delete(event.playerId);
  fallTrackers.delete(event.playerId);
  lastClutchTick.delete(event.playerId);
  activeWaterClutches.delete(event.playerId);
  activePowderSnowClutches.delete(event.playerId);
});

system.afterEvents.scriptEventReceive.subscribe((event) => {
  if (event.id !== "clutch:on" && event.id !== "clutch:off") return;
  const player = event.sourceEntity;
  if (!player || player.typeId !== "minecraft:player") return;
  const enable = event.id === "clutch:on";
  try {
    player.setDynamicProperty(ENABLED_PROPERTY, enable);
    player.sendMessage(enable ? "§aUniversal Clutch: ON" : "§cUniversal Clutch: OFF");
  } catch {}
});

# Universal Clutch — Intelligent Fall Mitigation Script

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**AUTOMATIC CLUTCH SYSTEM & FALL PROTECTION UTILITY**

Automatically attempts a clutch when you fall while looking toward the ground. Supported items can reduce or prevent fall damage, and placed water or powder snow may be collected again after landing.

---

## Overview

Universal Clutch provides a vanilla-faithful safety mechanic for Minecraft Bedrock players. The system checks your fall and whether you are looking toward the ground.

When aiming down and falling beyond the configured block height threshold, the add-on acts shortly before impact, triggers an authentic arm swing animation with placement audio, deploys the appropriate clutch item from your mainhand or offhand, and automatically re-scoops placed water upon touchdown.

---

## Key Features

### Multi-Item Clutch Support
- **Description**: Automatically identifies and uses Water Buckets, Powder Snow Buckets, Twisting Vines, Hay Bales, Cobwebs, Slime Blocks, Scaffolding, Boats/Rafts, or Sweet Berries.

### Dimension-Aware Clutches
- **Description**: Automatically disables evaporating Water Buckets in the Nether, prioritizing Twisting Vines & Powder Snow Buckets instead.

### Dual Hand & Inventory Detection
- **Description**: Evaluates mainhand items first, with failover to offhand, hotbar, and upper inventory.

### Totem of Undying Fail-Safe
- **Description**: When falling to certain death with no clutch item, automatically searches inventory and equips a Totem of Undying into the offhand slot before impact.

### Authentic Boat Placement
- **Description**: Places the boat on the ground below and seats the player in it at the same moment, from about 7 blocks up. The boat is removed if the player dies within a few seconds of the clutch.

### Timed Placement
- **Description**: Checks the blocks below you and triggers placements shortly before impact rather than snapping prematurely.

### Arm Placement Animation
- **Description**: Triggers vanilla arm swing animations (the standard arm-swing animation) synced directly to placement audio.

### Automatic Item Recovery
- **Description**: Automatically retrieves placed water or powder snow 3 ticks after touchdown; verifies powder snow wasn't consumed to extinguish fire before re-scooping.

---

### Water Bucket
- **Placement Target**: Impact Air Block
- **Effect**: Negates fall damage; auto-scooped 3 ticks after landing (disabled in Nether)

### Twisting Vines

### Hay Bale

### Slime Block
- **Placement Target**: Impact Air Block

- **Effect**: Auto-equipped when falling with no clutch items to attempt to prevent a fatal fall

## When It Activates

The player must be falling from the selected height threshold while looking down (not required in Extreme Mode), and the add-on attempts placement close to impact.

## Operating Modes & Subpack Configuration

Universal Clutch supports three distinct operating modes selectable via subpack settings slider in World Settings or customizable per-player in-game.

Each player can turn Universal Clutch off or on with the commands `/scriptevent clutch:off` and `/scriptevent clutch:on` (cheats required). The setting is saved per player.

### Operating Modes

- **Full Mode (Auto-Switch Inventory):**
  If a clutch item is not already held in hand, the system automatically scans your hotbar and upper inventory for available clutch items, switches or swaps it directly into your active hand, executes the clutch placement, and remains holding the switched item. Requires the player to aim downward (looking down toward the ground).
- **Simple Mode (Held Item Only):**
  Clutch placements only trigger if you are actively holding a supported clutch item in your mainhand or offhand slot. If holding a sword, pickaxe, or empty hand, no placement will occur. Requires the player to aim downward (looking down toward the ground).
- **Extreme Mode (Auto-Switch, No Look-Down Needed):**
  When falling from dangerous heights, the add-on does not require you to look down, scans and auto-switches to the best available clutch item in inventory, places it shortly before impact, and remains holding the switched item. **No-Item Limitation:** If no suitable clutch item is available, the add-on cannot assist with the fall.

### Built-in Subpack Presets

Configurable via the Behavior Pack subpack settings slider in World Settings:

1. **Full Mode (Auto-Switch Inventory):** Recommended default. Automatically detects, equips, and places clutch items from inventory, keeping the switched item active.
2. **Simple Mode (Held Item Only):** Traditional manual clutching. Only executes if a supported clutch item is already held in mainhand or offhand.
3. **Extreme Mode (Auto-Switch, No Look-Down Needed):** Hands-free safety system with auto item switching that works without looking down. If inventory has no clutch item, the add-on cannot assist.

#!/usr/bin/env python3
"""Regenerate the subpack main.js copies from behavior_pack/scripts/main.js.

Edit only behavior_pack/scripts/main.js, then run this script. The Full, Simple
and Extreme copies differ from it only in the DEFAULT_MODE line.
"""
import os
import sys

root = os.path.dirname(os.path.abspath(__file__))
base_path = os.path.join(root, "behavior_pack", "scripts", "main.js")

with open(base_path, encoding="utf-8", newline="") as f:
    src = f.read()

marker = 'const DEFAULT_MODE = "full";'
if src.count(marker) != 1:
    sys.exit("Base main.js must contain exactly one: " + marker)

for mode in ("full", "simple", "extreme"):
    out = src.replace(marker, 'const DEFAULT_MODE = "%s";' % mode)
    path = os.path.join(root, "behavior_pack", "subpacks", mode + "_mode", "scripts", "main.js")
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w", encoding="utf-8", newline="") as f:
        f.write(out)
    print("synced", mode)
