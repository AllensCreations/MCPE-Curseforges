const JUKEBOX = "minecraft:jukebox";
const STATE_KEY = "aln_morediscs:active";
const STATE_VERSION = 2;
const EVENT_STARTED = "aln:disc_started";
const EVENT_STOPPED = "aln:disc_stopped";
const EVENT_QUERY = "aln:disc_query";
const CHECK_INTERVAL = 40;
const SOUND_VOLUME = 4;
const MAINHAND = (mc.EquipmentSlot && mc.EquipmentSlot.Mainhand) || "Mainhand";

// Every disc that sits in a Jukebox, playing or finished, keyed by dimension and position.
const stored = new Map();
let loaded = false;

function keyOf(dimensionId, x, y, z) {
  return `${dimensionId}|${x}|${y}|${z}`;
}

function load() {
  if (loaded) return;
  loaded = true;
  try {
    const raw = world.getDynamicProperty(STATE_KEY);
    if (typeof raw !== "string") return;
    for (const entry of JSON.parse(raw)) {
      if (entry && entry.v === STATE_VERSION && typeof entry.id === "string" && DISCS[entry.disc]) {
        entry.playing = false;
        stored.set(entry.id, entry);
      }
    }
  } catch (error) {
    stored.clear();
  }
}

function save() {
  try {
    world.setDynamicProperty(STATE_KEY, JSON.stringify([...stored.values()]));
  } catch (error) {}
}

function send(eventId, payload) {
  try {
    world.getDimension("overworld").runCommand(`scriptevent ${eventId} ${JSON.stringify(payload)}`);
  } catch (error) {}
}

function stopSound(disc) {
  for (const other of stored.values()) {
    if (other.playing && other.disc === disc) return;
  }
  try {
    world.getDimension("overworld").runCommand(`stopsound @a ${DISCS[disc].sound}`);
  } catch (error) {}
}

function stopPlaying(entry, reason) {
  if (!entry.playing) return;
  entry.playing = false;
  save();
  send(EVENT_STOPPED, { id: entry.id, disc: entry.disc, track: entry.track, reason });
  if (reason !== "ended") stopSound(entry.disc);
}

function removeDisc(entry, reason, dimension, dropHeight) {
  stopPlaying(entry, reason);
  stored.delete(entry.id);
  save();
  try {
    dimension.spawnItem(new mc.ItemStack(entry.disc, 1), { x: entry.x + 0.5, y: entry.y + dropHeight, z: entry.z + 0.5 });
  } catch (error) {}
}

function startTrack(dimension, entry) {
  entry.playing = true;
  entry.startedAt = Date.now();
  save();
  send(EVENT_STARTED, entry);
  try {
    dimension.playSound(DISCS[entry.disc].sound, { x: entry.x + 0.5, y: entry.y + 1, z: entry.z + 0.5 }, { volume: SOUND_VOLUME, pitch: 1 });
  } catch (error) {}
}

function heldSlot(player) {
  try {
    return player.getComponent("minecraft:equippable").getEquipmentSlot(MAINHAND);
  } catch (error) {}
  try {
    return player.getComponent("minecraft:inventory").container.getSlot(player.selectedSlotIndex);
  } catch (error) {}
  return undefined;
}

function takeHeldDisc(player, discId) {
  try {
    const slot = heldSlot(player);
    if (!slot) return false;
    const item = slot.getItem();
    if (!item || item.typeId !== discId) return false;
    if (item.amount > 1) slot.amount = item.amount - 1;
    else slot.setItem(undefined);
    const after = slot.getItem();
    return !(after && after.typeId === discId && after.amount >= item.amount);
  } catch (error) {
    return false;
  }
}

function insertDisc(player, dimensionId, location, discId) {
  load();
  const disc = DISCS[discId];
  const dimension = world.getDimension(dimensionId);
  const id = keyOf(dimensionId, location.x, location.y, location.z);
  if (stored.has(id)) return;
  let block;
  try {
    block = dimension.getBlock(location);
  } catch (error) {
    return;
  }
  if (!block || block.typeId !== JUKEBOX) return;
  if (!takeHeldDisc(player, discId)) return;
  const entry = {
    v: STATE_VERSION,
    id,
    dim: dimensionId,
    x: location.x,
    y: location.y,
    z: location.z,
    disc: discId,
    track: disc.track,
    title: disc.title,
    seconds: disc.seconds,
    startedAt: 0,
    playing: false,
  };
  stored.set(id, entry);
  startTrack(dimension, entry);
}

function ejectDisc(id) {
  load();
  const entry = stored.get(id);
  if (!entry) return;
  removeDisc(entry, "ejected", world.getDimension(entry.dim), 1.1);
}

function vanillaRecordInside(block) {
  try {
    const recordPlayer = block.getComponent("minecraft:record_player");
    return !!(recordPlayer && recordPlayer.getRecord());
  } catch (error) {
    return false;
  }
}

function tick() {
  load();
  if (stored.size === 0) return;
  const now = Date.now();
  for (const entry of [...stored.values()]) {
    let dimension;
    let block;
    try {
      dimension = world.getDimension(entry.dim);
      block = dimension.getBlock({ x: entry.x, y: entry.y, z: entry.z });
    } catch (error) {
      continue;
    }
    if (!block) continue;
    if (block.typeId !== JUKEBOX) {
      removeDisc(entry, "broken", dimension, 0.5);
      continue;
    }
    if (!entry.playing) continue;
    if (now - entry.startedAt >= entry.seconds * 1000 + 2000) {
      stopPlaying(entry, "ended");
      continue;
    }
  }
}

export function getActiveDiscs() {
  load();
  return [...stored.values()].filter((entry) => entry.playing);
}

export function initDiscJukebox() {
  world.beforeEvents.playerInteractWithBlock.subscribe((event) => {
    const { block, player, itemStack, isFirstEvent } = event;
    if (!isFirstEvent || block.typeId !== JUKEBOX) return;
    if (player.isSneaking && itemStack) return;
    load();
    const dimensionId = block.dimension.id;
    const location = { x: block.location.x, y: block.location.y, z: block.location.z };
    const id = keyOf(dimensionId, location.x, location.y, location.z);
    if (stored.has(id)) {
      event.cancel = true;
      system.run(() => ejectDisc(id));
      return;
    }
    const held = itemStack ? itemStack.typeId : undefined;
    if (!held || !DISCS[held] || vanillaRecordInside(block)) return;
    event.cancel = true;
    system.run(() => insertDisc(player, dimensionId, location, held));
  });

  world.afterEvents.playerBreakBlock.subscribe((event) => {
    if (event.brokenBlockPermutation.type.id !== JUKEBOX) return;
    const { x, y, z } = event.block.location;
    const id = keyOf(event.block.dimension.id, x, y, z);
    system.run(() => {
      load();
      const entry = stored.get(id);
      if (entry) removeDisc(entry, "broken", world.getDimension(entry.dim), 0.5);
    });
  });

  system.afterEvents.scriptEventReceive.subscribe(
    (event) => {
      if (event.id !== EVENT_QUERY) return;
      load();
      for (const entry of stored.values()) {
        if (entry.playing) send(EVENT_STARTED, entry);
      }
    },
    { namespaces: ["aln"] }
  );

  system.runInterval(tick, CHECK_INTERVAL);
}

# ALN MoreDiscs: script notes (scripts/lib)

For the AI review. Keep this file in step with the scripts: update it whenever a file in `scripts/` changes.
Nothing here was run in game.

## Files

| File | Lines | Purpose |
| :-- | --: | :-- |
| `scripts/disc_jukebox.js` | 268 | Own Jukebox handling for the 30 discs: insert, play, eject, end, break, headphones events |
| `scripts/main.js` | 4 | Entry point, calls `initDiscJukebox()` |

## Imports

- `@minecraft/server` (in `disc_jukebox.js`, namespace import)
- `./disc_jukebox.js` (in `main.js`)

## Events

- `world.beforeEvents.playerInteractWithBlock`: a disc used on a Jukebox is cancelled and handled by the script; any use on a Jukebox that holds one of the discs ejects it
- `world.afterEvents.playerBreakBlock`: breaking a Jukebox that holds a disc ends the track and drops the disc
- `system.afterEvents.scriptEventReceive`: `aln:disc_query` resends `aln:disc_started` for every track that is playing
- `system.runInterval`: every 40 ticks ends finished tracks, drops the disc if the Jukebox is gone

## Custom ids

- Script events: `aln:disc_started`, `aln:disc_stopped`, `aln:disc_query`
- Dynamic property: `aln_morediscs:active` (stored discs, state version 2)
- Item ids: the 30 `aln:music_disc_<name>` ids listed in the `DISCS` table of `disc_jukebox.js`
- Sound events: `record.<name>` for each disc, defined in `resource_pack/sounds/sound_definitions.json`

## Features found

- cancel an interaction (beforeEvents) and finish the work in `system.run`
- dimension.playSound, `stopsound` through runCommand, spawnItem
- dynamic properties (saved data)
- repeating timer (system.runInterval)

## Notes (AI review)

- The discs have no `minecraft:record` component: its `sound_event` only accepts vanilla names (13, cat, blocks, chirp, far, mall, mellohi, stal, strad, ward, 11, wait, pigstep, otherside, 5, relic) unless the Beta APIs experiment is on (checked in the docs on 2026-11-22). The game log showed all 30 items failing on it. The script therefore plays the tracks itself, so no experiment is needed.
- Differences from a vanilla record: no spinning disc, no comparator signal, a Jukebox that already holds a vanilla record has to be emptied first, and a track does not resume after the world reloads (the disc stays inside).
- Not verified in the docs: `ContainerSlot.setItem`/`amount`, the `minecraft:equippable` and `minecraft:inventory` component ids, `minecraft:record_player` `getRecord()`, the `volume` and `pitch` fields of the sound options, and the `stopsound` command with a custom sound. Each use is wrapped so a failure does nothing instead of breaking the pack.
- Note particles were removed in v7: the vanilla note particle asked for color variables (`.r`, `.g`, `.b`) that the game log reported as missing, and its definition could not be checked in the docs.
- Not tested in game.

**Craft thirty original music discs, each with its own 55 to 92 second track, and play them in a Jukebox in Survival.**

> ALN MoreDiscs adds thirty music discs, Track 1 to Track 30, each crafted from Paper, Coal, Flint, a Note Block and one or two accent items. They give a Jukebox new music without any other add-on.

## Overview

Craft a disc at a Crafting Table, put it in a Jukebox and the matching track plays. Every track is an original piece made for this pack, so the thirty discs add music to your world without using any existing songs.
