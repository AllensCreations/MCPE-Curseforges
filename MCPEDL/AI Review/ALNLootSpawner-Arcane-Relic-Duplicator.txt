function getRarity(typeId) {
  if (EPIC_ITEMS.has(typeId))   return "epic";
  if (RARE_ITEMS.has(typeId))   return "rare";
  if (UNCOMMON_ITEMS.has(typeId)) return "uncommon";
  return "common";
}

let spawners = null;
let dirty = false;
let steps = 0;

function keyOf(dimId, x, y, z) {
  return `${dimId}|${x}|${y}|${z}`;
}

function loadSpawners() {
  if (spawners) return spawners;
  spawners = new Map();
  try {
    const raw = world.getDynamicProperty(REGISTRY_PROP);
    if (typeof raw === "string" && raw) {
      for (const s of JSON.parse(raw)) {
        if (s && typeof s.item === "string" && typeof s.d === "string") {
          spawners.set(keyOf(s.d, s.x, s.y, s.z), s);
        }
      }
    }
  } catch (e) {}
  return spawners;
}

function saveSpawners() {
  try {
    world.setDynamicProperty(REGISTRY_PROP, JSON.stringify([...loadSpawners().values()]));
    dirty = false;
  } catch (e) {}
}

function importLegacy(block) {
  const loc = block.location;
  const legacyKey = `${loc.x}:${loc.y}:${loc.z}:${block.dimension.id}`;
  try {
    const item = world.getDynamicProperty(`${LEGACY_ITEM}_${legacyKey}`);
    if (typeof item !== "string" || !item) return undefined;
    const rarityProp = world.getDynamicProperty(`${LEGACY_TYPE}_${legacyKey}`);
    const tickProp = world.getDynamicProperty(`${LEGACY_TICK}_${legacyKey}`);
    const rarity = typeof rarityProp === "string" && RARITY[rarityProp] ? rarityProp : getRarity(item);
    const entry = {
      d: block.dimension.id, x: loc.x, y: loc.y, z: loc.z,
      item, rarity,
      left: typeof tickProp === "number" && tickProp > 0 ? tickProp : RARITY[rarity].ticks
    };
    loadSpawners().set(keyOf(entry.d, entry.x, entry.y, entry.z), entry);
    world.setDynamicProperty(`${LEGACY_ITEM}_${legacyKey}`, "");
    saveSpawners();
    return entry;
  } catch (e) {
    return undefined;
  }
}

function prettyName(typeId) {
  return typeId.replace("minecraft:", "").split("_").map((w) => w.charAt(0).toUpperCase() + w.slice(1)).join(" ");
}

function ticksToTime(ticks) {
  const secs = Math.max(0, Math.ceil(ticks / 20));
  if (secs >= 60) {
    const m = Math.floor(secs / 60);
    const s = secs % 60;
    return s > 0 ? `${m}m ${s}s` : `${m}m`;
  }
  return `${secs}s`;
}

function nearbyPlayers(entry, range) {
  const out = [];
  try {
    for (const p of world.getAllPlayers()) {
      if (p.dimension.id !== entry.d) continue;
      const dx = p.location.x - entry.x;
      const dy = p.location.y - entry.y;
      const dz = p.location.z - entry.z;
      if (dx * dx + dy * dy + dz * dz <= range * range) out.push(p);
    }
  } catch (e) {}
  return out;
}

function sparks(dimension, entry, count) {
  for (let i = 0; i < count; i++) {
    try {
      dimension.spawnParticle("minecraft:basic_flame_particle", {
        x: entry.x + 0.15 + Math.random() * 0.7,
        y: entry.y + 0.2 + Math.random() * 0.7,
        z: entry.z + 0.15 + Math.random() * 0.7
      });
    } catch (e) {}
  }
}

function ejectDuplicate(dimension, entry) {
  let placed = false;
  try {
    const below = dimension.getBlock({ x: entry.x, y: entry.y - 1, z: entry.z });
    if (below && OUTPUT_CONTAINERS.has(below.typeId)) {
      const inv = below.getComponent("minecraft:inventory");
      if (inv && inv.container) {
        const leftover = inv.container.addItem(new ItemStack(entry.item, 1));
        placed = !leftover;
      }
    }
  } catch (e) {}
  if (!placed) {
    try {
      dimension.spawnItem(new ItemStack(entry.item, 1), { x: entry.x + 0.5, y: entry.y + 1.1, z: entry.z + 0.5 });
    } catch (e) {}
  }
  sparks(dimension, entry, 14);
  for (const p of nearbyPlayers(entry, SPARK_RANGE)) {
    try { p.playSound("random.orb"); } catch (e) {}
  }
}

function dropTemplate(dimension, entry) {
  try {
    dimension.spawnItem(new ItemStack(entry.item, 1), { x: entry.x + 0.5, y: entry.y + 0.5, z: entry.z + 0.5 });
  } catch (e) {}
}

function giveToPlayer(player, itemId) {
  let leftover = true;
  try {
    const inv = player.getComponent("minecraft:inventory");
    if (inv && inv.container) leftover = inv.container.addItem(new ItemStack(itemId, 1));
  } catch (e) {}
  if (leftover) {
    try {
      player.dimension.spawnItem(new ItemStack(itemId, 1), player.location);
    } catch (e) {}
  }
}

async function openSpawnerUI(player, key) {
  const entry = loadSpawners().get(key);
  if (!entry) return;
  const rar = RARITY[entry.rarity] || RARITY.common;
  const pct = Math.max(0, Math.min(100, Math.floor(((rar.ticks - entry.left) / rar.ticks) * 100)));

  const form = new ActionFormData()
    .title("Loot Spawner")
    .body(
      `Item: ${prettyName(entry.item)}\n` +
      `Rarity: ${rar.color}${rar.label}§r\n\n` +
      `Progress: ${pct}%\n` +
      `Next duplicate in: ${ticksToTime(entry.left)}`
    )
    .button("Retrieve Template Item")
    .button("Exit");

  let res;
  try { res = await form.show(player); } catch (e) { return; }
  if (!res || res.canceled || res.selection !== 0) return;

  const current = loadSpawners().get(key);
  if (!current) return;
  loadSpawners().delete(key);
  saveSpawners();
  giveToPlayer(player, current.item);
  try { player.playSound("random.orb"); } catch (e) {}
  player.onScreenDisplay.setActionBar("§aTemplate retrieved.");
}

function takeOneFromPlayer(player, typeId) {
  try {
    const inv = player.getComponent("minecraft:inventory");
    if (!inv || !inv.container) return false;
    const container = inv.container;
    for (let i = 0; i < container.size; i++) {
      const slot = container.getItem(i);
      if (slot && slot.typeId === typeId) {
        if (slot.amount > 1) {
          slot.amount -= 1;
          container.setItem(i, slot);
        } else {
          container.setItem(i, undefined);
        }
        return true;
      }
    }
  } catch (e) {}
  return false;
}

world.afterEvents.playerInteractWithBlock.subscribe(({ player, block, beforeItemStack }) => {
  if (block.typeId !== BLOCK_ID) return;
  const loc = block.location;
  const key = keyOf(block.dimension.id, loc.x, loc.y, loc.z);

  if (loadSpawners().has(key) || importLegacy(block)) {
    system.run(() => openSpawnerUI(player, key));
    return;
  }

  const held = beforeItemStack;
  if (!held || held.typeId === "minecraft:air" || held.typeId === BLOCK_ID) {
    player.onScreenDisplay.setActionBar("§eHold an item and interact to attune it.");
    return;
  }
  const typeId = held.typeId;
  if (BLOCKED_CONTAINERS.has(typeId)) {
    player.onScreenDisplay.setActionBar("§cFilled containers cannot be attuned.");
    return;
  }
  if (!takeOneFromPlayer(player, typeId)) return;

  const rarity = getRarity(typeId);
  loadSpawners().set(key, {
    d: block.dimension.id, x: loc.x, y: loc.y, z: loc.z,
    item: typeId, rarity, left: RARITY[rarity].ticks
  });
  saveSpawners();
  try { player.playSound("beacon.activate"); } catch (e) {}
  const rar = RARITY[rarity];
  player.onScreenDisplay.setActionBar(`§bAttuned: §r${prettyName(typeId)} §8[${rar.color}${rar.label}§8]`);
});

const misses = new Map();

system.runInterval(() => {
  const reg = loadSpawners();
  if (reg.size === 0) return;
  steps++;

  for (const [key, entry] of [...reg.entries()]) {
    let dimension;
    let block;
    try {
      dimension = world.getDimension(entry.d);
      block = dimension.getBlock({ x: entry.x, y: entry.y, z: entry.z });
    } catch (e) {
      continue;
    }
    if (!block) continue;

    if (block.typeId !== BLOCK_ID) {
      const count = (misses.get(key) || 0) + 1;
      if (count >= 2) {
        dropTemplate(dimension, entry);
        reg.delete(key);
        misses.delete(key);
        dirty = true;
      } else {
        misses.set(key, count);
      }
      continue;
    }
    misses.delete(key);

    const total = (RARITY[entry.rarity] || RARITY.common).ticks;
    entry.left -= STEP_TICKS;
    dirty = true;

    if (entry.left <= 0) {
      entry.left = total;
      ejectDuplicate(dimension, entry);
    } else if (nearbyPlayers(entry, SPARK_RANGE).length > 0) {
      const progress = 1 - entry.left / total;
      sparks(dimension, entry, 1 + Math.floor(progress * 5));
    }
  }

  if (dirty && steps % SAVE_EVERY_STEPS === 0) saveSpawners();
}, STEP_TICKS);

- **Duplication timer**: The old timer loop searched a 64-block volume every tick through a call whose signature could not be verified. Spawners are now tracked in a saved list and checked once per second, and they keep counting after the world is reloaded.
- **Container output**: Duplicates now go into a chest, barrel, or hopper below through the container's own add-item routine, and pop out on top of the block when the container is full.
- **Description accuracy**: The block renders as solid tinted glass, so the description no longer promises see-through panels.

## [1.0.3] - 2026-11-22 - Fix Black Rendering: Unify Render Method

### Fixed
- **Block rendering as solid black**: The glass and core material instances carried render_method "alpha_test" with a texture using partial-alpha pixels; alpha_test only supports fully transparent or fully opaque pixels, so every partial-alpha pixel in the glass texture was discarded outright, leaving the chamber faces unrendered. All four material instances (frame, glass, core, *) now use render_method "opaque", which draws the texture's RGB color regardless of alpha. This trades true see-through glass for a solid tinted glass look, but renders correctly and reliably.

## [1.0.2] - 2026-11-22 - Bug Fixes: Render Method and Script Crash

### Fixed
- **MaterialInstances render conflict**: All material instances now use "alpha_test" uniformly. Bedrock rejects any block where named instances mix opaque and alpha_test, causing a component error and warning on load.
- **Script crash in tick loop**: blockKey received block.location (a plain Vector3 with no dimension property) instead of the Block object. All four call sites now pass the Block directly so dimension.id resolves correctly.

## [1.0.1] - 2026-11-22 - Bug Fixes: Block Definition and Render Method

### Rarity-Based Duplication Engine
- **What Happens**: Players insert an item and the block classifies it into one of four tiers: Common (8-second cycle), Uncommon (20-second cycle), Rare (60-second cycle), or Epic (240-second cycle), then counts down and produces one copy on each completion.

### Interactive Inspection UI
- **What Happens**: Players interact with an active Loot Spawner to open a form showing the attuned item name, its rarity tier with a color code, the live progress percentage, and the countdown until the next duplicate.

### Template Retrieval
- **What Happens**: Players press Retrieve Template Item in the inspection UI to stop duplication and return the original item to their inventory, resetting the block to empty. Breaking the block also drops the template item.

### Automatic Container Output
- **What Happens**: When a duplication cycle completes, the block checks the space directly below for a chest, barrel, or hopper and inserts the duplicate there; if no container is present the item appears on top of the block instead.

### Survival Crafting Recipe
- **What Happens**: Players craft the Loot Spawner at a crafting table using Glass Panes, Obsidian, and Gold Ingots in a 3x3 ring pattern, with no experiment requirement.

### Nested Container Protection
- **What Happens**: Players cannot attune filled shulker boxes, chests, or bundles, preventing rarity timer bypasses through container nesting.

### Light-Emitting Chamber
- **What Happens**: The placed Loot Spawner emits light level 10 from its arcane core, illuminating the surrounding area.

## How to Use

1. Craft the Loot Spawner using the shaped recipe: Glass Panes in the corners and edges, Obsidian in the remaining outer slots, and a Gold Ingot in the center.
2. Place the Loot Spawner block in your world.
3. Hold any item you want to duplicate and interact (right-click or tap) the block to attune it. One unit of the item is consumed as the template.
4. Wait for the duplication timer to complete. Place a chest, barrel, or hopper directly below the block to collect duplicates automatically.
5. Interact with the active block at any time to open the inspection panel and check progress or retrieve your template.
