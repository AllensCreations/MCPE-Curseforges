## Authority

## Procedure

4. Make the smallest possible change. No unrelated rewrites. Never invent features, versions, links, assets, test results or compatibility. Never write comments in shipped `.js`, `.mcfunction` or `.json` files.
5. Write the spec (below), test it (below), then reply (below). For changes to an existing project, never deliver a .sh: this overrides any older "deliver the .sh" wording. A NEW pack still follows `mcpe-new-pack` and delivers its generated .sh.

## The spec

File name: `spec_vN.txt`, a downloadable file. N = CURRENT VERSION + 1, read from the attached repo export, never guessed. The user builds the .sh on their phone with `scripts/make_update_script.py`.

Format (`scripts/apply_update.py`):

- Start: `@@ version vN`, then `@@ project MCPEDL/<Category>/<Name>`, then one `@@ file <path relative to the project>` block per file.
- Directives: replace/with, replace-all, insert-after, insert-before, delete, append, create, overwrite, changelog. Remove files with `@@ delete-file <repo-root path>`.
- Small change: replace/insert only. Anchors are copied from the attached export, never from memory, and must occur exactly once. Do not end an anchor with a newline.
- Large redesign: `@@ overwrite` (new content must exceed 60% of the old file) or `@@ create` for new files. Never many fragile edits.
- A content line that starts with `@@` is written `@@@`.
- End with `@@ changelog <Update Name>` using `### Added`, `### Improved & Balanced`, `### Fixed` and `- **Title**: text`. Never touch manifest versions or README links.
- New pixel textures: `@@ create` the pixel JSON under `texture_sources/`, then add the PNG step to the Termux commands.
- Keep README.md and DESCRIPTION.md accurate (INSTRUCTIONS Part 2 template). Never claim an in-game test.

**Test before delivering.** Apply the spec to the export text in the sandbox with `apply_update.py` (`--dry-run`, then a real run on a copy). Check every anchor, validate each changed JSON, run `node --check` on JS. Report every check not run.

- Never edit manifest versions or the README version or download link. The auto-version-bump workflow does that on push.

import { world, system, EntityComponentTypes, GameMode } from "@minecraft/server";
import { PLAYER_ID } from "./info.js";
import { isOn } from "./store.js";
import { tune } from "./tuning.js";
import { isFriend } from "./friends.js";

const ARROW = "minecraft:arrow";

const GRAVITY = 0.05;
const MIN_SPEED = 0.8;

const minus = (a, b) => ({ x: a.x - b.x, y: a.y - b.y, z: a.z - b.z });
const length = (v) => Math.hypot(v.x, v.y, v.z);

const NEUTRAL = new Set([
  "minecraft:wolf",
  "minecraft:enderman",
  "minecraft:zombie_pigman",
  "minecraft:piglin",
  "minecraft:iron_golem",
  "minecraft:bee",
  "minecraft:llama",
  "minecraft:trader_llama",
  "minecraft:polar_bear",
  "minecraft:dolphin",
  "minecraft:panda",
  "minecraft:goat",
]);
const IGNORED = new Set(["minecraft:armor_stand"]);

function kindOf(entity) {
  if (NEUTRAL.has(entity.typeId)) return "aimNeutral";
  return entity.matches({ families: ["monster"] }) ? "aimHostile" : "aimPassive";
}

function candidates(owner) {
  const options = { location: owner.location, maxDistance: tune(owner, "aimRange") };
  const found = [];
  if (tune(owner, "aimPlayers") === 1) {
    found.push(...owner.dimension.getPlayers({ ...options, excludeGameModes: [GameMode.Creative, GameMode.Spectator] }));
  }
  for (const entity of owner.dimension.getEntities(options)) {
    if (entity.id === owner.id || entity.typeId === PLAYER_ID || IGNORED.has(entity.typeId)) continue;
    if (!entity.getComponent(EntityComponentTypes.Health)) continue;
    const kind = kindOf(entity);
    if (tune(owner, kind) !== 1) continue;
    if (kind !== "aimHostile" && entity.getComponent(EntityComponentTypes.IsTamed)) continue;
    found.push(entity);
  }
  return found.filter((e) => e.id !== owner.id && !isFriend(owner, e));
}

function bodyPoint(entity) {
  const base = entity.location;
  const head = entity.getHeadLocation();
  return { x: base.x, y: (base.y + head.y) / 2, z: base.z };
}

function seen(owner, entity) {
  const eye = owner.getHeadLocation();
  const points = [entity.getHeadLocation(), bodyPoint(entity)];
  for (const point of points) {
    const offset = minus(point, eye);
    const distance = length(offset);
    if (distance < 0.5) return true;
    const direction = { x: offset.x / distance, y: offset.y / distance, z: offset.z / distance };
    const hit = owner.dimension.getBlockFromRay(eye, direction, {
      maxDistance: distance - 0.3,
      includeLiquidBlocks: false,
      includePassableBlocks: false,
    });
    if (!hit) return true;
  }
  return false;
}

const CIRCLE_HEIGHT = 0.2;

const dot = (a, b) => a.x * b.x + a.y * b.y + a.z * b.z;

function circleRadius(owner) {
  const diameter = CIRCLE_HEIGHT * (tune(owner, "aimZoom") / 100);
  return diameter * Math.tan((tune(owner, "screenFov") * Math.PI) / 360);
}

export function pickTarget(owner) {
  const eye = owner.getHeadLocation();
  const view = owner.getViewDirection();
  const flat = Math.hypot(view.x, view.z);
  const right = flat < 1e-6 ? { x: 1, y: 0, z: 0 } : { x: -view.z / flat, y: 0, z: view.x / flat };
  const up = {
    x: right.y * view.z - right.z * view.y,
    y: right.z * view.x - right.x * view.z,
    z: right.x * view.y - right.y * view.x,
  };
  const radius = circleRadius(owner);
  let best;
  let bestDist = radius;
  for (const entity of candidates(owner)) {
    const points = [entity.getHeadLocation(), bodyPoint(entity)];
    let dist = Infinity;
    for (const point of points) {
      const offset = minus(point, eye);
      const depth = dot(offset, view);
      if (depth < 1) continue;
      dist = Math.min(dist, Math.hypot(dot(offset, right) / depth, dot(offset, up) / depth));
    }
    if (dist <= bestDist && seen(owner, entity)) {
      bestDist = dist;
      best = entity;
    }
  }
  return best;
}

function aimPoint(from, target, speed) {
  const body = bodyPoint(target);
  const motion = target.getVelocity();
  let wet = false;
  try {
    wet = target.isInWater;
  } catch (e) {}
  const drop = wet ? 0 : 0.5 * GRAVITY;
  let point = body;
  for (let i = 0; i < 2; i++) {
    const t = length(minus(point, from)) / speed;
    point = {
      x: body.x + motion.x * t,
      y: body.y + motion.y * t + drop * t * t,
      z: body.z + motion.z * t,
    };
  }
  return point;
}

function aim(arrow, projectile, target) {
  const speed = length(arrow.getVelocity());
  if (speed < MIN_SPEED) return false;
  const from = arrow.location;
  const direction = minus(aimPoint(from, target, speed), from);
  const size = length(direction);
  if (size < 0.001) return false;
  projectile.shoot({ x: (direction.x / size) * speed, y: (direction.y / size) * speed, z: (direction.z / size) * speed });
  return true;
}

const FOLLOW_TICKS = 80;

function steer(arrow) {
  try {
    if (!arrow.isValid) return;
    const projectile = arrow.getComponent(EntityComponentTypes.Projectile);
    const owner = projectile?.owner;
    if (!owner || owner.typeId !== PLAYER_ID || !isOn(owner, "bowAimbot")) return;
    const target = pickTarget(owner);
    if (!target || !aim(arrow, projectile, target)) return;
    let age = 0;
    const id = system.runInterval(() => {
      age += 2;
      try {
        if (age > FOLLOW_TICKS || !arrow.isValid || !target.isValid || !aim(arrow, projectile, target)) system.clearRun(id);
      } catch (e) {
        system.clearRun(id);
      }
    }, 2);
  } catch (e) {}
}

export function initBowAim() {
  world.afterEvents.entitySpawn.subscribe(({ entity }) => {
    if (entity.typeId !== ARROW) return;
    system.run(() => steer(entity));
  });
}

function drawZoom(p) {
  const base = tune(p, "aimZoom");
  const cut = tune(p, "bowZoom");
  const start = drawn.get(p.id);
  if (!cut || start === undefined) return base;
  const pull = Math.min(1, (system.currentTick - start) / 20);
  const fov = (tune(p, "screenFov") * Math.PI) / 180;
  return (base * Math.tan(fov / 2)) / Math.tan((fov * (1 - (cut / 100) * pull)) / 2);
}

function setCircle(p, name) {
  const same = (hud.get(p.id) ?? "") === name;
  if (same && (!name || system.currentTick - (hudAt.get(p.id) ?? 0) < HUD_REFRESH)) return;
  if (name) {
    hud.set(p.id, name);
    hudAt.set(p.id, system.currentTick);
  } else {
    hud.delete(p.id);
    hudAt.delete(p.id);
  }
  p.onScreenDisplay.setTitle(name, { fadeInDuration: 0, stayDuration: 72000, fadeOutDuration: 0 });
}

function mark(p) {
  const held = p.getComponent(EntityComponentTypes.Equippable)?.getEquipmentSlot(EquipmentSlot.Mainhand)?.getItem()?.typeId;
  if (!BOWS.has(held)) {
    locks.delete(p.id);
    setCircle(p, "");
    return;
  }
  const zoom = Math.min(300, Math.max(50, Math.round(drawZoom(p) / 10) * 10));
  setCircle(p, tune(p, "aimCircle") === 1 ? `alnb_z${zoom}` : "");

  const target = pickTarget(p);
  if (!target) {
    locks.delete(p.id);
    return;
  }
  const head = target.getHeadLocation();
  p.spawnParticle("allenscreations:aln_bow_target", { x: head.x, y: head.y + 0.8, z: head.z });
  const name = target.typeId === PLAYER_ID ? target.name : target.typeId.replace("minecraft:", "");
  const lock = locks.get(p.id);
  if (lock?.name === name && system.currentTick - lock.tick < LOCK_REFRESH) return;
  locks.set(p.id, { name, tick: system.currentTick });

const cache = new Map();

export function readSaved(player, key) {
  const id = `${key}|${player.id}`;
  if (cache.has(id)) return cache.get(id);
  let data = {};
  const raw = player.getDynamicProperty(key);
  if (typeof raw === "string") {
    try {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") data = parsed;
    } catch (e) {}
  }
  cache.set(id, data);
  return data;
}

export function writeSaved(player, key, data) {
  cache.set(`${key}|${player.id}`, data);
  player.setDynamicProperty(key, JSON.stringify(data));
}

  for (const id of [...cache.keys()]) if (id.endsWith(`|${playerId}`)) cache.delete(id);

You get an ALN: Bow Aimbot item, use it, and switch on Bow Aimbot in the ALNStudios menu. From then on, arrows you shoot turn toward the mob or player closest to your crosshair inside the aim circle. ## Key Features

### ALN: Bow Aimbot Item
- **What Happens**: A bow-and-arrow item that opens the menu when you use it. You craft one from a Bow, 2 Arrows, an Eye of Ender and a String, or take one from the Equipment tab of the Creative inventory.
- **How It Works**: Hold the item and use it, like right-clicking on PC or tapping on mobile. To craft one, put the Eye of Ender in the top middle square of a crafting table, the Bow in the center with an Arrow on each side of it, and the String in the bottom middle square.

- **What Happens**: Arrows you shoot from a bow or crossbow steer toward the mob or player closest to your crosshair within 48 blocks and inside the aim circle on your screen.
- **How It Works**: Aim roughly at a target and release. The arrow leads a moving target and allows for arrow drop, and it keeps correcting its aim for up to 4 seconds in flight, so long shots and moving targets are hit more often.
- **Good To Know**: If nothing is near your crosshair, the arrow flies as normal. A mob or player hidden behind blocks is never chosen, and players in Creative or Spectator mode are never chosen. Every arrow of a crossbow with Multishot aims at the same target.

- **What Happens**: Four On or Off choices under Bow Aimbot in Bow Aimbot Settings pick what it hits: hostile mobs, neutral mobs, passive mobs and players.
- **How It Works**: Open Bow Aimbot Settings and set each choice. Hostile mobs and players are on by default.

- **How It Works**: Switch on both Bow Aimbot and Target Highlight, hold a bow and aim near an enemy.
- **Good To Know**: It does nothing when Bow Aimbot is off, and the target picture only shows when a target is inside the circle. The circle can be switched off in Bow Aimbot Settings, and its size follows the Aim circle zoom and your Your game FOV setting value.

- **What Happens**: While you draw a bow the game zooms in, so the aim circle grows to cover the same area.
- **How It Works**: Raise Bow zoom while drawing, from 0 to 30 percent, until the circle matches what you see.
- **Good To Know**: It is 0 percent by default, so the circle stays the same size until you set it.

- **Good To Know**: The range, zoom and mob type choices are in Bow Aimbot Settings.
