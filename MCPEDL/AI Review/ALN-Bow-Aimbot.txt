import { world, system, EntityComponentTypes, GameMode } from "@minecraft/server";
import { PLAYER_ID } from "./info.js";
import { isOn } from "./store.js";
import { tune } from "./tuning.js";
import { isFriend } from "./friends.js";

const ARROW = "minecraft:arrow";

const GRAVITY = 0.05;
const MIN_SPEED = 0.8;

const minus = (a, b) => ({ x: a.x - b.x, y: a.y - b.y, z: a.z - b.z });
const length = (v) => Math.hypot(v.x, v.y, v.z);

const NEUTRAL = new Set([
  "minecraft:wolf",
  "minecraft:enderman",
  "minecraft:zombie_pigman",
  "minecraft:piglin",
  "minecraft:iron_golem",
  "minecraft:bee",
  "minecraft:llama",
  "minecraft:trader_llama",
  "minecraft:polar_bear",
  "minecraft:dolphin",
  "minecraft:panda",
  "minecraft:goat",
]);
const IGNORED = new Set(["minecraft:armor_stand"]);

function kindOf(entity) {
  if (NEUTRAL.has(entity.typeId)) return "aimNeutral";
  return entity.matches({ families: ["monster"] }) ? "aimHostile" : "aimPassive";
}

function candidates(owner) {
  const options = { location: owner.location, maxDistance: tune(owner, "aimRange") };
  const found = [];
  if (tune(owner, "aimPlayers") === 1) {
    found.push(...owner.dimension.getPlayers({ ...options, excludeGameModes: [GameMode.Creative, GameMode.Spectator] }));
  }
  for (const entity of owner.dimension.getEntities(options)) {
    if (entity.id === owner.id || entity.typeId === PLAYER_ID || IGNORED.has(entity.typeId)) continue;
    if (!entity.getComponent(EntityComponentTypes.Health)) continue;
    const kind = kindOf(entity);
    if (tune(owner, kind) !== 1) continue;
    if (kind !== "aimHostile" && entity.getComponent(EntityComponentTypes.IsTamed)) continue;
    found.push(entity);
  }
  return found.filter((e) => e.id !== owner.id && !isFriend(owner, e));
}

function bodyPoint(entity) {
  const base = entity.location;
  const head = entity.getHeadLocation();
  return { x: base.x, y: (base.y + head.y) / 2, z: base.z };
}

function seen(owner, entity) {
  const eye = owner.getHeadLocation();
  const points = [entity.getHeadLocation(), bodyPoint(entity)];
  for (const point of points) {
    const offset = minus(point, eye);
    const distance = length(offset);
    if (distance < 0.5) return true;
    const direction = { x: offset.x / distance, y: offset.y / distance, z: offset.z / distance };
    const hit = owner.dimension.getBlockFromRay(eye, direction, {
      maxDistance: distance - 0.3,
      includeLiquidBlocks: false,
      includePassableBlocks: false,
    });
    if (!hit) return true;
  }
  return false;
}

const CIRCLE_HEIGHT = 0.2;

const dot = (a, b) => a.x * b.x + a.y * b.y + a.z * b.z;

function circleRadius(owner) {
  const diameter = CIRCLE_HEIGHT * (tune(owner, "aimZoom") / 100);
  return diameter * Math.tan((tune(owner, "screenFov") * Math.PI) / 360);
}

export function pickTarget(owner) {
  const eye = owner.getHeadLocation();
  const view = owner.getViewDirection();
  const flat = Math.hypot(view.x, view.z);
  const right = flat < 1e-6 ? { x: 1, y: 0, z: 0 } : { x: -view.z / flat, y: 0, z: view.x / flat };
  const up = {
    x: right.y * view.z - right.z * view.y,
    y: right.z * view.x - right.x * view.z,
    z: right.x * view.y - right.y * view.x,
  };
  const radius = circleRadius(owner);
  let best;
  let bestDist = radius;
  for (const entity of candidates(owner)) {
    const points = [entity.getHeadLocation(), bodyPoint(entity)];
    let dist = Infinity;
    for (const point of points) {
      const offset = minus(point, eye);
      const depth = dot(offset, view);
      if (depth < 1) continue;
      dist = Math.min(dist, Math.hypot(dot(offset, right) / depth, dot(offset, up) / depth));
    }
    if (dist <= bestDist && seen(owner, entity)) {
      bestDist = dist;
      best = entity;
    }
  }
  return best;
}

function aimPoint(from, target, speed) {
  const body = bodyPoint(target);
  const motion = target.getVelocity();
  let wet = false;
  try {
    wet = target.isInWater;
  } catch (e) {}
  const drop = wet ? 0 : 0.5 * GRAVITY;
  let point = body;
  for (let i = 0; i < 2; i++) {
    const t = length(minus(point, from)) / speed;
    point = {
      x: body.x + motion.x * t,
      y: body.y + motion.y * t + drop * t * t,
      z: body.z + motion.z * t,
    };
  }
  return point;
}

function aim(arrow, projectile, target) {
  const speed = length(arrow.getVelocity());
  if (speed < MIN_SPEED) return false;
  const from = arrow.location;
  const direction = minus(aimPoint(from, target, speed), from);
  const size = length(direction);
  if (size < 0.001) return false;
  projectile.shoot({ x: (direction.x / size) * speed, y: (direction.y / size) * speed, z: (direction.z / size) * speed });
  return true;
}

const FOLLOW_TICKS = 80;

function steer(arrow) {
  try {
    if (!arrow.isValid) return;
    const projectile = arrow.getComponent(EntityComponentTypes.Projectile);
    const owner = projectile?.owner;
    if (!owner || owner.typeId !== PLAYER_ID || !isOn(owner, "bowAimbot")) return;
    const target = pickTarget(owner);
    if (!target || !aim(arrow, projectile, target)) return;
    let age = 0;
    const id = system.runInterval(() => {
      age += 2;
      try {
        if (age > FOLLOW_TICKS || !arrow.isValid || !target.isValid || !aim(arrow, projectile, target)) system.clearRun(id);
      } catch (e) {
        system.clearRun(id);
      }
    }, 2);
  } catch (e) {}
}

export function initBowAim() {
  world.afterEvents.entitySpawn.subscribe(({ entity }) => {
    if (entity.typeId !== ARROW) return;
    system.run(() => steer(entity));
  });
}

import { world, system, EntityComponentTypes, EquipmentSlot } from "@minecraft/server";
import { PLAYER_ID } from "./info.js";
import { isOn, anyOn } from "./store.js";
import { tune } from "./tuning.js";
import { pickTarget } from "./bowaim.js";

const BOWS = new Set(["minecraft:bow", "minecraft:crossbow"]);
const drawn = new Map();
const hud = new Map();
const hudAt = new Map();
const locks = new Map();
const HUD_REFRESH = 36000;
const LOCK_REFRESH = 30;

function drawZoom(p) {
  const base = tune(p, "aimZoom");
  const cut = tune(p, "bowZoom");
  const start = drawn.get(p.id);
  if (!cut || start === undefined) return base;
  const pull = Math.min(1, (system.currentTick - start) / 20);
  const fov = (tune(p, "screenFov") * Math.PI) / 180;
  return (base * Math.tan(fov / 2)) / Math.tan((fov * (1 - (cut / 100) * pull)) / 2);
}

function setCircle(p, name) {
  const same = (hud.get(p.id) ?? "") === name;
  if (same && (!name || system.currentTick - (hudAt.get(p.id) ?? 0) < HUD_REFRESH)) return;
  if (name) {
    hud.set(p.id, name);
    hudAt.set(p.id, system.currentTick);
  } else {
    hud.delete(p.id);
    hudAt.delete(p.id);
  }
  p.onScreenDisplay.setTitle(name, { fadeInDuration: 0, stayDuration: 72000, fadeOutDuration: 0 });
}

function mark(p) {
  const held = p.getComponent(EntityComponentTypes.Equippable)?.getEquipmentSlot(EquipmentSlot.Mainhand)?.getItem()?.typeId;
  if (!BOWS.has(held)) {
    locks.delete(p.id);
    setCircle(p, "");
    return;
  }
  const zoom = Math.min(300, Math.max(50, Math.round(drawZoom(p) / 10) * 10));
  setCircle(p, tune(p, "aimCircle") === 1 ? `alnb_z${zoom}` : "");

  const target = pickTarget(p);
  if (!target) {
    locks.delete(p.id);
    return;
  }
  const head = target.getHeadLocation();
  p.spawnParticle("allenscreations:aln_bow_target", { x: head.x, y: head.y + 0.8, z: head.z });
  const name = target.typeId === PLAYER_ID ? target.name : target.typeId.replace("minecraft:", "");
  const lock = locks.get(p.id);
  if (lock?.name === name && system.currentTick - lock.tick < LOCK_REFRESH) return;
  locks.set(p.id, { name, tick: system.currentTick });
  p.onScreenDisplay.setActionBar(`§cLocked on: ${name}`);
}

export function initHud() {
  world.afterEvents.itemStartUse.subscribe(({ source, itemStack }) => {
    if (BOWS.has(itemStack?.typeId)) drawn.set(source.id, system.currentTick);
  });
  world.afterEvents.itemStopUse.subscribe(({ source }) => drawn.delete(source.id));
  system.runInterval(() => {
    if (!anyOn("targetHighlight") && !hud.size) return;
    for (const p of world.getAllPlayers()) {
      if (!p.isValid) continue;
      try {
        if (isOn(p, "targetHighlight") && isOn(p, "bowAimbot")) mark(p);
        else if (hud.has(p.id)) setCircle(p, "");
      } catch (e) {}
    }
  }, 1);
  world.afterEvents.playerLeave.subscribe(({ playerId }) => {
    drawn.delete(playerId);
    hud.delete(playerId);
    hudAt.delete(playerId);
    locks.delete(playerId);
  });
}

import { world } from "@minecraft/server";

const cache = new Map();

export function readSaved(player, key) {
  const id = `${key}|${player.id}`;
  if (cache.has(id)) return cache.get(id);
  let data = {};
  const raw = player.getDynamicProperty(key);
  if (typeof raw === "string") {
    try {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") data = parsed;
    } catch (e) {}
  }
  cache.set(id, data);
  return data;
}

export function writeSaved(player, key, data) {
  cache.set(`${key}|${player.id}`, data);
  player.setDynamicProperty(key, JSON.stringify(data));
}

world.afterEvents.playerLeave.subscribe(({ playerId }) => {
  for (const id of [...cache.keys()]) if (id.endsWith(`|${playerId}`)) cache.delete(id);
});

export const FEATURES = [
  { id: "bowAimbot", name: "Bow Aimbot", info: "Arrows you shoot steer toward the target inside the aim circle." },
  { id: "targetHighlight", name: "Target Highlight", info: "Shows a target image over who Bow Aimbot picks and the aim circle." },

const KEY = "alnb_toggles";
const cache = new Map();

function load(player) {
  const raw = player.getDynamicProperty(KEY);
  let data = {};
  if (typeof raw === "string") {
    try {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") data = parsed;
    } catch (e) {}
  }
  cache.set(player.id, data);
  return data;
}

const read = (player) => cache.get(player.id) ?? load(player);

export function isOn(player, id) {
  return read(player)[id] === true;
}

export function activeNames(player) {
  return FEATURES.filter((f) => isOn(player, f.id)).map((f) => f.name);
}

export function applyValues(player, values) {
  const data = {};
  FEATURES.forEach((f, i) => {
    data[f.id] = values[i] === true;
  });
  cache.set(player.id, data);
  player.setDynamicProperty(KEY, JSON.stringify(data));
}

export function anyOn(id) {
  for (const player of world.getAllPlayers()) {
    if (isOn(player, id)) return true;
  }
  return false;
}

world.afterEvents.playerLeave.subscribe(({ playerId }) => {
  cache.delete(playerId);
