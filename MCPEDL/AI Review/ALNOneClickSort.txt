const ORE_EXACT = new Set([
  "diamond", "emerald", "coal", "redstone", "lapis_lazuli",
  "quartz", "amethyst_shard", "netherite_scrap",
]);
const REDSTONE_KEYS = [
  "hopper", "piston", "observer", "repeater", "comparator", "torch", "chest", "barrel",
];

function shortId(typeId) {
  const i = typeId.indexOf(":");
  return i >= 0 ? typeId.slice(i + 1) : typeId;
}

function safeHas(item, componentId) {
  try {
    return item.hasComponent(componentId);
  } catch {
    return false;
  }
}

function categoryOf(item) {
  const id = shortId(item.typeId);
  if (safeHas(item, "minecraft:durability")) return 0;
  if (id.endsWith("ore") || id.endsWith("ingot") || ORE_EXACT.has(id)) return 1;
  if (safeHas(item, "minecraft:food") || id.endsWith("potion")) return 2;
  if (REDSTONE_KEYS.some((k) => id.includes(k))) return 3;
  return 4;
}

function compareId(a, b) {
  return a.typeId < b.typeId ? -1 : a.typeId > b.typeId ? 1 : 0;
}

function makeComparator(sortMode) {
  if (sortMode === 1) {
    return (a, b) => compareId(a, b) || b.amount - a.amount;
  }
  if (sortMode === 2) {
    return (a, b) => b.amount - a.amount || compareId(a, b);
  }
  return (a, b) =>
    categoryOf(a) - categoryOf(b) || compareId(a, b) || b.amount - a.amount;
}

function consolidate(items) {
  const merged = [];
  for (const source of items) {
    const item = source.clone();
    let left = item.amount;
    for (const target of merged) {
      if (left <= 0) break;
      if (target.amount >= target.maxAmount) continue;
      if (!target.isStackableWith(item)) continue;
      const move = Math.min(target.maxAmount - target.amount, left);
      target.amount += move;
      left -= move;
    }
    if (left > 0) {
      item.amount = left;
      merged.push(item);
    }
  }
  return merged;
}

export function sortContainer(container, startSlot, endSlot, sortMode = 0) {
  const last = Math.min(endSlot, container.size - 1);
  const first = Math.max(0, startSlot);
  if (last < first) return;

  const items = [];
  for (let slot = first; slot <= last; slot++) {
    const item = container.getItem(slot);
    if (item) items.push(item);
  }

  const sorted = consolidate(items).sort(makeComparator(sortMode));

  for (let slot = first, i = 0; slot <= last; slot++, i++) {
    container.setItem(slot, sorted[i]);
  }
}

import { world, system } from "@minecraft/server";
import { sortContainer } from "./sort_engine.js";
import { getPrefs, isSettingsBook, openSettings } from "./sort_ui.js";
import { PARTICLES } from "./feedback_profile.js";

const CONTAINER_IDS = new Set(["minecraft:chest", "minecraft:trapped_chest", "minecraft:barrel"]);

const QUICK_STACK_RADIUS = 6;
const lastContainerSort = new Map();

function isSortableContainer(typeId) {
  if (typeId === "minecraft:ender_chest") return false;
  return CONTAINER_IDS.has(typeId) || typeId.endsWith("chest") || typeId.endsWith("shulker_box");
}

function isAutoSortFeather(item) {
  return (
    !!item &&
    item.typeId === "allenscreations:auto_sort_feather"
  );
}

function isRestockFeather(item) {
  return (
    !!item &&
    item.typeId === "allenscreations:restock_feather"
  );
}

function getBackpack(player) {
  return player.getComponent("minecraft:inventory").container;
}

function feedback(player, prefs, message, { sound = "random.orb", pitch = 1.3, block } = {}) {
  player.onScreenDisplay.setActionBar(message);
  if (!prefs.soundFeedback) return;
  try {
    player.playSound(sound, { pitch });
    if (block && PARTICLES) {
      player.playSound("block.barrel.close");
      block.dimension.spawnParticle("minecraft:villager_happy", {
        x: block.location.x + 0.5,
        y: block.location.y + 1.2,
        z: block.location.z + 0.5,
      });
    }
  } catch {
  }
}

function sortBackpack(player) {
  const prefs = getPrefs(player);
  const start = prefs.includeHotbar ? 0 : 9;
  sortContainer(getBackpack(player), start, 35, prefs.sortMode);
  feedback(player, prefs, "§a[OneClickSort] §fBackpack sorted!");
}

function sortBlockContainer(player, block) {
  try {
    const container = block.getComponent("minecraft:inventory")?.container;
    if (!container) return;
    const prefs = getPrefs(player);
    sortContainer(container, 0, container.size - 1, prefs.sortMode);
    feedback(player, prefs, "§a[OneClickSort] §fChest organized!", { block });
  } catch (e) {
    console.warn(`[ALN:OneClickSort] Container sort failed: ${e}`);
  }
}

function contentSignature(container) {
  let s = "";
  for (let i = 0; i < container.size; i++) {
    const it = container.getItem(i);
    s += it ? `${it.typeId}x${it.amount};` : ";";
  }
  return s;
}

function findNearbyContainers(player) {
  const dim = player.dimension;
  const o = player.location;
  const cx = Math.floor(o.x), cy = Math.floor(o.y), cz = Math.floor(o.z);
  const found = [];

  for (let dx = -QUICK_STACK_RADIUS; dx <= QUICK_STACK_RADIUS; dx++) {
    for (let dy = -QUICK_STACK_RADIUS; dy <= QUICK_STACK_RADIUS; dy++) {
      for (let dz = -QUICK_STACK_RADIUS; dz <= QUICK_STACK_RADIUS; dz++) {
        let block;
        try {
          block = dim.getBlock({ x: cx + dx, y: cy + dy, z: cz + dz });
        } catch {
          continue;
        }
        if (!block || !isSortableContainer(block.typeId)) continue;
        const container = block.getComponent("minecraft:inventory")?.container;
        if (!container) continue;

        if (container.size === 54) {
          const sig = contentSignature(container);
          const dupe = found.some(
            (f) =>
              f.container.size === 54 &&
              f.block.typeId === block.typeId &&
              Math.abs(f.block.x - block.x) + Math.abs(f.block.y - block.y) + Math.abs(f.block.z - block.z) === 1 &&
              f.sig === sig
          );
          if (dupe) continue;
          found.push({ block, container, sig });
        } else {
          found.push({ block, container, sig: null });
        }
      }
    }
  }
  return found;
}

function stockChest(player, block) {
  try {
    const container = block.getComponent("minecraft:inventory")?.container;
    if (!container) return;
    const prefs = getPrefs(player);
    const backpack = getBackpack(player);

    const types = new Set();
    for (let i = 0; i < container.size; i++) {
      const it = container.getItem(i);
      if (it) types.add(it.typeId);
    }

    let movedItems = 0;
    for (let slot = 9; slot <= 35; slot++) {
      let item = backpack.getItem(slot);
      if (!item || isAutoSortFeather(item) || isRestockFeather(item)) continue;
      if (!types.has(item.typeId)) continue;
      const before = item.amount;
      let leftover;
      try {
        leftover = container.addItem(item);
      } catch {
        continue;
      }
      const after = leftover ? leftover.amount : 0;
      if (after < before) movedItems += before - after;
      backpack.setItem(slot, leftover);
    }

    if (movedItems > 0) sortContainer(container, 0, container.size - 1, prefs.sortMode);
    feedback(
      player,
      prefs,
      `§b[Stock] §fStocked §e${movedItems} §fitems into the chest!`,
      { block }
    );
  } catch (e) {
    console.warn(`[ALN:OneClickSort] Chest stock failed: ${e}`);
  }
}

function countStackable(container, item) {
  let total = 0;
  for (let i = 0; i < container.size; i++) {
    const it = container.getItem(i);
    if (it && item.isStackableWith(it)) total += it.amount;
  }
  return total;
}

function pullEverything(backpack, targets, prefs, touched) {
  const templates = [];
  for (let i = 0; i <= 35; i++) {
    const it = backpack.getItem(i);
    if (!it || isAutoSortFeather(it) || isRestockFeather(it) || it.maxAmount <= 1) continue;
    if (!templates.some((tpl) => tpl.isStackableWith(it))) templates.push(it);
  }

  let pulled = 0;
  for (const tpl of templates) {
    let full = false;
    for (let t = 0; t < targets.length && !full; t++) {
      const container = targets[t].container;
      let allowed = prefs.keepOneInChest ? countStackable(container, tpl) - 1 : Infinity;
      for (let s = 0; s < container.size && allowed > 0 && !full; s++) {
        const stored = container.getItem(s);
        if (!stored || !tpl.isStackableWith(stored)) continue;
        const move = Math.min(stored.amount, allowed);
        const piece = stored.clone();
        piece.amount = move;
        let left;
        try {
          left = backpack.addItem(piece);
        } catch {
          full = true;
          continue;
        }
        const added = move - (left ? left.amount : 0);
        if (left) full = true;
        if (added <= 0) continue;
        allowed -= added;
        pulled += added;
        touched.add(t);
        if (stored.amount - added > 0) {
          stored.amount -= added;
          container.setItem(s, stored);
        } else {
          container.setItem(s, undefined);
        }
      }
    }
  }
  return pulled;
}

function restock(player) {
  const prefs = getPrefs(player);
  const backpack = getBackpack(player);
  const targets = findNearbyContainers(player);
  const slots = [];
  for (let i = 0; i <= 35; i++) slots.push(i);

  let pulledItems = 0;
  const touched = new Set();

  for (const slot of slots) {
    const held = backpack.getItem(slot);
    if (!held || isAutoSortFeather(held) || isRestockFeather(held)) continue;
    let need = held.maxAmount - held.amount;
    if (need <= 0) continue;

    for (let t = 0; t < targets.length && need > 0; t++) {
      const container = targets[t].container;
      let allowed = prefs.keepOneInChest ? countStackable(container, held) - 1 : Infinity;
      for (let s = 0; s < container.size && need > 0 && allowed > 0; s++) {
        const stored = container.getItem(s);
        if (!stored || !held.isStackableWith(stored)) continue;
        const move = Math.min(need, stored.amount, allowed);
        allowed -= move;
        held.amount += move;
        need -= move;
        pulledItems += move;
        touched.add(t);
        if (stored.amount - move > 0) {
          stored.amount -= move;
          container.setItem(s, stored);
        } else {
          container.setItem(s, undefined);
        }
      }
    }
    backpack.setItem(slot, held);
  }

  if (prefs.getEverything) pulledItems += pullEverything(backpack, targets, prefs, touched);

  feedback(
    player,
    prefs,
    `§d[Restock] §fPulled §e${pulledItems} §fitems from §a${touched.size} §fnearby chests!`
  );
}

export function registerTriggers() {
  world.afterEvents.itemUse.subscribe((event) => {
    const player = event.source;
    const item = event.itemStack;
    if (!player || !item) return;

    if (isSettingsBook(item)) {
      openSettings(player);
      return;
    }
    if (!isAutoSortFeather(item) && !isRestockFeather(item)) return;
    if (player.isSneaking) return;

    const last = lastContainerSort.get(player.id);
    if (last !== undefined && system.currentTick - last < 5) return;

    if (isRestockFeather(item)) restock(player);
    else sortBackpack(player);
  });

  world.beforeEvents.playerInteractWithBlock.subscribe((event) => {
    const { player, block, itemStack } = event;
    if (event.isFirstEvent === false) return;
    if (!player.isSneaking) return;
    if (!isSortableContainer(block.typeId)) return;
    const sortChest = isAutoSortFeather(itemStack);
    if (!sortChest && !isRestockFeather(itemStack)) return;

    event.cancel = true;
    lastContainerSort.set(player.id, system.currentTick);
    system.run(() => (sortChest ? sortBlockContainer(player, block) : stockChest(player, block)));
  });

  world.afterEvents.playerLeave.subscribe((e) => lastContainerSort.delete(e.playerId));
}

import { ModalFormData, FormCancelationReason } from "@minecraft/server-ui";

const PREFS_KEY = "aln_sort_prefs";
const SORT_MODES = [
  "Smart Category (Default)",
  "Alphabetical (A-Z)",
  "Stack Quantity (Highest First)",

const DEFAULTS = { sortMode: 0, includeHotbar: false, soundFeedback: true, keepOneInChest: false, getEverything: false };

export function getPrefs(player) {
  try {
    const raw = player.getDynamicProperty(PREFS_KEY);
    if (typeof raw === "string") {
      const p = JSON.parse(raw);
      return {
        sortMode: Number.isInteger(p.sortMode) && p.sortMode >= 0 && p.sortMode < 3 ? p.sortMode : 0,
        includeHotbar: p.includeHotbar === true,
        soundFeedback: p.soundFeedback !== false,
        keepOneInChest: p.keepOneInChest === true,
        getEverything: p.getEverything === true,
      };
    }
  } catch {
  }
  return { ...DEFAULTS };
}

function savePrefs(player, prefs) {
  player.setDynamicProperty(PREFS_KEY, JSON.stringify(prefs));
}

export function isSettingsBook(item) {
  return (
    !!item &&
    item.typeId === "allenscreations:sort_settings_book"
  );
}

export function openSettings(player, attempt = 0) {
  const prefs = getPrefs(player);
  const form = new ModalFormData()
    .title("§aOneClickSort §7Settings")
    .dropdown("Sort Order", SORT_MODES, { defaultValueIndex: prefs.sortMode })
    .toggle("Sort Hotbar Slots (0-8) When Sorting Backpack", { defaultValue: prefs.includeHotbar })
    .toggle("Audio & Particle Feedback", { defaultValue: prefs.soundFeedback })
    .toggle("Leave 1 Item in Chest When Restocking", { defaultValue: prefs.keepOneInChest })
    .toggle("Get Everything When Restocking (Not Just One Stack)", { defaultValue: prefs.getEverything });

  form
    .show(player)
    .then((res) => {
      if (res.canceled) {
        if (res.cancelationReason === FormCancelationReason.UserBusy && attempt < 10) {
          system.runTimeout(() => openSettings(player, attempt + 1), 20);
        }
        return;
      }
      const [mode, hotbar, sound, keepOne, getAll] = res.formValues;
      savePrefs(player, {
        sortMode: Number(mode),
        includeHotbar: !!hotbar,
        soundFeedback: !!sound,
        keepOneInChest: !!keepOne,
        getEverything: !!getAll,
      });
      player.onScreenDisplay.setActionBar("§a[OneClickSort] §fSettings saved!");
    })
    .catch((e) => console.warn(`[ALN:OneClickSort] Form error: ${e}`));
}

# ALN:OneClickSort — One-Click Backpack, Chest and Restock Sorter

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**Sort your 27 backpack slots, sort any chest, barrel or shulker box, and stock chests from your backpack with three custom items.**

> Hold the Auto-Sort Feather and one click sorts your backpack, or sneak and click a chest to sort it. Hold the Restock Feather to refill your stacks, or sneak and click a chest to stock it. Good for survival, OneBlock and multiplayer worlds.

## Overview

Hold the Auto-Sort Feather in your hand. Right-click to sort your backpack, or sneak and right-click a chest to sort it. Hold the Restock Feather to refill your stacks, or sneak and right-click a chest to stock it. The pack adds three items, no blocks or mobs, and never changes world generation, so it works in existing worlds.

## Key Features

### Backpack Sorting
- **What Happens**: Right-clicking the Auto-Sort Feather in the air merges partial stacks and sorts the 27 backpack slots. The hotbar stays untouched unless you turn on hotbar sorting in the settings.
- **How It Works**: Hold the Feather, right-click, and a message appears above your hotbar.
- **Good To Know**: Names, lore, enchantments and durability are kept, and only identical items merge.

### Chest Sorting
- **What Happens**: Sneak and right-click any chest (including copper chests), trapped chest, barrel or shulker box to sort it instantly without opening it. Double chests are sorted as one container.
- **How It Works**: Sneak, then right-click the container with the Auto-Sort Feather in your hand.
- **Good To Know**: A sound confirms the sort, and the Normal variant also shows a small particle effect above the container.

### Stock Chest
- **What Happens**: Sneak and right-click any chest (including copper chests), trapped chest, barrel or shulker box with the Restock Feather to send backpack items into it that it already holds.
- **How It Works**: Hold the Restock Feather, sneak, right-click the container, and a message shows how many items went in.
- **Good To Know**: Only the 27 backpack slots are used, the feathers and hotbar are never moved, and the chest is sorted afterward. Sneaking and right-clicking in the air does nothing.

### Restock
- **What Happens**: Right-click the Restock Feather to refill the partial stacks you already carry, hotbar included, from every chest, copper chest, trapped chest, barrel and shulker box within 6 blocks.
- **How It Works**: Run low on blocks, step next to your storage, right-click the Restock Feather, and a message shows how many items came from how many chests.
- **Good To Know**: Restock only tops up stacks you already hold and never fills empty slots, unless you turn on Get Everything in Sort Settings, which pulls every matching item the chests hold until your inventory is full. Sneak and right-click a chest with the Restock Feather to stock it from your backpack. Turn on Leave 1 Item in Chest in Sort Settings to keep one item of each type in a chest when restocking, so stocking that chest still works.

### Sort Settings
- **What Happens**: Right-click the Sort Settings Book to open a menu with five options: Sort Order, Sort Hotbar Slots, Audio and Particle Feedback, Leave 1 Item in Chest and Get Everything.
- **How It Works**: Choose Smart Category, Alphabetical (A-Z) or Stack Quantity (Highest First), then confirm.
- **Good To Know**: Each player keeps their own settings, which stay saved between sessions.

### Crafting Recipes
- **What Happens**: All three items can be crafted at a Crafting Table, and every recipe is unlocked from the start.
- **How It Works**: Auto-Sort Feather: Emerald, Feather and Hopper placed diagonally. Restock Feather: Gold Ingot, Feather and Chest placed diagonally. Sort Settings Book: Book in the center with Lapis Lazuli left and right and Redstone Dust above and below.
- **Good To Know**: The diagonal layout matches the feather on the icons, and the same items sit together in one OneClickSort group in the Creative inventory under Items.

### Pack Variants
### Pack Variants
- **What Happens**: Two variants are included: Simple, the default, shows an action bar message and sound only, and Normal (with particles) also shows happy-villager particles above a sorted chest.
- **How It Works**: Open the pack's settings in your world and move the slider to switch variant.
- **Good To Know**: Both variants sort the same way, and the Audio and Particle Feedback option in Sort Settings still applies.

## How to Use

1. Craft the Auto-Sort Feather, Restock Feather and Sort Settings Book with the recipes below, or take them from the Creative inventory under Items.
2. Right-click in the air to sort your backpack, or sneak and right-click a chest to sort it.
3. Right-click the Restock Feather next to your chests to refill your stacks, or sneak and right-click a chest with it to stock that chest.
4. Right-click the Sort Settings Book to change your options.
