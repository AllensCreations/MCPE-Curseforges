import { world } from "@minecraft/server";
import { LEDGER_ID, SPAWN_Y, STARTER_LEDGER } from "./config.js";
import { firstFreeSlot, assignSlot, slotCoords, slotOf, readSlots, recordOf } from "./slots.js";
import { pickTrait, traitInfo, traitLabel } from "./traits.js";
import { ensureIsland, currentPhase } from "./island.js";
import { phaseLabel } from "./phases.js";
import { overworld, giveItem, findPlayer } from "./util.js";

const claiming = new Set();

export function hasIsland(player) {
  return slotOf(player) !== undefined;
}

export function goHome(player, slot, islandReady) {
  const c = slotCoords(slot);
  const dim = overworld();
  try {
    player.teleport({ x: c.x + 0.5, y: SPAWN_Y, z: c.z + 0.5 }, { dimension: dim });
  } catch (e) {}
  try {
    player.setSpawnPoint({ dimension: dim, x: c.x, y: SPAWN_Y, z: c.z });
  } catch (e) {}
  if (!islandReady) {
    try {
      player.addEffect("slow_falling", 600, { amplifier: 0, showParticles: false });
    } catch (e) {}
  }
}

function welcomeToIsland(player, slot, islandReady) {
  const record = recordOf(slot);
  const info = traitInfo(record.trait);
  const c = slotCoords(slot);
  goHome(player, slot, islandReady);
  if (STARTER_LEDGER) giveItem(player, LEDGER_ID, 1);
  try {
    player.onScreenDisplay.setTitle(`§6Island #${slot}`);
  } catch (e) {}
  player.sendMessage(`§6[ALN] §eIsland #${slot} claimed at X: ${c.x}, Z: ${c.z}.`);
  player.sendMessage(`§6[ALN] §fRichness: ${traitLabel(record.trait)} §7- ${info.blurb}`);
  player.sendMessage(`§6[ALN] §fWorld phase: ${phaseLabel(currentPhase())}§f. Break your OneBlock to begin.`);
  if (STARTER_LEDGER) player.sendMessage("§6[ALN] §7You received a Global Pirate Ledger. Right-click it to trade with other islands.");
}

export function claimIsland(player) {
  const id = player.id;
  if (claiming.has(id)) return;
  const owned = slotOf(player);
  if (owned !== undefined) {
    goHome(player, owned, true);
    return;
  }
  const slot = firstFreeSlot();
  if (slot === undefined) {
    player.sendMessage("§c[ALN] Every island spot on this world is taken.");
    return;
  }
  const trait = pickTrait(readSlots());
  assignSlot(player, slot, trait);
  claiming.add(id);
  player.sendMessage(`§6[ALN] §7Preparing island #${slot}...`);
  world.sendMessage(`§6[ALN] §f${player.name} §7claimed island #${slot} (${traitLabel(trait)}§7).`);
  ensureIsland(slot, trait, (ready) => {
    claiming.delete(id);
    const p = findPlayer(id);
    if (p) welcomeToIsland(p, slot, ready);
  });
}

export function returnToIsland(player) {
  const slot = slotOf(player);
  if (slot === undefined) return;
  const record = recordOf(slot);
  if (!record.built) ensureIsland(slot, record.trait, () => {});
}

  const slot = slotAt(block.x, block.z);
  if (slot === undefined) return;
  const record = recordOf(slot);
  if (!record) return;

  const before = phaseIndexFor(globalBreaks());
  const total = globalBreaks() + 1;
  world.setDynamicProperty(PROP.GLOBAL, total);
  const after = phaseIndexFor(total);

  const breaker = event.player;
  const mine = (breaker.getDynamicProperty(PROP.BREAKS) ?? 0) + 1;
  breaker.setDynamicProperty(PROP.BREAKS, mine);

  const dim = block.dimension;
  const x = block.x;
  const z = block.z;
  system.run(() => {
    regenerate(dim, x, z, record.trait, after);
    dropBonus(dim, x, z, record.trait, after);
  });

  if (after !== before) announcePhase(after);
  const left = breaksToNextPhase(total);
  const tail = left > 0 ? `§7Next phase in §f${left}` : "§7Final phase";
  try {
    breaker.onScreenDisplay.setActionBar(`${phaseLabel(after)} §8| §7Island: ${traitLabel(record.trait)} §8| ${tail}`);
  } catch (e) {}
}

function guardIslands() {
  const dim = overworld();
  const phase = phaseIndexFor(globalBreaks());
  for (const player of world.getAllPlayers()) {
    const slot = slotOf(player);
    if (slot === undefined) continue;
    const record = recordOf(slot);
    if (!record || !record.built) continue;
    const c = slotCoords(slot);
    if (!isLoaded(dim, c.x, ONE_BLOCK_Y, c.z)) continue;
    if (blockIdAt(dim, c.x, ONE_BLOCK_Y, c.z) === "minecraft:air") {
      regenerate(dim, c.x, c.z, record.trait, phase);
    }
  }
}

export function startOneBlock() {
  world.afterEvents.playerBreakBlock.subscribe((event) => {
    try {
      handleBreak(event);
    } catch (e) {}
  });
  system.runInterval(() => {
    try {
      guardIslands();
    } catch (e) {}
  }, 40);
}

export const ONEBLOCK_PROGRESSION = ONEBLOCK_PROGRESSION_POOLS;


const DEFAULT_CURRENT_SHARE = 0.4;


export class OneblockPoolProvider {

  constructor(pools = ONEBLOCK_PROGRESSION_POOLS) {
    this.pools = pools;
    this._unlockCache = new Map();
  }


  getPhasePool(phaseNumber) {
    const clampedIndex = Math.max(1, Math.min(phaseNumber, this.pools.length)) - 1;
    return this.pools[clampedIndex];
  }


  getPhaseFromBreaks(totalBreaks) {
    let accumulated = 0;
    for (let i = 0; i < this.pools.length; i++) {
      const p = this.pools[i];
      if (totalBreaks < accumulated + p.blocksRequired) {
        return {
          phase: p,
          phaseIndex: i + 1,
          phaseBreaks: totalBreaks - accumulated,
          isLastPhase: false
        };
      }
      accumulated += p.blocksRequired;
    }
    const last = this.pools[this.pools.length - 1];
    return {
      phase: last,
      phaseIndex: this.pools.length,
      phaseBreaks: totalBreaks - accumulated,
      isLastPhase: true
    };
  }


  static sampleWeighted(pool) {
    const totalWeight = pool.reduce((sum, item) => sum + item.weight, 0);
    let rand = Math.random() * totalWeight;
    for (const item of pool) {
      if (rand < item.weight) return item;
      rand -= item.weight;
    }
    return pool[pool.length - 1];
  }


  getUnlockedPool(phaseNumber, kind, currentShare = DEFAULT_CURRENT_SHARE) {
    const upto = Math.max(1, Math.min(Math.floor(phaseNumber) || 1, this.pools.length));
    const key = `${kind}:${upto}:${currentShare}`;
    const cached = this._unlockCache.get(key);
    if (cached) return cached;

    const merged = new Map();
    for (let i = 0; i < upto; i++) {
      const list = this.pools[i][kind];
      const sum = list.reduce((s, e) => s + e.weight, 0);
      const share = upto === 1 ? 1 : (i === upto - 1 ? currentShare : (1 - currentShare) / (upto - 1));
      for (const e of list) {
        const w = (e.weight / sum) * share * 100;
        const prev = merged.get(e.id);
        if (!prev) {
          merged.set(e.id, { ...e, weight: w });
        } else {
          prev.weight += w;
          if (kind === "chests") {
            prev.min = Math.min(prev.min, e.min);
            prev.max = Math.max(prev.max, e.max);
          }
        }
      }
    }
    const out = [...merged.values()];
    this._unlockCache.set(key, out);
    return out;
  }


  _resolvePool(phaseNumber, kind, options = {}) {
    if (options.exclusive) return this.getPhasePool(phaseNumber)[kind];
    return this.getUnlockedPool(phaseNumber, kind, options.currentShare);
  }


  sampleBlock(phaseNumber, options) {
    return OneblockPoolProvider.sampleWeighted(this._resolvePool(phaseNumber, "blocks", options));
  }


  sampleChestItem(phaseNumber, options) {
    return OneblockPoolProvider.sampleWeighted(this._resolvePool(phaseNumber, "chests", options));
  }


  sampleMob(phaseNumber, options) {
    return OneblockPoolProvider.sampleWeighted(this._resolvePool(phaseNumber, "mobs", options));
  }
}

function spreadRing(ring) {
  if (cache.rings[ring]) return cache.rings[ring];
  const cells = [];
  for (let i = -ring; i <= ring; i++) {
    for (let j = -ring; j <= ring; j++) {
      if (Math.max(Math.abs(i), Math.abs(j)) === ring) cells.push([i, j]);
    }
  }

**Claim a private OneBlock island on a 2,000-block grid and trade surplus goods through the Global Pirate Ledger.**

> Every player gets a floating island, every island shares the same OneBlock phase, and no island has everything. Trading is how the world grows.

## Overview

ALN:MultiIslandOneBlock is a OneBlock world map for ten or more players. New players arrive on a lone bedrock pillar at the world spawn, read a short startup window and claim a personal island. Islands sit 2,000 blocks apart in a square grid around the central lobby.

All islands share one progression, and each island carries a Richness trait that shapes its blocks and bonus drops. Because the traits differ, players end up with surplus goods that other islands need. The Global Pirate Ledger lets them swap those goods from anywhere in the world.

The map is a void world and includes Creative, Survival and Hardcore versions. The OneBlock Trader item used by other ALNStudios OneBlock maps is included for compatibility and has no role in this map.

## Key Features

### Central Lobby
- **What Happens**: New players arrive on a single bedrock pillar at the world spawn, standing at the center of the grid. The map builds the pillar and sets the world spawn on its own when the first player enters.

### Startup Window
- **What Happens**: A first-time player sees the ALN:MultiIslandOneBlock window with Start / Claim Island and About / Rules buttons. The window warns that the island spot cannot be undone, and it reopens if it is closed before a choice is made.

### Permanent Island Claim
- **What Happens**: Pressing Start assigns the next free island spot, builds a small grass island with the starting OneBlock, sets the personal respawn point on top of it, and teleports the player there. The spot belongs to that player for good.

### 2,000-Block Square Grid
- **What Happens**: The first eight islands sit at the corners and edges of a square around the lobby, from X: -2000, Z: -2000 to X: 2000, Z: 2000. Spots nine and up fill outer perimeter rings, up to 120 islands in total.

### Shared OneBlock Phases
- **What Happens**: Every OneBlock broken on any island moves the whole world forward through eight phases: Forest, Underground, Desert, Frozen Tundra, Ocean Monument, Nether, Deepslate Depths and The End. All islands always draw blocks from the same phase.

### Island Richness Traits
- **What Happens**: Each island receives one trait: Ore Bounty, Crop Master, Treasure Angler, Timber Baron, Beast Herder or Ember Forager. Traits are handed out evenly, and each one changes which blocks appear most often on that island.

### Bonus Drops
- **What Happens**: Roughly three out of ten OneBlock breaks drop an extra item that matches the island's trait and the current phase. Ore Bounty islands earn metals and gems, Crop Master islands earn harvests, and the other traits earn their own goods.

### Global Pirate Ledger
- **What Happens**: A portable item crafted from a Book, a Gold Nugget and an Emerald in a crafting table. Every player also receives one when claiming an island, and right-clicking it opens the Global Trade Hub from anywhere.

### Global Trade Hub
- **What Happens**: Players post offers such as 32 Iron Ingots for 16 Pumpkins, browse and search offers from every island, and buy with a single confirmation. Offered items are held safely until the offer sells or is cancelled.

### Earnings Collection
- **What Happens**: Payment for a sold offer goes straight to the seller when online. Sellers who are offline receive their earnings the next time they join, or when they choose Collect Earnings in the hub.

## How to Use

1. Download the map and open the file with Minecraft to import it.
2. Open the imported world in the Creative, Survival or Hardcore version.
3. First-time players appear on the bedrock pillar and see the startup window.
4. Read About / Rules if you like, then choose Start / Claim Island. This choice cannot be undone.
5. Break the OneBlock under your feet. It regenerates every time, and the whole world advances through its phases as islands keep breaking blocks.
6. Right-click the Global Pirate Ledger to create offers, browse other islands' offers and collect earnings.
7. Craft a new ledger with a Book, a Gold Nugget and an Emerald if you ever lose yours.
8. Optional: the map includes a setup command that rebuilds the spawn pillar and resets the world spawn. It needs cheats, and cheats disable achievements in that world.
