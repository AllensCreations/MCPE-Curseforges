import { world, system, DisplaySlotId, ObjectiveSortOrder } from "@minecraft/server";
import { BOARD_ID, BOARD_TITLE, BOARD_TICKS } from "./config.js";
import { readSlots } from "./slots.js";
import { coinsOf } from "./market.js";

function ensureObjective() {
  const existing = world.scoreboard.getObjective(BOARD_ID);
  const objective = existing ?? world.scoreboard.addObjective(BOARD_ID, BOARD_TITLE);
  world.scoreboard.setObjectiveAtDisplaySlot(DisplaySlotId.Sidebar, {
    objective,
    sortOrder: ObjectiveSortOrder.Descending,
  });
  return objective;
}

function refresh(objective) {
  const slots = readSlots();
  for (const key of Object.keys(slots)) {
    const record = slots[key];
    if (!record || !record.id || !record.name) continue;
    objective.setScore(record.name, coinsOf(record.id));
  }
}

export function startCoinBoard() {
  let objective;
  try {
    objective = ensureObjective();
  } catch (e) {
    return;
  }
  system.runInterval(() => {
    try {
      refresh(objective);
    } catch (e) {}
  }, BOARD_TICKS);
}

import { world } from "@minecraft/server";
import { LEDGER_ID, SPAWN_Y, STARTER_LEDGER, PROP } from "./config.js";
import { randomFreeSlot, assignSlot, slotCoords, slotOf, recordOf, addCoopMember, clearInvite } from "./slots.js";
import { pickTrait, traitInfo, traitLabel } from "./traits.js";
import { ensureIsland } from "./island.js";
import { phaseLabel } from "./phases.js";
import { overworld, giveItem, findPlayer } from "./util.js";

const claiming = new Set();

export function hasIsland(player) {
  return slotOf(player) !== undefined;
}

export function goHome(player, slot, islandReady) {
  const c = slotCoords(slot);
  const dim = overworld();
  try {
    player.teleport({ x: c.x + 0.5, y: SPAWN_Y, z: c.z + 0.5 }, { dimension: dim });
  } catch (e) {}
  try {
    player.setSpawnPoint({ dimension: dim, x: c.x, y: SPAWN_Y, z: c.z });
  } catch (e) {}
  if (!islandReady) {
    try {
      player.addEffect("slow_falling", 600, { amplifier: 0, showParticles: false });
    } catch (e) {}
  }
}

function welcomeToIsland(player, slot, islandReady) {
  const record = recordOf(slot);
  const info = traitInfo(record.trait);
  const c = slotCoords(slot);
  goHome(player, slot, islandReady);
  if (STARTER_LEDGER) giveItem(player, LEDGER_ID, 1);
  try {
    player.onScreenDisplay.setTitle(`§6Island #${slot}`);
  } catch (e) {}
  player.sendMessage(`§6[ALN] §eIsland #${slot} claimed at X: ${c.x}, Z: ${c.z}.`);
  player.sendMessage(`§6[ALN] §fRichness: ${traitLabel(record.trait)} §7- ${info.blurb}`);
  player.sendMessage(`§6[ALN] §fIsland phase: ${phaseLabel(0)}§f. Break your OneBlock to begin.`);
  if (STARTER_LEDGER) player.sendMessage("§6[ALN] §7You received the TRADERS Item. Right-click it to trade with other islands.");
}

export function claimIsland(player) {
  const id = player.id;
  if (claiming.has(id)) return;
  const owned = slotOf(player);
  if (owned !== undefined) {
    goHome(player, owned, true);
    return;
  }
  const slot = randomFreeSlot();
  if (slot === undefined) {
    player.sendMessage("§c[ALN] Every island spot on this world is taken.");
    return;
  }
  const trait = pickTrait();
  assignSlot(player, slot, trait);
  claiming.add(id);
  player.sendMessage(`§6[ALN] §7Preparing island #${slot}...`);
  world.sendMessage(`§6[ALN] §f${player.name} §7claimed island #${slot} (${traitLabel(trait)}§7).`);
  ensureIsland(slot, trait, (ready) => {
    claiming.delete(id);
    const p = findPlayer(id);
    if (p) welcomeToIsland(p, slot, ready);
  });
}

export function returnToIsland(player) {
  const slot = slotOf(player);
  if (slot === undefined) return;
  const record = recordOf(slot);
  if (!record.built) ensureIsland(slot, record.trait, () => {});
}

export function joinCoopIsland(player, slot) {
  const record = recordOf(slot);
  clearInvite(player.id);
  if (!record || !addCoopMember(slot, player)) {
    player.sendMessage("§6[ALN] §cThat co-op island is full or no longer available.");
    return false;
  }
  player.setDynamicProperty(PROP.SLOT, slot);
  player.setDynamicProperty(PROP.TRAIT, record.trait);
  player.setDynamicProperty(PROP.BREAKS, 0);
  goHome(player, slot, record.built === true);
  if (STARTER_LEDGER) giveItem(player, LEDGER_ID, 1);
  player.sendMessage(`§6[ALN] §eYou joined ${record.name}'s island #${slot} as a co-op member.`);
  return true;
}

function legacyBreaks() {

}

export function islandBreaks(slot) {
  const v = world.getDynamicProperty(PROP.ISLAND_PREFIX + slot);
  return typeof v === "number" ? v : legacyBreaks();

export function addIslandBreak(slot) {
  const total = islandBreaks(slot) + 1;
  world.setDynamicProperty(PROP.ISLAND_PREFIX + slot, total);
  return total;
}

function placeIsland(dim, x, z, trait, slot) {
  const first = nextBlockId(phaseIndexFor(islandBreaks(slot)), trait);
  runCmd(dim, `fill ${x - 3} 63 ${z - 3} ${x + 3} 63 ${z + 3} grass_block`);
  runCmd(dim, `fill ${x - 2} 62 ${z - 2} ${x + 2} 62 ${z + 2} dirt`);
  runCmd(dim, `fill ${x - 1} 61 ${z - 1} ${x + 1} 61 ${z + 1} dirt`);
  runCmd(dim, `setblock ${x} 60 ${z} dirt`);
  runCmd(dim, `setblock ${x} ${ONE_BLOCK_Y} ${z} ${first}`);
  const top = blockIdAt(dim, x, ONE_BLOCK_Y, z);
  const base = blockIdAt(dim, x, ONE_BLOCK_Y - 1, z);
  return Boolean(top) && top !== "minecraft:air" && base === "minecraft:grass_block";
}

export function ensureIsland(slot, trait, onResult) {
  const c = slotCoords(slot);
  const dim = overworld();
  const area = `aln_i${slot}`;
  const hasArea = addArea(dim, area, c.x, ONE_BLOCK_Y, c.z);
  let tries = 0;
  let reported = false;
  const report = (ok) => {
    if (reported) return;
    reported = true;
    onResult(ok);
  };
  const step = () => {
    tries++;
    if (isLoaded(dim, c.x, ONE_BLOCK_Y, c.z) && placeIsland(dim, c.x, c.z, trait, slot)) {
      markBuilt(slot);
      if (hasArea) dropArea(dim, area);
      report(true);
      return;
    }
    if (tries === 40) report(false);
    if (tries < 400) {
      later(step, 10);
    } else if (hasArea) {
      dropArea(dim, area);
    }
  };
  step();
}

  try {
    const block = dim.getBlock({ x, y: ONE_BLOCK_Y, z });

function ledgerStackOf(entity) {

    const d = (l.x - at.x) ** 2 + (l.y - at.y) ** 2 + (l.z - at.z) ** 2;
    if (d <= bestDist) {
      best = p;
      bestDist = d;
    }
  }

}

# Script library

Scripts for ALN:MultiIslandOneBlock. All helper files live in this folder and are imported by relative path from scripts/main.js.

- lib/phases.js: the eight OneBlock phases (tracked per island), their thresholds and weighted block pools.

- lib/market.js: global listing storage, ALN Coins balances, purchases and cancellations.
- lib/prices.js: recommended price and amount for each item, and what the market pays for it.

function gcd(a, b) {
  return b === 0 ? a : gcd(b, a % b);

function spreadRing(ring) {
  if (cache.rings[ring]) return cache.rings[ring];
  const cells = [];
  for (let i = -ring; i <= ring; i++) {
    for (let j = -ring; j <= ring; j++) {
      if (Math.max(Math.abs(i), Math.abs(j)) === ring) cells.push([i, j]);
    }
  }
  cells.sort((a, b) => Math.atan2(a[1], a[0]) - Math.atan2(b[1], b[0]));
  const n = cells.length;
  let stride = Math.max(1, Math.round(n * 0.618));
  while (gcd(stride, n) !== 1) stride++;
  const ordered = [];
  for (let idx = 0; idx < n; idx++) ordered.push(cells[(idx * stride) % n]);
  cache.rings[ring] = ordered;
  return ordered;
}

function cellOf(slot) {
  if (!Number.isInteger(slot) || slot < 1) return undefined;
  if (slot <= 8) return CORE[slot - 1];
  if (slot <= 24) return RING_TWO[slot - 9];
  let offset = slot - 25;
  for (let ring = 3; ring <= LAST_RING; ring++) {
    const size = ring * 8;
    if (offset < size) return spreadRing(ring)[offset];
    offset -= size;
  }
  return undefined;
}

function allRequests() {
  const now = world.getAbsoluteTime();
  const ids = readReqIds();
  const found = [];
  const keep = [];
  for (const id of ids) {
    const req = readJson(world, reqKey(id), undefined);
    if (!req) continue;
    if (now - req.created >= OFFER_LIFETIME_TICKS) {
      addCoins(req.buyer, req.price);
      world.setDynamicProperty(reqKey(id), undefined);
      const buyer = findPlayer(req.buyer);

      else addEarned(req.buyer, req.price);
      continue;
    }
    found.push(req);

}

export function isPlain(item) {
  if (!item) return false;
  if (quiet(() => item.nameTag, undefined)) return false;
  if (quiet(() => item.getLore().length, 0) > 0) return false;
  const enchantable = quiet(() => item.getComponent("minecraft:enchantable"), undefined);
  if (enchantable && quiet(() => enchantable.getEnchantments().length, 0) > 0) return false;
  const durability = quiet(() => item.getComponent("minecraft:durability"), undefined);
  if (durability && quiet(() => durability.damage, 0) > 0) return false;
  return true;
}

export function plainInventory(player) {
  const totals = new Map();
  const inv = containerOf(player);
  if (!inv) return [];
  for (let i = 0; i < inv.size; i++) {
    const item = inv.getItem(i);
    if (!item || item.typeId === LEDGER_ID || !isPlain(item)) continue;
    totals.set(item.typeId, (totals.get(item.typeId) ?? 0) + item.amount);
  }
  return [...totals.entries()]
    .map(([id, amount]) => ({ id, amount }))
    .sort((a, b) => niceName(a.id).localeCompare(niceName(b.id)));
}

export function countPlain(player, typeId) {
  if (typeId === LEDGER_ID) return 0;
  const inv = containerOf(player);
  if (!inv) return 0;
  let total = 0;
  for (let i = 0; i < inv.size; i++) {
    const item = inv.getItem(i);
    if (item && item.typeId === typeId && isPlain(item)) total += item.amount;
  }
  return total;
}

export function removePlain(player, typeId, amount) {
  if (countPlain(player, typeId) < amount) return false;
  const inv = containerOf(player);
  let left = amount;
  for (let i = 0; i < inv.size && left > 0; i++) {
    const item = inv.getItem(i);
    if (!item || item.typeId !== typeId || !isPlain(item)) continue;
    const take = Math.min(left, item.amount);
    if (take === item.amount) {
      inv.setItem(i, undefined);
    } else {
      item.amount -= take;
      inv.setItem(i, item);
    }
    left -= take;
  }
  return left === 0;
}

export function giveItem(player, typeId, amount) {
  const inv = containerOf(player);
  const limit = stackLimit(typeId);
  let left = amount;
  while (left > 0) {
    const n = Math.min(left, limit);
    left -= n;
    const stack = new ItemStack(typeId, n);
    const rest = inv ? inv.addItem(stack) : stack;
    if (rest) player.dimension.spawnItem(rest, player.location);
  }
}

export function pickWeighted(list, weightOf) {
  let total = 0;
  for (const entry of list) total += weightOf(entry);
  if (total <= 0) return list[0];
  let roll = Math.random() * total;
  for (const entry of list) {
    roll -= weightOf(entry);
    if (roll < 0) return entry;
  }
  return list[list.length - 1];
}

      } catch (e) {
        return;
      }

        tries++;
        if (tries > WELCOME_RETRY_LIMIT) {
          p.sendMessage("§6[ALN] §7Rejoin the world or respawn to see the startup window again.");
          return;
        }
        await wait(res.cancelationReason === FormCancelationReason.UserBusy ? 10 : 100);
        continue;
      }
      if (res.selection === 2 && invite) {
        if (joinCoopIsland(p, invite.slot)) return;
        continue;
      }
      if (res.selection === 0) {

        return;

# ALN:MultiIslandOneBlock — A OneBlock world map with private islands on a 2,000-block grid and a global trade ledger

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**Claim a private OneBlock island on a 2,000-block grid and trade surplus goods for ALN Coins through the TRADERS Item.**

> Every player gets a floating island, every island climbs through its own OneBlock phases, and no island has everything. Trading is how the world grows.

## Overview

ALN:MultiIslandOneBlock is a OneBlock world map for ten or more players. New players arrive on a lone bedrock pillar at the world spawn, read a short startup window and claim a personal island. Islands sit 2,000 blocks apart in a square grid around the central lobby.

Each island has its own progression, and each island carries a Richness trait that shapes its blocks and bonus drops. Because the traits differ, players end up with surplus goods that other islands need. The TRADERS Item lets them swap those goods from anywhere in the world.

The map is a void world and includes Creative, Survival and Hardcore versions.

## Key Features

### Central Lobby
- **What Happens**: New players arrive on a single bedrock pillar at the world spawn, standing at the center of the grid. The map builds the pillar and sets the world spawn on its own when the first player enters.

### Startup Window
- **What Happens**: A first-time player sees the ALN:MultiIslandOneBlock window with Start / Claim Island and About / Rules buttons. The window warns that the island spot cannot be undone, and it reopens if it is closed before a choice is made.

### Permanent Island Claim
- **What Happens**: Pressing Start assigns a random free island spot, builds a small grass island with the starting OneBlock, sets the personal respawn point on top of it, and teleports the player there. The spot belongs to that player for good.

### 2,000-Block Square Grid
- **What Happens**: Islands are placed on a random free spot of the innermost ring that still has room. The first eight spots sit at the corners and edges of a square around the lobby, from X: -2000, Z: -2000 to X: 2000, Z: 2000. Spots nine and up fill outer perimeter rings, up to 120 islands in total.

### Individual OneBlock Phases
- **What Happens**: Every OneBlock broken on an island moves that island forward through eight phases: Forest, Underground, Desert, Frozen Tundra, Ocean Monument, Nether, Deepslate Depths and The End. Each island draws blocks from its own phase, so one player cannot rush everyone else.

### Island Richness Traits
- **What Happens**: Each island receives one trait: Ore Bounty, Crop Master, Treasure Angler, Timber Baron, Beast Herder or Ember Forager. Traits are picked at random, with lower odds for traits that many islands already have, and each one changes which blocks appear most often on that island.

### Bonus Drops
- **What Happens**: About one in twenty OneBlock breaks drops an extra item that matches the island's trait and the island's phase. Ore Bounty islands earn metals and gems, Crop Master islands earn harvests, and the other traits earn their own goods.

### TRADERS Item
- **What Happens**: A portable item every player receives when claiming an island. Right-clicking it opens the Global Trade Hub from anywhere. It is permanent: if you throw it away it returns to you, and if your inventory is full it returns as soon as a slot is free.

### Global Trade Hub
- **What Happens**: Players post offers priced in ALN Coins, such as 32 Iron Ingots for 160 ALN Coins, browse and search offers from every island, and buy with a single confirmation. Every item shows a recommended price and amount, and leaving the fields empty uses them. For items with a recommended price, an offer must stay between half and four times that price. The market also buys those items directly for half of the recommended price, or a quarter for very valuable items, and sellers pay a 5% fee when an offer sells. Players in Creative mode cannot sell. Offered items are held safely until the offer sells, is cancelled or expires after about 7 days and returns to you.

### ALN Coins
- **What Happens**: ALN Coins are the currency of the Global Trade Hub. Every player starts with 500 ALN Coins, and the balance is shown in the hub. A sold offer pays the seller right away, even when the seller is offline. Players also earn 1 ALN Coin for every 5 OneBlock breaks (not in Creative mode), and a sidebar scoreboard titled @ALNStudios ranks island owners by their ALN Coins.

### Go Home
- **What Happens**: The TRADERS Item menu has a Go Home button that teleports you back to your island. It has a short cooldown.

### Top Islands
- **What Happens**: A ranking page in the TRADERS Item menu lists the islands with the most OneBlock breaks, their phase and your own rank.

### Phase Rewards
- **What Happens**: The first time your island reaches a new phase you earn ALN Coins, and the reward grows with each phase.

### Bonus Drop Notice
- **What Happens**: When a bonus drop appears, its name shows on the action bar and you hear a sound.

### Island Protection
- **What Happens**: Only the island owner, co-op members and trusted players can build, break or use blocks on an island. Visitors can look around but cannot change anything.

### Visit Requests
- **What Happens**: Friends & Co-op lets you ask an online island owner for a visit. The owner accepts or declines, and trusted players and co-op members can visit directly. Go Home brings you back.

### Trusted Players
- **What Happens**: Island owners can trust up to eight players so they can build on the island.

### Daily Trade Bonus
- **What Happens**: Your first sale to another player each day earns bonus ALN Coins, and the bonus is bigger when the buyer's island has a different Richness trait.
