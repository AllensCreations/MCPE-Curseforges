// Alias for backward compatibility
export const ONEBLOCK_PROGRESSION = ONEBLOCK_PROGRESSION_POOLS;

/** Unlock mode: share of total weight given to the newest phase. Ramps from START (phase 2) to END (last phase). */
const CURRENT_SHARE_START = 0.4;
const CURRENT_SHARE_END = 0.6;

/** Mob tags skipped by default when sampling mobs. Pass { excludeTags: [] } to allow everything. */
const DEFAULT_MOB_EXCLUDE_TAGS = ["aquatic", "boss"];

/**
 * Pure Data Pool Accessor
 * Only provides sampling and lookup against the pools.
 * Pack rules (spawn rates, mechanics) are implemented by the pack.
 */
export class OneblockPoolProvider {
  /**
   * @param {PhaseDataPool[]} [pools]
   */
  constructor(pools = ONEBLOCK_PROGRESSION_POOLS) {
    this.pools = pools;
    this._unlockCache = new Map();
  }

  /**
   * Get phase data pool by 1-based number
   * @param {number} phaseNumber
   * @returns {PhaseDataPool}
   */
  getPhasePool(phaseNumber) {
    const clampedIndex = Math.max(1, Math.min(phaseNumber, this.pools.length)) - 1;
    return this.pools[clampedIndex];
  }

  /**
   * Calculate milestone phase from broken block count
   * @param {number} totalBreaks
   * @returns {{ phase: PhaseDataPool, phaseIndex: number, phaseBreaks: number, isLastPhase: boolean }}
   */
  getPhaseFromBreaks(totalBreaks) {
    let accumulated = 0;
    for (let i = 0; i < this.pools.length; i++) {
      const p = this.pools[i];
      if (totalBreaks < accumulated + p.blocksRequired) {
        return {
          phase: p,
          phaseIndex: i + 1,
          phaseBreaks: totalBreaks - accumulated,
          isLastPhase: false
        };
      }
      accumulated += p.blocksRequired;
    }
    const last = this.pools[this.pools.length - 1];
    return {
      phase: last,
      phaseIndex: this.pools.length,
      phaseBreaks: totalBreaks - accumulated,
      isLastPhase: true
    };
  }

  /**
   * Pick random item from a weighted pool
   * @template {{ weight: number }} T
   * @param {T[]} pool
   * @returns {T}
   */
  static sampleWeighted(pool) {
    const totalWeight = pool.reduce((sum, item) => sum + item.weight, 0);
    let rand = Math.random() * totalWeight;
    for (const item of pool) {
      if (rand < item.weight) return item;
      rand -= item.weight;
    }
    return pool[pool.length - 1];
  }

  /**
   * Cumulative "unlock" pool: everything introduced in phases 1..phaseNumber.
   * The current phase gets `currentShare` of the total weight so new content
   * feels fresh; the remainder is split evenly across earlier phases.
   * Duplicate ids are merged. Returns new objects; shared pools are never mutated.
   * @param {number} phaseNumber - 1-based phase (clamped; also valid past the last phase)
   * @param {"blocks"|"chests"|"mobs"} kind
   * @param {number} [currentShare] - 0..1 share for the current phase
   * @returns {Array<{ id: string, weight: number, min?: number, max?: number }>}
   */
  getUnlockedPool(phaseNumber, kind, currentShare) {
    const upto = Math.max(1, Math.min(Math.floor(phaseNumber) || 1, this.pools.length));
    if (currentShare === undefined) {
      const t = (upto - 1) / Math.max(1, this.pools.length - 1);
      currentShare = CURRENT_SHARE_START + (CURRENT_SHARE_END - CURRENT_SHARE_START) * t;
    }
    const key = `${kind}:${upto}:${currentShare}`;
    const cached = this._unlockCache.get(key);
    if (cached) return cached;

    const merged = new Map();
    for (let i = 0; i < upto; i++) {
      const list = this.pools[i][kind];
      const sum = list.reduce((s, e) => s + e.weight, 0);
      const share = upto === 1 ? 1 : (i === upto - 1 ? currentShare : (1 - currentShare) / (upto - 1));
      for (const e of list) {
        const w = (e.weight / sum) * share * 100;
        const prev = merged.get(e.id);
        if (!prev) {
          merged.set(e.id, { ...e, weight: w });
        } else {
          prev.weight += w;
          if (kind === "chests") {
            prev.min = Math.min(prev.min, e.min);
            prev.max = Math.max(prev.max, e.max);
          }
        }
      }
    }
    const out = [...merged.values()];
    this._unlockCache.set(key, out);
    return out;
  }

  /**
   * Resolve the pool to sample from.
   * Default = unlock mode (earlier phases stay available). `exclusive: true` = this phase only.
   */
  _resolvePool(phaseNumber, kind, options = {}) {
    const pool = options.exclusive
      ? this.getPhasePool(phaseNumber)[kind]
      : this.getUnlockedPool(phaseNumber, kind, options.currentShare);
    if (kind !== "mobs") return pool;
    const skip = options.excludeTags ?? DEFAULT_MOB_EXCLUDE_TAGS;
    if (!skip.length) return pool;
    const allowed = pool.filter((m) => !(m.tags ?? []).some((t) => skip.includes(t)));
    return allowed.length ? allowed : pool;
  }

  /** Sample a block (unlock mode by default) */
  sampleBlock(phaseNumber, options) {
    return OneblockPoolProvider.sampleWeighted(this._resolvePool(phaseNumber, "blocks", options));
  }

  /** Sample a chest item (unlock mode by default) */
  sampleChestItem(phaseNumber, options) {
    return OneblockPoolProvider.sampleWeighted(this._resolvePool(phaseNumber, "chests", options));
  }

  /** Sample a mob (unlock mode by default) */
  sampleMob(phaseNumber, options) {
    return OneblockPoolProvider.sampleWeighted(this._resolvePool(phaseNumber, "mobs", options));
  }
}

// Backward-compatible alias
export const OneBlockProgressionManager = OneblockPoolProvider;

function purchase(player, item) {
  if (item.tier > unlockedTier(player)) return tell(player, "Mine more central blocks to unlock this supply.");
  const inv = inventoryOf(player);
  if (!inv) return tell(player, "Your inventory is unavailable.");
  const funds = balance(player);
  const heat = itemHeat(item, loadMarket(), world.getAbsoluteTime());
  const totalPrice = quote(item, heat);
  if (funds < totalPrice) return tell(player, "Not enough Coins. You need " + totalPrice + " Coins.");
  let stack;
  try { stack = new ItemStack(item.id, item.amount); } catch { return tell(player, "This item is not available in the current game version."); }
  if (!setBalance(player, funds - totalPrice)) return tell(player, "Your balance could not be updated; no purchase was made.");
  let leftover;
  try { leftover = inv.addItem(stack); } catch { leftover = stack; }
  const delivered = item.amount - (leftover?.amount ?? 0);
  const charged = quote(item, heat, delivered);
  if (charged < totalPrice) setBalance(player, balance(player) + totalPrice - charged);
  if (delivered <= 0) return tell(player, "Inventory full. No Coins were taken.");
  recordPurchase(item, delivered);
  if (delivered < item.amount) tell(player, "Only " + delivered + " item(s) fit; charged " + charged + " Coins.");
  else tell(player, "Purchased " + delivered + " x " + item.name + " for " + charged + " Coins.");
  showBalance(player);
}
function scanSellable(inv, values) {
  const slots = [];
  let payout = 0, count = 0;
  for (let i = 0; i < inv.size; i++) {
    const stack = inv.getItem(i);
    if (!stack) continue;
    const unit = values.get(stack.typeId);
    if (!unit) continue;
    slots.push(i);
    payout += unit * stack.amount;
    count += stack.amount;
  }
  return { slots, payout, count };
}
const LOCK_PREFIX = "obm_locks_";
const DEFAULT_LOCKED = new Set(["trident", "elytra", "totem_of_undying"]);
function lockOverrides(player) {
  try {
    const raw = world.getDynamicProperty(LOCK_PREFIX + player.name);
    const data = typeof raw === "string" ? JSON.parse(raw) : {};
    return data && typeof data === "object" && !Array.isArray(data) ? data : {};
  } catch { return {}; }
}
function isLocked(overrides, item) {
  return Object.prototype.hasOwnProperty.call(overrides, item.raw) ? overrides[item.raw] === 1 : DEFAULT_LOCKED.has(item.raw);
}
function setLocked(player, items, locked) {
  const data = lockOverrides(player);
  for (const item of items) {
    if (locked === DEFAULT_LOCKED.has(item.raw)) delete data[item.raw];
    else data[item.raw] = locked ? 1 : 0;
  }
  try {
    world.setDynamicProperty(LOCK_PREFIX + player.name, JSON.stringify(data));
    return true;
  } catch (e) {
    logError("setLocked", e);
    tell(player, "Your lock list could not be saved.");
    return false;
  }
}
const addOptional = (form, method, ...args) => (typeof form[method] === "function" ? form[method](...args) : form);
function showLockForm(player, owned) {
  const locks = lockOverrides(player);
  let form = new ModalFormData().title("LOCK ITEMS");
  form = addOptional(form, "header", "Locked Items");
  form = addOptional(form, "label", "Switch an item on to lock it. Locked items are never sold.");
  for (const [item, amount] of owned) {
    form = form.toggle(item.name + " - " + amount + "x - " + item.sell + " Coins each", { defaultValue: isLocked(locks, item) });
  }
  form = form.submitButton("Save");
  showForm(form, player).then(r => {
    if (r.canceled) return showSellMenu(player);
    const values = (r.formValues ?? []).filter(v => typeof v === "boolean");
    if (values.length !== owned.length) {
      tell(player, "The lock list could not be read.");
      return showSellMenu(player);
    }
    const toLock = [], toUnlock = [];
    owned.forEach(([item], i) => (values[i] ? toLock : toUnlock).push(item));
    setLocked(player, toLock, true);
    setLocked(player, toUnlock, false);
    return showSellMenu(player);
  }).catch(e => { logError("showLockForm", e); tell(player, "The lock list could not open."); });
}
function showSellMenu(player) {
  const inv = inventoryOf(player);
  if (!inv) return tell(player, "Your inventory is unavailable.");
  const itemById = new Map(CATALOG.map(item => [item.id, item]));
  const totals = new Map();
  for (let i = 0; i < inv.size; i++) {
    const stack = inv.getItem(i);
    const item = stack && itemById.get(stack.typeId);
    if (item) totals.set(item, (totals.get(item) ?? 0) + stack.amount);
  }
  const owned = [...totals].sort((a, b) => a[0].name.localeCompare(b[0].name));
  if (!owned.length) return tell(player, "You have no market items to sell.");
  const locks = lockOverrides(player);
  let payout = 0, sellCount = 0, lockedCount = 0;
  for (const [item, amount] of owned) {
    if (isLocked(locks, item)) lockedCount += amount;
    else { payout += item.sell * amount; sellCount += amount; }
  }
  const form = new ActionFormData().title("SELL ITEMS")
    .body("Will sell: " + sellCount + " item(s) for " + payout + " Coins\nLocked: " + lockedCount + " item(s)")
    .button("§lLock / Unlock Items§r", MCICON_DIR + "iron_door")
    .button("§lSell All§r\nLocked items are kept", MCICON_DIR + "gold_ingot")
    .button("§lExit§r", MCICON_DIR + "oak_door");
  showForm(form, player).then(r => {
    if (r.canceled) return;
    if (r.selection === 0) return showLockForm(player, owned);
    if (r.selection === 1) return sellUnlocked(player);
  }).catch(e => { logError("showSellMenu", e); tell(player, "The sell menu could not open."); });
}
function sellUnlocked(player) {

    .body("Sell " + count + " unlocked item(s) for " + payout + " Coins? Locked items and items not listed in the market are kept.")
    .button1("Sell")
    .button2("Cancel");
  showForm(form, player).then(r => {
    if (r.canceled || r.selection !== 0) return;
    const again = scanSellable(inv, values);
    if (again.payout !== payout || again.count !== count) return tell(player, "Your inventory changed. Open the sell option again.");
    if (!setBalance(player, balance(player) + payout)) return tell(player, "Balance update failed; your items were kept.");
    for (const slot of again.slots) inv.setItem(slot, undefined);
    tell(player, "Sold " + count + " eligible item(s) for " + payout + " Coins.");
    showBalance(player);
  }).catch(e => { logError("sellEligible", e); tell(player, "The sell confirmation could not open."); });
}

function rescuePhased() {
  const dimension = world.getDimension("overworld");
  for (const player of world.getAllPlayers()) {
    try {
      if (player.dimension.id !== "minecraft:overworld" || !player.hasTag("void_cleaned")) continue;
      const p = player.location;
      if (p.y >= CORE.y + 0.95 || p.y + 1.75 <= CORE.y + 0.05) continue;
      let stuck = false;
      for (let x = Math.floor(p.x - 0.25); x <= Math.floor(p.x + 0.25) && !stuck; x++) {
        for (let z = Math.floor(p.z - 0.25); z <= Math.floor(p.z + 0.25) && !stuck; z++) {
          if (Math.abs(x - CORE.x) > 1 || Math.abs(z - CORE.z) > 1) continue;
          const block = dimension.getBlock({ x, y: CORE.y, z });
          if (block && !NOT_SOLID.has(block.typeId)) stuck = true;
        }
      }
      if (!stuck) continue;
      player.teleport({ x: CORE.x + 0.5, y: CORE.y + 1, z: CORE.z + 0.5 }, { keepVelocity: false });
      log("INFO", player.name + " was inside the core platform and was moved on top of the core.");
    } catch (e) { logError("rescuePhased", e); }
  }
}
function damageTool(player, heldId, brokenId) {
  try {
    let creative = false;
    try { creative = String(player.getGameMode()).toLowerCase() === "creative"; } catch {}
    if (creative || !heldId) return;
    const base = /_sword$/.test(heldId) ? 2 : /(pickaxe|_axe|_shovel|_hoe|shears)$/.test(heldId) ? 1 : 0;
    if (!base) return;
    const cost = base * (TIER_WEAR[Math.floor(coreTier(brokenId) / 5)] ?? 1);
    const slot = player.getComponent("minecraft:equippable")?.getEquipmentSlot(EquipmentSlot.Mainhand);
    const stack = slot?.getItem();
    const durability = stack?.getComponent("minecraft:durability");
    if (!stack || stack.typeId !== heldId || !durability) return;
    let level = 0;
    try { level = stack.getComponent("minecraft:enchantable")?.getEnchantment("unbreaking")?.level ?? 0; } catch {}
    let hits = 0;
    for (let i = 0; i < cost; i++) if (Math.random() < 1 / (level + 1)) hits++;
    if (!hits) return;
    if (durability.damage + hits >= durability.maxDurability) {
      slot.setItem(undefined);
      try { player.playSound("random.break"); } catch {}
      return;
    }
    durability.damage += hits;
    slot.setItem(stack);
  } catch (e) { logError("damageTool", e); }
}
function cooldownState() {
  try {
    const raw = world.getDynamicProperty(COOLDOWN_KEY);
    const data = typeof raw === "string" && raw ? JSON.parse(raw) : null;
    return data && typeof data.until === "number" ? data : null;
  } catch { return null; }
}
function coreCoolingDown() {
  const state = cooldownState();
  return !!state && world.getAbsoluteTime() < state.until;
}
function startCooldown(seconds, label) {
  try {
    world.setDynamicProperty(COOLDOWN_KEY, JSON.stringify({ until: world.getAbsoluteTime() + seconds * 20, label }));
    world.setDynamicProperty(BURST_KEY, 0);
    const block = world.getDimension("overworld").getBlock(CORE);
    if (block) block.setType("minecraft:bedrock");
    world.sendMessage("§6[Natural Market] §r" + label + ": the core is Bedrock for " + seconds + " seconds.");
    log("INFO", "Cooldown started: " + label + " " + seconds + "s.");
  } catch (e) { logError("startCooldown", e); }
}
function tickCooldown() {
  const state = cooldownState();
  if (!state) return;
  try {
    const left = state.until - world.getAbsoluteTime();
    if (left > 0) {
      const text = "§e" + state.label + ": the core returns in " + Math.ceil(left / 20) + "s";
      for (const p of world.getAllPlayers()) { try { p.onScreenDisplay.setActionBar(text); } catch {} }
      return;
    }
    world.setDynamicProperty(COOLDOWN_KEY, "");
    const block = world.getDimension("overworld").getBlock(CORE);
    if (block && ["minecraft:bedrock", "minecraft:air", "minecraft:grass_block"].includes(block.typeId)) placeCore(block);
    world.sendMessage("§6[Natural Market] §rThe core is back. Keep mining!");
  } catch (e) { logError("tickCooldown", e); }
}
let coreBreakPending = false;
function onCoreBreak(player, brokenId) {
  const personal = personalBreaks(player) + 1;
  const total = totalBreaks() + 1;
  writeScore(MINED, player, personal);
  writeScore(TOTAL, GLOBAL_KEY, total);
  tutorialStep(player, total);
  const value = blockValue(brokenId);
  writeScore(COINS, player, balance(player) + value);
  let cooldown = null;
  if (MILESTONES.some(m => m.unlock > 0 && m.unlock === total)) {
    world.sendMessage("§6[Natural Market] §rA new core milestone has been reached. More block types can now appear.");
    cooldown = { seconds: PHASE_SECONDS, label: "New phase" };
  } else {
    const burst = (Number(world.getDynamicProperty(BURST_KEY)) || 0) + 1;
    if (burst >= BURST_LIMIT) cooldown = { seconds: BURST_SECONDS, label: "Cooldown" };
    else world.setDynamicProperty(BURST_KEY, burst);
  }
  log("INFO", player.name + " mined the core: personal=" + personal + " shared=" + total + " block=" + brokenId + " reward=" + value + ".");
  tell(player, "Core mined: " + blockName(brokenId) + " +" + value + " Coins.");
  showBalance(player);
  return cooldown;
}

# OneBlock Natural Market — A OneBlock economy and supply shop

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**Use the Natural Market Ledger to buy supplies for a regenerating OneBlock survival challenge.**

## Overview

Start in the void on the central block at (0, 0, 0), with no bedrock beneath it. A short tutorial builds your platform: each of the first 8 breaks adds a block beside the core until a 3x3 platform surrounds it. Breaking it earns Coins and advances shared progression. Use the Natural Market Ledger to buy supplies with the Coins you earn.

## Key Features

### Regenerating OneBlock
- The central block returns after a successful break. It drops no items; breaking it only earns Coins.
- The core is a set of 30 custom Currency Blocks, numbered 1 to 30. Each has its own break time, the same with any tool or bare hands, and uses up more tool durability the higher it goes.
- After every 50 breaks the core turns into Bedrock for 30 seconds, and for 60 seconds at every new phase. A countdown shows on screen and the core comes back by itself.
- The central block is also restored automatically if something other than a player removes it.
- If a player ends up inside the central block or the 3x3 platform, they are teleported on top of the central block.
- The available natural block pool expands as shared mining milestones are reached.
- The starting core is Currency Block 1 and is managed by the map script after one-time void setup.

### Natural Market Ledger
- Opens the market in the game's standard form screen: a button list with item icons for each category and supply bundle.
- Tap a supply slot to purchase its displayed bundle directly; no separate purchase-confirmation screen.
- The Search button finds any unlocked supply by name, so you do not have to browse each section.
- Large categories use pages that reserve two slots for navigation controls.
- The catalogue only sells natural supplies (blocks, crops, raw food, raw ores, mob drops, Nether and End finds) plus mid-game Spawn Eggs. Crafted items are not sold. Entries unavailable in the current game version are hidden.
- Prices rise as supplies are bought and settle back down over time. Common supplies recover faster than rare ones.
- The market uses the standard server Action Form. It is **not a chest container screen** and does not support draggable slots or native container transfers.

### Coins and Selling
- Each central-block break awards Coins. Every block type has its own value, and higher tiers pay more.
- A Coins scoreboard is always shown on the sidebar. The About screen has a switch to hide your own name on it.
- Eligible catalogue items can be sold through the Ledger. Each item type can be locked so it is never sold; the Trident, Elytra and Totem of Undying start locked.

### Progression and Supply Pool
- Personal mining milestones unlock additional market supplies across 30 tiers, ending with Currency Block 30.
- The central block pool is set by the tiers only. Purchased blocks do not change it.

## How to Use

1. Import the Creative world first for initial checks.
2. Use the Natural Market Ledger from your inventory.
3. Tap a category, then tap a supply slot to buy its listed bundle.
4. Break the central block to earn Coins and progress.
5. Use the selling option in the market to exchange eligible catalogue items for Coins.

## Compatibility and Limitations

- Minecraft Bedrock Edition: minimum engine version 1.26.50.
- Stable Script API dependencies: `@minecraft/server` 2.10.0 and `@minecraft/server-ui` 2.2.0.
- Experimental features: none are turned on by the pack files. The custom Currency Blocks have not been tested in game.
- UI compatibility: The market uses the game's standard forms, so it does not depend on a custom screen and works with other add-ons that customize form screens.
- Achievements: Yes, achievement-friendly.
