```
import { world, system } from "@minecraft/server";

const STORE_KEY = "alnqol2:troughs";
const DIMENSION_IDS = ["minecraft:overworld", "minecraft:nether", "minecraft:the_end"];
const MAX_ENTRIES = 100;
const RUN_TICKS = 20;
const BREED_TICKS = 120;
const SAVE_TICKS = 200;
const RADIUS = 8;
const HERD_CAP = 24;
const COOLDOWN_KEY = "alnqol2:breed_until";
const COOLDOWN_TIME = 6000;
const TROUGHS = new Set(["minecraft:chest", "minecraft:barrel"]);
const BREEDS = [
  { type: "minecraft:cow", foods: ["minecraft:wheat"] },
  { type: "minecraft:sheep", foods: ["minecraft:wheat"] },
  { type: "minecraft:pig", foods: ["minecraft:carrot"] },
  { type: "minecraft:rabbit", foods: ["minecraft:carrot"] },
  {
    type: "minecraft:chicken",
    foods: [
      "minecraft:wheat_seeds",
      "minecraft:beetroot_seeds",
      "minecraft:melon_seeds",
      "minecraft:pumpkin_seeds",
      "minecraft:pitcher_pod",
      "minecraft:torchflower_seeds",
    ],
  },
];

const entries = [];
const known = new Set();
let dirty = false;

function keyOf(dimId, x, y, z) {
  return dimId + "|" + x + "|" + y + "|" + z;
}

function add(dimId, x, y, z) {
  const key = keyOf(dimId, x, y, z);
  if (known.has(key) || entries.length >= MAX_ENTRIES) return;
  known.add(key);
  entries.push({ dim: dimId, x, y, z, key, nextAt: system.currentTick + BREED_TICKS });
  dirty = true;
}

function removeAt(index) {
  known.delete(entries[index].key);
  entries.splice(index, 1);
  dirty = true;
}

function load() {
  try {
    const raw = world.getDynamicProperty(STORE_KEY);
    if (typeof raw !== "string") return;
    const list = JSON.parse(raw);
    if (!Array.isArray(list)) return;
    for (const item of list) {
      if (!Array.isArray(item) || item.length !== 4) continue;
      const dimId = DIMENSION_IDS[item[0]];
      if (dimId) add(dimId, item[1], item[2], item[3]);
    }
    dirty = false;
  } catch (e) {
    entries.length = 0;
    known.clear();
  }
}

function save() {
  if (!dirty) return;
  try {
    const payload = entries
      .map((e) => [DIMENSION_IDS.indexOf(e.dim), e.x, e.y, e.z])
      .filter((p) => p[0] >= 0);
    world.setDynamicProperty(STORE_KEY, JSON.stringify(payload));
    dirty = false;
  } catch (e) {}
}

function countFood(container, foods) {
  let total = 0;
  for (let slot = 0; slot < container.size; slot++) {
    const item = container.getItem(slot);
    if (item && foods.includes(item.typeId)) total += item.amount;
  }
  return total;
}

function takeFood(container, foods, amount) {
  let need = amount;
  for (let slot = 0; slot < container.size && need > 0; slot++) {
    const item = container.getItem(slot);
    if (!item || !foods.includes(item.typeId)) continue;
    const used = Math.min(item.amount, need);
    need -= used;
    if (used >= item.amount) {
      container.setItem(slot);
    } else {
      item.amount -= used;
      container.setItem(slot, item);
    }
  }
}

function isReady(animal, now) {
  if (!animal.isValid || animal.hasComponent("minecraft:is_baby")) return false;
  const health = animal.getComponent("minecraft:health");
  if (!health || health.currentValue < health.effectiveMax) return false;
  const until = animal.getDynamicProperty(COOLDOWN_KEY);
  return !(typeof until === "number" && until > now);
}

function hearts(dimension, animal) {
  try {
    const at = animal.location;
    dimension.spawnParticle("minecraft:heart_particle", { x: at.x, y: at.y + 1, z: at.z });
  } catch (e) {}
}

function breed(dimension, center, container, rule) {
  if (countFood(container, rule.foods) < 2) return;
  const herd = dimension.getEntities({ type: rule.type, location: center, maxDistance: RADIUS });
  if (herd.length >= HERD_CAP) return;
  const now = world.getAbsoluteTime();
  const ready = herd.filter((animal) => isReady(animal, now));
  if (ready.length < 2) return;
  const first = ready[0];
  const second = ready[1];
  const a = first.location;
  const b = second.location;
  const middle = { x: (a.x + b.x) / 2, y: Math.max(a.y, b.y), z: (a.z + b.z) / 2 };
  try {
    dimension.spawnEntity(rule.type, middle, { spawnEvent: "minecraft:entity_born" });
  } catch (e) {
    return;
  }
  takeFood(container, rule.foods, 2);
  first.setDynamicProperty(COOLDOWN_KEY, now + COOLDOWN_TIME);
  second.setDynamicProperty(COOLDOWN_KEY, now + COOLDOWN_TIME);
  hearts(dimension, first);
  hearts(dimension, second);
}

function process(entry, tick) {
  const dimension = world.getDimension(entry.dim);
  const block = dimension.getBlock({ x: entry.x, y: entry.y, z: entry.z });
  if (!block || !block.isValid) return true;
  if (!TROUGHS.has(block.typeId)) return false;
  if (tick < entry.nextAt) return true;
  entry.nextAt = tick + BREED_TICKS;
  const inventory = block.getComponent("minecraft:inventory");
  const container = inventory ? inventory.container : undefined;
  if (!container) return true;
  const center = block.center();
  for (const rule of BREEDS) breed(dimension, center, container, rule);
  return true;
}

function run() {
  const tick = system.currentTick;
  for (let i = entries.length - 1; i >= 0; i--) {
    let keep = true;
    try {
      keep = process(entries[i], tick);
    } catch (e) {
      keep = true;
    }
    if (!keep) removeAt(i);
  }
}

function onPlace(event) {
  try {
    const block = event.block;
    if (TROUGHS.has(block.typeId)) add(event.dimension.id, block.x, block.y, block.z);
  } catch (e) {}
}

world.afterEvents.worldLoad.subscribe(() => {
  load();
  world.afterEvents.playerPlaceBlock.subscribe(onPlace);
  system.runInterval(run, RUN_TICKS);
  system.runInterval(save, SAVE_TICKS);
});
```

```
const DIMENSION_IDS = ["minecraft:overworld", "minecraft:nether", "minecraft:the_end"];
const MAX_ENTRIES = 150;
const BUDGET = 30;
const REFRESH_TICKS = 20;
const SAVE_TICKS = 200;
const SIDES = [
  { x: 0, y: 0, z: -1 },
  { x: 1, y: 0, z: 0 },
  { x: 0, y: 0, z: 1 },
  { x: -1, y: 0, z: 0 },
];

const entries = [];

  return dimId + "|" + x + "|" + y + "|" + z;

  known.add(key);
  entries.push({ dim: dimId, x, y, z, key, filter: null, refreshAt: 0 });
  dirty = true;
}

function removeAt(index) {
  known.delete(entries[index].key);
  entries.splice(index, 1);
  dirty = true;
}

  try {

      .map((e) => [DIMENSION_IDS.indexOf(e.dim), e.x, e.y, e.z])
      .filter((p) => p[0] >= 0);
    world.setDynamicProperty(STORE_KEY, JSON.stringify(payload));
    dirty = false;
  } catch (e) {}
}

function parseFilter(text) {
  const joined = text.replace(/§./g, "").split("\n").join(",");
  const match = /^\s*filter\s*:(.*)$/i.exec(joined);
  if (!match) return null;
  const ids = new Set();
  for (const part of match[1].split(",")) {
    const id = part.trim().toLowerCase().replace(/\s+/g, "");
    if (!id) continue;
    ids.add(id.includes(":") ? id : "minecraft:" + id);
  }
  return ids.size > 0 ? ids : null;
}

function readSign(block) {
  const sign = block.getComponent("minecraft:sign");
  if (!sign) return undefined;
  const text = sign.getText(SignSide.Front);
  return typeof text === "string" ? text : "";
}

function scanSigns(block) {
  const info = { hasSign: false, unknown: false, filter: null };
  for (const side of SIDES) {
    const neighbour = block.offset(side);
    if (!neighbour) {
      info.unknown = true;
      continue;
    }
    const text = readSign(neighbour);
    if (text === undefined) continue;
    info.hasSign = true;
    if (!info.filter) info.filter = parseFilter(text);
  }
  return info;
}

function findDropSpot(block) {
  for (const side of SIDES) {
    const neighbour = block.offset(side);
    if (neighbour && neighbour.isAir) return neighbour.center();
  }
  return undefined;
}

function process(entry, tick) {
  const dimension = world.getDimension(entry.dim);
  const block = dimension.getBlock({ x: entry.x, y: entry.y, z: entry.z });
  if (!block || !block.isValid) return true;
  if (block.typeId !== "minecraft:hopper") return false;
  if (tick >= entry.refreshAt) {
    const info = scanSigns(block);
    if (!info.hasSign && !info.unknown) return false;
    entry.filter = info.filter;
    entry.refreshAt = tick + REFRESH_TICKS;
  }
  if (!entry.filter) return true;
  const inventory = block.getComponent("minecraft:inventory");
  const container = inventory ? inventory.container : undefined;
  if (!container) return true;
  for (let slot = 0; slot < container.size; slot++) {
    const item = container.getItem(slot);
    if (!item || entry.filter.has(item.typeId)) continue;
    const spot = findDropSpot(block);
    if (!spot) return true;
    container.setItem(slot);
    dimension.spawnItem(item, spot);
  }
  return true;
}

function step() {
  if (entries.length === 0) return;
  const tick = system.currentTick;
  let budget = Math.min(BUDGET, entries.length);
  while (budget-- > 0 && entries.length > 0) {
    if (cursor >= entries.length) cursor = 0;
    let keep = true;
    try {
      keep = process(entries[cursor], tick);
    } catch (e) {
      keep = true;
    }
    if (keep) cursor++;
    else removeAt(cursor);
  }
}

function onPlace(event) {
  try {
    const block = event.block;
    if (block.typeId === "minecraft:hopper") {
      if (scanSigns(block).hasSign) add(event.dimension.id, block.x, block.y, block.z);
      return;
    }
    if (!block.getComponent("minecraft:sign")) return;
    for (const side of SIDES) {
      const hopper = block.offset(side);
      if (hopper && hopper.typeId === "minecraft:hopper") add(event.dimension.id, hopper.x, hopper.y, hopper.z);
    }
  } catch (e) {}
}

world.afterEvents.worldLoad.subscribe(() => {
  load();
  world.afterEvents.playerPlaceBlock.subscribe(onPlace);
  system.runInterval(step, 1);
  system.runInterval(save, SAVE_TICKS);
});
```

```
import { world, system, BlockTypes, EntityInitializationCause } from "@minecraft/server";

const COOLDOWN_TICKS = 3;
const PLAYER_CLEARANCE = 1.5;
const MIN_ALONG = 0.45;
const MAX_ALONG = 1.05;
const MAX_ACROSS = 0.5;
const TOOL_SUFFIXES = ["_pickaxe", "_axe", "_shovel"];
const UNSAFE_PLACE = /(^|_)door$|bed$|sign$|banner$|head$|skull$/;
const PROTECTED = new Set([
  "minecraft:bedrock",
  "minecraft:barrier",
  "minecraft:command_block",
  "minecraft:chain_command_block",
  "minecraft:repeating_command_block",
  "minecraft:structure_block",
  "minecraft:structure_void",
  "minecraft:jigsaw",
  "minecraft:end_portal",
  "minecraft:end_portal_frame",
  "minecraft:end_gateway",
  "minecraft:portal",
  "minecraft:reinforced_deepslate",
  "minecraft:allow",
  "minecraft:deny",
  "minecraft:border_block",
  "minecraft:light_block",
]);
const FRONTS = [
  { x: 1, y: 0, z: 0 },
  { x: -1, y: 0, z: 0 },
  { x: 0, y: 1, z: 0 },
  { x: 0, y: -1, z: 0 },
  { x: 0, y: 0, z: 1 },
  { x: 0, y: 0, z: -1 },
];

const cooldowns = new Map();

function isDigTool(stack) {
  if (!TOOL_SUFFIXES.some((suffix) => stack.typeId.endsWith(suffix))) return false;
  return stack.hasComponent("minecraft:durability");
}

function isPlaceable(stack) {
  if (UNSAFE_PLACE.test(stack.typeId)) return false;
  return BlockTypes.get(stack.typeId) !== undefined;
}

function findDispenser(dimension, at) {
  const cell = { x: Math.floor(at.x), y: Math.floor(at.y), z: Math.floor(at.z) };
  let best;
  for (const front of FRONTS) {
    const origin = { x: cell.x - front.x, y: cell.y - front.y, z: cell.z - front.z };
    const block = dimension.getBlock(origin);
    if (!block || block.typeId !== "minecraft:dispenser") continue;
    const dx = at.x - (origin.x + 0.5);
    const dy = at.y - (origin.y + 0.5);
    const dz = at.z - (origin.z + 0.5);
    const along = dx * front.x + dy * front.y + dz * front.z;
    const across = Math.max(
      Math.abs(front.x === 0 ? dx : 0),
      Math.abs(front.y === 0 ? dy : 0),
      Math.abs(front.z === 0 ? dz : 0)
    );
    if (along < MIN_ALONG || along > MAX_ALONG || across > MAX_ACROSS) continue;
    if (!best || along < best.along) best = { block, front, along };
  }
  return best;
}

function giveBack(dimension, dispenser, front, stack) {
  if (!stack || stack.amount <= 0) return;
  const inventory = dispenser.getComponent("minecraft:inventory");
  const container = inventory ? inventory.container : undefined;
  const rest = container ? container.addItem(stack) : stack;
  if (!rest) return;
  const mouth = dispenser.offset(front);
  dimension.spawnItem(rest, mouth ? mouth.center() : dispenser.center());
}

function destroy(dimension, block) {
  try {
    dimension.runCommand("setblock " + block.x + " " + block.y + " " + block.z + " air destroy");
  } catch (e) {}
  if (block.isAir) return true;
  try {
    const drop = block.getItemStack(1, true);
    const spot = block.center();
    block.setType("minecraft:air");
    if (drop) dimension.spawnItem(drop, spot);
    return true;
  } catch (e) {
    return false;
  }
}

function breakFront(dimension, dispenser, front, stack) {
  const tool = stack.clone();
  const target = dispenser.offset(front);
  const blocked =
    !target ||
    target.isAir ||
    target.isLiquid ||
    PROTECTED.has(target.typeId) ||
    target.hasComponent("minecraft:inventory");
  if (blocked || !destroy(dimension, target)) {
    giveBack(dimension, dispenser, front, tool);
    return;
  }
  const durability = tool.getComponent("minecraft:durability");
  if (durability) {
    durability.damage += 1;
    if (durability.damage >= durability.maxDurability) return;
  }
  giveBack(dimension, dispenser, front, tool);
}

function placeFront(dimension, dispenser, front, stack) {
  const target = dispenser.offset(front);
  if (!target || !target.isAir) {
    giveBack(dimension, dispenser, front, stack.clone());
    return;
  }
  try {
    target.setType(stack.typeId);
  } catch (e) {
    giveBack(dimension, dispenser, front, stack.clone());
    return;
  }
  if (stack.amount > 1) {
    const rest = stack.clone();
    rest.amount = stack.amount - 1;
    giveBack(dimension, dispenser, front, rest);
  }
}

function pruneCooldowns(tick) {
  if (cooldowns.size < 300) return;
  for (const [key, until] of cooldowns) {
    if (until <= tick) cooldowns.delete(key);
  }
}

function onSpawn(event) {
  try {
    if (event.cause === EntityInitializationCause.Loaded) return;
    const entity = event.entity;
    if (!entity.isValid || entity.typeId !== "minecraft:item") return;
    const dimension = entity.dimension;
    const at = entity.location;
    const found = findDispenser(dimension, at);
    if (!found) return;
    const component = entity.getComponent("minecraft:item");
    const stack = component ? component.itemStack : undefined;
    if (!stack) return;
    const breaker = isDigTool(stack);
    if (!breaker && !isPlaceable(stack)) return;
    const tick = system.currentTick;
    const key = dimension.id + "|" + found.block.x + "|" + found.block.y + "|" + found.block.z;
    if ((cooldowns.get(key) || 0) > tick) return;
    if (dimension.getPlayers({ location: at, maxDistance: PLAYER_CLEARANCE }).length > 0) return;
    cooldowns.set(key, tick + COOLDOWN_TICKS);
    pruneCooldowns(tick);
    entity.remove();
    if (breaker) breakFront(dimension, found.block, found.front, stack);
    else placeFront(dimension, found.block, found.front, stack);
  } catch (e) {}
}

world.afterEvents.worldLoad.subscribe(() => {
  world.afterEvents.entitySpawn.subscribe(onSpawn);
});
```

```
# Script Library Notes

Each file in this folder registers its own behavior when it is imported by main.js. Nothing is exported, and every file starts its work inside the world load event.

- filter_hopper.js: tracks Hoppers that have a Sign beside them, reads a Sign starting with "Filter:" and pushes any item not on the list out onto a free side of the Hopper. Registers the player place block event and two intervals (1 tick, 200 ticks). Depends on the server module. Saves its Hopper list in a world property named alnqol2:filter_hoppers (150 entries at most).
- vacuum_hopper.js: tracks Hoppers that have a Sign beside them, and for a Sign line reading "Vacuum Hopper" moves loose item entities from a 5 by 5 by 5 box into the Hopper every 10 ticks. Registers the player place block event and two intervals (10 ticks, 200 ticks). Depends on the server module. Saves its Hopper list in a world property named alnqol2:vacuum_hoppers (150 entries at most).
- mechanical_dispenser.js: listens to the entity spawn event for item entities that appear at the front of a Dispenser. A Pickaxe, Axe or Shovel breaks the block in front and loses 1 durability; a placeable block item is placed in an empty space in front. The ejected item is removed and returned to the Dispenser when nothing happens. Depends on the server module. Keeps no saved data.
- auto_breed.js: tracks Chests and Barrels, and every 120 ticks spawns a baby for two healthy adult animals within 8 blocks when the container holds their food. Registers the player place block event and two intervals (20 ticks, 200 ticks). Depends on the server module. Saves its container list in a world property named alnqol2:troughs (100 entries at most) and a breed cooldown dynamic property on each animal named alnqol2:breed_until.

Script events and custom ids: none. The pack adds no custom items, blocks or entities.

Known limits:
- Hoppers, Chests and Barrels placed before the pack was installed are tracked only after they are placed again, or after a Sign is placed beside a Hopper.
- Experience orbs are never collected by the Vacuum Hopper.
- The Dispenser logic reacts to the item a Dispenser drops, so a player standing within 1.5 blocks of that item stops it from firing.
- Breeding spawns a baby directly; animals do not enter love mode.
- Source created, not tested in game.
```

```
import { world, system, SignSide } from "@minecraft/server";

const STORE_KEY = "alnqol2:vacuum_hoppers";
const DIMENSION_IDS = ["minecraft:overworld", "minecraft:nether", "minecraft:the_end"];
const MAX_ENTRIES = 150;
const SWEEP_TICKS = 10;
const REFRESH_SWEEPS = 4;
const SAVE_TICKS = 200;
const BOX = 5;
const REACH = 2;
const MAX_PER_SWEEP = 32;
const SIDES = [
  { x: 0, y: 0, z: -1 },
  { x: 1, y: 0, z: 0 },
  { x: 0, y: 0, z: 1 },
  { x: -1, y: 0, z: 0 },
];

const entries = [];
const known = new Set();
let sweepCount = 0;
let dirty = false;

function keyOf(dimId, x, y, z) {
  return dimId + "|" + x + "|" + y + "|" + z;
}

function add(dimId, x, y, z) {
  const key = keyOf(dimId, x, y, z);
  if (known.has(key) || entries.length >= MAX_ENTRIES) return;
  known.add(key);
  entries.push({ dim: dimId, x, y, z, key, active: false, refreshAt: 0 });
  dirty = true;
}

function removeAt(index) {
  known.delete(entries[index].key);
  entries.splice(index, 1);
  dirty = true;
}

function load() {
  try {
    const raw = world.getDynamicProperty(STORE_KEY);
    if (typeof raw !== "string") return;
    const list = JSON.parse(raw);
    if (!Array.isArray(list)) return;
    for (const item of list) {
      if (!Array.isArray(item) || item.length !== 4) continue;
      const dimId = DIMENSION_IDS[item[0]];
      if (dimId) add(dimId, item[1], item[2], item[3]);
    }
    dirty = false;
  } catch (e) {
    entries.length = 0;
    known.clear();
  }
}

function save() {
  if (!dirty) return;
  try {
    const payload = entries
      .map((e) => [DIMENSION_IDS.indexOf(e.dim), e.x, e.y, e.z])
      .filter((p) => p[0] >= 0);
    world.setDynamicProperty(STORE_KEY, JSON.stringify(payload));
    dirty = false;
  } catch (e) {}
}

function isVacuumText(text) {
  const lines = text.replace(/§./g, "").split("\n");
  for (const line of lines) {
    if (line.trim().toLowerCase() === "vacuum hopper") return true;
  }
  return false;
}

function readSign(block) {
  const sign = block.getComponent("minecraft:sign");
  if (!sign) return undefined;
  const text = sign.getText(SignSide.Front);
  return typeof text === "string" ? text : "";
}

function scanSigns(block) {
  const info = { hasSign: false, unknown: false, vacuum: false };
  for (const side of SIDES) {
    const neighbour = block.offset(side);
    if (!neighbour) {
      info.unknown = true;
      continue;
    }
    const text = readSign(neighbour);
    if (text === undefined) continue;
    info.hasSign = true;
    if (isVacuumText(text)) info.vacuum = true;
  }
  return info;
}

function collect(dimension, block, container) {
  const origin = { x: block.x - REACH, y: block.y - REACH, z: block.z - REACH };
  const found = dimension.getEntities({
    type: "minecraft:item",
    location: origin,
    volume: { x: BOX, y: BOX, z: BOX },
  });
  let moved = 0;
  for (const entity of found) {
    if (moved >= MAX_PER_SWEEP) break;
    if (!entity.isValid) continue;
    const at = entity.location;
    if (at.x < origin.x || at.x >= origin.x + BOX) continue;
    if (at.y < origin.y || at.y >= origin.y + BOX) continue;
    if (at.z < origin.z || at.z >= origin.z + BOX) continue;
    let stack;
    try {
      const component = entity.getComponent("minecraft:item");
      stack = component ? component.itemStack : undefined;
    } catch (e) {
      continue;
    }
    if (!stack) continue;
    const rest = container.addItem(stack);
    if (rest && rest.amount >= stack.amount) continue;
    entity.remove();
    if (rest) dimension.spawnItem(rest, at);
    moved++;
  }
}

function process(entry) {
  const dimension = world.getDimension(entry.dim);
  const block = dimension.getBlock({ x: entry.x, y: entry.y, z: entry.z });
  if (!block || !block.isValid) return true;
  if (block.typeId !== "minecraft:hopper") return false;
  if (sweepCount >= entry.refreshAt) {
    const info = scanSigns(block);
    if (!info.hasSign && !info.unknown) return false;
    entry.active = info.vacuum;
    entry.refreshAt = sweepCount + REFRESH_SWEEPS;
  }
  if (!entry.active) return true;
  const inventory = block.getComponent("minecraft:inventory");
  const container = inventory ? inventory.container : undefined;
  if (!container) return true;
  collect(dimension, block, container);
  return true;
}

function sweep() {
  sweepCount++;
  for (let i = entries.length - 1; i >= 0; i--) {
    let keep = true;
    try {
      keep = process(entries[i]);
    } catch (e) {
      keep = true;
    }
    if (!keep) removeAt(i);
  }
}

function onPlace(event) {
  try {
    const block = event.block;
    if (block.typeId === "minecraft:hopper") {
      if (scanSigns(block).hasSign) add(event.dimension.id, block.x, block.y, block.z);
      return;
    }
    if (!block.getComponent("minecraft:sign")) return;
    for (const side of SIDES) {
      const hopper = block.offset(side);
      if (hopper && hopper.typeId === "minecraft:hopper") add(event.dimension.id, hopper.x, hopper.y, hopper.z);
    }
  } catch (e) {}
}

world.afterEvents.worldLoad.subscribe(() => {
  load();
  world.afterEvents.playerPlaceBlock.subscribe(onPlace);
  system.runInterval(sweep, SWEEP_TICKS);
  system.runInterval(save, SAVE_TICKS);
});
```

```
# ALN QualityOfLife Part 2 — Smart Redstone Automation and Quality of Life Suite

**Filter and Vacuum Hoppers set by Sign, Block Breaker and Block Placer Dispensers, and troughs that breed 5 farm animals.**

> This Behavior Pack adds sorting, collecting, mining, building and breeding automation to Hoppers, Dispensers, Chests and Barrels. It needs no custom items, so it works in existing survival and multiplayer worlds.

## Overview

Place a Sign next to a Hopper to turn it into a Filter Hopper or a Vacuum Hopper. Load a Dispenser with a tool or with blocks to make it break or place a block on each redstone pulse. Stock a Chest or Barrel with food near your pens to breed animals on a timer. Everything uses vanilla blocks, so no new items are added.

## Key Features

### Filter Hopper
- **What Happens**: A Hopper with a Sign beside it that starts with "Filter:" keeps only the items listed on the Sign. Any other item is pushed out onto a free side of the Hopper within a tick or two.
- **How It Works**: Write "Filter: diamond, iron_ingot" on a Sign placed next to the Hopper. Item names use the game's item names with underscores, and the lines of the Sign are read together.
- **Good To Know**: The pack tracks up to 150 Filter Hoppers. A Hopper with no free side keeps its items.

### Block Breaker Dispenser
- **What Happens**: A Dispenser holding a Pickaxe, Axe or Shovel breaks the block in front of it on every redstone pulse. The block drops as an item and the tool loses 1 durability.
- **How It Works**: Load the tool, point the Dispenser at a block and pulse it. The tool goes back into the Dispenser until it wears out.
- **Good To Know**: Bedrock, command blocks, liquids and blocks with an inventory are never broken. Stand more than 1.5 blocks away from the front of the Dispenser while it fires.

### Block Placer Dispenser
- **What Happens**: A Dispenser holding placeable blocks puts 1 block into the empty space in front of it on every redstone pulse.
- **How It Works**: Load the blocks, point the Dispenser at an empty space and pulse it. If the space is not empty, the block stays in the Dispenser.
- **Good To Know**: Placed blocks use their default direction. Doors, beds, signs, banners and heads are not placed.

### Vacuum Hopper
- **What Happens**: A Hopper with a Sign beside it that reads "Vacuum Hopper" pulls loose item drops from a 5 by 5 by 5 area into its own slots every 10 ticks.
- **How It Works**: Place the Hopper, put a Sign next to it with "Vacuum Hopper" on any line, and drop or let items fall nearby.
- **Good To Know**: Experience orbs are not collected, because a Hopper cannot hold them. The pack tracks up to 150 Vacuum Hoppers.

### Auto-Breed Trough
- **What Happens**: A Chest or Barrel stocked with food breeds healthy adult animals within 8 blocks every 120 ticks. Each breeding uses 2 food and spawns one baby with heart particles.
- **How It Works**: Wheat breeds cows and sheep, Carrots breed pigs and rabbits, and Wheat Seeds, Beetroot Seeds, Melon Seeds, Pumpkin Seeds, Pitcher Pods and Torchflower Seeds breed chickens.
- **Good To Know**: The baby is spawned directly, so animals do not enter love mode. Each parent waits 5 minutes before breeding again, and a trough stops when 24 animals of that kind are already in range.

## How to Use

1. Place a Hopper, then place a Sign beside it and write "Filter:" followed by item names, or write "Vacuum Hopper".
2. Load a Dispenser with a Pickaxe, Axe or Shovel to break blocks, or with building blocks to place them, and pulse it with redstone.
3. Place a Chest or Barrel within 8 blocks of your animals and fill it with their food. Watch for hearts and babies every 6 seconds.

- **Do I need a Resource Pack?**: No. The pack is a Behavior Pack only.
```
