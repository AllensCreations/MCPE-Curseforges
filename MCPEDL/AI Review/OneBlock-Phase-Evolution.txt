function getBreakCount() {
  const value = world.getDynamicProperty(BREAKS_KEY);
  return typeof value === "number" && Number.isFinite(value) ? Math.max(0, Math.floor(value)) : 0;
}

function phaseForCount(count) {
  let phase = 1;
  for (let i = 0; i < PHASES.length; i++) if (count >= PHASES[i].threshold) phase = i + 1;
  return phase;
}

function weightedPick(entries) {
  const total = entries.reduce((sum, entry) => sum + entry.weight, 0);
  let roll = Math.random() * total;
  for (const entry of entries) {
    roll -= entry.weight;
    if (roll < 0) return entry;
  }
  return entries[entries.length - 1];
}

function phaseWeights(unlocked) {
  const weights = [];
  for (let phase = 1; phase <= unlocked; phase++) {
    const age = unlocked - phase;
    if (age === 0) weights.push({ phase, weight: 45 });
    else if (age === 1) weights.push({ phase, weight: 25 });
    else if (age === 2) weights.push({ phase, weight: 15 });
    else if (age === 3) weights.push({ phase, weight: 10 });
    else weights.push({ phase, weight: 5 / (unlocked - 4) });
  }
  return weights;
}

function chooseBlock(unlocked) {
  const selectedPhase = weightedPick(phaseWeights(unlocked)).phase;
  return weightedPick(PHASES[selectedPhase - 1].blocks).id;

# OneBlock Phase Evolution — A cumulative twelve-phase OneBlock map

**Automatic Team Inventory**

The map automatically combines eligible players’ 36-slot inventories, armor, offhand, and Ender Chests into one persistent shared inventory. - **What Happens**: Every unlocked phase remains eligible, with the newest phase receiving the highest phase-selection weight.

- **What Happens**: Blocks inside each phase have individual weights so common materials appear more often than rare materials.

- **What Happens**: The world stores the total number of OneBlock breaks and restores the correct phase after rejoining.

### Mixed Biome Families
- **What Happens**: Each phase adds resource families from several Overworld biomes or from the Nether and End, rather than locking generation to one biome.

### Natural Special Resources
- **What Happens**: Rare blocks enter the pool at later phases. The map does not force-spawn mobs or fabricate mob drops; mob drops remain tied to naturally available mobs.

### Three World Modes
- **What Happens**: The repository workflow creates Creative, Survival, and Hardcore world variants from their corresponding foundations.

Break the single block at the center of the island. Each successful player break increases the shared counter.
