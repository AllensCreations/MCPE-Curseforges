import {
  world,
  system,
  MolangVariableMap,
  EquipmentSlot,
  EntityComponentTypes,
} from "@minecraft/server";
import { isActive, getBattery } from "./visor_toggle.js";

const VISOR_ID      = "aln:xray_visor";
const SCAN_INTERVAL = 10;
const RADIUS_MIN    = 3;
const RADIUS_MAX    = 7;
const ORE_LIMIT     = 20;
const STONE_LIMIT   = 10;

const ORE_TIERS = [
  {
    suffixes: ["diamond_ore", "emerald_ore"],
    particle: "aln:ore_diamond",
    color: { red: 0.0, green: 1.0, blue: 0.93 },
  },
  {
    suffixes: ["gold_ore", "ancient_debris"],
    particle: "aln:ore_gold",
    color: { red: 1.0, green: 0.84, blue: 0.0 },
  },
  {
    suffixes: ["iron_ore", "copper_ore"],
    particle: "aln:ore_iron",
    color: { red: 1.0, green: 0.55, blue: 0.26 },
  },
  {
    suffixes: ["redstone_ore"],
    particle: "aln:ore_redstone",
    color: { red: 1.0, green: 0.13, blue: 0.13 },
  },
  {
    suffixes: ["lapis_ore", "coal_ore", "nether_quartz_ore", "nether_gold_ore"],
    particle: "aln:ore_common",
    color: { red: 0.53, green: 0.6, blue: 0.8 },
  },
];

const STONE_IDS = new Set([
  "minecraft:stone",
  "minecraft:deepslate",
  "minecraft:cobbled_deepslate",
  "minecraft:dirt",
  "minecraft:grass_block",
  "minecraft:gravel",
  "minecraft:tuff",
  "minecraft:andesite",
  "minecraft:granite",
  "minecraft:diorite",
  "minecraft:netherrack",
  "minecraft:end_stone",
]);

function getOreTier(typeId) {
  for (const tier of ORE_TIERS) {
    for (const suffix of tier.suffixes) {
      if (typeId.endsWith(suffix)) return tier;
    }
  }
  return null;
}

function isStone(typeId) {
  return STONE_IDS.has(typeId);
}

function buildShellOffsets() {
  const offsets = [];
  const rMaxSq  = RADIUS_MAX * RADIUS_MAX;
  const rMinSq  = RADIUS_MIN * RADIUS_MIN;
  for (let x = -RADIUS_MAX; x <= RADIUS_MAX; x++) {
    for (let y = -RADIUS_MAX; y <= RADIUS_MAX; y++) {
      for (let z = -RADIUS_MAX; z <= RADIUS_MAX; z++) {
        const dSq = x * x + y * y + z * z;
        if (dSq >= rMinSq && dSq <= rMaxSq) offsets.push([x, y, z]);
      }
    }
  }
  return offsets;
}
const SHELL_OFFSETS = buildShellOffsets();

function shuffled(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function getActiveVisorStack(player) {
  try {
    const equip = player.getComponent(EntityComponentTypes.Equippable);
    if (!equip) return null;
    const stack = equip.getEquipment(EquipmentSlot.Head);
    if (!stack || stack.typeId !== VISOR_ID) return null;
    if (!isActive(stack)) return null;
    if (getBattery(stack) <= 0) return null;
    return stack;
  } catch {
    return null;
  }
}

function spawnOreMarker(dimension, bx, by, bz, tier) {
  try {
    const loc   = { x: bx + 0.5, y: by + 0.5, z: bz + 0.5 };
    const molang = new MolangVariableMap();
    molang.setColorRGB("variable.color", tier.color);
    dimension.spawnParticle(tier.particle, loc, molang);
  } catch {  }
}

function spawnStoneHaze(dimension, bx, by, bz) {
  try {
    const loc = { x: bx + 0.5, y: by + 0.5, z: bz + 0.5 };
    dimension.spawnParticle("aln:stone_haze", loc);
  } catch {  }
}

function scanForPlayer(player) {
  const loc       = player.location;
  const dimension = player.dimension;
  const bx0 = Math.floor(loc.x);
  const by0 = Math.floor(loc.y);
  const bz0 = Math.floor(loc.z);

  let oreCount   = 0;
  let stoneCount = 0;
  const offsets  = shuffled(SHELL_OFFSETS);

  for (const [ox, oy, oz] of offsets) {
    if (oreCount >= ORE_LIMIT && stoneCount >= STONE_LIMIT) break;

    const bx = bx0 + ox;
    const by = by0 + oy;
    const bz = bz0 + oz;

    let block;
    try {
      block = dimension.getBlock({ x: bx, y: by, z: bz });
    } catch {
      continue;
    }
    if (!block) continue;

    const typeId = block.typeId;
    const tier   = getOreTier(typeId);

    if (tier && oreCount < ORE_LIMIT) {
      spawnOreMarker(dimension, bx, by, bz, tier);
      oreCount++;
    } else if (!tier && stoneCount < STONE_LIMIT && isStone(typeId)) {
      spawnStoneHaze(dimension, bx, by, bz);
      stoneCount++;
    }
  }
}

export function startScannerLoop() {
  system.runInterval(() => {
    for (const player of world.getAllPlayers()) {
      if (!getActiveVisorStack(player)) continue;
      scanForPlayer(player);
    }
  }, SCAN_INTERVAL);
}

import {
  world,
  system,
  EquipmentSlot,
  EntityComponentTypes,
} from "@minecraft/server";

const VISOR_ID = "aln:xray_visor";

const PROP_ACTIVE  = "aln_visor_active";
const PROP_BATTERY = "aln_visor_battery";

const DRAIN_INTERVAL_TICKS = 100;

function getVisorStack(player) {
  try {
    const equip = player.getComponent(EntityComponentTypes.Equippable);
    if (!equip) return null;
    const stack = equip.getEquipment(EquipmentSlot.Head);
    if (!stack || stack.typeId !== VISOR_ID) return null;
    return stack;
  } catch {
    return null;
  }
}

function setVisorStack(player, stack) {
  try {
    const equip = player.getComponent(EntityComponentTypes.Equippable);
    if (!equip) return;
    equip.setEquipment(EquipmentSlot.Head, stack);
  } catch {

  }
}

export function getBattery(stack) {
  const val = stack.getDynamicProperty(PROP_BATTERY);
  return typeof val === "number" ? Math.max(0, Math.min(100, Math.floor(val))) : 100;
}

export function isActive(stack) {
  const val = stack.getDynamicProperty(PROP_ACTIVE);
  return val === true;
}

function buildStatusBar(active, battery) {
  if (!active) {
    return "§b[X-Ray Visor] §cOFFLINE §8(Battery: " + battery + "%)";
  }
  const battColour = battery > 30 ? "§a" : battery > 10 ? "§e" : "§c";
  return "§b[X-Ray Visor] §aONLINE §8(Battery: " + battColour + battery + "%§8)";
}

export function forceVisorOff(player) {
  const stack = getVisorStack(player);
  if (!stack) return null;
  stack.setDynamicProperty(PROP_ACTIVE, false);
  setVisorStack(player, stack);
  try {
    player.onScreenDisplay.setActionBar("§b[X-Ray Visor] §cBATTERY EMPTY — recharge with Redstone Dust!");
    player.playSound("random.click", { pitch: 0.5, volume: 1.0 });
  } catch {  }
  return stack;
}

export function registerToggleListener() {

  system.runInterval(() => {
    for (const player of world.getAllPlayers()) {
      const stack = getVisorStack(player);
      if (!stack) continue;
      if (!isActive(stack)) continue;

      let battery = getBattery(stack);
      battery = Math.max(0, battery - 1);
      stack.setDynamicProperty(PROP_BATTERY, battery);

      if (battery <= 0) {

        stack.setDynamicProperty(PROP_ACTIVE, false);
        setVisorStack(player, stack);
        try {
          player.onScreenDisplay.setActionBar(
            "§b[X-Ray Visor] §cBATTERY EMPTY — recharge with Redstone Dust!"
          );
          player.playSound("random.click", { pitch: 0.5, volume: 0.8 });
        } catch {  }
      } else {

        setVisorStack(player, stack);

        try {
          player.onScreenDisplay.setActionBar(buildStatusBar(true, battery));
        } catch {  }
      }
    }
  }, DRAIN_INTERVAL_TICKS);

  world.afterEvents.itemUse.subscribe((ev) => {
    const player = ev.source;

    if (ev.itemStack.typeId !== VISOR_ID) return;

    const stack = getVisorStack(player);
    if (!stack) return;

    const battery = getBattery(stack);

    if (!isActive(stack) && battery <= 0) {
      try {
        player.onScreenDisplay.setActionBar(
          "§b[X-Ray Visor] §cNo charge — repair with Redstone Dust!"
        );
        player.playSound("random.click", { pitch: 0.4, volume: 0.8 });
      } catch {  }
      return;
    }

    const newState = !isActive(stack);
    stack.setDynamicProperty(PROP_ACTIVE, newState);
    setVisorStack(player, stack);

    try {
      player.onScreenDisplay.setActionBar(buildStatusBar(newState, battery));

      player.playSound("random.click", {
        pitch: newState ? 1.4 : 0.8,
        volume: 1.0,
      });
    } catch {  }
  });
}

**Wear the craftable X-Ray Visor helmet to mark ores with glowing particles up to 7 blocks away, powered by a draining battery.**

> This add-on adds the X-Ray Visor, a helmet that highlights nearby ores through solid walls. It suits miners who want a faster way to find Diamond, Gold and Iron without digging blind.

## Overview

Craft the X-Ray Visor, put it on and switch it on. While it is online, it scans the area around you twice a second and places colored glowing markers on valuable ores, while a faint haze hangs over the surrounding stone and dirt. Each ore type has its own color, and the battery slowly drains while the visor is on.

## Key Features

### X-Ray Visor Helmet
- **What Happens**: Adds a wearable sci-fi helmet with its own 3D model, an inventory icon and 520 durability.
- **How It Works**: Craft it with four Glass Panes, one Gold Ingot, one Redstone Comparator and one Iron Helmet.

### Ore Scanner
- **What Happens**: Marks Diamond, Emerald, Gold, Ancient Debris, Iron, Copper, Redstone, Lapis and Coal ores with colored glowing particles you can see through walls.
- **Good To Know**: The scan covers a ring from 3 to 7 blocks around you, so ores closer than 3 blocks are not marked.

### Stone Haze
- **What Happens**: Spawns faint, low-opacity smoke over Stone, Deepslate, Dirt, Gravel, Netherrack and similar filler blocks so ore markers stand out.

### Battery and Toggle
- **What Happens**: Right-click to switch the visor between ONLINE and OFFLINE with a click sound. The action bar shows the battery percentage.
- **Good To Know**: The battery drops 1% every 5 seconds while the visor is online and turns it off automatically at 0%.

## How to Use

1. Craft the X-Ray Visor at a crafting table and wear it in the helmet slot.
2. Right-click to switch it on and read the battery level on the action bar.
3. Look for the colored glowing markers on ores around you.
