5. Write the spec (below), test it (below), then reply (below). For changes to an existing project, never deliver a .sh: this overrides any older "deliver the .sh" wording. A NEW pack still follows `mcpe-new-pack` and delivers its generated .sh.

## The spec

File name: `spec_vN.txt`, a downloadable file. N = CURRENT VERSION + 1, read from the attached repo export, never guessed. The user builds the .sh on their phone with `scripts/make_update_script.py`.

Format (`scripts/apply_update.py`):

- Start: `@@ version vN`, then `@@ project MCPEDL/<Category>/<Name>`, then one `@@ file <path relative to the project>` block per file.
- Directives: replace/with, replace-all, insert-after, insert-before, delete, append, create, overwrite, changelog. Remove files with `@@ delete-file <repo-root path>`.
- Small change: replace/insert only. Anchors are copied from the attached export, never from memory, and must occur exactly once. Do not end an anchor with a newline.
- Large redesign: `@@ overwrite` (new content must exceed 60% of the old file) or `@@ create` for new files. Never many fragile edits.
- A content line that starts with `@@` is written `@@@`.
- End with `@@ changelog <Update Name>` using `### Added`, `### Improved & Balanced`, `### Fixed` and `- **Title**: text`. Never touch manifest versions or README links.
- New pixel textures: `@@ create` the pixel JSON under `texture_sources/`, then add the PNG step to the Termux commands.
- Keep README.md and DESCRIPTION.md accurate (INSTRUCTIONS Part 2 template). Never claim an in-game test.

**Test before delivering.** Apply the spec to the export text in the sandbox with `apply_update.py` (`--dry-run`, then a real run on a copy). Check every anchor, validate each changed JSON, run `node --check` on JS. Report every check not run.

## Versions

- Never edit manifest versions or the README version or download link. The auto-version-bump workflow does that on push.

const STONE_IDS = new Set([
  "minecraft:stone",
  "minecraft:deepslate",
  "minecraft:cobbled_deepslate",
  "minecraft:dirt",
  "minecraft:grass_block",
  "minecraft:gravel",

  "minecraft:granite",
  "minecraft:diorite",
  "minecraft:netherrack",
  "minecraft:end_stone",
]);

function getOreTier(typeId) {
  for (const tier of ORE_TIERS) {
    for (const suffix of tier.suffixes) {
      if (typeId.endsWith(suffix)) return tier;
    }
  }
  return null;
}

function isStone(typeId) {
  return STONE_IDS.has(typeId);
}

function buildShellOffsets() {
  const offsets = [];
  const rMaxSq  = RADIUS_MAX * RADIUS_MAX;
  const rMinSq  = RADIUS_MIN * RADIUS_MIN;
  for (let x = -RADIUS_MAX; x <= RADIUS_MAX; x++) {
    for (let y = -RADIUS_MAX; y <= RADIUS_MAX; y++) {
      for (let z = -RADIUS_MAX; z <= RADIUS_MAX; z++) {
        const dSq = x * x + y * y + z * z;
        if (dSq >= rMinSq && dSq <= rMaxSq) offsets.push([x, y, z]);
      }
    }
  }
  return offsets;
}
const SHELL_OFFSETS = buildShellOffsets();

function shuffled(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function getActiveVisorStack(player) {
  try {
    const equip = player.getComponent(EntityComponentTypes.Equippable);
    if (!equip) return null;
    const stack = equip.getEquipment(EquipmentSlot.Head);
    if (!stack || stack.typeId !== VISOR_ID) return null;
    if (!isActive(stack)) return null;
    if (getBattery(stack) <= 0) return null;
    return stack;
  } catch {
    return null;
  }
}

function spawnOreMarker(dimension, bx, by, bz, tier) {
  try {
    const loc   = { x: bx + 0.5, y: by + 0.5, z: bz + 0.5 };
    const molang = new MolangVariableMap();
    molang.setColorRGB("variable.color", tier.color);
    dimension.spawnParticle(tier.particle, loc, molang);
  } catch {  }
}

function spawnStoneHaze(dimension, bx, by, bz) {
  try {
    const loc = { x: bx + 0.5, y: by + 0.5, z: bz + 0.5 };
    dimension.spawnParticle("aln:stone_haze", loc);
  } catch {  }
}

function scanForPlayer(player) {
  const loc       = player.location;
  const dimension = player.dimension;
  const bx0 = Math.floor(loc.x);
  const by0 = Math.floor(loc.y);
  const bz0 = Math.floor(loc.z);

  let oreCount   = 0;
  let stoneCount = 0;
  const offsets  = shuffled(SHELL_OFFSETS);

  for (const [ox, oy, oz] of offsets) {
    if (oreCount >= ORE_LIMIT && stoneCount >= STONE_LIMIT) break;

    const bx = bx0 + ox;
    const by = by0 + oy;
    const bz = bz0 + oz;

    let block;
    try {
      block = dimension.getBlock({ x: bx, y: by, z: bz });
    } catch {
      continue;
    }
    if (!block) continue;

    const typeId = block.typeId;
    const tier   = getOreTier(typeId);

    if (tier && oreCount < ORE_LIMIT) {
      spawnOreMarker(dimension, bx, by, bz, tier);
      oreCount++;
    } else if (!tier && stoneCount < STONE_LIMIT && isStone(typeId)) {
      spawnStoneHaze(dimension, bx, by, bz);
      stoneCount++;
    }
  }
}

export function startScannerLoop() {
  system.runInterval(() => {
    for (const player of world.getAllPlayers()) {
      if (!getActiveVisorStack(player)) continue;
      scanForPlayer(player);
    }
  }, SCAN_INTERVAL);
}

- **What Happens**: Spawns faint, low-opacity smoke over Stone, Deepslate, Dirt, Gravel, Netherrack and similar filler blocks so ore markers stand out.

- **What Happens**: Right-click to switch the visor between ONLINE and OFFLINE with a click sound. The action bar shows the battery percentage.
