import { world, system, ItemStack, EntityComponentTypes, GameMode } from "@minecraft/server";
import { ActionFormData, FormCancelationReason } from "@minecraft/server-ui";

const CONTROLLER_ID = "aln:random_drop_chaos_controller";
const STATE_KEY = "game_state";
const OPERATOR_LEVEL = 2;

let config = null;
let registered = false;

const getState = () => {
  const v = world.getDynamicProperty(STATE_KEY);
  return typeof v === "string" ? v : "idle";
};
const setState = (s) => world.setDynamicProperty(STATE_KEY, s);
const isCreative = (player) => player.getGameMode() === GameMode.Creative;
const isHost = (player) => (typeof player.isOp === "function" ? player.isOp() : player.playerPermissionLevel === OPERATOR_LEVEL);
const containerOf = (player) => player.getComponent(EntityComponentTypes.Inventory)?.container;

function hasController(player) {
  const inv = containerOf(player);
  if (!inv) return false;
  for (let i = 0; i < inv.size; i++) {
    if (inv.getItem(i)?.typeId === CONTROLLER_ID) return true;
  }
  return false;
}

function giveController(player) {
  if (hasController(player)) return;
  containerOf(player)?.addItem(new ItemStack(CONTROLLER_ID, 1));
}

function consumeController(player) {
  if (isCreative(player)) return;
  const inv = containerOf(player);
  if (!inv) return;
  for (let i = 0; i < inv.size; i++) {
    if (inv.getItem(i)?.typeId === CONTROLLER_ID) {
      inv.setItem(i, undefined);
      player.playSound("random.break");
      player.sendMessage("The Controller broke.");
      return;
    }
  }
}

function runStart(player) {
  if (getState() === "running") {
    player.sendMessage("Already running.");
    return;
  }
  setState("running");
  if (config.consumeOnStart) consumeController(player);
  try {
    config.onStart(player);
  } catch (e) {
    setState("idle");
    giveController(player);
    player.sendMessage("The game could not start.");
  }
}

export function endGame() {
  if (getState() !== "running") return;
  setState("ended");
  try {
    config?.onEnd();
  } catch (e) {
    world.sendMessage("The game ended with an error.");
  }
  for (const p of world.getAllPlayers()) if (isHost(p)) giveController(p);
}

async function showHowToPlay(player) {
  const form = new ActionFormData().title("How to play").body(config.howToPlay).button("Back");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (!res.canceled && res.selection === 0) openController(player);
}

async function confirmEnd(player) {
  if (getState() !== "running") {
    player.sendMessage("No game is running.");
    return;
  }
  const form = new ActionFormData()
    .title("End game?")
    .body("This stops the game that is running now.")
    .button("End Game")
    .button("Back");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (res.canceled) return;
  if (res.selection === 0) {
    if (getState() !== "running") {
      player.sendMessage("The game already ended.");
      return;
    }
    world.sendMessage(`${player.name} ended the game.`);
    endGame();
  } else {
    openController(player);
  }
}

async function openController(player, attempt = 0) {
  const running = getState() === "running";
  const note = running
    ? "End Game stops the current game."
    : config.consumeOnStart
      ? (isCreative(player) ? "Creative: the Controller is kept." : "Start uses up the Controller.")
      : "The Controller is kept.";
  const form = new ActionFormData()
    .title(config.title)
    .body(`${config.gameInfo}\n\nState: ${getState()}\n${note}`)
    .button(running ? "End Game" : "Start")
    .button("How to play");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (res.canceled) {
    if (res.cancelationReason === FormCancelationReason.UserBusy && attempt < 5) {
      system.runTimeout(() => openController(player, attempt + 1), 10);
    }
    return;
  }
  if (res.selection === 0) {
    if (running) confirmEnd(player);
    else runStart(player);
  } else if (res.selection === 1) {
    showHowToPlay(player);
  }
}

export function registerController(options) {
  if (registered) return;
  registered = true;
  config = {
    title: options?.title ?? "Controller",
    gameInfo: options?.gameInfo ?? "",
    howToPlay: options?.howToPlay ?? "",
    consumeOnStart: options?.consumeOnStart === true,
    onStart: options?.onStart ?? (() => {}),
    onEnd: options?.onEnd ?? (() => {}),
  };

  world.afterEvents.itemUse.subscribe(({ source, itemStack }) => {
    if (itemStack?.typeId === CONTROLLER_ID && isHost(source)) openController(source);
  });

  system.afterEvents.scriptEventReceive.subscribe(({ id }) => {
    if (id === "allenscreations:end") endGame();
  });

  world.afterEvents.worldLoad.subscribe(() => {
    if (getState() === "running") setState("idle");
    for (const p of world.getAllPlayers()) if (isHost(p)) giveController(p);
  });

  world.afterEvents.playerSpawn.subscribe(({ player, initialSpawn }) => {
    if (!isHost(player)) return;
    if (config.consumeOnStart) {
      if (initialSpawn && (isCreative(player) || getState() !== "running")) giveController(player);
      return;
    }
    giveController(player);
  });
}

import { world, system, ItemStack } from "@minecraft/server";
import { getReplacement } from "./random_loot_table.js";

const PROCESSED_TAG = "aln_randomized";
const ITEM_ENTITY = "minecraft:item";
const PLAYER_ENTITY = "minecraft:player";
const RADIUS_SQUARED = 1.5 * 1.5;
const ZONE_TICKS = 3;
const MAX_STACKS = 64;

let active = false;
let zones = [];
let pending = [];
const queue = [];
let jobRunning = false;
const maxAmounts = new Map();

export function setActive(value) {
  active = value === true;
  if (!active) {
    zones = [];
    pending = [];
    queue.length = 0;
  }
}

export function isActive() {
  return active;
}

function isNear(a, b) {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  const dz = a.z - b.z;
  return dx * dx + dy * dy + dz * dz <= RADIUS_SQUARED;
}

function getMaxAmount(typeId) {
  let amount = maxAmounts.get(typeId);
  if (amount === undefined) {
    amount = Math.max(1, new ItemStack(typeId, 1).maxAmount);
    maxAmounts.set(typeId, amount);
  }
  return amount;
}

function processItem(entity) {
  try {
    if (!entity.isValid) return;
    if (entity.hasTag(PROCESSED_TAG)) return;
    const component = entity.getComponent("minecraft:item");
    if (!component) return;
    const original = component.itemStack;
    const replacementId = getReplacement(original.typeId);
    if (!replacementId || replacementId === original.typeId) {
      entity.addTag(PROCESSED_TAG);
      return;
    }
    const dimension = entity.dimension;
    const location = entity.location;
    const maxAmount = getMaxAmount(replacementId);
    let remaining = original.amount;
    let stacks = 0;
    while (remaining > 0 && stacks < MAX_STACKS) {
      const amount = Math.min(maxAmount, remaining);
      const spawned = dimension.spawnItem(new ItemStack(replacementId, amount), location);
      spawned.addTag(PROCESSED_TAG);
      remaining -= amount;
      stacks += 1;
    }
    entity.remove();
  } catch (e) {}
}

function* drain() {
  try {
    let index = 0;
    while (index < queue.length) {
      processItem(queue[index]);
      index += 1;
      yield;
    }
  } finally {
    queue.length = 0;
    jobRunning = false;
  }
}

function enqueue(entity) {
  queue.push(entity);
  if (jobRunning) return;
  jobRunning = true;
  system.runJob(drain());
}

function prune(tick) {
  if (zones.length > 0) zones = zones.filter((zone) => tick - zone.tick <= ZONE_TICKS);
  if (pending.length > 0) pending = pending.filter((item) => tick - item.tick <= ZONE_TICKS);
}

function openZone(dimensionId, location) {
  const tick = system.currentTick;
  prune(tick);
  zones.push({ dimensionId, location, tick });
  if (pending.length === 0) return;
  const rest = [];
  for (const item of pending) {
    if (item.dimensionId === dimensionId && isNear(item.location, location)) enqueue(item.entity);
    else rest.push(item);
  }
  pending = rest;
}

function onItemSpawned(entity) {
  const tick = system.currentTick;
  prune(tick);
  const dimensionId = entity.dimension.id;
  const location = entity.location;
  for (const zone of zones) {
    if (zone.dimensionId === dimensionId && isNear(zone.location, location)) {
      enqueue(entity);
      return;
    }
  }
  pending.push({ entity, dimensionId, location, tick });
}

export function registerDropHooks() {
  world.afterEvents.entitySpawn.subscribe(({ entity }) => {
    if (!active) return;
    try {
      if (entity.typeId === ITEM_ENTITY) onItemSpawned(entity);
    } catch (e) {}
  });

  world.afterEvents.playerBreakBlock.subscribe(({ block, dimension }) => {
    if (!active) return;
    try {
      const center = { x: block.location.x + 0.5, y: block.location.y + 0.5, z: block.location.z + 0.5 };
      openZone(dimension.id, center);
    } catch (e) {}
  });

  world.afterEvents.entityDie.subscribe(({ deadEntity }) => {
    if (!active) return;
    try {
      if (deadEntity.typeId === PLAYER_ENTITY) return;
      openZone(deadEntity.dimension.id, deadEntity.location);
    } catch (e) {}
  });
}

let pool = [];
let mapping = new Map();
let seed = 0;
let ready = false;

function createRandom(initial) {
  let state = initial | 0;
  return function () {
    state = (state + 0x6d2b79f5) | 0;
    let t = Math.imul(state ^ (state >>> 15), 1 | state);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function hashId(text, salt) {
  let h = (2166136261 ^ salt) >>> 0;
  for (let i = 0; i < text.length; i++) {
    h ^= text.charCodeAt(i);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h >>> 0;
}

function freshSeed() {
  return Math.floor(Math.random() * 2147483646) + 1;
}

function resolvePool() {
  const found = [];
  const seen = new Set();
  for (const name of POOL_NAMES) {
    const id = NAMESPACE + name;
    if (seen.has(id)) continue;
    seen.add(id);
    let type;
    try {
      type = ItemTypes.get(id);
    } catch (e) {
      type = undefined;
    }
    if (type) found.push(id);
  }
  return found;
}

function build(seedValue) {
  seed = seedValue;
  if (pool.length === 0) pool = resolvePool();
  const count = pool.length;
  const random = createRandom(seedValue);
  const order = pool.map((_, index) => index);
  for (let i = count - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    const swap = order[i];
    order[i] = order[j];
    order[j] = swap;
  }
  if (count > 1) {
    for (let i = 0; i < count; i++) {
      if (order[i] === i) {
        const j = (i + 1) % count;
        const swap = order[i];
        order[i] = order[j];
        order[j] = swap;
      }
    }
  }
  mapping = new Map();
  for (let i = 0; i < count; i++) mapping.set(pool[i], pool[order[i]]);
}

function ensureReady() {
  if (ready) return;
  const stored = world.getDynamicProperty(SEED_KEY);
  const value = typeof stored === "number" ? stored : freshSeed();
  if (typeof stored !== "number") world.setDynamicProperty(SEED_KEY, value);
  build(value);
  ready = true;
}

export function initLootTable() {
  try {
    ensureReady();
  } catch (e) {
    ready = false;
  }
}

export function reshuffle() {
  const value = freshSeed();
  world.setDynamicProperty(SEED_KEY, value);
  build(value);
  ready = true;
  return value;
}

export function getReplacement(typeId) {
  ensureReady();
  const direct = mapping.get(typeId);
  if (direct) return direct;
  if (pool.length === 0) return typeId;
  return pool[hashId(typeId, seed) % pool.length];
}

# Minecraft, But Every Drop is Random: Chaos Edition — Randomized Drop Challenge

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)
[![Format](https://img.shields.io/badge/Format-.MCPACK%20%7C%20.MCADDON%20%7C%20.MCWORLD-blue?style=flat-square)](https://github.com/AllensCreations/MCPE-Curseforges)

**Every block and mob drop turns into a different item, and defeating the Ender Dragon wins the run.**

> This minigame swaps every drop, from Dirt to Rotten Flesh, for another item from a shuffled table. It suits survival challenge runs where nobody knows what the next block will give.

## Overview

The host opens the Controller and presses Start. From then on, each item that drops from a broken block or a defeated mob is replaced by another item, and the stack size stays the same. The same item always turns into the same replacement during one run, so players can learn the table as they go. The Controller is used up when the game starts, and the run ends when the Ender Dragon falls.

## Key Features

### Randomized Drops
- **What Happens**: Blocks and mobs drop a different item than normal, and the number of items stays the same. Items that cannot stack as high as the original are split into several stacks.
- **How It Works**: If Oak Log is swapped for a Golden Apple in your run, every Oak Log you break gives Golden Apples for the rest of that run.
- **Good To Know**: Items dropped by players on death are not swapped.

### Fresh Table Every Run
- **What Happens**: Pressing Start shuffles a new table, and no item is swapped for itself.
- **Good To Know**: The table cannot be reshuffled while a game is running.

### Reshuffle Outside a Game
- **What Happens**: Any player can reshuffle the table while no game is running, by using the command or by sneaking while using a Book.
- **Good To Know**: Starting a game always shuffles a new table anyway.

### Controller
- **What Happens**: The host and operators receive the Controller. It opens a menu with Start and How to play.
- **Good To Know**: Start uses up the Controller, except in Creative mode, and a new one is given to the host and operators when the game ends.

## How to Use

1. Enable the Behavior Pack and the Resource Pack in a world and take the Controller.
2. Use the Controller and press Start.
3. Break blocks and defeat mobs, then defeat the Ender Dragon to win.

## Commands

- `/aln:reshuffle`: Reshuffles the random drop table while no game is running.

## Common Questions

- **Can I reshuffle during a game?**: No. The table stays fixed while a game runs, so each run keeps one set of swaps.
- **What happens to the Controller when a game starts?**: It is used up unless you are in Creative mode, and a new one is given to the host and operators when the game ends.

- Minecraft Bedrock Edition: The Behavior Pack declares 1.26.60 as its minimum, and Minecraft's pack importer accepts it.
- Experimental features: No experiment requirement detected in the inspected files.
- Achievements: Yes, achievement-friendly. The optional recovery commands for giving the Controller and ending the game need cheats, and cheats disable achievements in that world.
