## FILE: MCPEDL/Addons/ALNSpiritSphere/behavior_pack/scripts/main.js (88 lines, 2817 bytes)

```
import { world, system, EntityComponentTypes } from "@minecraft/server";
import { FOCUS_ID, MODES, MIN_HOLD, OVER_TICKS, state, modeIndexOf } from "./elements.js";
import { renderCharge, phaseOf } from "./visuals.js";
import { launchOrb, tickOrbs, registerHits } from "./orbs.js";

const chargers = new Map();
const PHASES = ["", "Gathering", "Compressing", "Max Charge", "Overcharged"];

function syncLore(player, index) {
  const mode = MODES[index];
  const lore = [`${mode.code}Mode: ${mode.name}`, `§7${mode.lore}`, "§8Sneak + use to switch element"];
  try {
    const inv = player.getComponent(EntityComponentTypes.Inventory)?.container;
    if (!inv) return;
    for (let i = 0; i < inv.size; i++) {
      const item = inv.getItem(i);
      if (item?.typeId === FOCUS_ID) {
        item.setLore(lore);
        inv.setItem(i, item);
      }
    }
  } catch (e) {}
}

function cycleMode(player) {
  const next = (modeIndexOf(player) + 1) % MODES.length;
  state.modes.set(player.id, next);
  syncLore(player, next);
  const mode = MODES[next];
  try {
    player.onScreenDisplay.setActionBar(`${mode.code}Element: ${mode.name}`);
    player.playSound("random.orb");
  } catch (e) {}
}

function finish(source, itemStack) {
  if (!itemStack || itemStack.typeId !== FOCUS_ID) return;
  const entry = chargers.get(source.id);
  if (!entry || entry.fired) return;
  entry.fired = true;
  entry.doneTick = state.tick;
  const held = state.tick - entry.start;
  if (held >= MIN_HOLD) {
    try {
      launchOrb(source, modeIndexOf(source), held);
    } catch (e) {}
  }
}

world.afterEvents.itemStartUse.subscribe((ev) => {
  const player = ev.source;
  if (!ev.itemStack || ev.itemStack.typeId !== FOCUS_ID) return;
  if (player.isSneaking) {
    cycleMode(player);
    chargers.set(player.id, { player, start: state.tick, fired: true, doneTick: state.tick });
    return;
  }
  chargers.set(player.id, { player, start: state.tick, fired: false, doneTick: -1 });
});

world.afterEvents.itemReleaseUse.subscribe((ev) => finish(ev.source, ev.itemStack));
world.afterEvents.itemStopUse.subscribe((ev) => finish(ev.source, ev.itemStack));

registerHits();

system.runInterval(() => {
  state.tick++;
  for (const [id, entry] of chargers) {
    if (entry.fired) {
      if (state.tick - entry.doneTick > 3) chargers.delete(id);
      continue;
    }
    try {
      const held = state.tick - entry.start;
      const mode = MODES[modeIndexOf(entry.player)];
      renderCharge(entry.player, held, mode);
      if (held % 5 === 0) {
        const phase = phaseOf(held);
        const tail = held >= OVER_TICKS ? "§c" : "§f";
        entry.player.onScreenDisplay.setActionBar(`${mode.code}${mode.name} §7| ${tail}${PHASES[phase]}`);
      }
    } catch (e) {
      chargers.delete(id);
    }
  }
  tickOrbs();
}, 1);
```

import { MODES, ORB_ID, FULL_TICKS, OVER_TICKS, MAX_ORB_AGE } from "./elements.js";
import { sound, impactBurst, orbTrail, handPoint } from "./visuals.js";

const orbs = new Map();
const PLAYER_ID = "minecraft:player";

function nearby(dimension, at, distance) {
  try {
    return dimension.getEntities({ location: at, maxDistance: distance });
  } catch (e) {
    return [];
  }
}

function explode(dimension, at, radius, breaks, fire) {
  try {
    dimension.createExplosion(at, radius, { breaksBlocks: breaks, causesFire: fire, allowUnderwater: true });
  } catch (e) {}
}

function shove(entity, from, strength, lift) {
  try {
    const l = entity.location;
    const dx = l.x - from.x;
    const dz = l.z - from.z;
    const d = Math.hypot(dx, dz) || 1;
    entity.applyKnockback({ x: (dx / d) * strength, z: (dz / d) * strength }, lift);
  } catch (e) {}
}

function shoveAll(dimension, at, reach, strength, lift) {
  for (const e of nearby(dimension, at, reach)) {
    if (e.typeId === ORB_ID) continue;
    try {
      const l = e.location;
      const d = Math.hypot(l.x - at.x, l.y - at.y, l.z - at.z);
      const falloff = Math.max(0.15, 1 - d / reach);
      shove(e, at, strength * falloff, lift * falloff);
    } catch (err) {}
  }
}

function freeze(dimension, at, radius) {
  const R = Math.min(7, Math.ceil(radius));
  const bx = Math.floor(at.x);
  const by = Math.floor(at.y);
  const bz = Math.floor(at.z);
  for (let dx = -R; dx <= R; dx++) {
    for (let dy = -R; dy <= R; dy++) {
      for (let dz = -R; dz <= R; dz++) {
        if (dx * dx + dy * dy + dz * dz > R * R) continue;
        const p = { x: bx + dx, y: by + dy, z: bz + dz };
        try {
          const b = dimension.getBlock(p);
          if (b && (b.typeId === "minecraft:water" || b.typeId === "minecraft:flowing_water")) {
            dimension.setBlockType(p, "minecraft:ice");
          }
        } catch (e) {}
      }
    }
  }
}

function detonate(s, at) {
  const d = s.dimension;
  const f = s.f;
  const r = (1.6 + 4.4 * f) * (s.over ? 1.25 : 1);
  const reach = r * 2.2;
  const strength = 0.8 + 2.6 * f;
  const lift = 0.35 + 0.8 * f;
  impactBurst(d, at, r, s.mode.color);
  sound(d, "random.explode", at);
  if (s.mode.id === "aero") {
    explode(d, at, r, true, false);
    shoveAll(d, at, reach, strength * 1.6, lift * 1.3);
  } else if (s.mode.id === "blaze") {
    explode(d, at, r * 0.9, true, true);
    for (const e of nearby(d, at, r * 1.5)) {
      try {
        if (e.typeId !== ORB_ID) e.setOnFire(6, true);
      } catch (err) {}
    }
    shoveAll(d, at, reach, strength * 0.8, lift);
  } else if (s.mode.id === "frost") {
    explode(d, at, r * 0.6, false, false);
    freeze(d, at, r * 1.2);
    for (const e of nearby(d, at, r * 1.6)) {
      try {
        if (e.typeId !== ORB_ID) e.addEffect("slowness", 160 + Math.round(160 * f), { amplifier: 3 });
      } catch (err) {}
    }
    shoveAll(d, at, reach, strength * 0.5, lift * 0.5);
  } else if (s.mode.id === "storm") {
    explode(d, at, r * 0.7, true, false);
    try {
      d.spawnEntity("minecraft:lightning_bolt", at);
    } catch (e) {}
    shoveAll(d, at, reach, strength, lift);
  } else {
    explode(d, at, r * 0.8, true, false);
    shoveAll(d, at, reach, strength * 0.6, lift * 0.6);
  }
}

function pull(s, at) {
  for (const e of nearby(s.dimension, at, 12)) {
    if (e.id === s.owner || e.id === s.orb.id || e.typeId === PLAYER_ID || e.typeId === ORB_ID) continue;
    try {
      const l = e.location;
      const dx = at.x - l.x;
      const dy = at.y - l.y;
      const dz = at.z - l.z;
      const d = Math.hypot(dx, dy, dz) || 1;
      const k = 0.1 + 0.12 * s.f;
      e.applyImpulse({ x: (dx / d) * k, y: (dy / d) * k + 0.02, z: (dz / d) * k });
    } catch (err) {}
  }
}

export function launchOrb(player, modeIndex, held) {
  const mode = MODES[modeIndex];
  const f = Math.min(1, held / FULL_TICKS);
  const dimension = player.dimension;
  const view = player.getViewDirection();
  const start = handPoint(player, view);
  const orb = dimension.spawnEntity(ORB_ID, start);
  const speed = (0.55 + 2.45 * f) * mode.speed;
  orb.applyImpulse({ x: view.x * speed, y: view.y * speed, z: view.z * speed });
  orbs.set(orb.id, {
    orb,
    dimension,
    mode,
    f,
    over: held >= OVER_TICKS,
    owner: player.id,
    age: 0,
    last: start,
    lob: f < 0.35 ? 0.035 * (1 - f / 0.35) : 0,
  });
  sound(dimension, "mob.ghast.fireball", start);
}

export function tickOrbs() {
  for (const [id, s] of orbs) {
    try {
      const loc = s.orb.location;
      s.last = loc;
      s.age++;
      orbTrail(s.dimension, loc, s.mode.color, s.age);
      if (s.lob > 0) s.orb.applyImpulse({ x: 0, y: -s.lob, z: 0 });
      if (s.mode.id === "void") pull(s, loc);
      if (s.age > MAX_ORB_AGE) {
        orbs.delete(id);
        detonate(s, loc);
        s.orb.kill();
      }
    } catch (e) {
      orbs.delete(id);
    }
  }
}

function onHit(projectile, location) {
  let id;
  try {
    id = projectile.id;
  } catch (e) {
    return;
  }
  const s = orbs.get(id);
  if (!s) return;
  orbs.delete(id);
  detonate(s, location ?? s.last);
  try {
    projectile.kill();
  } catch (e) {}
}

export function registerHits() {
  world.afterEvents.projectileHitBlock.subscribe((ev) => onHit(ev.projectile, ev.location));
  world.afterEvents.projectileHitEntity.subscribe((ev) => onHit(ev.projectile, ev.location));
}

import { MolangVariableMap } from "@minecraft/server";
import { BURST_TICK, FULL_TICKS, OVER_TICKS } from "./elements.js";

const SMALL = "aln:ss_small";
const MID = "aln:ss_mid";
const BIG = "aln:ss_big";

export function puff(dimension, id, location, color) {
  try {
    const vars = new MolangVariableMap();
    vars.setColorRGB("variable.color", color);
    dimension.spawnParticle(id, location, vars);
  } catch (e) {}
}

export function sound(dimension, id, location) {
  try {
    dimension.playSound(id, location);
  } catch (e) {}
}

function cross(a, b) {
  return { x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x };
}

export function basisOf(view) {
  const rl = Math.hypot(view.z, view.x) || 1;
  const right = { x: -view.z / rl, y: 0, z: view.x / rl };
  const up = cross(right, view);
  return { right, up };
}

export function handPoint(player, view) {
  const loc = player.location;
  const { right } = basisOf(view);
  return {
    x: loc.x + view.x * 0.9 + right.x * 0.4,
    y: loc.y + 1.2 + view.y * 0.9,
    z: loc.z + view.z * 0.9 + right.z * 0.4,
  };
}

function ring(dimension, center, view, radius, count, spin, id, color) {
  const { right, up } = basisOf(view);
  for (let i = 0; i < count; i++) {
    const a = spin + (i * Math.PI * 2) / count;
    const c = Math.cos(a) * radius;
    const s = Math.sin(a) * radius;
    puff(
      dimension,
      id,
      {
        x: center.x + right.x * c + up.x * s,
        y: center.y + right.y * c + up.y * s,
        z: center.z + right.z * c + up.z * s,
      },
      color
    );
  }
}

export function shell(dimension, center, radius, count, id, color) {
  for (let i = 0; i < count; i++) {
    const u = Math.random() * 2 - 1;
    const t = Math.random() * Math.PI * 2;
    const r = Math.sqrt(1 - u * u);
    puff(
      dimension,
      id,
      { x: center.x + Math.cos(t) * r * radius, y: center.y + u * radius, z: center.z + Math.sin(t) * r * radius },
      color
    );
  }
}

export function phaseOf(held) {
  if (held >= OVER_TICKS) return 4;
  if (held >= BURST_TICK) return 3;
  if (held >= 24) return 2;
  return 1;
}

export function renderCharge(player, held, mode) {
  const dimension = player.dimension;
  const view = player.getViewDirection();
  const center = handPoint(player, view);
  const phase = phaseOf(held);
  if (phase === 1) {
    if (held % 2 === 0) ring(dimension, center, view, 0.55, 8, held * 0.25, SMALL, mode.color);
  } else if (phase === 2) {
    const k = (held - 24) / (FULL_TICKS - 24);
    ring(dimension, center, view, 0.55 * (1 - k) + 0.04, 6, held * (0.5 + k * 1.5), SMALL, mode.color);
    puff(dimension, MID, center, { red: 1, green: 1, blue: 1 });
  } else {
    if (held === BURST_TICK) {
      shell(dimension, center, 0.9, 30, MID, mode.color);
      ring(dimension, center, view, 0.9, 16, 0, BIG, mode.color);
      sound(dimension, "mob.warden.sonic_boom", center);
    }
    puff(dimension, BIG, center, mode.color);
    shell(dimension, center, 0.55, 8, SMALL, mode.color);
    if (phase === 4 && held % 3 === 0) shell(dimension, center, 0.8, 4, MID, { red: 1, green: 1, blue: 1 });
  }
}

export function impactBurst(dimension, center, radius, color) {
  shell(dimension, center, radius * 0.6, 40, MID, color);
  shell(dimension, center, radius * 0.3, 12, BIG, color);
  for (let i = 0; i < 24; i++) {
    const a = (i * Math.PI * 2) / 24;
    puff(dimension, SMALL, { x: center.x + Math.cos(a) * radius, y: center.y + 0.2, z: center.z + Math.sin(a) * radius }, color);
  }
}

export function orbTrail(dimension, center, color, age) {
  puff(dimension, BIG, center, color);
  const a = age * 0.9;
  puff(dimension, SMALL, { x: center.x + Math.cos(a) * 0.5, y: center.y + Math.sin(a * 1.3) * 0.5, z: center.z + Math.sin(a) * 0.5 }, color);
  puff(dimension, SMALL, { x: center.x - Math.cos(a) * 0.5, y: center.y - Math.sin(a * 1.3) * 0.5, z: center.z - Math.sin(a) * 0.5 }, color);
}

# ALN:SpiritSphere — Chargeable Elemental Energy Orbs

> Craft one item, charge it in your hand, and throw an energy sphere that behaves like an enhanced Ghast fireball.

## Overview

ALN:SpiritSphere adds the Spirit Focus, a craftable item that charges like a bow. While you hold use, energy swirls in your hand through three visual phases. On release it launches the Spirit Orb, a projectile whose speed and explosion size depend on how long you charged.

Sneaking while using the item switches between five elements, each with its own color and impact behavior.

## Key Features

### Spirit Focus Item
- **What Happens**: Craft the Spirit Focus with an Ender Pearl, Blaze Powder, a Wind Charge (or a Feather) and an Amethyst Shard in any order on the crafting grid. Two recipes are included, one for each of the Wind Charge and Feather versions.

### Hold to Charge
- **What Happens**: Hold use with the Spirit Focus like a bow. The longer you hold, the faster the orb flies and the larger its blast.

### Phase 1: Loose Ring
- **What Happens**: For the first 24 ticks (about 1.2 seconds) a wide ring of energy particles swirls around your hand in the color of the active element.

### Phase 2: Compression
- **What Happens**: From tick 24 the ring tightens and spins faster until it is a dense point of light in your palm.

### Phase 3: Expansion
- **What Happens**: At 50 ticks (2.5 seconds) the core bursts into a large glowing sphere with a sonic pulse. Holding to 70 ticks overcharges the shot for a 25 percent larger blast.

### Charge-Scaled Orb
- **What Happens**: Releasing fires the Spirit Orb along your aim. A quick release lobs a slow orb with a small blast, a medium charge flies steadily, and a full charge flies almost straight and fast with a cratering explosion.

### Five Elements
- **What Happens**: Sneak and use the Spirit Focus to cycle Aero Vortex, Blaze Solar, Frost Nova, Storm Surge and Void Singularity. The item lore, particle colors and impact effect change with the element.

### Element Impacts
- **What Happens**: Aero Vortex adds heavy knockback and a fire-free blast. Blaze Solar ignites the area. Frost Nova freezes nearby water and applies Slowness IV. Storm Surge flies 40 percent faster and calls lightning. Void Singularity pulls mobs toward the orb while it flies.

## How to Use

1. Craft the Spirit Focus with an Ender Pearl, Blaze Powder, a Wind Charge or Feather, and an Amethyst Shard.
2. Hold use and watch the three charge phases build in your hand.
3. Release to fire the orb along your aim. Release early for a slow lob, hold to 50 ticks or more for a fast, large blast.
4. Sneak and use the item to switch the element. The item lore and the action bar show the active element.
