import { powerOf } from "./elements.js";

const ZOOM_FOV = [70, 62, 50, 38, 46];

export function applyZoom(entry, phase) {
  if (entry.zoom === phase) return;
  entry.zoom = phase;
  try {
    entry.player.runCommand(`camera @s fov_set ${ZOOM_FOV[phase]} 0.35 linear`);
  } catch (e) {}
}

export function clearZoom(player) {
  try {
    player.runCommand("camera @s fov_clear 0.2 linear");
  } catch (e) {}
}

export function chargeSound(player, held, mode) {
  if (held < 2 || held % 3 !== 0) return;
  const k = powerOf(held);
  try {
    player.playSound(`${mode.sfx}.charge`, { pitch: 0.6 + 1.4 * k, volume: 0.5 + 0.4 * k });
  } catch (e) {}
}

import { world, system } from "@minecraft/server";
import { MODES, MODE_BY_ITEM, MIN_HOLD, OVER_TICKS, FULL_TICKS, COOLDOWN_TICKS, powerOf, state } from "./elements.js";
import { renderCharge, phaseOf } from "./visuals.js";
import { launchOrb, tickOrbs, registerHits } from "./orbs.js";
import { applyZoom, clearZoom, chargeSound } from "./charge_fx.js";

const chargers = new Map();
const PHASES = ["", "Gathering", "Compressing", "Sweet Spot", "Fading"];

function finish(source, itemStack) {
  if (!itemStack || !MODE_BY_ITEM.has(itemStack.typeId)) return;
  const entry = chargers.get(source.id);
  if (!entry || entry.fired) return;
  entry.fired = true;
  entry.doneTick = state.tick;
  clearZoom(source);
  const held = state.tick - entry.start;
  if (held >= MIN_HOLD) {
    try {
      launchOrb(source, entry.mode, held);
      state.cooldowns.set(`${source.id}:${entry.mode}`, state.tick + COOLDOWN_TICKS);
    } catch (e) {}
    try {
      itemStack.getComponent("minecraft:cooldown")?.startCooldown(source);
    } catch (e) {}
  }
}

world.afterEvents.itemStartUse.subscribe((ev) => {
  const player = ev.source;
  const mode = MODE_BY_ITEM.get(ev.itemStack?.typeId);
  if (mode === undefined) return;
  const ready = state.cooldowns.get(`${player.id}:${mode}`) ?? 0;
  if (state.tick < ready) {
    try {
      player.onScreenDisplay.setActionBar(`§cRecharging §f${((ready - state.tick) / 20).toFixed(1)}s`);
    } catch (e) {}
    return;
  }
  chargers.set(player.id, { player, mode, start: state.tick, fired: false, doneTick: -1, zoom: -1 });
});

world.afterEvents.itemReleaseUse.subscribe((ev) => finish(ev.source, ev.itemStack));
world.afterEvents.itemStopUse.subscribe((ev) => finish(ev.source, ev.itemStack));

registerHits();

system.runInterval(() => {
  state.tick++;
  for (const [id, entry] of chargers) {
    if (entry.fired) {
      if (state.tick - entry.doneTick > 3) chargers.delete(id);
      continue;
    }
    try {
      const held = state.tick - entry.start;
      const mode = MODES[entry.mode];
      const phase = phaseOf(held);
      renderCharge(entry.player, held, mode);
      applyZoom(entry, phase);
      chargeSound(entry.player, held, mode);
      if (held % 2 === 0) {
        const power = Math.round(powerOf(held) * 100);
        const tail = held >= OVER_TICKS ? "§c" : held >= FULL_TICKS ? "§a" : "§f";
        entry.player.onScreenDisplay.setActionBar(`${mode.code}${mode.name} §7| ${tail}${PHASES[phase]} §7| §f${power}%`);
      }
    } catch (e) {
      clearZoom(entry.player);
      chargers.delete(id);
    }
  }
  tickOrbs();
}, 1);

    const d = Math.hypot(dx, dz) || 1;
    entity.applyKnockback({ x: (dx / d) * strength, z: (dz / d) * strength }, lift);
  } catch (e) {}
}

function shoveAll(dimension, at, reach, strength, lift) {
  for (const e of nearby(dimension, at, reach)) {
    if (e.typeId === ORB_ID) continue;
    try {
      const l = e.location;
      const d = Math.hypot(l.x - at.x, l.y - at.y, l.z - at.z);
      const falloff = Math.max(0.15, 1 - d / reach);
      shove(e, at, strength * falloff, lift * falloff);
    } catch (err) {}
  }
}

function freeze(dimension, at, radius) {
  const R = Math.min(7, Math.ceil(radius));
  const bx = Math.floor(at.x);
  const by = Math.floor(at.y);
  const bz = Math.floor(at.z);
  for (let dx = -R; dx <= R; dx++) {
    for (let dy = -R; dy <= R; dy++) {
      for (let dz = -R; dz <= R; dz++) {
        if (dx * dx + dy * dy + dz * dz > R * R) continue;
        const p = { x: bx + dx, y: by + dy, z: bz + dz };
        try {
          const b = dimension.getBlock(p);
          if (b && (b.typeId === "minecraft:water" || b.typeId === "minecraft:flowing_water")) {
            dimension.setBlockType(p, "minecraft:ice");
          }
        } catch (e) {}
      }
    }
  }
}

function affect(dimension, at, distance, effectId, ticks, amplifier) {
  for (const e of nearby(dimension, at, distance)) {
    try {
      if (e.typeId !== ORB_ID) e.addEffect(effectId, ticks, { amplifier });
    } catch (err) {}
  }
}

function detonate(s, at) {
  const d = s.dimension;
  const f = s.f;
  const r = (1.6 + 4.4 * f) * (s.over ? 1.25 : 1);
  const reach = r * 2.2;
  const strength = 0.8 + 2.6 * f;
  const lift = 0.35 + 0.8 * f;
  impactBurst(d, at, r, s.mode.color);
  sound(d, `${s.mode.sfx}.impact`, at);
  if (s.mode.id === "aero") {
    explode(d, at, r, true, false);
    shoveAll(d, at, reach, strength * 1.6, lift * 1.3);
  } else if (s.mode.id === "blaze") {
    explode(d, at, r * 0.9, true, true);
    for (const e of nearby(d, at, r * 1.5)) {
      try {
        if (e.typeId !== ORB_ID) e.setOnFire(6, true);
      } catch (err) {}
    }
    shoveAll(d, at, reach, strength * 0.8, lift);
  } else if (s.mode.id === "frost") {
    explode(d, at, r * 0.6, false, false);
    freeze(d, at, r * 1.2);
    for (const e of nearby(d, at, r * 1.6)) {
      try {
        if (e.typeId !== ORB_ID) e.addEffect("slowness", 160 + Math.round(160 * f), { amplifier: 3 });
      } catch (err) {}
    }
    shoveAll(d, at, reach, strength * 0.5, lift * 0.5);
  } else if (s.mode.id === "storm") {
    explode(d, at, r * 0.7, true, false);
    try {
      d.spawnEntity("minecraft:lightning_bolt", at);
    } catch (e) {}
    shoveAll(d, at, reach, strength, lift);
  } else if (s.mode.id === "void") {
    explode(d, at, r * 1.1, true, false);
    shoveAll(d, at, reach, strength * 0.6, lift * 0.6);
  } else if (s.mode.id === "terra") {
    explode(d, at, r * 1.2, true, false);
    shoveAll(d, at, reach, strength * 0.4, lift * 2.2);
    affect(d, at, r * 1.8, "slowness", 80, 1);
  } else if (s.mode.id === "toxic") {
    explode(d, at, r * 0.5, false, false);
    affect(d, at, r * 2, "poison", 200 + Math.round(200 * f), 1);
    shoveAll(d, at, reach, strength * 0.3, lift * 0.4);
  } else if (s.mode.id === "gravity") {
    explode(d, at, r * 0.4, false, false);
    affect(d, at, r * 2.2, "levitation", 60 + Math.round(80 * f), 1);
  } else if (s.mode.id === "life") {
    affect(d, at, r * 2, "regeneration", 120 + Math.round(120 * f), 2);
  } else {
    explode(d, at, r * 0.4, false, false);
    affect(d, at, r * 2, "slowness", 80 + Math.round(80 * f), 6);
  }
}

const VOID_PULL_RADIUS = 10;
const VOID_HOLD_TICKS = 100;
const anchors = new Map();

function pull(a) {
  let found;
  try {
    found = a.dimension.getEntities({ location: a.at, maxDistance: VOID_PULL_RADIUS, families: ["mob"] });
  } catch (e) {
    return;
  }
  for (const e of found) {
    if (e.id === a.owner || e.typeId === PLAYER_ID || e.typeId === ORB_ID) continue;
    try {
      const l = e.location;
      const dx = a.at.x - l.x;
      const dy = a.at.y - l.y;
      const dz = a.at.z - l.z;
      const d = Math.hypot(dx, dy, dz) || 1;
      const k = (0.16 + 0.1 * a.f) * Math.min(1, d / 2.5);
      e.applyImpulse({ x: (dx / d) * k, y: (dy / d) * k + 0.03, z: (dz / d) * k });
    } catch (err) {}
  }
}

function anchorOrb(id, s, at) {
  anchors.set(id, {
    dimension: s.dimension,
    mode: s.mode,
    f: s.f,
    over: s.over,
    owner: s.owner,
    at: { x: at.x, y: at.y + 0.5, z: at.z },
    age: 0,
  });
  sound(s.dimension, `${s.mode.sfx}.burst`, at);
}

function tickAnchors() {
  for (const [id, a] of anchors) {
    a.age++;
    try {
      voidSwirl(a.dimension, a.at, VOID_PULL_RADIUS, a.mode.color, a.age);
      pull(a);
      if (a.age >= VOID_HOLD_TICKS) {
        anchors.delete(id);
        detonate(a, a.at);
      }
    } catch (e) {
      anchors.delete(id);
    }
  }
}

export function launchOrb(player, modeIndex, held) {
  const mode = MODES[modeIndex];
  const f = powerOf(held);
  const dimension = player.dimension;
  const view = player.getViewDirection();
  const start = handPoint(player, view);
  const orb = dimension.spawnEntity(ORB_ID, start);
  const speed = (0.55 + 2.45 * f) * mode.speed;
  orb.applyImpulse({ x: view.x * speed, y: view.y * speed, z: view.z * speed });
  orbs.set(orb.id, {
    orb,
    dimension,
    mode,
    f,
    over: held >= FULL_TICKS && held <= OVER_TICKS,
    owner: player.id,
    age: 0,
    last: start,
    lob: f < 0.35 ? 0.035 * (1 - f / 0.35) : 0,
  });
  sound(dimension, `${mode.sfx}.launch`, start);
}

export function tickOrbs() {
  tickAnchors();
  for (const [id, s] of orbs) {
    try {
      const loc = s.orb.location;
      s.last = loc;
      s.age++;
      orbTrail(s.dimension, loc, s.mode.color, s.age);
      if (s.lob > 0) s.orb.applyImpulse({ x: 0, y: -s.lob, z: 0 });
      if (s.age > MAX_ORB_AGE) {
        orbs.delete(id);
        if (s.mode.id === "void") anchorOrb(id, s, loc);
        else detonate(s, loc);
        s.orb.kill();
      }
    } catch (e) {
      orbs.delete(id);
    }
  }
}

function onHit(projectile, location) {
  let id;
  try {
    id = projectile.id;
  } catch (e) {
    return;
  }
  const s = orbs.get(id);
  if (!s) return;
  orbs.delete(id);
  const at = location ?? s.last;
  if (s.mode.id === "void") anchorOrb(id, s, at);
  else detonate(s, at);
  try {
    projectile.kill();
  } catch (e) {}
}

export function registerHits() {
  world.afterEvents.projectileHitBlock.subscribe((ev) => onHit(ev.projectile, ev.location));
  world.afterEvents.projectileHitEntity.subscribe((ev) => onHit(ev.projectile, ev.location));
}

function cross(a, b) {
  return { x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x };
}

export function basisOf(view) {
  const rl = Math.hypot(view.z, view.x) || 1;
  const right = { x: -view.z / rl, y: 0, z: view.x / rl };
  const up = cross(right, view);
  return { right, up };
}

export function handPoint(player, view) {
  const loc = player.location;
  const { right } = basisOf(view);
  return {
    x: loc.x + view.x * 0.9 + right.x * 0.4,
    y: loc.y + 1.2 + view.y * 0.9,
    z: loc.z + view.z * 0.9 + right.z * 0.4,
  };
}

function ring(dimension, center, view, radius, count, spin, id, color) {

  for (let i = 0; i < count; i++) {

    const c = Math.cos(a) * radius;
    const s = Math.sin(a) * radius;
    puff(
      dimension,
      id,
      {
        x: center.x + right.x * c + up.x * s,
        y: center.y + right.y * c + up.y * s,
        z: center.z + right.z * c + up.z * s,
      },
      color
    );
  }
}

export function shell(dimension, center, radius, count, id, color) {
  for (let i = 0; i < count; i++) {
    const u = Math.random() * 2 - 1;
    const t = Math.random() * Math.PI * 2;
    const r = Math.sqrt(1 - u * u);
    puff(
      dimension,
      id,
      { x: center.x + Math.cos(t) * r * radius, y: center.y + u * radius, z: center.z + Math.sin(t) * r * radius },
      color
    );
  }

export function phaseOf(held) {
  if (held >= OVER_TICKS) return 4;
  if (held >= BURST_TICK) return 3;
  if (held >= 24) return 2;
  return 1;
}

export function renderCharge(player, held, mode) {
  const dimension = player.dimension;
  const view = player.getViewDirection();
  const center = handPoint(player, view);
  const phase = phaseOf(held);
  if (phase === 1) {
    if (held % 2 === 0) ring(dimension, center, view, 0.55, 8, held * 0.25, SMALL, mode.color);
    if (held % 6 === 0) puff(dimension, PULL_HAND, center, mode.color);
  } else if (phase === 2) {
    const k = (held - 24) / (FULL_TICKS - 24);
    ring(dimension, center, view, 0.55 * (1 - k) + 0.04, 6, held * (0.5 + k * 1.5), SMALL, mode.color);
    puff(dimension, MID, center, { red: 1, green: 1, blue: 1 });
    if (held % 2 === 0) puff(dimension, PULL_HAND, center, mode.color);
  } else {
    if (held === BURST_TICK) {
      shell(dimension, center, 0.9, 30, MID, mode.color);
      ring(dimension, center, view, 0.9, 16, 0, BIG, mode.color);
      sound(dimension, `${mode.sfx}.burst`, center);
    }

    shell(dimension, center, 0.55 * Math.max(powerOf(held), 0.5), 8, SMALL, mode.color);

  }
}
