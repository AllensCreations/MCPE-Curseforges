```
import { system } from "@minecraft/server";
import { ModalFormData, ActionFormData, FormCancelationReason } from "@minecraft/server-ui";
import { MAX_NAME, DIM_STYLE, FALLBACK_STYLE } from "./config.js";

function wait(ticks) {
  return new Promise((resolve) => system.runTimeout(resolve, ticks));
}

function cleanName(value) {
  return value
    .replace(/§./g, "")
    .replace(/[\u0000-\u001f]/g, "")
    .trim()
    .slice(0, MAX_NAME)
    .trim();
}

async function showWhenFree(player, makeForm) {
  for (let attempt = 0; attempt < 12; attempt++) {
    const response = await makeForm().show(player);
    if (!response.canceled) return response;
    if (response.cancelationReason !== FormCancelationReason.UserBusy) return undefined;
    await wait(10);
  }
  return undefined;
}

export async function askGateName(player, fallback) {
  try {
    const response = await showWhenFree(player, () =>
      new ModalFormData().title("Name Your Beacon Gate").textField("Waypoint name", "Home Base")
    );
    const value = response && response.formValues ? response.formValues[0] : "";
    const name = typeof value === "string" ? cleanName(value) : "";
    return name || fallback;
  } catch {
    return fallback;
  }
}

export async function chooseGate(player, gates) {
  try {
    const response = await showWhenFree(player, () => {
      const form = new ActionFormData().title("Warp Scroll").body("Choose a Beacon Gate to warp to.");
      for (const gate of gates) {
        const style = DIM_STYLE[gate.d] || FALLBACK_STYLE;
        form.button(style.color + gate.n + "\n§r§8" + style.label);
      }
      return form;
    });
    return response ? response.selection : undefined;
  } catch {
    return undefined;
  }
}
```

```
import { world } from "@minecraft/server";
import { STORE_KEY, MAX_GATES } from "./config.js";

export function shortDim(dimension) {
  return String(dimension.id).replace("minecraft:", "");
}

function isGate(entry) {
  return (
    entry !== null &&
    typeof entry === "object" &&
    typeof entry.n === "string" &&
    typeof entry.d === "string" &&
    Number.isFinite(entry.x) &&
    Number.isFinite(entry.y) &&
    Number.isFinite(entry.z)
  );
}

export function getGates() {
  const raw = world.getDynamicProperty(STORE_KEY);
  if (typeof raw !== "string") return [];
  try {
    const data = JSON.parse(raw);
    return Array.isArray(data) ? data.filter(isGate) : [];
  } catch {
    return [];
  }
}

function saveGates(list) {
  try {
    world.setDynamicProperty(STORE_KEY, JSON.stringify(list));
    return true;
  } catch {
    return false;
  }
}

function same(gate, dim, x, y, z) {
  return gate.d === dim && gate.x === x && gate.y === y && gate.z === z;
}

export function findGate(dim, x, y, z) {
  return getGates().find((gate) => same(gate, dim, x, y, z));
}

export function isFull() {
  return getGates().length >= MAX_GATES;
}

export function addGate(name, dim, x, y, z) {
  const list = getGates().filter((gate) => !same(gate, dim, x, y, z));
  if (list.length >= MAX_GATES) return false;
  list.push({ n: name, d: dim, x, y, z });
  return saveGates(list);
}

export function removeGateAt(dim, x, y, z) {
  const list = getGates();
  const gate = list.find((entry) => same(entry, dim, x, y, z));
  if (!gate) return undefined;
  saveGates(list.filter((entry) => entry !== gate));
  return gate;
}
```

```
import { world, system, ItemComponentTypes, EntityInventoryComponent } from "@minecraft/server";
import {
  GATE_ID,
  SCROLL_ID,
  WARP_TICKS,
  PARTICLE_STEP,
  SOUND,
  NAUSEA_AMPLIFIER,
  EYES_MARKER,
  EYES_HOLD_TICKS,
  SPHERE_RADIUS,
  SPHERE_RINGS,
  SPHERE_LEAD,
  BEAM_HEIGHT,
  BEAM_RISE_TICKS,
  BEAM_LINGER_STEPS,
  MOVE_LIMIT,
} from "./config.js";
import { removeGateAt } from "./registry.js";

const active = new Set();

export function isWarping(player) {
  return active.has(player.id);
}

export function gateStillStands(gate) {
  try {
    const block = world.getDimension(gate.d).getBlock({ x: gate.x, y: gate.y, z: gate.z });
    if (block && block.typeId !== GATE_ID) {
      removeGateAt(gate.d, gate.x, gate.y, gate.z);
      return false;
    }
  } catch {
    return true;
  }
  return true;
}

function tell(player, text) {
  try {
    player.sendMessage(text);
  } catch {
    return;
  }
}

function findScroll(player) {
  const inventory = player.getComponent(EntityInventoryComponent.componentId);
  if (!inventory) return { slot: -1 };
  const container = inventory.container;
  const selected = player.selectedSlotIndex;
  const order = [selected];
  for (let i = 0; i < 36; i++) if (i !== selected) order.push(i);
  for (const i of order) {
    try {
      const item = container.getItem(i);
      if (item && item.typeId === SCROLL_ID) return { slot: i, container };
    } catch {
      if (i !== selected) break;
    }
  }
  return { slot: -1 };
}

function landingSpot(dimension, gate) {
  const fallback = { x: gate.x + 0.5, y: gate.y + 1, z: gate.z + 0.5 };
  for (let dy = 1; dy <= 4; dy++) {
    try {
      const feet = dimension.getBlock({ x: gate.x, y: gate.y + dy, z: gate.z });
      const head = dimension.getBlock({ x: gate.x, y: gate.y + dy + 1, z: gate.z });
      if ((!feet || feet.isAir) && (!head || head.isAir)) {
        return { x: gate.x + 0.5, y: gate.y + dy, z: gate.z + 0.5 };
      }
    } catch {
      return fallback;
    }
  }
  return fallback;
}

function spawn(dimension, id, location) {
  try {
    dimension.spawnParticle(id, location);
  } catch {
    return;
  }
}

function sphereFrame(dimension, center, progress, tick) {
  const radius = SPHERE_RADIUS;
  const reach = Math.min(progress, 1) * 2 * radius;
  for (let ring = 0; ring <= SPHERE_RINGS; ring++) {
    const rise = (ring / SPHERE_RINGS) * 2 * radius;
    if (rise > reach) break;
    const lat = rise - radius;
    const ringRadius = Math.sqrt(Math.max(radius * radius - lat * lat, 0));
    const points = Math.max(1, Math.round(ringRadius * 5));
    const spin = tick * 0.35 + ring * 0.7;
    for (let p = 0; p < points; p++) {
      const angle = spin + (p / points) * Math.PI * 2;
      spawn(dimension, "minecraft:basic_portal_particle", {
        x: center.x + Math.cos(angle) * ringRadius,
        y: center.y + rise,
        z: center.z + Math.sin(angle) * ringRadius,
      });
    }
  }
  const frontLat = reach - radius;
  const frontRadius = Math.sqrt(Math.max(radius * radius - frontLat * frontLat, 0));
  for (let p = 0; p < 18; p++) {
    const angle = tick * 0.5 + (p / 18) * Math.PI * 2;
    spawn(dimension, "minecraft:endrod", {
      x: center.x + Math.cos(angle) * frontRadius,
      y: center.y + reach,
      z: center.z + Math.sin(angle) * frontRadius,
    });
  }
}

function chargeParticles(player, tick) {
  const base = player.location;
  sphereFrame(player.dimension, { x: base.x, y: base.y - 0.5, z: base.z }, tick / (WARP_TICKS - SPHERE_LEAD), tick);
  beamFrame(player.dimension, base, tick, tick / BEAM_RISE_TICKS);
}

function risingSphere(dimension, spot) {
  let step = 0;
  const handle = system.runInterval(() => {
    step += 1;
    sphereFrame(dimension, { x: spot.x, y: spot.y - 0.5, z: spot.z }, step / 6, step * PARTICLE_STEP);
    if (step >= 6) system.clearRun(handle);
  }, PARTICLE_STEP);
}

function beamFrame(dimension, base, tick, grown) {
  const height = Math.min(BEAM_HEIGHT, Math.round(Math.min(grown, 1) * BEAM_HEIGHT));
  for (let y = 0; y < height; y++) {
    spawn(dimension, "minecraft:endrod", { x: base.x, y: base.y + y, z: base.z });
    if (y % 2 === 0) {
      const angle = tick * 0.6 + y * 0.5;
      spawn(dimension, "minecraft:basic_portal_particle", {
        x: base.x + Math.cos(angle) * 0.45,
        y: base.y + y,
        z: base.z + Math.sin(angle) * 0.45,
      });
    }
  }
}

function lingeringBeam(dimension, spot) {
  let step = 0;
  const handle = system.runInterval(() => {
    step += 1;
    beamFrame(dimension, spot, step * PARTICLE_STEP, step / 5);
    if (step >= BEAM_LINGER_STEPS) system.clearRun(handle);
  }, PARTICLE_STEP);
}

function moved(player, origin) {
  const at = player.location;
  const dx = at.x - origin.x;
  const dy = at.y - origin.y;
  const dz = at.z - origin.z;
  return Math.sqrt(dx * dx + dy * dy + dz * dz) > MOVE_LIMIT;
}

function cancelWarp(player) {
  openEyes(player);
  try {
    player.removeEffect("nausea");
  } catch {
    return;
  }
  try {
    player.runCommand("stopsound @s " + SOUND.charge);
  } catch {
    return;
  }
}

function closeEyes(player) {
  try {
    player.onScreenDisplay.setTitle(EYES_MARKER, {
      fadeInDuration: 0,
      stayDuration: WARP_TICKS + EYES_HOLD_TICKS,
      fadeOutDuration: 0,
    });
  } catch {
    return;
  }
}

function openEyes(player) {
  try {
    player.onScreenDisplay.setTitle("");
  } catch {
    return;
  }
}

export function arrivalBurst(dimension, at) {
  for (let i = 0; i < 30; i++) {
    const angle = (i / 30) * Math.PI * 2;
    const radius = 0.4 + (i % 4) * 0.25;
    const point = {
      x: at.x + Math.cos(angle) * radius,
      y: at.y + 0.2 + (i % 7) * 0.28,
      z: at.z + Math.sin(angle) * radius,
    };
    spawn(dimension, "minecraft:basic_portal_particle", point);
    if (i % 3 === 0) spawn(dimension, "minecraft:endrod", point);
    if (i % 5 === 0) spawn(dimension, "minecraft:totem_particle", point);
  }
}

function breakEffects(player) {
  try {
    const at = player.location;
    player.playSound(SOUND.tear, { volume: 1, pitch: 1 });
    for (let i = 0; i < 12; i++) {
      spawn(player.dimension, "minecraft:totem_particle", {
        x: at.x + (Math.random() - 0.5) * 0.8,
        y: at.y + 0.8 + Math.random() * 0.8,
        z: at.z + (Math.random() - 0.5) * 0.8,
      });
    }
  } catch {
    return;
  }
}

function complete(player, gate) {
  if (!player.isValid) return;
  const found = findScroll(player);
  if (found.slot === -1) {
    tell(player, "§cThe Warp Scroll fizzled out because it left your inventory.");
    return;
  }
  let dimension;
  let spot;
  try {
    dimension = world.getDimension(gate.d);
    spot = landingSpot(dimension, gate);
    player.teleport(spot, { dimension });
  } catch {
    tell(player, "§cThe Warp Scroll could not reach that Beacon Gate. It was not used up.");
    return;
  }
  try {
    found.container.setItem(found.slot, undefined);
  } catch {
    tell(player, "§cThe Warp Scroll could not be consumed.");
  }
  player.playSound(SOUND.arrive, { volume: 1, pitch: 1 });
  arrivalBurst(dimension, spot);
  risingSphere(dimension, spot);
  lingeringBeam(dimension, spot);
  system.runTimeout(() => {
    if (player.isValid) breakEffects(player);
  }, 6);
  tell(player, "§bWarped to §f" + gate.n + "§b.");
}

export function beginWarp(player, item, gate) {
  if (active.has(player.id)) return;
  active.add(player.id);
  try {
    const cooldown = item.getComponent(ItemComponentTypes.Cooldown);
    if (cooldown) cooldown.startCooldown(player);
  } catch {
    tell(player, "§7The cooldown overlay could not start.");
  }
  try {
    player.addEffect("nausea", WARP_TICKS, { amplifier: NAUSEA_AMPLIFIER, showParticles: false });
  } catch {
    tell(player, "§7The nausea effect could not start.");
  }
  player.playSound(SOUND.charge, { volume: 1, pitch: 1 });
  closeEyes(player);
  const origin = { x: player.location.x, y: player.location.y, z: player.location.z };
  let tick = 0;
  const handle = system.runInterval(() => {
    if (!player.isValid) {
      system.clearRun(handle);
      active.delete(player.id);
      return;
    }
    if (moved(player, origin)) {
      system.clearRun(handle);
      cancelWarp(player);
      tell(player, "§cYou moved. The warp was interrupted.");
      active.delete(player.id);
      return;
    }
    tick += PARTICLE_STEP;
    if (tick < WARP_TICKS) {
      chargeParticles(player, tick);
      return;
    }
    system.clearRun(handle);
    openEyes(player);
    try {
      complete(player, gate);
    } finally {
      active.delete(player.id);
    }
  }, PARTICLE_STEP);
}
```

```
import { world, system } from "@minecraft/server";
import { GATE_ID, SCROLL_ID, SOUND } from "./lib/config.js";
import { getGates, addGate, removeGateAt, shortDim, isFull } from "./lib/registry.js";
import { askGateName, chooseGate } from "./lib/menus.js";
import { beginWarp, isWarping, gateStillStands, arrivalBurst } from "./lib/warp.js";

const pendingMenus = new Set();
const pendingNames = new Set();

async function registerGate(player, dimension, block) {
  const dim = shortDim(dimension);
  const x = block.x;
  const y = block.y;
  const z = block.z;
  const key = dim + ":" + x + ":" + y + ":" + z;
  if (pendingNames.has(key)) return;
  pendingNames.add(key);
  try {
    if (isFull()) {
      player.sendMessage("§cThis world already has the maximum number of Beacon Gates. This one was not registered.");
      return;
    }
    const fallback = "Beacon Gate " + (getGates().length + 1);
    const name = await askGateName(player, fallback);
    if (!addGate(name, dim, x, y, z)) {
      player.sendMessage("§cThe Beacon Gate could not be saved.");
      return;
    }
    player.sendMessage("§aBeacon Gate §b" + name + "§a registered.");
    player.playSound(SOUND.register, { volume: 1, pitch: 1 });
    arrivalBurst(dimension, { x: x + 0.5, y: y + 1, z: z + 0.5 });
  } catch {
    return;
  } finally {
    pendingNames.delete(key);
  }
}

async function openScroll(player, item) {
  if (isWarping(player) || pendingMenus.has(player.id)) return;
  pendingMenus.add(player.id);
  try {
    const gates = getGates();
    if (gates.length === 0) {
      player.sendMessage("§eNo Beacon Gates are registered yet. Place a Beacon Gate and name it first.");
      return;
    }
    const picked = await chooseGate(player, gates);
    if (picked === undefined) return;
    const gate = gates[picked];
    if (!gate) return;
    if (!gateStillStands(gate)) {
      player.sendMessage("§cThat Beacon Gate no longer exists and was removed from the list.");
      return;
    }
    beginWarp(player, item, gate);
  } catch {
    return;
  } finally {
    pendingMenus.delete(player.id);
  }
}

world.afterEvents.playerPlaceBlock.subscribe((event) => {
  if (event.block.typeId !== GATE_ID) return;
  const { player, block, dimension } = event;
  system.runTimeout(() => {
    registerGate(player, dimension, block);
  }, 5);
});

world.afterEvents.playerBreakBlock.subscribe((event) => {
  const { player, block, dimension } = event;
  const gate = removeGateAt(shortDim(dimension), block.x, block.y, block.z);
  if (gate) player.sendMessage("§eBeacon Gate §f" + gate.n + "§e removed from the destination list.");
});

world.afterEvents.itemUse.subscribe((event) => {
  const item = event.itemStack;
  if (!item || item.typeId !== SCROLL_ID) return;
  openScroll(event.source, item);
});
```

```
# ALN:WarpScrolls — Teleport Scrolls and Named Waypoint Gates

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)
[![Format](https://img.shields.io/badge/Format-.MCPACK%20%7C%20.MCADDON%20%7C%20.MCWORLD-blue?style=flat-square)](https://github.com/AllensCreations/MCPE-Curseforges)

**Craft a Warp Scroll, pick a named Beacon Gate, and teleport after a 3-second Nausea charge that uses the scroll up.**

> This add-on adds the Beacon Gate block and the single-use Warp Scroll, both with 32x32 textures. Players name their own waypoints and jump between them across the Overworld, Nether and End.

## Overview

Place a Beacon Gate wherever you want a destination and give it a name. Later, right-click a Warp Scroll, choose a gate from the colored list and wait out the 3-second charge. You arrive on top of the gate and the scroll is gone, so every trip costs one scroll.

## Key Features

### Beacon Gate
- **What Happens**: Place a Beacon Gate and a form asks for a waypoint name such as Home Base. The gate glows at light level 10 and is saved with its name and dimension. Its custom 3D model has a gold plinth, four obsidian pillars with amethyst tips, gold rails and a floating crystal.
- **How It Works**: Craft it with 5 Obsidian, 1 Amethyst Shard, 2 Gold Ingots and 1 Glass in a crafting table.
- **Good To Know**: Breaking a gate removes it from the destination list and drops the block. The world keeps up to 100 gates.

### Warp Scroll
- **What Happens**: Right-click a Warp Scroll to open a menu of every Beacon Gate. Names are green for the Overworld, red for the Nether and purple for The End, with no coordinates shown.
- **How It Works**: Craft it with 4 Gold Nuggets, 3 Paper, 1 Ender Pearl and 1 Lapis Lazuli in a crafting table.
- **Good To Know**: A Warp Scroll stacks to 1 and is used up when the teleport finishes.

### 3-Second Warp Sequence
- **What Happens**: After you pick a gate, the scroll shows a 3-second cooldown sweep, you get 3 seconds of stronger Nausea, your eyelids slowly close over the screen, a hollow sphere of portal particles rises from your feet to above your head, and a pillar of light shoots up from you while a rising charge sound plays.
- **How It Works**: When the 3 seconds end, your eyes snap open, you land on the block directly above the Beacon Gate, a second sphere rises around you, a light beam shines over the gate for a couple of seconds, an arrival chime and particle burst play, and the scroll tears apart.
- **Good To Know**: Moving during the charge cancels the warp, and so does the scroll leaving your inventory. A cancelled warp keeps the scroll. The eyelid screen comes from the Resource Pack, so keep it enabled with the Behavior Pack.

### Custom Sound Effects
- **What Happens**: Four new sounds ship with the pack: the warp charge, the crystal arrival, the paper and glass scroll break, and the gate registration chime.
- **Good To Know**: Minecraft needs a full restart after installing a pack with new sound files.

## How to Use

1. Craft a Beacon Gate and place it, then type a waypoint name in the form.
2. Craft a Warp Scroll and right-click it to open the destination menu.
3. Pick a gate and stand still for 3 seconds until you land above it and the scroll breaks.

## Common Questions

- **Does a Warp Scroll stack?**: No. Each scroll stacks to 1 and is used up on arrival.
- **What happens when I break a Beacon Gate?**: It leaves the destination list and drops as a block you can place again.

- Minecraft Bedrock Edition: 1.26.50 or newer.
- Experimental features: None declared in the manifest, stable script modules only.
```
