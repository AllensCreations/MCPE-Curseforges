function countMatching(container, match) {
  let total = 0;
  for (let i = 0; i < container.size; i++) {
    const item = container.getItem(i);
    if (item && match(item)) total += item.amount;
  }
  return total;
}

function removeMatching(container, match, amount) {
  let left = amount;
  for (let i = 0; i < container.size && left > 0; i++) {
    const item = container.getItem(i);
    if (!item || !match(item)) continue;
    const taken = Math.min(item.amount, left);
    left -= taken;
    if (item.amount === taken) {
      container.setItem(i, undefined);
    } else {
      item.amount -= taken;
      container.setItem(i, item);
    }
  }
  return left === 0;
}

export function checkRecipe(container, recipe) {
  const needs = recipe.needs.map((need) => ({ need, have: countMatching(container, need.match) }));
  return { needs, ready: needs.every((entry) => entry.have >= entry.need.amount) };
}

function burst(dimension, center, particles, count) {
  for (let i = 0; i < count; i++) {

export function craft(player, recipe, dimension, center) {

  if (!status.ready) {

    player.playSound("note.bass", { volume: 0.8, pitch: 0.7 });

async function showForm(player, form) {

export async function openWorkbench(player, block) {

  active.add(player.id);
  const dimension = block.dimension;

  try {

      const container = player.getComponent("minecraft:inventory").container;

        await wait(6);
      } else {
        view = -1;
      }

    console.warn(`[Fletching Table Arsenal] ${error}`);

    active.delete(player.id);
  }
}

function offsetByFace(location, face, scale) {

  return { x: location.x + normal.x * scale, y: location.y + normal.y * scale, z: location.z + normal.z * scale };
}

const effects = {
  explosive({ dimension, location, source, projectile }) {
    const options = { breaksBlocks: EXPLOSION_BREAKS_BLOCKS, causesFire: false };
    if (source) options.source = source;
    dimension.createExplosion(location, EXPLOSION_RADIUS, options);
    projectile.remove();
  },

  teleport({ dimension, location, source, projectile, hitEntity, face }) {
    if (!source || source.typeId !== "minecraft:player") return;
    const origin = source.location;
    const originDimension = source.dimension;
    let destination = hitEntity ? hitEntity.location : offsetByFace(location, face, face === "Down" ? 1.9 : 0.6);
    source.teleport(destination, { dimension });
    particlesAt(originDimension, origin, ["minecraft:endrod", "minecraft:villager_happy"], 14);
    particlesAt(dimension, destination, ["minecraft:endrod", "minecraft:villager_happy"], 14);
    soundAt(originDimension, origin, "mob.endermen.portal", 0.8, 1);
    soundAt(dimension, destination, "mob.endermen.portal", 0.8, 1.2);
    projectile.remove();
  },

  frost({ dimension, location, hitEntity }) {
    if (hitEntity) {
      hitEntity.addEffect("slowness", 160, { amplifier: 3 });
      hitEntity.addEffect("weakness", 160, { amplifier: 0 });
      hitEntity.extinguishFire();
    }
    particlesAt(dimension, location, ["minecraft:ice_evaporation_emitter", "minecraft:critical_hit_emitter"], 12);
    soundAt(dimension, location, "random.glass", 0.8, 1.4);
  },

  incendiary({ dimension, location, hitEntity, block, face }) {
    if (hitEntity) {
      hitEntity.setOnFire(8, true);
    } else if (block) {
      const normal = FACE_NORMALS[face] ?? FACE_NORMALS.Up;
      const target = block.offset(normal);
      if (target && target.typeId === "minecraft:air") target.setType("minecraft:fire");
    }
    particlesAt(dimension, location, ["minecraft:basic_flame_particle", "minecraft:lava_particle"], 12);
    soundAt(dimension, location, "fire.ignite", 1, 1);
  }
};

function handleHit(event, hitEntity, blockHit) {
  const projectile = event.projectile;
  const recipe = RECIPES.find((entry) => projectile.hasTag(entry.tag));
  if (!recipe) return;
  projectile.removeTag(recipe.tag);
  tracked.delete(projectile);
  effects[recipe.key]({
    dimension: event.dimension,
    location: event.location,
    source: event.source,
    projectile,
    hitEntity,
    block: blockHit?.block,
    face: blockHit?.face
  });
}

export function registerImpactSystem() {
  world.afterEvents.itemStartUse.subscribe((event) => {
    const player = event.source;
    if (player.typeId !== "minecraft:player" || event.itemStack?.typeId !== BOW_ID) return;
    const recipe = detectLoadedArrow(player);
    if (recipe) pending.set(player.id, { recipe, tick: system.currentTick });
    else pending.delete(player.id);
  });

  world.afterEvents.entitySpawn.subscribe((event) => {
    const arrow = event.entity;
    if (arrow.typeId !== ARROW_ID || pending.size === 0) return;
    let best;
    let bestDistance = 5;
    for (const player of arrow.dimension.getPlayers({ location: arrow.location, maxDistance: 5 })) {
      const entry = pending.get(player.id);
      if (!entry || system.currentTick - entry.tick > PENDING_TTL) continue;
      const dx = player.location.x - arrow.location.x;
      const dy = player.location.y - arrow.location.y;
      const dz = player.location.z - arrow.location.z;
      const distance = Math.sqrt(dx * dx + dy * dy + dz * dz);
      if (distance < bestDistance) {
        bestDistance = distance;
        best = { id: player.id, recipe: entry.recipe };
      }
    }
    if (!best) return;
    pending.delete(best.id);
    arrow.addTag(best.recipe.tag);
    if (tracked.size < MAX_TRACKED) tracked.add(arrow);
  });

  world.afterEvents.projectileHitEntity.subscribe((event) => {
    try {
      handleHit(event, event.getEntityHit()?.entity, undefined);
    } catch (error) {
      console.warn(`[Fletching Table Arsenal] ${error}`);
    }
  });

  world.afterEvents.projectileHitBlock.subscribe((event) => {
    try {
      handleHit(event, undefined, event.getBlockHit());
    } catch (error) {
      console.warn(`[Fletching Table Arsenal] ${error}`);
    }
  });

  system.runInterval(() => {
    for (const arrow of tracked) {
      try {
        const recipe = RECIPES.find((entry) => arrow.hasTag(entry.tag));
        if (!recipe) {
          tracked.delete(arrow);
          continue;
        }
        arrow.dimension.spawnParticle(recipe.particles[0], arrow.location);
      } catch {
        tracked.delete(arrow);
      }
    }
  }, 2);

  world.afterEvents.playerLeave.subscribe((event) => pending.delete(event.playerId));
}

# Fletching Table Arsenal — Tactical Arrow Workstation

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

Right-click a Fletching Table to open the Fletcher's Workbench and convert ordinary Arrows into four specialized types of ammunition, using only vanilla items.

## Overview

The Fletching Table becomes a working archery station. A menu shows your current supplies and whether each conversion is ready. Every conversion turns 16 plain Arrows into 16 specialized Arrows, renamed and color-coded so they are easy to spot in your inventory.

## Key Features

### Explosive Arrows
- **What Happens**: Costs 16 Arrows and 2 Gunpowder. On impact the arrow detonates with a small blast that hurts nearby targets but does not break blocks.

### Teleportation Arrows
- **What Happens**: Costs 16 Arrows and 1 Ender Pearl. On impact the archer is teleported to where the arrow landed.

### Frost Arrows
- **What Happens**: Costs 16 Arrows and 4 Snowballs. Targets that are hit are slowed and weakened for a short time, and any fire on them is put out.

### Incendiary Arrows
- **What Happens**: Costs 16 Arrows and 1 Fire Charge. Targets that are hit are set on fire, and arrows that land on a block can ignite the open space beside it.

## How to Use

- Right-click a Fletching Table to open the Fletcher's Workbench. Sneak while right-clicking to place blocks against the table as usual.
- Pick a conversion. If you have the materials, they are removed from your inventory and the new Arrows are added. If not, a warning appears above your hotbar.
- Only plain Arrows with no custom name count toward the cost.
- Enable the Resource Pack to get the stone-style workbench screen with slot-framed icons. Without it, the menu uses the standard Minecraft form look.
- Fire the specialized Arrows from a Bow. Crossbows are not supported.
- Specialized Arrows are chosen from the first matching stack in your inventory when you start drawing the Bow.
