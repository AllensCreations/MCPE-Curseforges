const IGNORED_TYPES = new Set([
  "minecraft:player",
  "minecraft:item",
  "minecraft:xp_orb",
  "minecraft:arrow",
  "minecraft:snowball",
  "minecraft:egg",

  "minecraft:shulker_bullet",
  "minecraft:wither_skull",
  "minecraft:wither_skull_dangerous",
  "minecraft:dragon_fireball",
  "minecraft:small_fireball",
  "minecraft:large_fireball",
  "minecraft:boat",
  "minecraft:chest_boat",
  "minecraft:minecart",
  "minecraft:chest_minecart",
  "minecraft:hopper_minecart",
  "minecraft:tnt_minecart",
  "minecraft:command_block_minecart"
]);

export function registerSpawnFilter() {
  world.afterEvents.entitySpawn.subscribe((event) => {
    const entity = event.entity;
    if (!entity || !entity.isValid) {
      return;
    }

    const typeId = entity.typeId;
    if (IGNORED_TYPES.has(typeId)) {
      return;
    }

    const rules = getSpawnRules(world);
    if (rules[typeId] !== false) {
      return;
    }

    const config = getSpawnConfig(world);
    if (!config.blockSpawnEggs) {
      const cause = event.cause;
      const isNaturalOrWorldGen =
        cause === EntityInitializationCause.Spawned ||
        cause === EntityInitializationCause.Born;

      if (!isNaturalOrWorldGen) {
        return;
      }
    }

    try {
      entity.remove();
      blockedCount++;
    } catch (e) {
    }
  });
}

import { world, system } from "@minecraft/server";
import { ActionFormData, ModalFormData, MessageFormData } from "@minecraft/server-ui";
import {
  MOB_CATEGORIES,
  getSpawnRules,
  saveSpawnRules,
  getSpawnConfig,
  saveSpawnConfig,
  getAllMobIds
} from "./spawn_registry.js";
import { getBlockedCount } from "./spawn_filter.js";

const ADMIN_TAG = "aln_admin";

function hasAdminPermission(player) {
  const config = getSpawnConfig(world);
  if (!config.adminOnly) {
    return true;
  }
  const allPlayers = world.getAllPlayers();
  const hasAnyAdmin = allPlayers.some((p) => p.hasTag(ADMIN_TAG));
  if (!hasAnyAdmin) {
    player.addTag(ADMIN_TAG);
    player.onScreenDisplay.setActionBar("§a[SpawnControl] §fYou were granted Admin permissions.");
    return true;
  }
  return player.hasTag(ADMIN_TAG);
}

export function registerSpawnUI() {
  world.afterEvents.itemUse.subscribe((event) => {
    const item = event.itemStack;
    const player = event.source;

    if (!item || !player) {
      return;
    }

    const typeId = item.typeId;
    if (typeId !== "minecraft:book") {
      return;
    }

    const nameTag = item.nameTag;
    if (!nameTag || nameTag.trim().toLowerCase() !== "spawn control") {
      return;
    }

    if (!hasAdminPermission(player)) {
      player.onScreenDisplay.setActionBar("§c[SpawnControl] §fOnly admins can open Spawn Control!");
      return;
    }

    system.run(() => {
      openMainMenu(player);
    });
  });
}

function openMainMenu(player) {
  const rules = getSpawnRules(world);
  const config = getSpawnConfig(world);
  const allIds = getAllMobIds();

  let allowedCount = 0;
  let blockedCount = 0;

  for (const id of allIds) {
    if (rules[id] === false) {
      blockedCount++;
    } else {
      allowedCount++;
    }

function isTamedEntity(entity) {
  try {
    if (entity.getComponent("minecraft:is_tamed")) {
      return true;
    }
  } catch (e) {
  }
  try {
    const tameable = entity.getComponent("minecraft:tameable");
    if (tameable && tameable.isTamed === true) {
      return true;
    }
  } catch (e) {
  }
  return false;
}

const PURGE_PROTECTIONS = [
  { key: "tamed", label: "Include Tamed Mobs", kept: "tamed", test: isTamedEntity },
  { key: "named", label: "Include Name-Tagged Mobs", kept: "name-tagged", test: isNameTaggedEntity },
  { key: "leashed", label: "Include Leashed Mobs", kept: "leashed", test: isLeashedEntity },
  { key: "ridden", label: "Include Mobs With Riders", kept: "with riders", test: isRiddenEntity },
  { key: "chested", label: "Include Mobs Carrying Chests", kept: "carrying chests", test: isChestedEntity }
];

function getMobNameMap() {
  const map = {};
  for (const key of Object.keys(MOB_CATEGORIES)) {
    for (const mob of MOB_CATEGORIES[key].mobs) {
      map[mob.id] = mob.name;
    }
  }
  return map;
}

function getGroupMobIds(groupIndex) {
  const keys = Object.keys(MOB_CATEGORIES);
  if (groupIndex <= 0 || groupIndex > keys.length) {
    return getAllMobIds();
  }
  return MOB_CATEGORIES[keys[groupIndex - 1]].mobs.map((mob) => mob.id);
}

function loadPurgeOptions() {
  const options = {
    dimension: 0,
    group: 0,
    tamed: false,
    named: false,
    leashed: false,
    ridden: false,
    chested: false
  };
  try {
    const raw = world.getDynamicProperty(PURGE_OPTIONS_KEY);
    if (typeof raw === "string") {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") {
        const dim = Number(parsed.dimension);
        const grp = Number(parsed.group);
        if (Number.isInteger(dim) && dim >= 0 && dim < PURGE_DIMENSIONS.length) {
          options.dimension = dim;
        }
        if (Number.isInteger(grp) && grp >= 0 && grp <= Object.keys(MOB_CATEGORIES).length) {
          options.group = grp;
        }
        for (const p of PURGE_PROTECTIONS) {
          options[p.key] = Boolean(parsed[p.key]);
        }
      }
    }
  } catch (e) {
  }
  return options;
}

function savePurgeOptions(options) {
  try {
    world.setDynamicProperty(PURGE_OPTIONS_KEY, JSON.stringify(options));
  } catch (e) {
  }
}

function scanPurge(options, selectedIds) {
  const rules = getSpawnRules(world);
  const groupIds = new Set(getGroupMobIds(options.group));
  const dimension = PURGE_DIMENSIONS[options.dimension] || PURGE_DIMENSIONS[0];
  const types = {};
  const kept = {};
  for (const p of PURGE_PROTECTIONS) {
    kept[p.key] = 0;
  }

  for (const dimId of dimension.ids) {
    try {
      const dim = world.getDimension(dimId);
      if (!dim) {
        continue;
      }
      for (const entity of dim.getEntities()) {
        if (!entity || !entity.isValid) {
          continue;
        }
        const id = entity.typeId;
        if (rules[id] !== false || !groupIds.has(id)) {
          continue;
        }
        if (selectedIds && !selectedIds.has(id)) {
          continue;
        }
        let protectedBy = null;
        for (const p of PURGE_PROTECTIONS) {
          if (!options[p.key] && p.test(entity)) {
            protectedBy = p.key;
            break;
          }
        }
        if (protectedBy) {
          kept[protectedBy]++;
          continue;
        }
        if (!types[id]) {
          types[id] = [];
        }
        types[id].push(entity);
      }
    } catch (e) {
    }
  }

  return { types, kept };
}

function describeKept(kept) {
  const parts = [];
  for (const p of PURGE_PROTECTIONS) {
    if (kept[p.key] > 0) {
      parts.push(`${kept[p.key]} ${p.kept}`);
    }
  }
  return parts.length > 0 ? parts.join(", ") : "none";
}

function countDisabledLoaded() {
  const rules = getSpawnRules(world);
  let count = 0;
  for (const dimId of PURGE_DIMENSIONS[0].ids) {
    try {
      const dim = world.getDimension(dimId);
      if (!dim) {
        continue;
      }
      for (const entity of dim.getEntities()) {
        if (entity && entity.isValid && rules[entity.typeId] === false) {
          count++;
        }
      }
    } catch (e) {
    }
  }
  return count;
}

function openPurgeMobPicker(player, options) {

  const ids = Object.keys(scan.types);

  });

function openPurgeConfirm(player, options, selectedIds) {
  const scan = scanPurge(options, selectedIds);
  const names = getMobNameMap();
  const ids = Object.keys(scan.types);
  let total = 0;
  const lines = [];
  for (const id of ids) {
    total += scan.types[id].length;
    lines.push(`§f${names[id] || id}: §c${scan.types[id].length}`);
  }

  if (total === 0) {
    showPurgeResult(player, 0, scan.kept, [], "No disabled mobs matched your choices.");
    return;
  }

  const form = new MessageFormData();
  form.title("§cConfirm Purge");
  form.body(

    `${lines.join("\n")}\n\n` +
    `§7Kept safe: §f${describeKept(scan.kept)}\n\n` +
    `§cThis cannot be undone.`
  );
  form.button1("Cancel");
  form.button2("Purge Now");

  form.show(player).then((response) => {
    if (response.canceled || response.selection !== 1) {
      player.onScreenDisplay.setActionBar("§e[SpawnControl] §fPurge cancelled.");
      return;
    }
    system.run(() => runPurge(player, options, selectedIds));
  });
}

function runPurge(player, options, selectedIds) {
  const scan = scanPurge(options, selectedIds);
  const names = getMobNameMap();
  const removed = [];
  let purged = 0;

  for (const id of Object.keys(scan.types)) {
    let typeCount = 0;
    for (const entity of scan.types[id]) {
      try {
        if (entity.isValid) {
          entity.remove();
          typeCount++;
          purged++;
        }
      } catch (e) {
      }
    }
    if (typeCount > 0) {
      removed.push(`${names[id] || id}: ${typeCount}`);
    }
  }

  showPurgeResult(player, purged, scan.kept, removed, null);
}

function showPurgeResult(player, purged, kept, removed, note) {
  const form = new ActionFormData();
  form.title("§aPurge Summary");
  let body = `§fMobs purged: §a${purged}\n§7Kept safe: §f${describeKept(kept)}`;
  if (removed.length > 0) {
    body += `\n\n§fRemoved:\n§7${removed.join("\n")}`;
  }
  if (note) {
    body += `\n\n§e${note}`;
  }
  form.body(body);
  form.button("Close");
  form.show(player).then(() => {
  });
}

# ALN:SpawnControl — Custom Mob Spawn Whitelist & Toggle UI

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**Control world mob spawns with individual switches for 58 creatures, presets, and filter rules opened from a Spawn Control book.**

> ALN:SpawnControl adds an interactive in-game control panel to configure individual mob spawning rules across 4 categorized rosters. It suits world builders, realm administrators, and survival players wanting a fine-tuned environment without custom entities.

## Overview

Rename a vanilla Book to "Spawn Control" on an Anvil and use it to open the control panel. From the dashboard, players can inspect active mob counts, configure 4 categorized mob rosters with individual toggles, apply one-tap presets, see how many disabled mobs are currently loaded, configure global filter modes, or open the separate purge menu to remove existing disabled mobs. All preferences are saved persistently to world storage.

## Key Features

### Categorized Mob Rosters
- **What Happens**: Individual toggle switches govern 58 standard mobs organized across 4 distinct groups: Overworld Hostiles, Passive & Neutral Mobs, Nether & End Mobs, and Bosses & Rare Mobs.
- **How It Works**: Open any category to adjust individual toggles, such as turning off Creeper or Phantom while keeping Zombie and Skeleton active. Close a menu to leave it and use the book again to reopen the dashboard.
- **Good To Know**: All mobs are enabled by default. Mobs that are toggled off are removed immediately when spawning occurs. Each group menu also has a Group Action choice to enable or disable every mob in that group at once.

### Custom Mob Spawner & Farm Control
- **What Happens**: Fine-tune mob farms and monster spawners by permitting only chosen mob types to spawn while blocking all unwanted background spawns.
- **Good for Chosen Mob Spawner**: Perfect for mob spawner chambers and specialized drop farms (such as Skeleton, Blaze, Witch, or Creeper farms). Disabling other competing mob spawns maximizes spawner efficiency, frees up mob caps, and prevents unwanted entities from crowding your spawn areas.
- **How It Works**: Disable unneeded hostiles to maximize spawn rates for chosen creatures, or block specific mob types around complex farms without breaking the target mob grinder.

### Quick Presets
- **What Happens**: One-tap buttons allow bulk configuration: Enable All Mobs, Disable All Hostiles, Disable All Passive Mobs, or Disable Everything for an empty mob world.
- **How It Works**: Selecting "Disable All Hostiles" immediately switches off all Overworld monsters, Nether threats, and bosses simultaneously.

### Global Filter Rules
- **What Happens**: Global settings let you choose whether spawn restrictions apply only to natural world-gen spawns or also block Spawn Eggs and summon commands.
- **How It Works**: Leaving natural spawn mode active allows administrators to test mobs using Spawn Eggs while preventing wild spawns.

### Entity Purge
- **What Happens**: A dedicated Purge Disabled Mobs menu scans loaded chunks and removes mobs whose toggles are currently disabled.
- **How It Works**: Pick a dimension and a mob group, then choose whether tamed mobs, name-tagged mobs, leashed mobs, mobs with riders, and mobs carrying chests are included. A second screen lists each disabled mob type that is loaded with its count so you can purge only the types you pick.
- **Confirmation and Summary**: Before anything is removed, a confirmation screen shows how many mobs will be removed and how many are kept safe. After the purge, a summary screen lists what was removed.
- **Good To Know**: Tamed, name-tagged, leashed, ridden and chested mobs are kept by default, and your last purge choices are remembered.

### Admin Tag Lock
- **What Happens**: An optional permission lock restricts opening the control panel to players carrying the administrator tag.
- **How It Works**: When Admin Tag Lock is turned on, a player must carry the tag to open the book; the first player to open the book automatically receives the tag if none exists.
