```
import { world, EntityInitializationCause } from "@minecraft/server";
import { getSpawnRules, getSpawnConfig } from "./spawn_registry.js";

let blockedCount = 0;

export function getBlockedCount() {
  return blockedCount;
}

export function resetBlockedCount() {
  blockedCount = 0;
}

const IGNORED_TYPES = new Set([
  "minecraft:player",
  "minecraft:item",
  "minecraft:xp_orb",
  "minecraft:arrow",
  "minecraft:snowball",
  "minecraft:egg",
  "minecraft:splash_potion",
  "minecraft:lingering_potion",
  "minecraft:ender_pearl",
  "minecraft:eye_of_ender_signal",
  "minecraft:firework_rocket",
  "minecraft:trident",
  "minecraft:wind_charge_projectile",
  "minecraft:breeze_wind_charge_projectile",
  "minecraft:shulker_bullet",
  "minecraft:wither_skull",
  "minecraft:wither_skull_dangerous",
  "minecraft:dragon_fireball",
  "minecraft:small_fireball",
  "minecraft:large_fireball",
  "minecraft:boat",
  "minecraft:chest_boat",
  "minecraft:minecart",
  "minecraft:chest_minecart",
  "minecraft:hopper_minecart",
  "minecraft:tnt_minecart",
  "minecraft:command_block_minecart"
]);

export function registerSpawnFilter() {
  world.afterEvents.entitySpawn.subscribe((event) => {
    const entity = event.entity;
    if (!entity || !entity.isValid) {
      return;
    }

    const typeId = entity.typeId;
    if (IGNORED_TYPES.has(typeId)) {
      return;
    }

    const rules = getSpawnRules(world);
    if (rules[typeId] !== false) {
      return;
    }

    const config = getSpawnConfig(world);
    if (!config.blockSpawnEggs) {
      const cause = event.cause;
      const isNaturalOrWorldGen =
        cause === EntityInitializationCause.Spawned ||
        cause === EntityInitializationCause.Born;

      if (!isNaturalOrWorldGen) {
        return;
      }
    }

    try {
      entity.remove();
      blockedCount++;
    } catch (e) {
    }
  });
}
```

```
export function getSpawnRules(world) {
  try {
    const raw = world.getDynamicProperty(RULES_KEY);
    if (typeof raw === "string") {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") {
        return parsed;
      }
    }
  } catch (e) {
  }
  const defaultRules = {};
  for (const catKey of Object.keys(MOB_CATEGORIES)) {
    for (const mob of MOB_CATEGORIES[catKey].mobs) {
      defaultRules[mob.id] = true;
    }
  }
  saveSpawnRules(world, defaultRules);
  return defaultRules;
}

export function saveSpawnRules(world, rules) {
  world.setDynamicProperty(RULES_KEY, JSON.stringify(rules));
}

export function getSpawnConfig(world) {
  try {
    const raw = world.getDynamicProperty(CONFIG_KEY);
    if (typeof raw === "string") {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") {
        return {
          blockSpawnEggs: Boolean(parsed.blockSpawnEggs),
          adminOnly: Boolean(parsed.adminOnly)
        };
      }
    }
  } catch (e) {
  }
  const defaultConfig = { blockSpawnEggs: false, adminOnly: false };
  saveSpawnConfig(world, defaultConfig);
  return defaultConfig;
}

export function saveSpawnConfig(world, config) {
  world.setDynamicProperty(CONFIG_KEY, JSON.stringify(config));
}

export function getAllMobIds() {

  for (const catKey of Object.keys(MOB_CATEGORIES)) {
    for (const mob of MOB_CATEGORIES[catKey].mobs) {
      ids.push(mob.id);
    }
  }
  return ids;
```

```
import { ActionFormData, ModalFormData } from "@minecraft/server-ui";

const ADMIN_TAG = "aln_admin";

function hasAdminPermission(player) {
  const config = getSpawnConfig(world);
  if (!config.adminOnly) {
    return true;
  }
  const allPlayers = world.getAllPlayers();
  const hasAnyAdmin = allPlayers.some((p) => p.hasTag(ADMIN_TAG));
  if (!hasAnyAdmin) {
    player.addTag(ADMIN_TAG);
    player.onScreenDisplay.setActionBar("§a[SpawnControl] §fYou were granted Admin permissions.");
    return true;
  }
  return player.hasTag(ADMIN_TAG);
}

export function registerSpawnUI() {
  world.afterEvents.itemUse.subscribe((event) => {
    const item = event.itemStack;
    const player = event.source;

    if (!item || !player) {
      return;
    }

    const typeId = item.typeId;
    if (typeId !== "minecraft:book") {
      return;
    }

    const nameTag = item.nameTag;
    if (!nameTag || nameTag.trim().toLowerCase() !== "spawn control") {
      return;
    }

    if (!hasAdminPermission(player)) {
      player.onScreenDisplay.setActionBar("§c[SpawnControl] §fOnly admins can open Spawn Control!");
      return;

    system.run(() => {
      openMainMenu(player);
    });
  });
}

function openMainMenu(player) {
  const rules = getSpawnRules(world);
  const config = getSpawnConfig(world);
  const allIds = getAllMobIds();

  let allowedCount = 0;

  for (const id of allIds) {

function openCategoryMenu(player, categoryKey) {

    form.toggle(`${mob.name}`, { defaultValue: isAllowed });

      currentRules[mob.id] = Boolean(val);

    } else if (response.selection === 3) {

  const form = new ModalFormData();
  form.title("§3⚙ Global Filter Settings");

  form.toggle("§e« Back to Main Menu §7(Turn ON to return)", { defaultValue: false });
  form.toggle("Include Spawn Eggs & Summon Commands", { defaultValue: config.blockSpawnEggs });
  form.toggle("Admin Tag Lock (player tag: aln_admin)", { defaultValue: config.adminOnly });
  form.toggle("Purge Loaded Disabled Mobs Now", { defaultValue: false });

  form.show(player).then((response) => {
    if (response.canceled || !response.formValues) {
      return;
    }

    const backSelected = Boolean(response.formValues[0]);
    if (backSelected) {
      system.run(() => openMainMenu(player));
      return;
    }

    const blockSpawnEggs = Boolean(response.formValues[1]);
    const adminOnly = Boolean(response.formValues[2]);
    const purgeNow = Boolean(response.formValues[3]);

    saveSpawnConfig(world, { blockSpawnEggs, adminOnly });

    if (purgeNow) {
      purgeDisabledEntities(player);
    } else {
      player.onScreenDisplay.setActionBar("§a[SpawnControl] §fSaved global filter settings!");
    }
  });
}

function purgeDisabledEntities(player) {
  const rules = getSpawnRules(world);
  let purgedCount = 0;

  const dimensions = ["overworld", "nether", "the_end"];
  for (const dimId of dimensions) {
    try {
      const dim = world.getDimension(dimId);
      if (!dim) {
        continue;
      }
      const entities = dim.getEntities();
      for (const entity of entities) {
        if (!entity || !entity.isValid) {
          continue;
        }
        if (rules[entity.typeId] === false) {
          try {
            entity.remove();
            purgedCount++;
          } catch (e) {
          }
        }
      }
    } catch (e) {
    }
  }

  player.onScreenDisplay.setActionBar(`§a[SpawnControl] §fPurged ${purgedCount} disabled mobs!`);
}
```

```
**By ALNStudios**

**Control world mob spawns with individual switches for 58 creatures, presets, and filter rules opened from a Spawn Control book.**

> ALN:SpawnControl adds an interactive in-game control panel to configure individual mob spawning rules across 4 categorized rosters. It suits world builders, realm administrators, and survival players wanting a fine-tuned environment without custom entities.

## Overview

Rename a vanilla Book to "Spawn Control" on an Anvil and use it to open the control panel. From the dashboard, players can inspect active mob counts, configure 4 categorized mob rosters with individual toggles, apply one-tap presets, configure global filter modes, or purge existing disabled mobs. All preferences are saved persistently to world storage.

## Key Features

### Categorized Mob Rosters
- **What Happens**: Individual toggle switches govern 58 standard mobs organized across 4 distinct groups: Overworld Hostiles, Passive & Neutral Mobs, Nether & End Mobs, and Bosses & Rare Mobs.
- **How It Works**: Open any category to adjust individual toggles, such as turning off Creeper or Phantom while keeping Zombie and Skeleton active. Each category includes a dedicated back switch to return to the dashboard.
- **Good To Know**: All mobs are enabled by default. Mobs that are toggled off are removed immediately when spawning occurs.

### Custom Mob Spawner & Farm Control
- **What Happens**: Fine-tune mob farms and monster spawners by permitting only chosen mob types to spawn while blocking all unwanted background spawns.
- **Good for Chosen Mob Spawner**: Perfect for mob spawner chambers and specialized drop farms (such as Skeleton, Blaze, Witch, or Creeper farms). Disabling other competing mob spawns maximizes spawner efficiency, frees up mob caps, and prevents unwanted entities from crowding your spawn areas.
- **How It Works**: Disable unneeded hostiles to maximize spawn rates for chosen creatures, or block specific mob types around complex farms without breaking the target mob grinder.

### Quick Presets
- **What Happens**: One-tap buttons allow bulk configuration: Enable All Mobs, Disable All Hostiles, Disable All Passive Mobs, or Disable Everything for an empty mob world.
- **How It Works**: Selecting "Disable All Hostiles" immediately switches off all Overworld monsters, Nether threats, and bosses simultaneously.

### Global Filter Rules
- **What Happens**: Global settings let you choose whether spawn restrictions apply only to natural world-gen spawns or also block Spawn Eggs and summon commands.
- **How It Works**: Leaving natural spawn mode active allows administrators to test mobs using Spawn Eggs while preventing wild spawns.

### Entity Purge
- **What Happens**: An optional purge action scans loaded chunks in all three dimensions and removes mobs whose toggles are currently disabled.
- **Good To Know**: Toggling this option inside Global Filter Settings runs a single-pass removal and reports the number of purged entities.

### Admin Tag Lock
- **What Happens**: An optional permission lock restricts opening the control panel to players carrying the administrator tag.
- **How It Works**: When Admin Tag Lock is turned on, a player must carry the tag to open the book; the first player to open the book automatically receives the tag if none exists.

## How to Use

1. Import the .mcpack file and activate ALN:SpawnControl in your world's Behavior Packs.
2. Place a Book (not a Book and Quill) in an Anvil and rename it to "Spawn Control".
3. Hold the renamed book and use it to open the ALN:SpawnControl control panel.

## Common Questions

- **Do I need experimental toggles?**: No experimental toggles or cheats are required for normal menu usage.
- **Does it stop mob spawners?**: When set to block natural spawns or all sources, spawner-generated entities of disabled mob types are intercepted upon spawning.
- **Will turning off a mob remove existing ones?**: New spawns are blocked immediately; use the Purge Loaded Disabled Mobs option in Global Filter Settings to clear existing mobs.
```
