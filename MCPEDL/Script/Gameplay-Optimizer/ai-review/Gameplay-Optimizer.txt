```
import { world, system } from "@minecraft/server";
import { getSettings } from "./settings.js";

const PLAYER_TYPE = "minecraft:player";
const MIN_DURATION_TICKS = 10;
const SKIPPED = new Set(["minecraft:instant_health", "minecraft:instant_damage", "minecraft:saturation"]);
const pending = new Set();

export function startEffectParticles() {
  world.afterEvents.effectAdd.subscribe(({ entity, effect }) => {
    if (!getSettings().hideParticles) return;
    if (!entity || entity.typeId !== PLAYER_TYPE || !effect) return;
    if (SKIPPED.has(effect.typeId) || effect.duration < MIN_DURATION_TICKS) return;
    const key = `${entity.id}|${effect.typeId}`;
    if (pending.has(key)) {
      pending.delete(key);
      return;
    }
    pending.add(key);
    system.runTimeout(() => pending.delete(key), 40);
    try {
      entity.addEffect(effect.typeId, effect.duration, {
        amplifier: effect.amplifier,
        showParticles: false,
      });
    } catch (e) {
      pending.delete(key);
    }
  });
}
```

```
import { world } from "@minecraft/server";
import { getSettings } from "./settings.js";

const ITEM_TYPE = "minecraft:item";
const tracked = new Map();

function isLive(entity) {
  try {
    const flag = entity.isValid;
    return typeof flag === "function" ? flag.call(entity) : flag === true;
  } catch (e) {
    return false;
  }
}

export function resetItemTracking() {
  tracked.clear();
}

export function startItemCap() {
  world.afterEvents.entitySpawn.subscribe(({ entity }) => {
    const settings = getSettings();
    if (!settings.itemCap) return;
    if (!isLive(entity) || entity.typeId !== ITEM_TYPE) return;
    const key = entity.dimension.id;
    const live = (tracked.get(key) ?? []).filter(isLive);
    live.push(entity);
    while (live.length > settings.itemLimit) {
      const oldest = live.shift();
      try {
        oldest.remove();
      } catch (e) {
        continue;
      }
    }
    tracked.set(key, live);
  });
}
```

```
import { world, system } from "@minecraft/server";
import { getSettings } from "./settings.js";

const PORTAL_BLOCK = "minecraft:portal";
const OVERWORLD = "minecraft:overworld";
const NETHER = "minecraft:nether";
const CHECK_TICKS = 10;
const MAX_TICKS = 400;
const HOLD_TICKS = 100;
const HALF_SPAN = 16;
const OUTSIDE_LIMIT = 3;
const AREA_PREFIX = "aln_portal_";
const active = new Map();

function manager() {
  return world.tickingAreaManager;
}

function release(playerId) {
  const state = active.get(playerId);
  if (!state) return;
  active.delete(playerId);
  try {
    const mgr = manager();
    if (mgr && mgr.hasTickingArea(state.area)) mgr.removeTickingArea(state.area);
  } catch (e) {
    return;
  }
}

function inPortal(player) {
  try {
    return player.dimension.getBlock(player.location)?.typeId === PORTAL_BLOCK;
  } catch (e) {
    return false;
  }
}

function destinationOf(player) {
  const from = player.dimension.id;
  const loc = player.location;
  const y = Math.min(120, Math.max(0, Math.floor(loc.y)));
  if (from === OVERWORLD) {
    return { dimensionId: NETHER, x: Math.floor(loc.x / 8), y, z: Math.floor(loc.z / 8) };
  }
  if (from === NETHER) {
    return { dimensionId: OVERWORLD, x: Math.floor(loc.x * 8), y, z: Math.floor(loc.z * 8) };
  }
  return null;
}

function begin(player, now) {
  const mgr = manager();
  if (!mgr) return;
  const target = destinationOf(player);
  if (!target) return;
  const options = {
    dimension: world.getDimension(target.dimensionId),
    from: { x: target.x - HALF_SPAN, y: target.y, z: target.z - HALF_SPAN },
    to: { x: target.x + HALF_SPAN, y: target.y, z: target.z + HALF_SPAN },
  };
  if (!mgr.hasCapacity(options)) return;
  const area = `${AREA_PREFIX}${player.id}`;
  if (mgr.hasTickingArea(area)) return;
  active.set(player.id, { area, started: now, outside: 0, holding: false });
  mgr.createTickingArea(area, options).catch(() => release(player.id));
}

function sweep() {
  const settings = getSettings();
  if (!settings.portalPreload && active.size === 0) return;
  const now = system.currentTick;
  const present = new Set();
  for (const player of world.getAllPlayers()) {
    present.add(player.id);
    const state = active.get(player.id);
    const inside = inPortal(player);
    if (!state) {
      if (settings.portalPreload && inside) {
        try {
          begin(player, now);
        } catch (e) {
          release(player.id);
        }
      }
      continue;
    }
    if (state.holding) continue;
    state.outside = inside ? 0 : state.outside + 1;
    if (state.outside >= OUTSIDE_LIMIT || now - state.started > MAX_TICKS) release(player.id);
  }
  for (const id of [...active.keys()]) {
    if (!present.has(id)) release(id);
  }
}

export function startPortalPreload() {
  world.afterEvents.worldLoad.subscribe(() => {
    try {
      manager()?.removeAllTickingAreas();
    } catch (e) {
      active.clear();
    }
    active.clear();
    system.runInterval(sweep, CHECK_TICKS);
  });

  world.afterEvents.playerDimensionChange.subscribe(({ player }) => {
    const state = active.get(player.id);
    if (!state) return;
    state.holding = true;
    system.runTimeout(() => release(player.id), HOLD_TICKS);
  });

  world.afterEvents.playerLeave.subscribe(({ playerId }) => release(playerId));
}
```

```
import { world } from "@minecraft/server";

const KEY = "aln_gameplay_optimizer_settings";

export const ITEM_LIMIT_MIN = 50;
export const ITEM_LIMIT_MAX = 500;
export const ITEM_LIMIT_STEP = 50;

const DEFAULTS = {
  itemCap: false,
  itemLimit: 200,
  hideParticles: true,
  portalPreload: true,
};

let cache = null;

function sanitize(raw) {
  const out = { ...DEFAULTS };
  if (raw && typeof raw === "object") {
    if (typeof raw.itemCap === "boolean") out.itemCap = raw.itemCap;
    if (typeof raw.hideParticles === "boolean") out.hideParticles = raw.hideParticles;
    if (typeof raw.portalPreload === "boolean") out.portalPreload = raw.portalPreload;
    if (typeof raw.itemLimit === "number" && Number.isFinite(raw.itemLimit)) {
      const stepped = Math.round(raw.itemLimit / ITEM_LIMIT_STEP) * ITEM_LIMIT_STEP;
      out.itemLimit = Math.min(ITEM_LIMIT_MAX, Math.max(ITEM_LIMIT_MIN, stepped));
    }
  }
  return out;
}

export function getSettings() {
  if (cache) return cache;
  let saved = null;
  try {
    const raw = world.getDynamicProperty(KEY);
    if (typeof raw === "string") saved = JSON.parse(raw);
  } catch (e) {
    saved = null;
  }
  cache = sanitize(saved);
  return cache;
}

export function saveSettings(next) {
  cache = sanitize(next);
  try {
    world.setDynamicProperty(KEY, JSON.stringify(cache));
  } catch (e) {
    return false;
  }
  return true;
}
```

```
# Gameplay Optimizer — Item limits, quiet potions and portal preloading

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)
[![Format](https://img.shields.io/badge/Format-.MCPACK%20%7C%20.MCADDON%20%7C%20.MCWORLD-blue?style=flat-square)](https://github.com/AllensCreations/MCPE-Curseforges)

**Limits dropped items to between 50 and 500 per dimension, hides potion particles, and preloads Nether portal destinations from a Clock menu.**

> Gameplay Optimizer adds one settings menu, opened by sneaking and using a Clock, with three switches. It suits worlds with item farms, frequent potion effects, or Nether portal trips that stall on arrival.

## Overview

Sneak and use a Clock to open the menu, then pick Optimizer Settings. The menu has a switch for the dropped item limit, one for potion particle hiding, and one for portal preloading. Settings are saved in the world and apply to every player in it.

## Key Features

### Dropped Item Limit
- **What Happens**: When switched on, the pack keeps the number of dropped items in each dimension at or below a limit you choose from 50 to 500, in steps of 50. When a new drop passes the limit, the oldest tracked dropped item is removed.
- **How It Works**: With the limit set to 200, the 201st dropped item in the Overworld makes the oldest dropped item in the Overworld disappear.
- **Good To Know**: This switch is off by default. Removed items are gone for good, including valuable ones. Only items dropped after the world loaded and after the switch was turned on are counted.

### Potion Particle Hiding
- **What Happens**: When switched on, effects that players receive from potions, beacons and similar sources are reapplied with the swirling particles turned off.
- **Good To Know**: This switch is on by default. Only players are affected. Instant effects such as Instant Health are skipped, and every other particle in the game is unchanged.

### Portal Preloading
- **What Happens**: While you stand inside a Nether portal, the pack asks the game to start loading the matching spot on the other side: Overworld coordinates divided by 8, or Nether coordinates multiplied by 8. The loaded area is released about 5 seconds after you arrive, or after 20 seconds at most.
- **Good To Know**: This switch is on by default. The effect on portal trip time was not measured, and loading an area early adds work while you wait. If the game has no room for another loaded area, the step is skipped.

### Settings Menu

- **Good To Know**: Operators can change the settings, and so can a player who is alone in the world. Other players can open the menu and read the status.
```
