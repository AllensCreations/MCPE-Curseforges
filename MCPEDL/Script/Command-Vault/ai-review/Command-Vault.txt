```
const PROP_KEY = "aln_saved_commands";

export function getSavedCommands(player) {
  try {
    const raw = player.getDynamicProperty(PROP_KEY);
    if (!raw || typeof raw !== "string") {
      return [];
    }
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

export function sanitizeCommand(commandStr) {
  if (!commandStr || typeof commandStr !== "string") return "";
  let trimmed = commandStr.trim();
  while (trimmed.startsWith("/")) {
    trimmed = trimmed.substring(1).trim();
  }
  return trimmed;
}

export function saveCommand(player, label, commandStr) {
  const sanitized = sanitizeCommand(commandStr);
  if (!sanitized) {
    return { success: false, command: "", message: "Command cannot be empty." };
  }

  const cleanLabel = (label && typeof label === "string" && label.trim().length > 0)
    ? label.trim()
    : sanitized;

  const current = getSavedCommands(player);
  current.push({
    label: cleanLabel,
    command: sanitized,
  });

  try {
    player.setDynamicProperty(PROP_KEY, JSON.stringify(current));
    return { success: true, command: sanitized };
  } catch (err) {
    return { success: false, command: sanitized, message: String(err) };
  }
}
export function deleteCommandsByIndex(player, indicesToDelete) {
  const current = getSavedCommands(player);
  const deleteSet = new Set(indicesToDelete);
  const remaining = current.filter((_, idx) => !deleteSet.has(idx));

  try {
    player.setDynamicProperty(PROP_KEY, JSON.stringify(remaining));
    return { success: true, deletedCount: current.length - remaining.length };
  } catch (err) {
    return { success: false, deletedCount: 0, message: String(err) };
  }
}
```

```
import { system } from "@minecraft/server";
import { ActionFormData, ModalFormData } from "@minecraft/server-ui";
import { deleteCommandsByIndex, getSavedCommands, saveCommand } from "./command_storage.js";

export function openMainMenu(player) {
  const commands = getSavedCommands(player);
  const form = new ActionFormData()
    .title("§3§lALN:CommandVault")
    .body(`§7Saved Commands: §b${commands.length}`)
    .button("§aCreate Command", "textures/items/book_writable")
    .button("§eExecute Command", "textures/items/repeater")
    .button("§cDelete Command", "textures/ui/cancel")
    .button("§bAbout", "textures/items/paper")
    .button("§8Exit", "textures/ui/cancel");

  form.show(player).then((res) => {
    if (res.canceled) return;

    system.run(() => {
      switch (res.selection) {
        case 0:
          openCreateCommandScreen(player);
          break;
        case 1:
          openExecuteCommandScreen(player);
          break;
        case 2:
          openDeleteCommandScreen(player);
          break;
        case 3:
          openAboutScreen(player);
          break;
        case 4:
        default:
          break;
      }
    });
  }).catch(() => {});
}

export function openCreateCommandScreen(player) {
  const form = new ModalFormData()
    .title("§a§lCreate Command")
    .textField("Command Name (Optional)", "e.g., Clear Weather / Full Heal")
    .textField("Enter Command", "e.g., /weather clear or time set day")
    .submitButton("Save Command");

  form.show(player).then((res) => {
    if (res.canceled) {
      system.run(() => openMainMenu(player));
      return;
    }

    const [labelInput, commandInput] = res.formValues;
    const result = saveCommand(player, labelInput, commandInput);

    if (result.success) {
      player.playSound("random.orb");
      player.onScreenDisplay.setActionBar(`§a[CommandVault] §fSaved command: §e/${result.command}`);
    } else {
      player.onScreenDisplay.setActionBar(`§c[CommandVault] §fError: ${result.message || "Failed to save command."}`);
    }

    system.run(() => openMainMenu(player));
  }).catch(() => {});
}

export function openExecuteCommandScreen(player) {
  const commands = getSavedCommands(player);
  if (commands.length === 0) {
    player.onScreenDisplay.setActionBar("§c[CommandVault] §fNo saved commands yet! Use Create Command first.");
    system.run(() => openMainMenu(player));
    return;
  }

  const dropdownItems = commands.map((c) => {
    if (c.label && c.label !== c.command) {
      return `${c.label} (/${c.command})`;
    }
    return `/${c.command}`;
  });

  const form = new ModalFormData()
    .title("§e§lExecute Command")
    .dropdown("Select Saved Command", dropdownItems)
    .submitButton("Execute Command");

  form.show(player).then((res) => {
    if (res.canceled) {
      system.run(() => openMainMenu(player));
      return;
    }

    const [selectedIndex] = res.formValues;
    const target = commands[selectedIndex];
    if (!target || !target.command) {
      system.run(() => openMainMenu(player));
      return;
    }

    try {
      player.runCommand(target.command);
      player.onScreenDisplay.setActionBar(`§a[CommandVault] §fExecuted: §e/${target.command}`);
    } catch (err) {
      player.onScreenDisplay.setActionBar(`§c[CommandVault] §fError: ${err?.message || "Execution failed."}`);
    }
  }).catch(() => {});
}

export function openDeleteCommandScreen(player) {
  const commands = getSavedCommands(player);
  if (commands.length === 0) {
    player.onScreenDisplay.setActionBar("§c[CommandVault] §fNo saved commands to delete!");
    system.run(() => openMainMenu(player));
    return;
  }

  const form = new ModalFormData()
    .title("§c§lDelete Commands");

  for (const c of commands) {
    const itemLabel = (c.label && c.label !== c.command)
      ? `${c.label} (/${c.command})`
      : `/${c.command}`;
    form.toggle(`Delete ${itemLabel}`, false);
  }

  form.submitButton("Delete Selected");

  form.show(player).then((res) => {
    if (res.canceled) {
      system.run(() => openMainMenu(player));
      return;
    }

    const indicesToDelete = [];
    if (Array.isArray(res.formValues)) {
      res.formValues.forEach((val, idx) => {
        if (val === true) {
          indicesToDelete.push(idx);
        }
      });
    }

    if (indicesToDelete.length === 0) {
      player.onScreenDisplay.setActionBar("§e[CommandVault] §fNo commands selected for deletion.");
    } else {
      const result = deleteCommandsByIndex(player, indicesToDelete);
      if (result.success) {
        player.playSound("random.break");
        player.onScreenDisplay.setActionBar(`§c[CommandVault] §fDeleted §e${result.deletedCount} §fcommand(s).`);
      } else {
        player.onScreenDisplay.setActionBar(`§c[CommandVault] §fError: ${result.message || "Failed to delete commands."}`);
      }
    }

    system.run(() => openMainMenu(player));
  }).catch(() => {});
}
```

```
world.afterEvents.itemUse.subscribe((event) => {
  const { itemStack, source: player } = event;
  if (!itemStack || !player) return;

  const typeId = itemStack.typeId;
  if (typeId !== "minecraft:book" && typeId !== "minecraft:writable_book") {
    return;
  }

  const nameTag = itemStack.nameTag;
  if (nameTag && nameTag.trim().toLowerCase() === "command vault") {
    openMainMenu(player);
  }
});
```

```
# Command Vault — Saved Command Macro and Quick-Execute Menu

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**Save and execute your favorite commands directly from an interactive in-game menu.**

> Use an Anvil-renamed book to open your vault, store custom commands, and run them with one click.

## Overview

Command Vault gives players an intuitive, menu-driven way to store and run command macros in Minecraft Bedrock Edition without typing long slash commands repeatedly. Everything is stored persistently per player, and the entire system operates through a standard renamed book.

## Key Features

### Anvil Item Key
- **What Happens**: Rename any vanilla Book or Quill to Command Vault on an Anvil. Using the book immediately opens your personal command menu.

### Create Command Form
- **What Happens**: Add custom command macros with friendly labels such as Clear Weather or Spawn Point. Leading slashes and extra whitespace are automatically cleaned up.

### Quick-Execute Dropdown
- **What Happens**: Browse all saved commands in a dropdown list and execute them instantly with full command authority.

### Persistent Player Storage
- **What Happens**: Saved commands remain saved on your player profile across world sessions and reloads.

### Clean Action Bar Feedback
- **What Happens**: Receive instant audio-visual confirmations when commands are saved or successfully triggered.

## How to Use

1. Take a standard Book or Quill to an Anvil and rename it to Command Vault.
2. Hold or right-click the book to open the Command Vault menu.
3. Select Create Command to enter a label and command string.
4. Select Execute Command from the menu to run your saved macro.
```
