function getMainhandTool(player) {

function checkToolSafety(player) {
  if (!player || !player.isValid()) {
    return { safe: false, remaining: 0 };
  }

  const tool = getMainhandTool(player);
  if (!tool) {
    return { safe: false, remaining: 0 };
  }

  const durabilityComp = tool.getComponent("minecraft:durability");
  if (!durabilityComp) {
    return { safe: true, tool, remaining: 9999 };
  }

  const remaining = durabilityComp.maxDurability - durabilityComp.damage;
  if (remaining <= CONFIG.DURABILITY_SAFETY_THRESHOLD) {
    return { safe: false, tool, remaining };
  }

  return { safe: true, tool, remaining };
}

function applyToolDurabilityDamage(player) {
  if (!player || !player.isValid()) return false;

  const equippable = player.getComponent("minecraft:equippable");
  if (!equippable) return false;

  const tool = equippable.getEquipment(EquipmentSlot.Mainhand);
  if (!tool) return false;

  const durabilityComp = tool.getComponent("minecraft:durability");
  if (!durabilityComp) return true;

  let unbreakingLevel = 0;
  try {
    const enchantable = tool.getComponent("minecraft:enchantable");
    if (enchantable) {
      const unbreaking = enchantable.getEnchantment("unbreaking");
      if (unbreaking && typeof unbreaking.level === "number") {
        unbreakingLevel = unbreaking.level;
      }
    }
  } catch (err) {}

  const shouldTakeDamage = unbreakingLevel > 0
    ? (Math.random() < 1.0 / (unbreakingLevel + 1))
    : true;

  if (shouldTakeDamage) {
    durabilityComp.damage += 1;

    if (durabilityComp.damage >= durabilityComp.maxDurability) {
      equippable.setEquipment(EquipmentSlot.Mainhand, undefined);
      try {
        player.playSound("random.break", { volume: 1.0, pitch: 0.9 });
      } catch (e) {}
      return false;
    }

    equippable.setEquipment(EquipmentSlot.Mainhand, tool);
  }

  return true;
}

function calculateFortuneDropCount(fortuneLevel, minBase = 1, maxBase = 1) {
  const baseCount = minBase === maxBase
    ? minBase
    : Math.floor(Math.random() * (maxBase - minBase + 1)) + minBase;

  if (fortuneLevel <= 0) return baseCount;
  const multiplier = Math.max(1, Math.floor(Math.random() * (fortuneLevel + 2)));
  return Math.min(baseCount * multiplier, 64);
}

function isTreeCanopy(dimension, blocks) {
  if (!blocks || blocks.length === 0) return false;
  const candidates = [...blocks].sort((a, b) => b.y - a.y).slice(0, 8);
  for (const pos of candidates) {
    for (let dx = -2; dx <= 2; dx++) {
      for (let dz = -2; dz <= 2; dz++) {
        for (let dy = -1; dy <= 2; dy++) {
          try {
            const b = dimension.getBlock({ x: pos.x + dx, y: pos.y + dy, z: pos.z + dz });
            if (!b) continue;

            // Nether tree canopies (wart blocks and shroomlights)
            if (b.typeId.includes("wart_block") || b.typeId.includes("shroomlight")) {
              return true;
            }

            // Overworld leaves: ignore player-placed leaves (persistent_bit: true)
            if (b.typeId.includes("leaves")) {
              const permutation = b.permutation;
              const isPersistent = permutation?.getState("persistent_bit");
              // If persistent_bit is true, the leaves were placed by a player (decorative)
              if (isPersistent === true) {
                continue;
              }
              // Natural tree leaf detected
              return true;
            }
          } catch (e) {}
        }
      }
    }
  }
  return false;
}

// ─── Tree Capitator Execution ─────────────────────────────────────────────────

function executeTreeCapitator(player, dimension, startLoc, startBlockId, onComplete) {
  if (typeof startBlockId === "function") {
    onComplete = startBlockId;
    startBlockId = null;
  }
  if (!CONFIG.TREE_CAPITATOR_ENABLED) {
    if (onComplete) onComplete();
    return;
  }

  const targetSpecies = startBlockId ? getWoodSpecies(startBlockId) : null;
  const visited = new Set();
  visited.add(`${startLoc.x},${startLoc.y},${startLoc.z}`);

  const blocksToBreak = [];
  const queue = [{ x: startLoc.x, y: startLoc.y, z: startLoc.z }];

  // 26-neighbor 3D flood fill (detecting all surrounding connected blocks dx, dy, dz in [-1, 0, 1])
  // Upward-biased (dy >= 0) to fell entire trunk and branching canopy without digging below broken stump
  while (queue.length > 0 && blocksToBreak.length < CONFIG.MAX_TREE_BLOCKS) {
    const cur = queue.shift();

    for (let dy = 0; dy <= 1; dy++) {
      for (let dx = -1; dx <= 1; dx++) {
        for (let dz = -1; dz <= 1; dz++) {
          if (dx === 0 && dy === 0 && dz === 0) continue;

          const nx = cur.x + dx;
          const ny = cur.y + dy;
          const nz = cur.z + dz;

          if (ny < startLoc.y) continue;
          if (ny > startLoc.y + CONFIG.MAX_TREE_HEIGHT) continue;

          const key = `${nx},${ny},${nz}`;
          if (visited.has(key)) continue;
          visited.add(key);

          try {
            const block = dimension.getBlock({ x: nx, y: ny, z: nz });
            if (block && isMatchingLog(block.typeId, targetSpecies)) {
              const logPos = { x: nx, y: ny, z: nz };
              blocksToBreak.push(logPos);
              queue.push(logPos);
              if (blocksToBreak.length >= CONFIG.MAX_TREE_BLOCKS) break;
            }
          } catch (e) {}
        }
        if (blocksToBreak.length >= CONFIG.MAX_TREE_BLOCKS) break;
      }
      if (blocksToBreak.length >= CONFIG.MAX_TREE_BLOCKS) break;
    }
  }

  if (blocksToBreak.length === 0) {
    if (onComplete) onComplete();
    return;
  }

  // House & build protection: require natural tree canopy (leaves with persistent_bit: false or nether canopy)
  if (!isTreeCanopy(dimension, blocksToBreak)) {
    if (onComplete) onComplete();
    return;
  }

  // Break felled logs
  for (const pos of blocksToBreak) {
    if (!player || !player.isValid()) break;

    const safety = checkToolSafety(player);
    if (!safety.safe) {
      notifyToolProtection(player, safety.remaining);
      break;
    }

    try {
      dimension.runCommand(`setblock ${pos.x} ${pos.y} ${pos.z} air destroy`);
      applyToolDurabilityDamage(player);
    } catch (err) {}
  }

  if (onComplete) onComplete();
}

// ─── Domino Ore Miner Execution ───────────────────────────────────────────────

function executeDominoOreMiner(player, dimension, startLoc, startBlockId, onComplete) {
  if (!CONFIG.DOMINO_ORE_MINER_ENABLED) {
    if (onComplete) onComplete();
    return;
  }

  const startOreInfo = getOreInfo(startBlockId);
  if (!startOreInfo) {
    if (onComplete) onComplete();
    return;
  }

  const initialTool = getMainhandTool(player);
  const toolTier = getPickaxeTier(initialTool?.typeId);
  if (toolTier < startOreInfo.minTier) {
    if (onComplete) onComplete();
    return;
  }

  const queue = [{ x: startLoc.x, y: startLoc.y, z: startLoc.z }];
  const visited = new Set();
  visited.add(`${startLoc.x},${startLoc.y},${startLoc.z}`);

  const cluster = [];

  while (queue.length > 0 && cluster.length < CONFIG.MAX_ORE_CLUSTER_SIZE) {
    const current = queue.shift();

    for (const offset of NEIGHBOR_OFFSETS_26) {
      const nx = current.x + offset.x;
      const ny = current.y + offset.y;
      const nz = current.z + offset.z;
      const key = `${nx},${ny},${nz}`;

      if (visited.has(key)) continue;
      visited.add(key);

      if (ny < -64 || ny > 320) continue;

      try {
        const neighborBlock = dimension.getBlock({ x: nx, y: ny, z: nz });
        if (!neighborBlock) continue;

        const neighborOreInfo = getOreInfo(neighborBlock.typeId);
        if (neighborOreInfo && neighborOreInfo.category === startOreInfo.category) {
          cluster.push({
            x: nx,
            y: ny,
            z: nz,
            blockId: neighborBlock.typeId,
            oreInfo: neighborOreInfo,
          });
          queue.push({ x: nx, y: ny, z: nz });

          if (cluster.length >= CONFIG.MAX_ORE_CLUSTER_SIZE) break;
        }
      } catch (err) {}
    }
  }

  if (cluster.length === 0) {
    if (onComplete) onComplete();
    return;
  }

  for (const target of cluster) {
    if (!player || !player.isValid()) break;

    const safety = checkToolSafety(player);
    if (!safety.safe) {
      notifyToolProtection(player, safety.remaining);
      break;
    }

    const currentTool = safety.tool;
    const currentTier = getPickaxeTier(currentTool?.typeId);
    if (currentTier < target.oreInfo.minTier) break;

    let silkTouch = false;
    let fortuneLevel = 0;

    try {
      const enchantable = currentTool.getComponent("minecraft:enchantable");
      if (enchantable) {
        silkTouch = Boolean(enchantable.getEnchantment("silk_touch"));
        const fortune = enchantable.getEnchantment("fortune");
        if (fortune && typeof fortune.level === "number") {
          fortuneLevel = fortune.level;
        }
      }
    } catch (e) {}

    try {
      const centerPos = { x: target.x + 0.5, y: target.y + 0.5, z: target.z + 0.5 };

      if (silkTouch) {
        dimension.spawnItem(new ItemStack(target.blockId, 1), centerPos);
        dimension.getBlock({ x: target.x, y: target.y, z: target.z })?.setType("minecraft:air");
        try {
          dimension.playSound("dig.stone", centerPos, { volume: 0.8, pitch: 1.0 });
        } catch (pe) {}
      } else if (fortuneLevel > 0 && target.oreInfo.dropItem && target.oreInfo.dropItem !== target.blockId) {
        const minDrop = target.oreInfo.minDrop || 1;
        const maxDrop = target.oreInfo.maxDrop || 1;
        const count = calculateFortuneDropCount(fortuneLevel, minDrop, maxDrop);

        dimension.spawnItem(new ItemStack(target.oreInfo.dropItem, count), centerPos);
        dimension.getBlock({ x: target.x, y: target.y, z: target.z })?.setType("minecraft:air");
        try {
          dimension.playSound("dig.stone", centerPos, { volume: 0.8, pitch: 1.0 });
        } catch (pe) {}
      } else {
        dimension.runCommand(`setblock ${target.x} ${target.y} ${target.z} air destroy`);
      }

      applyToolDurabilityDamage(player);
    } catch (err) {}
  }

  if (onComplete) onComplete();
}

// ─── Event Dispatcher ─────────────────────────────────────────────────────────

const activeOperations = new Set();

world.afterEvents.playerBreakBlock.subscribe((event) => {
  const { block, brokenBlockPermutation, player, dimension, itemStackBeforeBreak } = event;

  if (!player || !player.isValid()) return;
  if (!player.isSneaking) return;
  if (activeOperations.has(player.id)) return;

  const startLoc = {
    x: Math.floor(block.location.x),
    y: Math.floor(block.location.y),
    z: Math.floor(block.location.z),
  };
  const brokenTypeId = brokenBlockPermutation?.type?.id;
  if (!brokenTypeId) return;

  const tool = getMainhandTool(player) || itemStackBeforeBreak;
  if (!tool) return;

  const toolTypeId = tool.typeId;

  // Tree Capitator
  if (isAxe(toolTypeId) && isEligibleLog(brokenTypeId)) {
    activeOperations.add(player.id);
    try {
      executeTreeCapitator(player, dimension, startLoc, brokenTypeId, () => {
        activeOperations.delete(player.id);
      });
    } catch (err) {
      activeOperations.delete(player.id);
    }
    return;
  }

  // Domino Ore Miner
  if (isPickaxe(toolTypeId) && getOreInfo(brokenTypeId)) {
    activeOperations.add(player.id);
    try {
      executeDominoOreMiner(player, dimension, startLoc, brokenTypeId, () => {
        activeOperations.delete(player.id);
      });
    } catch (err) {
      activeOperations.delete(player.id);
    }
    return;
  }
});

# Smart Tree Capitator and Domino Ore Miner — Vein and Tree Miner

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

Fell entire trees and mine contiguous ore veins in a single sneak-break, protected by automated durability safety stops.

---

## Overview

Mining large underground ore clusters and chopping tall trees one block at a time can take time. **Smart Tree Capitator & Domino Ore Miner** adds sneak-activated chain mining to Minecraft Bedrock Edition . Sneak while breaking a log or ore to fell the entire tree or harvest the connected ore vein in one action.

## Key Features

### Sneak-Activated Mining

### Connected Tree Felling

### Connected Ore Mining

### Mineral Category Normalization

### Durability Safety Guard

### Unbreaking Respect

### Fortune & Silk Touch Support

### Zero Hunger Penalties

## How It Works

### Standard Mining
- **Requirement**: Standing or walking
- **What Happens**: Breaks single block with standard vanilla drops.

### Tree Capitation
- **Requirement**: Sneak with an Axe
- **What Happens**: Fells connected trunk logs and branches upward up to 128 blocks during the same mining action.

### Domino Ore Mining
- **Requirement**: Sneak with a Pickaxe
- **What Happens**: Sweeps contiguous ore veins across 26 spatial neighbors up to 64 blocks.

### Durability Protection
- **Requirement**: Tool has 5 durability points remaining
- **What Happens**: Halts chain break immediately, preserves tool, and plays alert sound.

### Tool Switch Safety
- **Requirement**: Switch or unequip tool
- **What Happens**: Terminates chain process instantly before processing the next block.

---

## How Tree Felling Works

When you break an eligible log while sneaking with an axe, the add-on scans through the tree:

1. **Connected Branches**: Follows connected logs and branches, including diagonal connections.
2. **Matching Wood Types**: Only the same wood type is felled, helping protect nearby trees.
3. **The Leaf Scan (Player-Placed Leaves Ignored)**: Ignores player-placed leaves (sheared decorative leaves), ensuring custom hedges, bushes, and decorative roof foliage are never falsely detected as natural tree canopies.
4. **Build Protection**: Uses nearby natural canopy clues to reduce accidental damage to wooden builds.
5. **Safety Limits**: Capped at 128 logs per tree and bounded by the world height range to prevent runaway chain reactions.

### Supported Trees
- **Overworld**: Oak, Spruce, Birch, Jungle, Acacia, Dark Oak, Mangrove (including roots), Cherry, and Pale Oak.
- **Nether**: Crimson Stem, Warped Stem, Crimson Hyphae, Warped Hyphae.
- **Wood Variants**: All stripped logs and all 6-sided bark wood blocks.
- **Canopy Leaves**: Left intact for natural leaf decay so saplings, sticks, and apples drop normally over time.

---

## How Ore Mining Works

Domino Ore Miner follows connected matching ores, including diagonal connections, up to the 64-block cluster limit. Regular and deepslate versions of the same ore can be mined together:

- **Drops**: Nether Quartz

- **Drops**: Ancient Debris Block

## Durability Protection System

- Checks the tool's remaining durability before breaking each additional block.
- Halts immediately when remaining durability  reaches 5 or less.
- Plays an audible warning sound (warning sound) and displays an action bar alert:
  > **[Tool Protection]** Stopped mining to save tool! (5 durability left)
- Revalidates the equipped tool before each block break. If the tool is switched or removed, chain mining stops instantly.
- Respects the normal durability-saving effect of the Unbreaking enchantment.
