```
import { world, system } from "@minecraft/server";
import { PHASE_COUNT, phaseName, phaseIndexFor, pickBlock } from "./phases.js";

/* ------------------------------ Configuration ------------------------------ */
const ORIGIN = { x: 0, y: 64, z: 0 };            // the central block
const ABOVE_ORIGIN = { x: 0.5, y: 65, z: 0.5 };  // merchant spawn and void rescue point
const COUNT_KEY = "total_blocks_broken";
const MERCHANT_EVERY = 250;
const MERCHANT_TICKS = 1200;                      // 60 seconds
const MERCHANT_TAG = "ob_merchant";
const VOID_Y = -50;
const VOID_DAMAGE = 10;

/* --------------------------------- State ----------------------------------- */
let total = 0;
let ready = false;
const hudPause = new Map(); // player id -> tick until which the HUD stays quiet

const overworld = () => world.getDimension("overworld");

function saveTotal() {
  try {
    world.setDynamicProperty(COUNT_KEY, total);
  } catch (e) {}
}

function pauseHud(player, ticks) {
  hudPause.set(player.id, system.currentTick + ticks);
}

/* ----------------------- Regeneration and progression ----------------------- */
// Puts the next block for the current phase on the origin. Returns false if the chunk is unloaded.
function placeNext() {
  const block = overworld().getBlock(ORIGIN);
  if (!block) return false;
  block.setType(pickBlock(phaseIndexFor(total)));
  return true;
}

function announcePhase(index) {
  const title = "§6[PHASE UNLOCKED] §fPhase " + (index + 1) + ": " + phaseName(index);
  for (const player of world.getAllPlayers()) {
    try {
      player.onScreenDisplay.setTitle(title);
      player.playSound("random.levelup");
    } catch (e) {}
  }
}

/* ---------------------------------- Merchant --------------------------------- */
function spawnMerchant() {
  let trader;
  try {
    trader = overworld().spawnEntity("minecraft:wandering_trader", ABOVE_ORIGIN);
    trader.addTag(MERCHANT_TAG);
    // Slowness 255 keeps the trader standing on the origin instead of wandering into the void.
    trader.addEffect("slowness", MERCHANT_TICKS + 100, { amplifier: 255, showParticles: false });
  } catch (e) {
    return;
  }
  for (const player of world.getAllPlayers()) {
    try {
      player.onScreenDisplay.setActionBar("§a[MERCHANT] §fA Wandering Trader has arrived for 60 seconds!");
      pauseHud(player, 100);
    } catch (e) {}
  }
  system.runTimeout(() => {
    try {
      trader.remove();
    } catch (e) {}
  }, MERCHANT_TICKS);
}

/* -------------------------------- Event hooks -------------------------------- */
world.afterEvents.playerBreakBlock.subscribe((event) => {
  if (!ready) return;
  const b = event.block;
  if (b.dimension.id !== "minecraft:overworld" || b.x !== ORIGIN.x || b.y !== ORIGIN.y || b.z !== ORIGIN.z) return;

  const before = phaseIndexFor(total);
  total += 1;
  saveTotal();
  const after = phaseIndexFor(total);

  // Regenerate on the next tick.
  system.run(() => {
    try {
      placeNext();
      if (total % MERCHANT_EVERY === 0) spawnMerchant();
    } catch (e) {}
  });
  if (after !== before) announcePhase(after);
});

world.afterEvents.playerLeave.subscribe((event) => {
  hudPause.delete(event.playerId);
});

/* ----------------------------- Void safeguard (10 ticks) ----------------------------- */
system.runInterval(() => {
  if (!ready) return;
  for (const player of world.getAllPlayers()) {
    try {
      if (player.location.y >= VOID_Y) continue;
      player.teleport(ABOVE_ORIGIN, { dimension: overworld() });
      try {
        player.clearVelocity();
      } catch (e) {}
      // The safety tax never kills: it leaves at least half a heart.
      const health = player.getComponent("minecraft:health");
      const damage = health ? Math.min(VOID_DAMAGE, Math.max(0, health.currentValue - 1)) : VOID_DAMAGE;
      if (damage > 0) player.applyDamage(damage);
      player.playSound("mob.enderdragon.hit");
      player.onScreenDisplay.setActionBar("§c[VOID SAVIOR] §fSaved from the abyss! Safety tax applied.");
      pauseHud(player, 60);
    } catch (e) {}
  }
}, 10);

/* ----------------------------- HUD and origin check (20 ticks) ----------------------------- */
system.runInterval(() => {
  if (!ready) return;
  const now = system.currentTick;

  // Fallback: if the origin vanished some other way (explosion, piston...), bring it back without counting it.
  try {
    const block = overworld().getBlock(ORIGIN);
    if (block && block.isAir) placeNext();
  } catch (e) {}

  const index = phaseIndexFor(total);
  const text =
    "§6[ ONE-BLOCK ECONOMY ] §fBlocks: " + total + " §8| §ePhase " + (index + 1) + "/" + PHASE_COUNT + " §8| §b" + phaseName(index);
  for (const player of world.getAllPlayers()) {
    try {
      if ((hudPause.get(player.id) || 0) > now) continue;
      player.onScreenDisplay.setActionBar(text);
    } catch (e) {}
  }
}, 20);

/* ---------------------------------- Startup ---------------------------------- */
// Runs on the first tick, when world state can be read safely.
system.run(() => {
  try {
    const saved = world.getDynamicProperty(COUNT_KEY);
    total = typeof saved === "number" ? saved : 0;
  } catch (e) {}
  try {
    world.setDefaultSpawnLocation({ x: 0, y: 65, z: 0 });
  } catch (e) {}
  try {
    // Clear any trader left over from before a reload.
    for (const e of overworld().getEntities({ tags: [MERCHANT_TAG] })) e.remove();
  } catch (e) {}
  ready = true;
});
```

```
  { name: "Cosmic Chaos", from: 23001, blocks: [] }
];

export const PHASE_COUNT = POOLS.length;

export function phaseName(index) {
  return POOLS[index].name;
}

// Index (0-19) of the phase that applies once `total` blocks have been broken.
export function phaseIndexFor(total) {
  for (let i = POOLS.length - 1; i >= 0; i--) {
    if (total >= POOLS[i].from) return i;
  }
  return 0;
}

function randomFrom(list) {
  return list[Math.floor(Math.random() * list.length)];
}

// Phase 20 draws from every earlier pool; later pools are weighted heavier.
function cosmicPick() {
  const last = POOLS.length - 1;
  let weight = 0;
  for (let i = 0; i < last; i++) weight += i + 1;
  let roll = Math.random() * weight;
  for (let i = 0; i < last; i++) {
    roll -= i + 1;
    if (roll < 0) return randomFrom(POOLS[i].blocks);
  }
  return randomFrom(POOLS[last - 1].blocks);
}

// Returns a full block id for the given phase index.
export function pickBlock(index) {
  const block = index === POOLS.length - 1 ? cosmicPick() : randomFrom(POOLS[index].blocks);
  return "minecraft:" + block;
}
```

```
# One-Block Hardcore Survival Economy — 20-Phase Progression Add-on

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)
[![Format](https://img.shields.io/badge/Format-.MCPACK%20%7C%20.MCADDON%20%7C%20.MCWORLD-blue?style=flat-square)](https://github.com/AllensCreations/MCPE-Curseforges)

This Minecraft Bedrock add-on adds central-block progression, recurring Wandering Trader visits, and a void-rescue feature for a compatible OneBlock world.

## Features

### OneBlock Regeneration
- The central block returns after it is broken.
- Each break advances the saved block counter and can unlock a new phase.

### 20 Progression Phases
- The block pool changes as the total number of blocks broken increases.
- An action-bar display shows the total, current phase, and phase name.

### Wandering Trader
- A Wandering Trader arrives every 250 blocks broken.
- Each visit lasts about 60 seconds.

### Void Rescue
- Players who fall below Y = -50 are teleported back above the OneBlock.
- The rescue applies damage but is designed to leave at least one health point.

## Important World Requirement

This add-on does not generate a complete OneBlock world. Use it in a world with the central block at **(0, 64, 0)** in the Overworld.

## Compatibility

- Minimum Minecraft Bedrock version: 1.21.0.
- Minecraft Bedrock Edition: 1.21.0 or newer.
- Experimental features: Not verified.
- Achievements: Likely achievement-friendly.
- Verification: Source created, not tested in game.
```
