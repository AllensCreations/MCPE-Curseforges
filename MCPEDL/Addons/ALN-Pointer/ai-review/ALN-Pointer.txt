import { world, system, EntityComponentTypes, GameMode } from "@minecraft/server";
import { PLAYER_ID } from "./info.js";
import { isOn } from "./store.js";

const ARROW = "minecraft:arrow";
const RANGE = 48;
const MIN_COS = Math.cos((25 * Math.PI) / 180);
const GRAVITY = 0.05;
const MIN_SPEED = 0.8;

const minus = (a, b) => ({ x: a.x - b.x, y: a.y - b.y, z: a.z - b.z });
const length = (v) => Math.hypot(v.x, v.y, v.z);

function candidates(owner) {
  const options = { location: owner.location, maxDistance: RANGE };
  const mobs = owner.dimension.getEntities({ ...options, families: ["monster"] });
  const players = owner.dimension.getPlayers({ ...options, excludeGameModes: [GameMode.Creative, GameMode.Spectator] });
  return [...mobs, ...players].filter((e) => e.id !== owner.id);
}

function pickTarget(owner) {
  const eye = owner.getHeadLocation();
  const view = owner.getViewDirection();
  let best;
  let bestCos = MIN_COS;
  for (const entity of candidates(owner)) {
    const offset = minus(entity.getHeadLocation(), eye);
    const distance = length(offset);
    if (distance < 1) continue;
    const cos = (view.x * offset.x + view.y * offset.y + view.z * offset.z) / distance;
    if (cos > bestCos) {
      bestCos = cos;
      best = entity;
    }
  }
  return best;
}

function aimPoint(from, target, speed) {
  const head = target.getHeadLocation();
  const motion = target.getVelocity();
  let point = head;
  for (let i = 0; i < 2; i++) {
    const t = length(minus(point, from)) / speed;
    point = {
      x: head.x + motion.x * t,
      y: head.y + motion.y * t + 0.5 * GRAVITY * t * t,
      z: head.z + motion.z * t,
    };
  }
  return point;
}

function steer(arrow) {
  try {
    if (!arrow.isValid) return;
    const projectile = arrow.getComponent(EntityComponentTypes.Projectile);
    const owner = projectile?.owner;
    if (!owner || owner.typeId !== PLAYER_ID || !isOn(owner, "bowAimbot")) return;
    const speed = length(arrow.getVelocity());
    if (speed < MIN_SPEED) return;
    const target = pickTarget(owner);
    if (!target) return;
    const from = arrow.location;
    const direction = minus(aimPoint(from, target, speed), from);
    const size = length(direction);
    if (size < 0.001) return;
    projectile.shoot({ x: (direction.x / size) * speed, y: (direction.y / size) * speed, z: (direction.z / size) * speed });
  } catch (e) {}
}

export function initBowAim() {
  world.afterEvents.entitySpawn.subscribe(({ entity }) => {
    if (entity.typeId !== ARROW) return;
    system.run(() => steer(entity));
  });
}

import { world, system, EntityComponentTypes, EntityDamageCause, EquipmentSlot, ItemStack } from "@minecraft/server";
import { PLAYER_ID } from "./info.js";
import { isOn } from "./store.js";

const TOTEM = "minecraft:totem_of_undying";
const SHIELD = "minecraft:shield";
const SPARKS = "minecraft:critical_hit_emitter";
const CRIT = 1.5;
const NO_TOTEM = new Set([EntityDamageCause.void, EntityDamageCause.suicide]);
const BLOCKED = new Set([EntityDamageCause.entityAttack, EntityDamageCause.projectile]);
const replaying = new Set();

const slot = (p, which) => p.getComponent(EntityComponentTypes.Equippable)?.getEquipmentSlot(which);
const bag = (p) => p.getComponent(EntityComponentTypes.Inventory)?.container;
const hands = (p) => [slot(p, EquipmentSlot.Mainhand)?.getItem()?.typeId, slot(p, EquipmentSlot.Offhand)?.getItem()?.typeId];

function totemIndex(p) {
  const inv = bag(p);
  for (let i = 0; inv && i < inv.size; i++) if (inv.getItem(i)?.typeId === TOTEM) return i;
  return -1;
}

function equipTotem(p) {
  const off = slot(p, EquipmentSlot.Offhand);
  if (!off || hands(p).includes(TOTEM)) return;
  const inv = bag(p);
  const i = totemIndex(p);
  const old = off.getItem();
  if (i >= 0) {
    off.setItem(inv.getItem(i));
    inv.setItem(i, old);
    return;
  }
  off.setItem(new ItemStack(TOTEM, 1));
  const left = old && (inv ? inv.addItem(old) : old);
  if (left) p.dimension.spawnItem(left, p.location);
}

function facing(p, src) {
  const o = src.damagingProjectile?.location ?? src.damagingEntity?.location;
  if (!o) return false;
  const v = p.getViewDirection();
  const dx = o.x - p.location.x;
  const dz = o.z - p.location.z;
  return dx * v.x + dz * v.z > 0;
}

function replayOptions(src) {

function aura(p) {
  const opts = { location: p.location, maxDistance: AURA_RANGE };
  const found = new Map();
  const mobs = p.dimension.getEntities({ ...opts, families: ["monster"] });
  const foes = p.dimension.getPlayers({ ...opts, excludeGameModes: [GameMode.Creative, GameMode.Spectator] });
  for (const e of [...mobs, ...foes]) if (e.id !== p.id) found.set(e.id, e);
  [...found.values()]
    .sort((a, b) => dist(p.location, a) - dist(p.location, b))
    .slice(0, AURA_TARGETS)
    .forEach((t) => t.applyDamage(AURA_DAMAGE, { cause: EntityDamageCause.entityAttack, damagingEntity: p }));
}

function clutch(p) {
  if (skip(p) || p.isOnGround || !p.isFalling || p.isGliding || p.isFlying || p.isInWater) {
    falls.delete(p.id);
    return;
  }
  const y = p.location.y;
  const peak = Math.max(falls.get(p.id) ?? y, y);
  falls.set(p.id, peak);
  const vy = p.getVelocity().y;
  if (peak - y < MIN_FALL || vy > -0.2) return;
  const hit = p.dimension.getBlockFromRay(p.location, { x: 0, y: -1, z: 0 }, {
    maxDistance: Math.max(4, -vy * 3),
    includeLiquidBlocks: false,
    includePassableBlocks: false,
  });
  const spot = hit?.block.above();
  if (!spot?.isAir) return;
  spot.setType("minecraft:water");
  falls.delete(p.id);
  const dim = p.dimension;
  const at = spot.location;
  system.runTimeout(() => {
    const b = dim.getBlock(at);
    if (b?.typeId === "minecraft:water") b.setType("minecraft:air");
  }, WATER_LIFETIME);
}

function each(id, fn) {
  if (!anyOn(id)) return;
  for (const p of world.getAllPlayers()) {
    if (!p.isValid || !isOn(p, id)) continue;

      fn(p);
    } catch (e) {}
  }
}

You get an ALN Pointer, use it, and pick the toggles you want in the ALNStudios menu. The toggles then run by themselves whenever you fight, shoot or fall, and vanilla Minecraft has no equivalent switches. The features are cheat-style helpers, so use them only on worlds where everyone agrees.
