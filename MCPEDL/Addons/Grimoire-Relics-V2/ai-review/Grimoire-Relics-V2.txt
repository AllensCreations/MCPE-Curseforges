function consumeVoidRelic(p, s) {
  if (s.voidRelicConsumed) return false;

  try {

    const off = eq && eq.getEquipment(EquipmentSlot.Offhand);
    if (off && off.typeId === "custom:tome_void") {
      eq.setEquipment(EquipmentSlot.Offhand, undefined);
      p.playSound("random.break");
      p.onScreenDisplay.setActionBar("§cYour Void Phantom has shattered!");
      return true;
    }

    const inventory = p.getComponent("minecraft:inventory");
    const container = inventory && inventory.container;
    if (container) {
      for (let i = 0; i < container.size; i++) {
        const item = container.getItem(i);
        if (!item || item.typeId !== "custom:tome_void") continue;
        if (item.amount > 1) {
          item.amount -= 1;
          container.setItem(i, item);
        } else {
          container.setItem(i, undefined);
        }
        p.playSound("random.break");
        p.onScreenDisplay.setActionBar("§cYour Void Phantom has shattered!");
        return true;
      }
    }
  } catch (err) {}
  return false;
}

/* Offhand gate: every effect and trigger reads the live offhand slot */
function offhandItem(p) {
  const eq = p.getComponent("minecraft:equippable");
  return eq ? eq.getEquipment(EquipmentSlot.Offhand) : undefined;
}

function damageOffhand(p, amount = 1) {
  try {
    const eq = p.getComponent("minecraft:equippable");
    if (!eq) return;
    const item = eq.getEquipment(EquipmentSlot.Offhand);
    if (!item) return;
    const dur = item.getComponent("minecraft:durability");
    if (!dur) return;
    
    // Honor Unbreaking (script damage bypasses vanilla handling): 1/(level+1) chance to take damage.
    try {
      const ench = item.getComponent("minecraft:enchantable");
      const ub = ench && ench.getEnchantment("unbreaking");
      if (ub && ub.level > 0 && Math.random() >= 1 / (ub.level + 1)) return;
    } catch (err) {}
    if (dur.damage + amount >= dur.maxDurability) {
      // Break the item
      eq.setEquipment(EquipmentSlot.Offhand, undefined);
      p.playSound("random.break");
      p.onScreenDisplay.setActionBar("§cYour Grimoire has shattered!");
      particle(p.dimension, "minecraft:item_break", { x: p.location.x, y: p.location.y + 1, z: p.location.z });
    } else {
      dur.damage += amount;
      eq.setEquipment(EquipmentSlot.Offhand, item);
    }
  } catch (err) {}
}

function relicOf(p) {
  const item = offhandItem(p);
  return item ? RELICS[item.typeId] : undefined;
}

function loreFor(cfg) {
  return ["§8Offhand Relic", ...cfg.info.map((t) => "§7• " + t)];

function ensureLore(p) {
  const eq = p.getComponent("minecraft:equippable");
  const off = eq && eq.getEquipment(EquipmentSlot.Offhand);
  if (off && RELICS[off.typeId] && off.getLore().length === 0) {
    off.setLore(loreFor(RELICS[off.typeId]));
    eq.setEquipment(EquipmentSlot.Offhand, off);
  }
  const c = p.getComponent("minecraft:inventory").container;
  for (let i = 0; i < c.size; i++) {
    const item = c.getItem(i);
    if (item && RELICS[item.typeId] && item.getLore().length === 0) {
      item.setLore(loreFor(RELICS[item.typeId]));
      c.setItem(i, item);
    }
  }
}

## Durability, Enchanting & World Generation

### Durability, Enchanting, and Repairing
- **100 durability points**: Each Grimoire possesses a durability health bar.
- **Event-Only Wear**: Passive stat buffs (such as Water Breathing, Speed, or Haste) do **not** drain durability. Durability is consumed strictly upon active mechanical execution:
  - *Combat triggers* (e.g. Phoenix burn on hit/hurt, Vampire life-steal, Iron Golem absorption & launch, Hydra poison, Lich wither): 1 durability per event.
  - *Environmental abilities* (e.g. Druid crop boost, Frost Weaver water freeze, Stormcaller lightning strike, Mirror Trickster blink): 1 to 2 durability per event.
  - *Emergency Lifesavers* (Void Phantom): single use; the relic is consumed after a safe touchdown.
- **Enchanting Table & Anvil Compatible**:
  - Can be enchanted via the Enchanting Table or Anvil with enchantments like **Mending**, **Efficiency**, and **Unbreaking**.
  - **Mending** restores tome durability automatically as you collect experience orbs.
- **Anvil Repairing**:
  - Combine two of the same tomes on an Anvil to restore their remaining durability.
  - Can also be repaired on an Anvil using standard **Books** at a strict **1:1 ratio** (1 book restores 1 durability point).
- **Shatter**: If durability reaches 0 without being repaired or protected, the tome shatters with an item break sound and particles and is removed from your equipment.

### World Chest Distribution
The pack includes exploration loot entries for Ancient Relics. Their intended locations are:
- **Ancient Cities**: Chance for **Void Phantom** and **Mirror Trickster**.
- **Bastion Remnants (Treasure)**: **Infernal Phoenix** and **Necrotic Lich**.
- **End Cities**: **Cataclysm Dragon**.
- **Stronghold Libraries**: **Arcane Scholar** and **Cursed Alchemist**.
- **Shipwreck Treasuries**: **Abyssal Tide**.
- **Monster Rooms / Simple Dungeons**: **Blood Vampire** and **Deep Delver**.
- **Enchanted Golden Apples**: No longer appear in Ancient City and Bastion Treasure chests; the tomes take their place.

<h2 style="font-size:17px;text-align:left"><span style="background-color:#15803d;color:#ffffff;border-radius:20px">&nbsp;&nbsp;Durability, Enchanting &amp; World Generation&nbsp;&nbsp;</span></h2>
<div class="spoiler" title="Durability, Enchanting &amp; World Generation" data-title="Durability, Enchanting &amp; World Generation">
<h3>Durability, Enchanting, and Repairing</h3><ul><li><strong>100 durability points</strong>: Each Grimoire possesses a durability health bar.</li><li><strong>Event-Only Wear</strong>: Passive stat buffs (such as Water Breathing, Speed, or Haste) do <strong>not</strong> drain durability. Durability is consumed strictly upon active mechanical execution:</li><li><em>Combat triggers</em> (e.g. Phoenix burn on hit/hurt, Vampire life-steal, Iron Golem absorption &amp; launch, Hydra poison, Lich wither): 1 durability per event.</li><li><em>Environmental abilities</em> (e.g. Druid crop boost, Frost Weaver water freeze, Stormcaller lightning strike, Mirror Trickster blink): 1 to 2 durability per event.</li><li><em>Emergency Lifesavers</em> (Void Phantom): single use; the relic is consumed after a safe touchdown.</li><li><strong>Enchanting Table &amp; Anvil Compatible</strong>:</li><li>Can be enchanted via the Enchanting Table or Anvil with enchantments like <strong>Mending</strong>, <strong>Efficiency</strong>, and <strong>Unbreaking</strong>.</li><li><strong>Mending</strong> restores tome durability automatically as you collect experience orbs.</li><li><strong>Anvil Repairing</strong>:</li><li>Combine two of the same tomes on an Anvil to restore their remaining durability.</li><li>Can also be repaired on an Anvil using standard <strong>Books</strong> at a strict <strong>1:1 ratio</strong> (1 book restores 1 durability point).</li><li><strong>Shatter</strong>: If durability reaches 0 without being repaired or protected, the tome shatters with an item break sound and particles and is removed from your equipment.</li></ul><h3>World Chest Distribution</h3><p>The pack includes exploration loot entries for Ancient Relics. Their intended locations are:</p><ul><li><strong>Ancient Cities</strong>: Chance for <strong>Void Phantom</strong> and <strong>Mirror Trickster</strong>.</li><li><strong>Bastion Remnants (Treasure)</strong>: <strong>Infernal Phoenix</strong> and <strong>Necrotic Lich</strong>.</li><li><strong>End Cities</strong>: <strong>Cataclysm Dragon</strong>.</li><li><strong>Stronghold Libraries</strong>: <strong>Arcane Scholar</strong> and <strong>Cursed Alchemist</strong>.</li><li><strong>Shipwreck Treasuries</strong>: <strong>Abyssal Tide</strong>.</li><li><strong>Monster Rooms / Simple Dungeons</strong>: <strong>Blood Vampire</strong> and <strong>Deep Delver</strong>.</li><li><strong>Enchanted Golden Apples</strong>: No longer appear in Ancient City and Bastion Treasure chests; the tomes take their place.</li></ul></div>
