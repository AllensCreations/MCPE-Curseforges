```
import { ItemStack } from "@minecraft/server";

export const QUIVER_IDS = ["aln:quiver", "aln:quiver_half", "aln:quiver_full"];
export const CAPACITY = 256;
export const ARROW_ID = "minecraft:arrow";
const DATA_KEY = "aln_data";

export function isQuiver(item) {
  return !!item && QUIVER_IDS.includes(item.typeId);
}

export function tierFor(count) {
  if (count <= 0) return QUIVER_IDS[0];
  if (count < 128) return QUIVER_IDS[1];
  return QUIVER_IDS[2];
}

export function readCount(item) {
  try {
    const raw = item.getDynamicProperty(DATA_KEY);
    if (typeof raw !== "string") return 0;
    const n = Math.floor(Number(JSON.parse(raw)?.items?.[ARROW_ID]));
    return Number.isFinite(n) ? Math.max(0, Math.min(CAPACITY, n)) : 0;
  } catch {
    return 0;
  }
}

export function buildQuiver(count, source) {
  const n = Math.max(0, Math.min(CAPACITY, count));
  const stack = new ItemStack(tierFor(n), 1);
  stack.setDynamicProperty(DATA_KEY, JSON.stringify({ v: 1, items: { [ARROW_ID]: n } }));
  stack.setLore([`§7Capacity: §e${n} / ${CAPACITY}`]);
  if (source && source.nameTag) stack.nameTag = source.nameTag;
  return stack;
}

export function isPlainArrow(item) {
  if (!item || item.typeId !== ARROW_ID) return false;
  try {
    if (item.getComponent("minecraft:potion")) return false;
  } catch {}
  const key = String(item.localizationKey || "").toLowerCase();
  return !(key.includes("tipped") || key.includes("effect"));
}

export function findQuiverSlot(container) {
  for (let i = 0; i < container.size; i++) {
    if (isQuiver(container.getItem(i))) return i;
  }
  return -1;
}

export function setQuiverCount(container, slot, count) {
  const old = container.getItem(slot);
  if (!isQuiver(old)) return false;
  container.setItem(slot, buildQuiver(count, old));
  return true;
}

export function countPlainArrows(container) {
  let total = 0;
  for (let i = 0; i < container.size; i++) {
    const it = container.getItem(i);
    if (isPlainArrow(it)) total += it.amount;
  }
  return total;
}
```

## FILE: MCPEDL/Addons/ALN-Quiver/behavior_pack/scripts/quiver_feed.js (68 lines, 1993 bytes)

```
import { world, system, ItemStack, GameMode } from "@minecraft/server";
import { ARROW_ID, findQuiverSlot, readCount, setQuiverCount, countPlainArrows, isPlainArrow, CAPACITY } from "./quiver_data.js";

const state = new Map();

function removeOnePlainArrow(c) {
  for (let i = 0; i < c.size; i++) {
    const it = c.getItem(i);
    if (!isPlainArrow(it)) continue;
    if (it.amount <= 1) c.setItem(i, undefined);
    else {
      it.amount -= 1;
      c.setItem(i, it);
    }
    return true;
  }
  return false;
}

function giveBack(c, id) {
  const entry = state.get(id);
  state.delete(id);
  if (!entry || countPlainArrows(c) < entry.expected) return;
  const slot = findQuiverSlot(c);
  if (slot < 0) return;
  const count = readCount(c.getItem(slot));
  if (count >= CAPACITY || !removeOnePlainArrow(c)) return;
  setQuiverCount(c, slot, count + 1);
}

function step(player) {
  const c = player.getComponent("inventory")?.container;
  if (!c) return;
  const held = c.getItem(player.selectedSlotIndex);
  const armed = !!held && (held.typeId === "minecraft:bow" || held.typeId === "minecraft:crossbow");
  if (!armed) {
    if (state.has(player.id)) giveBack(c, player.id);
    return;
  }
  if (player.getGameMode() === GameMode.Creative) return;
  const slot = findQuiverSlot(c);
  if (slot < 0) {
    state.delete(player.id);
    return;
  }
  const have = countPlainArrows(c);
```

```
import { ActionFormData } from "@minecraft/server-ui";
import { CAPACITY, ARROW_ID, isQuiver, readCount, isPlainArrow, setQuiverCount } from "./quiver_data.js";

const VERSION = "1.0.0";
const TITLE = `§l§6@ALNStudios §eQuiver§r (v${VERSION})`;
const open = new Set();

function bar(player, text) {
  try {
    player.onScreenDisplay.setActionBar(text);
  } catch {}
}

function container(player) {
  return player.getComponent("inventory")?.container;
}

function liveCount(player, slot) {
  const c = container(player);
  const item = c?.getItem(slot);
  return isQuiver(item) ? readCount(item) : -1;
}

function deposit(player, slot, limit) {
  const c = container(player);
  const count = liveCount(player, slot);
  if (!c || count < 0) return 0;
  let room = Math.min(limit, CAPACITY - count);
  let moved = 0;
  for (let i = 0; i < c.size && room > 0; i++) {
    if (i === slot) continue;
    const it = c.getItem(i);
    if (!isPlainArrow(it)) continue;
    const take = Math.min(it.amount, room);
    if (take === it.amount) c.setItem(i, undefined);
    else {
      it.amount -= take;
      c.setItem(i, it);
    }
    room -= take;
    moved += take;
  }
  if (moved > 0) setQuiverCount(c, slot, count + moved);
  return moved;
}

function withdraw(player, slot, limit) {
  const c = container(player);
  const count = liveCount(player, slot);
  if (!c || count <= 0) return 0;
  let left = Math.min(limit, count);
  const total = left;
  while (left > 0) {
    const n = Math.min(64, left);
    const rest = c.addItem(new ItemStack(ARROW_ID, n));
    if (rest) player.dimension.spawnItem(rest, player.location);
    left -= n;
  }
  setQuiverCount(c, slot, count - total);
  return total;
}

async function manage(player, slot) {
  for (;;) {
    const count = liveCount(player, slot);
    if (count < 0) return bar(player, "§cThe quiver is no longer in that slot.");
    const form = new ActionFormData()
      .title(TITLE)
      .body(`§lStored: ${count} / ${CAPACITY}§r\n§lOnly standard arrows can be stored.§r`)
      .button("§lDeposit All Arrows§r", "textures/items/arrow")
      .button("§lDeposit 64 Arrows§r", "textures/items/arrow")
      .button("§lWithdraw 64 Arrows§r", "textures/items/arrow")
      .button("§lWithdraw All Arrows§r", "textures/items/arrow")
      .button("§lBack§r", "textures/items/door_wood");
    const res = await form.show(player);
    if (res.canceled || res.selection === 4) return res.canceled ? "exit" : "back";
    let n = 0;
    if (res.selection === 0) n = deposit(player, slot, CAPACITY);
    if (res.selection === 1) n = deposit(player, slot, 64);
    if (res.selection === 2) n = -withdraw(player, slot, 64);
    if (res.selection === 3) n = -withdraw(player, slot, CAPACITY);
    if (n > 0) bar(player, `§aDeposited ${n} arrows`);
    else if (n < 0) bar(player, `§aWithdrew ${-n} arrows`);
    else bar(player, res.selection < 2 ? "§eNo standard arrows to deposit or quiver is full" : "§eThe quiver is empty");
  }
}

  const form = new ActionFormData()

    .body(`§lCreator: ALNStudios§r\n§lQuiver v${VERSION}§r\n§lHolds up to ${CAPACITY} arrows and feeds them to bows and crossbows.§r\n§lCurseForge: ALNStudios (allenscreations)§r\n§lMCPEDL: alnstudios§r\n§lNot affiliated with Mojang or Microsoft.§r`)
    .button("§lBack§r", "textures/items/door_wood");
  await form.show(player);
}

async function mainMenu(player, slot) {
  for (;;) {
    const count = liveCount(player, slot);
    if (count < 0) return;
    const form = new ActionFormData()
      .title(TITLE)
      .body(`§lStored: ${count} / ${CAPACITY}§r`)
      .button("§lManage Arrows§r", "textures/items/quiver")
      .button("§lAbout§r", "textures/items/name_tag")
      .button("§lExit§r", "textures/items/door_wood");
    const res = await form.show(player);
    if (res.canceled || res.selection === 2) return;
    if (res.selection === 0 && (await manage(player, slot)) === "exit") return;
    if (res.selection === 1) await about(player);
  }
}

export function registerStorage() {
  world.afterEvents.itemUse.subscribe((ev) => {
    const player = ev.source;
    if (!isQuiver(ev.itemStack) || open.has(player.id)) return;
    const slot = player.selectedSlotIndex;
    open.add(player.id);
    system.run(() => {
      mainMenu(player, slot)
        .catch(() => {})
        .finally(() => open.delete(player.id));
    });
  });
}
```

```
# ALN Quiver — Arrow-Only Storage With Auto-Feed

**By ALNStudios**

[![Minecraft Bedrock](https://img.shields.io/badge/Minecraft%20Bedrock-Stable%20Release-brightgreen?style=flat-square&logo=minecraft)](https://github.com/AllensCreations/MCPE-Curseforges)

**The Quiver holds up to 256 arrows and feeds your Bow or Crossbow one arrow at a time.**

> This add-on adds the Quiver, a craftable item that only accepts standard arrows.

## Overview

Craft a Quiver from Leather and String, then use it to move arrows in and out. While you hold a Bow or Crossbow with the Quiver in your inventory, it lends you an arrow so you can keep shooting.

## Key Features

### Arrow-Only Storage
- **What Happens**: The Quiver stores up to 256 standard arrows and refuses every other item.
- **Good To Know**: Tipped arrows are not stored, because their effect cannot be restored from the Quiver.

### Auto-Feed
- **What Happens**: With a Bow or Crossbow in hand and no loose arrows, one arrow is taken from the Quiver and placed in your inventory.
- **Good To Know**: Putting the weapon away returns an unused arrow to the Quiver.

### Fill Icons
- **What Happens**: The Quiver shows an empty icon at 0 arrows, a half icon at 1 to 127, and a full icon at 128 or more.
- **Good To Know**: All three icons currently use the same Quiver picture, so the change is not yet visible.

### Crafting
- **What Happens**: Combine 3 Leather and 2 String in any arrangement on a crafting table.

## How to Use

1. Craft a Quiver and hold it.
2. Use it, choose Manage Arrows, and deposit your arrows.
3. Hold a Bow or Crossbow with the Quiver in your inventory and shoot.

## Common Questions

- **Can I put other items in the Quiver?**: No. Only standard arrows are accepted.
```
