## FILE: MCPEDL/Addons/Magnetic-Satchel/behavior_pack/scripts/main.js (147 lines, 2883 bytes)

```
import { world, system, EquipmentSlot } from "@minecraft/server";

const SATCHEL_ID = "custom:magnetic_satchel";

const CONFIG = {
  interval: 2,
  radius: 8,
  chestOffset: 0.5,
  minDistance: 1.0,
  baseStrength: 0.04,
  distanceStrength: 0.01,
  maxStrength: 0.12,
  maxItemsPerPlayer: 48,
  particleEvery: 3,
  maxParticlesPerPlayer: 4
};

let cycle = 0;

function hasSatchelEquipped(player) {
  try {
    const equippable = player.getComponent("minecraft:equippable");
    if (!equippable) return false;

    const mainhand = equippable.getEquipment(EquipmentSlot.Mainhand);
    const offhand = equippable.getEquipment(EquipmentSlot.Offhand);

    return (
      mainhand?.typeId === SATCHEL_ID ||
      offhand?.typeId === SATCHEL_ID
    );
  } catch {
    return false;
  }
}

function pullItems(player, withParticles) {
  let dimension;
  let origin;

  try {
    dimension = player.dimension;
    origin = player.location;
  } catch {
    return;
  }

  const target = {
    x: origin.x,
    y: origin.y + CONFIG.chestOffset,
    z: origin.z
  };

  let items;

  try {
    items = dimension.getEntities({
      location: origin,
      maxDistance: CONFIG.radius,
      type: "minecraft:item"
    });
  } catch {
    return;
  }

  let pulled = 0;
  let particles = 0;

  for (const item of items) {
    if (pulled >= CONFIG.maxItemsPerPlayer) break;

    try {
      const loc = item.location;

      const dx = target.x - loc.x;
      const dy = target.y - loc.y;
      const dz = target.z - loc.z;

      const distanceSquared =
        dx * dx +
        dy * dy +
        dz * dz;

      if (
        distanceSquared <
        CONFIG.minDistance * CONFIG.minDistance
      ) {
        continue;
      }

      const distance = Math.sqrt(distanceSquared);

      if (distance === 0) continue;

      const strength = Math.min(
        CONFIG.maxStrength,
        CONFIG.baseStrength +
          distance * CONFIG.distanceStrength
      );

      item.applyImpulse({
        x: (dx / distance) * strength,
        y: (dy / distance) * strength,
        z: (dz / distance) * strength
      });

      pulled++;

      if (
        withParticles &&
        particles < CONFIG.maxParticlesPerPlayer
      ) {
        try {
          dimension.spawnParticle(
            "minecraft:basic_smoke_particle",
            loc
          );

          particles++;
        } catch {
          // Particle effects are optional.
        }
      }
    } catch {
      // Skip invalid or despawned items.
      continue;
    }
  }
}

system.runInterval(() => {
  cycle++;

  const withParticles =
    cycle % CONFIG.particleEvery === 0;

  for (const player of world.getAllPlayers()) {
    try {
      if (!hasSatchelEquipped(player)) continue;

      pullItems(player, withParticles);
    } catch {
      // Skip players that are temporarily unavailable.
    }
  }
}, CONFIG.interval);
```

## Overview

Collect nearby drops more conveniently while mining, exploring, or fighting mobs.

**Magnetic Satchel** introduces a crafted utility satchel that equips into your offhand slot.  it attracts nearby dropped items within an 8-block radius and draws them toward you.

Leave your main hand free to hold your pickaxe, weapon, or torches while the satchel gathers all dropped resources directly to your inventory.

---

## Key Features

### Offhand Equippable
- **Description**: Equips in the offhand slot, keeping your main hand free for tools and weapons.

### 8-Block Vacuum Radius
- **Description**: Detects and pulls all nearby dropped item entities within an 8-block radius.

### Item Attraction
- **Description**: Uses smooth gentle movement toward the player to glide items toward your chest level.

### Particle Trails
- **Description**: Emits subtle portal particle trails behind moving items as they glide inward.

### Quick Toggle Switch
- **Description**: Sneak and use the satchel while holding it in your main hand to instantly toggle magnetism ON or OFF.

### Action Bar Feedback
- **Description**: Clear action bar HUD indicator displays an on-screen message indicating whether magnetism is active.

### Flexible Recipes
- **Description**: Craftable with either an Ender Pearl or a Compass depending on your survival stage.

## How It Works

### Equip in Offhand
- **Control / Trigger**: Place into offhand shield slot
- **Outcome**: Automatically activates the vacuum pull for all items within 8 blocks.

### Toggle Active/Disabled
- **Control / Trigger**: Sneak and use the satchel while holding it in your main hand
- **Outcome**: Toggles magnetic pull on/off without needing to unequip.

### Automatic Collection
- **Control / Trigger**: Stand within 8 blocks of dropped items
- **Outcome**: Items are drawn toward your player until normal item pickup occurs.

### Safe Inventory Handling
- **Control / Trigger**: Normal inventory behavior
- **Outcome**: When full, items remain nearby when your inventory is full.
