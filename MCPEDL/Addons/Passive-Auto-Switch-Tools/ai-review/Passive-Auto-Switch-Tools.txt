    score += enchants.sharpness;

      blockId === "glass" ||
      blockId.endsWith("_glass") ||
      blockId.endsWith("glass_pane") ||

      blockId === "bookshelf" ||

function switchPlayerSlot1Tool(player, category, blockTypeId = "") {
  if (!player || player.selectedSlotIndex !== CONFIG.slot) {
    return false;
  }

  const container = getInventory(player);

  if (!container) return false;

  const blockId = normalizeId(blockTypeId);

  const currentScore = scoreItem(
    container.getItem(CONFIG.slot),
    category,
    CONFIG.slot,
    blockId
  );

  let bestSlot = -1;
  let bestScore = -Infinity;

  for (let slot = 1; slot < container.size; slot++) {
    const item = container.getItem(slot);

    if (!item) continue;

    const score = scoreItem(item, category, slot, blockId);

    if (score > bestScore) {
      bestScore = score;
      bestSlot = slot;
    }
  }

  if (bestSlot === -1 || bestScore === -Infinity) {
    return false;
  }

  if (
    currentScore !== -Infinity &&
    bestScore < currentScore + CONFIG.switchMargin
  ) {
    return false;
  }

  try {
    container.swapItems(bestSlot, CONFIG.slot, container);
    return true;
  } catch {
    return false;
  }
}

    if (player.selectedSlotIndex !== CONFIG.slot) return;

    const block = event.hitBlock;

    if (!block) return;

    if (!category) return;

    const lastCombat = recentCombat.get(player.id) ?? -Infinity;

    if (now - lastCombat < CONFIG.combatCooldownTicks) {
      return;

      key: blockKey(block),
      lastInteractionTick: now,
    });

    if (!switchPlayerSlot1Tool(player, category, block.typeId)) {
      scheduleSwapRetry(player, category, block.typeId);

// Cobwebs: watch for interaction, not merely looking at them.
system.runInterval(() => {
  const now = system.currentTick;

  for (const player of world.getAllPlayers()) {
    try {
      if (player.selectedSlotIndex !== CONFIG.slot) {
        webInteraction.delete(player.id);
        continue;
      }

      const lastCombat = recentCombat.get(player.id) ?? -Infinity;

      if (now - lastCombat < CONFIG.webCombatCooldown) {
        continue;
      }

      const block = getTargetBlock(player);

      if (!isWeb(block)) {
        webInteraction.delete(player.id);
        continue;
      }

      const key = blockKey(block);
      const state = webInteraction.get(player.id);

      if (!state || state.key !== key) {
        webInteraction.set(player.id, {
          key,
          lastInteractionTick: now,
          initialTick: now,
        });

        continue;
      }

      const elapsed = now - state.initialTick;

      if (elapsed < CONFIG.webInteractionWindow) {
        continue;
      }

      // Aiming alone is not sufficient: require a recent block interaction.
      if (
        now - state.lastInteractionTick >
        CONFIG.webInteractionWindow
      ) {
        webInteraction.delete(player.id);
        continue;
      }

      const inventory = getInventory(player);
      const held = inventory?.getItem(CONFIG.slot);

      if (held && normalizeId(held.typeId).includes("shears")) {
        continue;
      }

      switchPlayerSlot1Tool(
        player,
        ToolCategory.WEB,
        block.typeId
      );

      webInteraction.set(player.id, {
        key,
        initialTick: now,
        lastInteractionTick: now,
      });
    } catch {}
  }
}, CONFIG.webCheckInterval);

// Mob combat: switch to the best sword or axe.
world.afterEvents.entityHitEntity.subscribe((event) => {
  try {
    const player = event.damagingEntity;

    if (!player || player.typeId !== "minecraft:player") return;
    if (player.selectedSlotIndex !== CONFIG.slot) return;
    if (!isCombatTarget(event.hitEntity)) return;

    recentCombat.set(player.id, system.currentTick);

    clearPendingSwap(player.id);
    webInteraction.delete(player.id);

    switchPlayerSlot1Tool(player, ToolCategory.WEAPON);
  } catch {}
});

world.afterEvents.playerLeave.subscribe((event) => {
  recentCombat.delete(event.playerId);
  webInteraction.delete(event.playerId);
  clearPendingSwap(event.playerId);
});

console.warn(
  `[Passive Auto-Switch Tools v${VERSION}] Initialized.`
);

## Overview

Passive Auto-Switch Tools is an survival add-on that automates tool and weapon selection in Minecraft Bedrock Edition. When you select hotbar slot 1, the add-on activates when you **start to break/hit a block** (avoiding annoying swaps when merely looking around), and **when attacking a mob** by automatically equipping your highest-tier axe or sword.

Hotbar slots 2 through 9 remain completely unmanaged, ensuring uninterrupted building, offhand usage, and consumable item management.

---

## Mechanics and Operation

### Hotbar Slot 1 Activation

Automatic selection is active only while hotbar slot 1 is selected:
- Hotbar Slot 1: Active auto-switch slot. Striking blocks triggers tool selection; attacking mobs triggers weapon selection.
- Hotbar Slots 2 through 9: Standard manual operation. the add-on does not swap items while these slots are selected.

### Block-to-Tool & Combat Matching

- **Pickaxe**: Stone, deepslate, ores, obsidian, terracotta, cobblestone, and metal blocks.
- **Axe**: Logs, wood, planks, fences, fence gates, chests, crafting tables, melons, and pumpkins.
- **Shovel**: Dirt, grass, sand, gravel, clay, mud, soul sand, snow, and concrete powder.
- **Shears / Sword**: Leaves, wool, cobwebs, and vines.
- **Hoe**: Hay bales, target blocks, sponges, sculk, moss, and nether wart blocks.
- **Combat (Sword or Axe)**: When attacking any mob or combat entity, automatically equips the best available sword or axe.

### Tier Priority Hierarchy

When multiple matching tools or weapons are present in inventory, the system selects the highest tier available:

**Netherite > Diamond > Iron > Stone > Golden / Wooden**

If items of equal tier exist across hotbar and main inventory, the item already present in the hotbar is given priority. For combat, swords have a slight damage preference in Bedrock, while existing axes or swords already in Slot 1 are retained.

### Item Integrity and Preservation

The add-on aims to preserve tool details when switching items:
- Durability damage values remain exact.
- Custom anvil names and lore strings are preserved.
- Enchantments (Efficiency, Fortune, Silk Touch, Unbreaking, Mending, Sharpness, etc.) are preserved intact.
- Active tool locking: If the currently held tool or weapon is already valid for the target block or combat, no redundant inventory swaps occur.
- Tools switch when you begin breaking a block, not merely when you look at it.
