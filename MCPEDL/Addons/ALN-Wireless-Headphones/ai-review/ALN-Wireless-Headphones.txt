  const safe = Math.max(0, Math.floor(totalSeconds));

import { world, system, EquipmentSlot } from "@minecraft/server";
import {
  HEADPHONES_ID,
  PAIR_KEY,
  JUKEBOX_ID,
  PREFIX,
  formatTime,
  readJukebox,
  resetSession
} from "./headphone_common.js";

const STREAM_INTERVAL_TICKS = 10;
const listening = new Set();

function parsePairing(item) {
  const raw = item.getDynamicProperty(PAIR_KEY);
  if (typeof raw !== "string") {
    return undefined;
  }
  try {
    const data = JSON.parse(raw);
    if (
      typeof data.x === "number" &&
      typeof data.y === "number" &&
      typeof data.z === "number" &&
      typeof data.dimension === "string"
    ) {
      return data;
    }
  } catch (error) {
    return undefined;
  }
  return undefined;
}

function renderFeed(item) {
  const pairing = parsePairing(item);
  if (!pairing) {
    return PREFIX + "\u00a77Not paired. Sneak and use on a Jukebox with a disc.";
  }
  let dimension;
  try {
    dimension = world.getDimension(pairing.dimension);
  } catch (error) {
    return PREFIX + "\u00a7cPaired Jukebox dimension is unavailable.";
  }
  const location = { x: pairing.x, y: pairing.y, z: pairing.z };
  if (!dimension.isChunkLoaded(location)) {
    resetSession(pairing.dimension, pairing.x, pairing.y, pairing.z);
    return PREFIX + "\u00a77Signal lost. Jukebox chunk is not loaded.";
  }
  const block = dimension.getBlock(location);
  if (!block || block.typeId !== JUKEBOX_ID) {
    return PREFIX + "\u00a7cPaired Jukebox no longer exists.";
  }
  const status = readJukebox(block, pairing.dimension);
  if (status.state === "empty") {
    return PREFIX + "\u00a77Jukebox has no disc.";
  }
  if (status.state === "stopped") {
    return PREFIX + "\u00a77Jukebox is not playing. \u00a78| \u00a7e" + status.info.name;
  }
  if (status.state !== "playing") {
    return PREFIX + "\u00a7cJukebox data unavailable.";
  }
  const total = status.info.totalSeconds === undefined ? "--:--" : formatTime(status.info.totalSeconds);
  return (
    PREFIX +
    "\u00a7fNow Playing: \u00a7e" +
    status.info.name +
    " \u00a78| \u00a7b[ " +
    formatTime(status.elapsedSeconds) +
    " / " +
    total +
    " ]"
  );
}

system.runInterval(() => {
  const present = new Set();
  for (const player of world.getAllPlayers()) {
    present.add(player.id);
    try {
      const equippable = player.getComponent("minecraft:equippable");
      const head = equippable ? equippable.getEquipment(EquipmentSlot.Head) : undefined;
      if (head && head.typeId === HEADPHONES_ID) {
        listening.add(player.id);
        player.onScreenDisplay.setActionBar(renderFeed(head));
      } else if (listening.has(player.id)) {
        listening.delete(player.id);
        player.onScreenDisplay.setActionBar("");
      }
    } catch (error) {
      continue;
    }
  }
  for (const id of listening) {
    if (!present.has(id)) {
      listening.delete(id);
    }
  }
}, STREAM_INTERVAL_TICKS);

**Pair Wireless Headphones to a Jukebox once, then wear them to see its track name and time on your Action Bar.**

> This pack adds one helmet, Wireless Headphones, that links to a Jukebox holding a music disc. Players who build far from their music room can still see what is playing and how far along it is.

## Overview

Craft the Wireless Headphones, hold them, and sneak while using a Jukebox that has a music disc inside. Put the headphones on your head and the Action Bar shows the track and a timer for as long as you wear them. Take them off and the display clears.

## Key Features

### Wireless Headphones
- **What Happens**: A helmet that stacks to 1 and goes in the head slot. It gives no armor protection.
- **How It Works**: Craft it with 2 Iron Ingots, 1 Redstone Dust and 2 Leather. Row 1 is Iron Ingot, Redstone Dust, Iron Ingot. Row 2 is Leather, empty, Leather.
- **Good To Know**: The recipe is unlocked from the start. The helmet has no worn model, so it appears only in the inventory and hand.

### Jukebox Pairing
- **What Happens**: Sneak and use the headphones on a Jukebox that holds a music disc. A chime plays and the Action Bar confirms the pairing.
- **How It Works**: Hold the headphones, crouch, then use them on the Jukebox. The paired position is saved on that item and shown in its lore.
- **Good To Know**: A Jukebox with no disc is refused. Pairing again overwrites the old link.

### Live Action Bar Player
- **What Happens**: While worn, the Action Bar shows the disc as artist and title, plus elapsed and total time.
- **How It Works**: A paired Jukebox playing cat shows the track name with a time such as 0:42 out of 3:05.
- **Good To Know**: The display needs the Jukebox area to be loaded. If it is not, the bar says the signal is lost. Total times are listed for the 22 vanilla discs; other discs show no total.

## How to Use

1. Craft the Wireless Headphones at a Crafting Table.
2. Hold them, sneak, and use them on a Jukebox that has a music disc.
3. Wear them in the helmet slot and watch the Action Bar for the track and timer.

## Common Questions

- **Does the timer start from the beginning of the song?**: The timer starts when the Jukebox is seen playing. A disc started by a player counts from about the moment it began, and a disc that was already playing before pairing starts at 0:00.
- **What happens when I take the headphones off?**: The Action Bar feed is cleared right away. Wear them again to resume the display.
