# Changelog — Alphabet Scavenger Hunt

## [v2] - 2026-11-14

### Added
- **Victory Celebration**: Finishing all 20 rounds now shows a Congratulations title and a fireworks display, then the Controller comes back so you can play again.

### Improved & Balanced
- **Smoother Performance**: The challenge now updates once per second instead of every tick, which greatly reduces lag.
- **Co-op Eliminations**: A player who dies is eliminated, and the challenge only ends when every player is out.
- **Fair Matching**: Renaming an item in an anvil no longer counts toward a round.

### Fixed
- **Round One Start**: The challenge no longer stops by itself when the first round begins.

```
import { world, system, ItemStack, EntityComponentTypes, GameMode } from "@minecraft/server";
import { ActionFormData, FormCancelationReason } from "@minecraft/server-ui";

const CONTROLLER_ID = "aln:alphabet_scavenger_hunt_controller";
const STATE_KEY = "game_state";
const OPERATOR_LEVEL = 2;

let config = null;
let registered = false;

const getState = () => {
  const v = world.getDynamicProperty(STATE_KEY);
  return typeof v === "string" ? v : "idle";
};
const setState = (s) => world.setDynamicProperty(STATE_KEY, s);
const isCreative = (player) => player.getGameMode() === GameMode.Creative;
const isHost = (player) => (typeof player.isOp === "function" ? player.isOp() : player.playerPermissionLevel === OPERATOR_LEVEL);
const containerOf = (player) => player.getComponent(EntityComponentTypes.Inventory)?.container;

function hasController(player) {
  const inv = containerOf(player);
  if (!inv) return false;
  for (let i = 0; i < inv.size; i++) {
    if (inv.getItem(i)?.typeId === CONTROLLER_ID) return true;
  }
  return false;
}

function giveController(player) {
  if (hasController(player)) return;
  containerOf(player)?.addItem(new ItemStack(CONTROLLER_ID, 1));
}

function consumeController(player) {
  if (isCreative(player)) return;
  const inv = containerOf(player);
  if (!inv) return;
  for (let i = 0; i < inv.size; i++) {
    if (inv.getItem(i)?.typeId === CONTROLLER_ID) {
      inv.setItem(i, undefined);
      player.playSound("random.break");
      player.sendMessage("The Controller broke.");
      return;
    }
  }
}

function runStart(player) {
  if (getState() === "running") {
    player.sendMessage("Already running.");
    return;
  }
  setState("running");
  if (config.consumeOnStart) consumeController(player);
  try {
    config.onStart(player);
  } catch (e) {
    setState("idle");
    giveController(player);
    player.sendMessage("The game could not start.");
  }
}

export function endGame() {
  if (getState() !== "running") return;
  setState("ended");
  try {
    config?.onEnd();
  } catch (e) {
    world.sendMessage("The game ended with an error.");
  }
  for (const p of world.getAllPlayers()) if (isHost(p)) giveController(p);
}

async function showHowToPlay(player) {
  const form = new ActionFormData().title("How to play").body(config.howToPlay).button("Back");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (!res.canceled && res.selection === 0) openController(player);
}

async function confirmEnd(player) {
  if (getState() !== "running") {
    player.sendMessage("No game is running.");
    return;
  }
  const form = new ActionFormData()
    .title("End game?")
    .body("This stops the game that is running now.")
    .button("End Game")
    .button("Back");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (res.canceled) return;
  if (res.selection === 0) {
    if (getState() !== "running") {
      player.sendMessage("The game already ended.");
      return;
    }
    world.sendMessage(`${player.name} ended the game.`);
    endGame();
  } else {
    openController(player);
  }
}

async function openController(player, attempt = 0) {
  const running = getState() === "running";
  const note = running
    ? "End Game stops the current game."
    : config.consumeOnStart
      ? (isCreative(player) ? "Creative: the Controller is kept." : "Start uses up the Controller.")
      : "The Controller is kept.";
  const form = new ActionFormData()
    .title(config.title)
    .body(`${config.gameInfo}\n\nState: ${getState()}\n${note}`)
    .button(running ? "End Game" : "Start")
    .button("How to play");
  let res;
  try {
    res = await form.show(player);
  } catch (e) {
    return;
  }
  if (res.canceled) {
    if (res.cancelationReason === FormCancelationReason.UserBusy && attempt < 5) {
      system.runTimeout(() => openController(player, attempt + 1), 10);
    }
    return;
  }
  if (res.selection === 0) {
    if (running) confirmEnd(player);
    else runStart(player);
  } else if (res.selection === 1) {
    showHowToPlay(player);
  }
}

export function registerController(options) {
  if (registered) return;
  registered = true;
  config = {
    title: options?.title ?? "Controller",
    gameInfo: options?.gameInfo ?? "",
    howToPlay: options?.howToPlay ?? "",
    consumeOnStart: options?.consumeOnStart === true,
    onStart: options?.onStart ?? (() => {}),
    onEnd: options?.onEnd ?? (() => {}),
  };

  world.afterEvents.itemUse.subscribe(({ source, itemStack }) => {
    if (itemStack?.typeId === CONTROLLER_ID && isHost(source)) openController(source);
  });

  system.afterEvents.scriptEventReceive.subscribe(({ id }) => {
    if (id === "allenscreations:end") endGame();
  });

  world.afterEvents.worldLoad.subscribe(() => {
    if (getState() === "running") setState("idle");
    for (const p of world.getAllPlayers()) if (isHost(p)) giveController(p);
  });

  world.afterEvents.playerSpawn.subscribe(({ player, initialSpawn }) => {
    if (!isHost(player)) return;
    if (config.consumeOnStart) {
      if (initialSpawn && (isCreative(player) || getState() !== "running")) giveController(player);
      return;
    }
    giveController(player);
  });
}
```

/**
 * Extracts all searchable word tokens for an item from:
 * 1) Raw item identifier tokens (split by underscore)
 * 2) Official in-game display name tokens from PrismarineJS/minecraft-data
 * 3) Custom anvil nametag tokens (if renamed by the player)
 *
 * @param {string} typeId e.g. "minecraft:cooked_beef"
 * @param {string|null} customNameTag optional custom anvil name
 * @returns {string[]} array of uppercase word tokens
 */
export function getItemSearchTokens(typeId, customNameTag = null) {
  const tokens = new Set();
  const rawId = typeId.replace(/^.*:/, "").toLowerCase();

  // 1. Raw ID full string & underscore-split words
  tokens.add(rawId);
  for (const part of rawId.split("_")) {
    if (part.length > 0) tokens.add(part);
  }

  // 2. Official display name from PrismarineJS/minecraft-data
  const mappedDisplayName = BEDROCK_ITEM_DISPLAY_NAMES[rawId];
  if (mappedDisplayName) {
    const cleanDisplay = mappedDisplayName.toLowerCase();
    tokens.add(cleanDisplay);
    for (const word of cleanDisplay.split(/[\s_\-']+/)) {
      if (word.length > 0) tokens.add(word);
    }
  } else {
    // Fallback: title-cased words from rawId
    for (const part of rawId.split("_")) {
      if (part.length > 0) tokens.add(part);
    }
  }

  // 3. Custom anvil nametag (if renamed)
  if (customNameTag && typeof customNameTag === "string") {
    const cleanCustom = customNameTag.trim().toLowerCase();
    tokens.add(cleanCustom);
    for (const word of cleanCustom.split(/[\s_\-']+/)) {
      if (word.length > 0) tokens.add(word);
    }
  }

  return Array.from(tokens);
}

/**
 * Determines whether an item matches the target letter.
 * Returns true if ANY search token (word) starts with the target letter.
 *
 * @param {string} typeId e.g. "minecraft:cooked_beef"
 * @param {string} letter e.g. "S" (Steak) or "B" (Beef) or "C" (Cooked)
 * @param {string|null} customNameTag
 * @returns {boolean}
 */
export function itemMatchesTargetLetter(typeId, letter, customNameTag = null) {
  if (!typeId || !letter) return false;
  const targetChar = letter.trim().charAt(0).toLowerCase();
  const tokens = getItemSearchTokens(typeId, customNameTag);

  for (const token of tokens) {
    if (token.length > 0 && token.charAt(0) === targetChar) {
      return true;
    }
  }

  return false;
}

/**
 * Produces clean display name for notifications (no emojis)
 * @param {string} typeId
 * @returns {string}
 */
export function getFriendlyItemName(typeId) {
  const rawId = typeId.replace(/^.*:/, "").toLowerCase();
  if (BEDROCK_ITEM_DISPLAY_NAMES[rawId]) {
    return BEDROCK_ITEM_DISPLAY_NAMES[rawId];
  }
  return rawId
    .split("_")
    .map((w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase())
    .join(" ");
}

      const x = base.x + Math.cos(angle) * distance;

      if (rockets < CELEBRATION_MAX_ROCKETS_PER_BURST) {
        try {
          dimension.spawnEntity(FIREWORK_ROCKET_ID, { x, y: base.y, z });
          rockets++;
        } catch (e) {}
      }

      if (spots < CELEBRATION_MAX_SPOTS_PER_BURST) {
        const y = base.y + 7 + Math.random() * 4;
        for (const name of CELEBRATION_PARTICLES) {
          try {
            dimension.spawnParticle(name, { x, y, z });
          } catch (e) {}
        }
        spots++;
      }

      try {
        const sound = SOUND_FIREWORKS[Math.floor(Math.random() * SOUND_FIREWORKS.length)];
        player.playSound(sound, { volume: 1.0, pitch: 0.9 + Math.random() * 0.3 });
      } catch (e) {}
    }
  }

  stopFireworks() {
    if (this.fireworksRun !== null) {
      system.clearRun(this.fireworksRun);
      this.fireworksRun = null;
    }
  }

  stopCelebration() {
    this.stopFireworks();
    if (this.celebrationEnd !== null) {
      system.clearRun(this.celebrationEnd);
      this.celebrationEnd = null;
    }
  }

  /**
   * Triggers mission failure state
   */
  triggerGameOver() {
    if (this.state === GameState.FAILED) return;
    this.state = GameState.FAILED;

    world.sendMessage("[GAME OVER] MISSION FAILED!");

// 3. INVENTORY VERIFICATION USING PRISMARINEJS MINECRAFT-DATA
// ============================================================================

// Match results are cached per item type and letter, so each item is only analysed once.
const matchCache = new Map();

function typeMatchesLetter(typeId, letter) {
  const key = `${letter}|${typeId}`;
  let result = matchCache.get(key);
  if (result === undefined) {
    // Anvil renames are intentionally ignored so a renamed item can not win a round
    result = itemMatchesTargetLetter(typeId, letter, null);
    matchCache.set(key, result);
  }
  return result;
}

/**
 * Scans player inventory for an item starting with the target letter.
 * Supports identifier tokens and official display names.
 *
 * @param {import("@minecraft/server").Player} player
 * @param {string} letter
 * @returns {string|null} matching item typeId or null
 */
export function scanPlayerInventoryForLetter(player, letter) {
  try {
    const inv = player.getComponent("minecraft:inventory");
    if (!inv || !inv.container) return null;

    const container = inv.container;
    for (let slot = 0; slot < container.size; slot++) {
      const item = container.getItem(slot);
      if (item && item.typeId && typeMatchesLetter(item.typeId, letter)) {
        return item.typeId;
      }
    }
  } catch (e) {}
  return null;
}

// Game logic and UI: once per second (was every tick)
system.runInterval(() => {
  const players = world.getAllPlayers();
  if (players.length === 0) return;
  onSecondTick(players);
  renderTopScreenUI(players);
}, 20);

// Health clamp: twice per second, and it exits instantly for players without a penalty
system.runInterval(() => {
  if (manager.state === GameState.IDLE) return;
  for (const player of world.getAllPlayers()) {
    manager.enforceHealthClamp(player);
  }
}, 10);

/**
 * Main 1-second game timer routine
 * @param {import("@minecraft/server").Player[]} players
 */
function onSecondTick(players) {
  if (manager.state === GameState.IDLE) return;

  // ACTIVE ROUND STATE
  if (manager.state === GameState.ACTIVE) {
    // 1. Scan player inventories for target letter item
    for (const player of players) {
      if (!player.isValid() || manager.eliminated.has(player.id)) continue;
      const matchedTypeId = scanPlayerInventoryForLetter(player, manager.targetLetter);
      if (matchedTypeId) {
        manager.completeRound(player, matchedTypeId);
        return;
      }
    }

    // 2. Decrement countdown
    if (manager.timerSeconds > 0) {
      manager.timerSeconds--;

      // Sound countdown cue for final 5 seconds
      if (manager.timerSeconds <= 5 && manager.timerSeconds > 0) {
        for (const player of players) {
          try {
            player.playSound(SOUND_COUNTDOWN, { volume: 0.8, pitch: 1.5 });
          } catch (e) {}
        }
      }

      // Timer expired: apply penalty!
      if (manager.timerSeconds <= 0) {
        manager.handleRoundFailure();
      }
    }
    return;
  }

  // INTERMISSION REST STATE
  if (manager.state === GameState.INTERMISSION) {
    if (manager.timerSeconds > 0) {
      manager.timerSeconds--;
      if (manager.timerSeconds <= 0) {
        manager.startNextRound();
      }
    }
  }
}

/**
 * Displays objective at top of screen via Bossbar (once, shared) and Action Bar (per player).
 * ZERO emojis used in all display strings.
 *
 * @param {import("@minecraft/server").Player[]} players
 */
function renderTopScreenUI(players) {
  const round = manager.currentRound;
  const time = manager.timerSeconds;
  const letter = manager.targetLetter;

  let bossbarTitle = "";
  let bossbarColor = "yellow";
  let bossbarVal = time;
  let bossbarMax = ROUND_DURATION_SECONDS;

  switch (manager.state) {
    case GameState.IDLE:
      bossbarTitle = "Alphabet Scavenger Hunt";
      bossbarColor = "white";
      bossbarVal = 0;
      bossbarMax = 1;
      break;
    case GameState.ACTIVE:
      bossbarTitle = `Round ${round}/${TOTAL_ROUNDS} | Letter: [ ${letter} ] | Time: ${time}s`;
      bossbarColor = time <= 15 ? "red" : time <= 30 ? "yellow" : "green";
      break;
    case GameState.INTERMISSION:
      bossbarTitle = round === 0
        ? `Get Ready | First Letter in ${time}s`
        : `Intermission | Next Letter in ${time}s | Round ${round}/${TOTAL_ROUNDS}`;
      bossbarColor = "blue";
      bossbarMax = round === 0 ? 10 : INTERMISSION_DURATION_SECONDS;
      break;
    case GameState.VICTORY:
      bossbarTitle = `[ VICTORY ] All ${TOTAL_ROUNDS} Rounds Conquered!`;
      bossbarColor = "green";
      bossbarVal = 100;
      bossbarMax = 100;
      break;
    case GameState.FAILED:
      bossbarTitle = "[ MISSION FAILED ] Every player was eliminated.";
      bossbarColor = "red";
      bossbarVal = 0;
      bossbarMax = 100;
      break;
  }

  // One shared bossbar update (skipped automatically when nothing changed)
  manager.updateBossbar(bossbarTitle, bossbarVal, bossbarMax, bossbarColor);

  for (const player of players) {
    if (!player.isValid()) continue;

    const penaltyHp = manager.getPlayerPenaltyHp(player);
    const penaltyText = penaltyHp > 0 ? `-${penaltyHp} HP` : `0 HP`;
    let topText = "";

    switch (manager.state) {
      case GameState.IDLE:
        topText = "§e[ Alphabet Scavenger ] §fUse the Controller to start";
        break;
      case GameState.ACTIVE:
        // Format: [ Round 5/20 ] Letter: [ C ] | Time: 42s | Penalty: -2 HP
        topText = `§e[ Round ${round}/${TOTAL_ROUNDS} ] §fLetter: §b[ ${letter} ] §f| Time: §c${time}s §f| Penalty: §c${penaltyText}`;
        break;
      case GameState.INTERMISSION:
        topText = round === 0
          ? `§a[ Get Ready ] §fFirst Letter In: §e${time}s`
          : `§a[ Round ${round}/${TOTAL_ROUNDS} Complete ] §fNext Letter In: §e${time}s §f| Penalty: §c${penaltyText}`;
        break;
      case GameState.VICTORY:
        topText = `§6[ CHALLENGE COMPLETE (${TOTAL_ROUNDS}/${TOTAL_ROUNDS}) ] §aAll ${TOTAL_ROUNDS} Rounds Conquered!`;
        break;
      case GameState.FAILED:
        topText = "§4[ MISSION FAILED ] §cUse the Controller to restart.";
        break;
    }

    try {
      player.onScreenDisplay.setActionBar(topText);
    } catch (e) {}
  }
}

// ============================================================================
// Player Death Check: a death eliminates that player; the run ends when all are out
world.afterEvents.entityDie.subscribe((event) => {
  const dead = event.deadEntity;
  if (dead && dead.typeId === "minecraft:player") {
    manager.handlePlayerDeath(dead);
  }
});

// Welcome / instructions on player join (Zero emojis)
world.afterEvents.playerSpawn.subscribe((event) => {
  const { player, initialSpawn } = event;
  if (initialSpawn) {

      player.sendMessage("Survive 20 Rounds! Find items matching the target letter.");
      player.sendMessage("Use the Controller item to start, end, or restart the challenge.");
      player.sendMessage("========================================");
    }, 40);
  }
});
