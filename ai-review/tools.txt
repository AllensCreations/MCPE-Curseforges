ACHIEVEMENT = ("likely achievement-friendly", "not achievement-friendly", "unknown / needs confirmation")
COMMON_REQUIRED = ("project.json", "README.md", "DESCRIPTION.md", "CHANGELOG.md", "behavior_pack/manifest.json")

                "behavior_pack/functions/void_cleaner.mcfunction", "behavior_pack/functions/oneblock_setup.mcfunction",
                "behavior_pack/items/trader_ledger.json", "resource_pack/textures/item_texture.json",

                       "resource_pack/textures/item_texture.json", "resource_pack/textures/items/controller.png",
                       "behavior_pack/functions/give_controller.mcfunction", "behavior_pack/functions/end_game.mcfunction")

def _png(b64, sha):
    data = base64.b64decode(b64, validate=True)
    assert hashlib.sha256(data).hexdigest() == sha, "embedded PNG is corrupt"
    return data

def pack_slug(name):
    slug = re.sub(r"[^a-z0-9]+", "_", str(name).lower()).strip("_")
    return slug or "oneblock"

def read_project(root):
    try:
        return json.loads((Path(root) / "project.json").read_text(encoding="utf-8-sig"))
    except Exception:
        return None

def wants_controller(pj):
    return isinstance(pj, dict) and pj.get("type") == "minigame"

def item_format_version(root):
    try:
        m = json.loads((Path(root) / "behavior_pack/manifest.json").read_text(encoding="utf-8-sig"))
        mev = m["header"]["min_engine_version"]
        if isinstance(mev, list) and len(mev) == 3 and all(isinstance(x, int) and not isinstance(x, bool) for x in mev) and mev != [0, 0, 0]:
            return ".".join(str(x) for x in max(mev, MIN_ITEM_FORMAT))
    except Exception:
        pass
    return None

def _item(template, root):
    fv = item_format_version(root)
    if fv is None:
        sys.exit("Set behavior_pack/manifest.json header.min_engine_version to the looked-up game version before running this script: the shipped item JSON takes its format_version from it.")
    return {**template, "format_version": fv}

def _write_json(p, data):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(data, indent=2) + "\n", encoding="utf-8")

def _install_icon(root, key, png_bytes, entry, label, base_texture):
    png = root / f"resource_pack/textures/items/{key}.png"
    if not png.exists():
        png.parent.mkdir(parents=True, exist_ok=True)
        png.write_bytes(png_bytes)
        print(f"installed  {png.relative_to(root).as_posix()}")
    t = root / "resource_pack/textures/item_texture.json"
    if not t.exists():
        _write_json(t, base_texture)
        print("installed  resource_pack/textures/item_texture.json")
        return
    try:
        data = json.loads(t.read_text(encoding="utf-8-sig"))
    except Exception:
        return  # run_checks reports invalid JSON
    if data.get("texture_name") != "atlas.items" or data.get("texture_data", {}).get(key) != entry:
        data["texture_name"] = "atlas.items"
        data.setdefault("texture_data", {})[key] = entry
        _write_json(t, data)
        print(f"repaired   item_texture.json {key} entry")

def install_skin(root, repo_root):
    """Controller projects: copy the ALN UI Redesign font and About icon from the repo into the Resource Pack
    (copy, never link). Existing files stay. Returns the list of problems (empty = fine)."""
    pj = read_project(root)
    if not isinstance(pj, dict) or not wants_controller(pj):
        return []
    src = Path(repo_root) / SKIN_REL
    pairs = list(SKIN_FILES)
    about = next((n for n in SKIN_ABOUT if (src / n).is_file()), None)
    if about:
        pairs.append((about, f"resource_pack/textures/ui/aln/{about}"))
    problems = []
    for s, d in pairs:
        if not (src / s).is_file():
            problems.append(f"repo skin file {SKIN_REL}/{s} is missing")
            continue
        out = root / d
        if not out.exists():
            out.parent.mkdir(parents=True, exist_ok=True)
            out.write_bytes((src / s).read_bytes())
            print("installed  " + d)
    if not about:
        problems.append(f"repo skin file {SKIN_REL}/about_icon.png or about_icon.jpg is missing")
    return problems

def install_assets(root):
    """Create the mandatory icon + item files if missing. Never overwrites an existing PNG."""
    pj = read_project(root)
    if not isinstance(pj, dict):
        return
    if pj.get("type") == "map":
        base = {"resource_pack_name": pack_slug(pj.get("name")), "texture_name": "atlas.items",
                "texture_data": {"trader_ledger": LEDGER_ENTRY}}
        _install_icon(root, "trader_ledger", _png(LEDGER_PNG_B64, LEDGER_SHA256), LEDGER_ENTRY, "ledger", base)
        i = root / "behavior_pack/items/trader_ledger.json"
        if not i.exists():
            _write_json(i, _item(LEDGER_ITEM, root))
            print("installed  behavior_pack/items/trader_ledger.json")
    elif wants_controller(pj):
        base = {"resource_pack_name": "allenscreations", "texture_name": "atlas.items",
                "texture_data": {"controller": CONTROLLER_ENTRY}}
        _install_icon(root, "controller", _png(CONTROLLER_PNG_B64, CONTROLLER_SHA256), CONTROLLER_ENTRY, "controller", base)
        i = root / "behavior_pack/items/controller.json"
        if not i.exists():
            _write_json(i, _item(CONTROLLER_ITEM, root))
            print("installed  behavior_pack/items/controller.json")
        g = root / "behavior_pack/functions/give_controller.mcfunction"
        if not g.exists():
            g.parent.mkdir(parents=True, exist_ok=True)
            g.write_text(GIVE_LINE + "\n", encoding="utf-8")
            print("installed  behavior_pack/functions/give_controller.mcfunction")
        e = root / "behavior_pack/functions/end_game.mcfunction"
        if not e.exists():
            e.parent.mkdir(parents=True, exist_ok=True)
            e.write_text(END_LINE + "\n", encoding="utf-8")
            print("installed  behavior_pack/functions/end_game.mcfunction")
        mjs = root / "behavior_pack/scripts/main.js"
        cjs = root / CONTROLLER_JS_REL
        pjs = root / PIXELIFY_JS_REL
        if not pjs.exists() and mjs.is_file() and "registerController(" in mjs.read_text(encoding="utf-8"):
            pjs.parent.mkdir(parents=True, exist_ok=True)
            pjs.write_text(PIXELIFY_JS, encoding="utf-8", newline="\n")
            print("installed  " + PIXELIFY_JS_REL)
        if not cjs.exists() and mjs.is_file() and "registerController(" in mjs.read_text(encoding="utf-8"):
            cjs.parent.mkdir(parents=True, exist_ok=True)
            cjs.write_text(CONTROLLER_JS, encoding="utf-8", newline="\n")
            print("installed  " + CONTROLLER_JS_REL)

def _png_size(d):
    if d[:8] != b"\x89PNG\r\n\x1a\n" or len(d) < 24:
        return None
    return struct.unpack(">II", d[16:24])

def _has_comment(src):
    i, n, q = 0, len(src), None
    while i < n:
        c = src[i]
        if q:
            if c == "\\":
                i += 2
                continue
            if c == q or (q != "`" and c == "\n"):
                q = None
            i += 1
            continue
        if c in "\"'`":
            q = c
        elif c == "/" and i + 1 < n and src[i + 1] in "/*":
            return True
        i += 1
    return False

def _node_syntax(path):
    """Return (ran, error). Uses node --check on a .mjs copy; ran is False when node is missing."""
    node = shutil.which("node")
    if not node:
        return False, ""
    with tempfile.TemporaryDirectory() as t:
        tmp = Path(t) / "check.mjs"
        shutil.copyfile(path, tmp)
        r = subprocess.run([node, "--check", str(tmp)], capture_output=True, text=True)
    if r.returncode == 0:
        return True, ""
    if "SyntaxError" not in r.stderr:
        return False, ""
    errs = [ln for ln in r.stderr.splitlines() if "Error" in ln]
    return True, (errs[-1] if errs else "syntax error").strip()

def run_checks(root, repo_root=None, folder=None):
    """Return (fails, warns, file_count, total_bytes).
    repo_root (importer only): compare icons to the repo copies.
    folder (importer only): the folder name the project will get."""
    root = Path(root)
    fails, warns = [], []

    def check(ok, msg):
        if not ok:
            fails.append(msg)

    def jload(rel, strict=True):
        try:
            return json.loads((root / rel).read_text(encoding="utf-8-sig"))
        except Exception as exc:
            (fails if strict else warns).append(f"{rel}: missing or invalid JSON ({exc})")
            return None

    pj = jload("project.json")
    ptype = pj.get("type") if isinstance(pj, dict) else None
    check(ptype in TYPES, f"project.json: type must be one of {', '.join(TYPES)}")
    if isinstance(pj, dict):
        check(isinstance(pj.get("name"), str) and bool(pj["name"].strip()), "project.json needs a non-empty name")
        check(pj.get("controller", True) in (True, False), "project.json: controller must be true or false")
        if ptype == "minigame":
            check(pj.get("controller", True) is not False, "minigame projects must use the Controller item")
    strict = {"project.json", "behavior_pack/manifest.json", "resource_pack/manifest.json", "behavior_pack/functions/tick.json",
              "resource_pack/textures/item_texture.json", "behavior_pack/items/trader_ledger.json",
              "behavior_pack/items/controller.json", "oneblock.json"}

    files = [p for p in root.rglob("*") if p.is_file()]
    total = 0
    check(len(files) <= MAX_FILES, f"{len(files)} files (max {MAX_FILES})")
    for p in files:
        rel = p.relative_to(root).as_posix()
        low = PurePosixPath(rel.lower())
        size = p.stat().st_size
        total += size
        check(size <= MAX_FILE, f"{rel}: over 5 MB")
        check(low.suffix in ALLOWED, f"{rel}: file type not allowed")
        check("db" not in low.parts and low.name != "level.dat", f"{rel}: world data not allowed")
        check(low.name not in ("world_behavior_packs.json", "world_resource_packs.json"), f"{rel}: link files not allowed")
        check(not rel.lower().endswith(BLOCKED), f"{rel}: nested archive")
        check(not p.is_symlink(), f"{rel}: symlink")
        check(not rel.startswith("Mcpeworlds/"), f"{rel}: Mcpeworlds paths must not be shipped")
        check(low.name != "test_world.json", f"{rel}: test worlds were removed; delete test_world.json")
        if low.suffix == ".json" and rel not in strict and not rel.startswith("behavior_pack/recipes/"):
            jload(rel, strict=False)
    check(total <= MAX_TOTAL, f"total {total} bytes (max 25 MB)")
    for rel in COMMON_REQUIRED:
        check((root / rel).is_file(), f"missing required file: {rel}")
    controller = wants_controller(pj)
    if ptype == "map":
        for rel in MAP_REQUIRED:
            check((root / rel).is_file(), f"missing required file: {rel}")
    if controller:
        for rel in CONTROLLER_REQUIRED:
            check((root / rel).is_file(), f"missing required file: {rel}")
    if ptype in ("script", "minigame"):
        check((root / "behavior_pack/scripts/main.js").is_file(), "missing required file: behavior_pack/scripts/main.js")
    if ptype == "script":
        check(not (root / "resource_pack").exists(), "script projects must be Behavior Pack-only so the package is a .mcpack")
    if ptype == "minigame":
        check((root / "resource_pack/manifest.json").is_file(), "minigame projects need a Resource Pack for the Controller item")

    # oneblock.json is optional in the payload (the importer writes it for maps); if present it must agree
    if (root / "oneblock.json").is_file() and ptype == "map":
        cfg = jload("oneblock.json")
        if isinstance(cfg, dict):
            check(cfg.get("enabled") is True, "oneblock.json enabled must be true")
            if isinstance(pj, dict):
                check(cfg.get("name") == pj.get("name"), "oneblock.json name must equal project.json name")

    # tick.json: required for maps, validated whenever present
    tick_p = root / "behavior_pack/functions/tick.json"
    if ptype == "map" or tick_p.is_file():
        tick = jload("behavior_pack/functions/tick.json")
        vals = (tick or {}).get("values")
        check(isinstance(vals, list) and bool(vals), "tick.json needs a non-empty values list")
        if isinstance(vals, list) and vals:
            if ptype == "map":
                check(vals[0] == "void_cleaner", "tick.json: first entry must be void_cleaner")
            for v in vals:
                check((root / f"behavior_pack/functions/{v}.mcfunction").is_file(), f"tick.json names {v} but its .mcfunction is missing")
            if "oneblock_setup" in vals:
                warns.append("oneblock_setup is listed in tick.json and will run every tick; intended?")
    if ptype == "map":
        cl = root / "behavior_pack/functions/void_cleaner.mcfunction"
        if cl.is_file():
            lines = cl.read_text(encoding="utf-8").splitlines()
            check(CLEANER in lines, "void_cleaner.mcfunction lacks the exact required fill line")
            cmds = [x for x in lines if x.strip() and not x.lstrip().startswith("#")]
            check(bool(cmds) and cmds[0] == CLEANER, "the required fill line must be the first command")
            check(bool(cmds) and re.search(r"\btag\b.*\badd\s+void_cleaned\b", cmds[-1]) is not None,
                  "void_cleaner.mcfunction: the last command must add the void_cleaned tag, otherwise every command repeats every tick")
            check(bool(cmds) and CORE_GUARD.search(cmds[-1]) is not None,
                  "void_cleaner.mcfunction: the last command must add the void_cleaned tag only if the core Bedrock exists "
                  "(execute as @a[tag=!void_cleaned] if block <x> <y> <z> bedrock run tag @s add void_cleaned), "
                  "otherwise a fill on unloaded chunks is never retried")

    uuids, versions, huuid, bpm, mevs = [], {}, {}, {}, {}
    labels = ["behavior_pack"] + (["resource_pack"] if (root / "resource_pack/manifest.json").is_file() else [])
    for label in labels:
        m = jload(f"{label}/manifest.json")
        if not isinstance(m, dict):
            continue
        if label == "behavior_pack":
            bpm = m
        check(m.get("format_version") == 2, f"{label}: format_version must be 2")
        h = m.get("header") or {}
        uuid, ver, name, desc = h.get("uuid"), h.get("version"), h.get("name", ""), h.get("description", "")
        check(isinstance(uuid, str) and bool(UUID_V4.match(uuid)), f"{label}: header.uuid missing or not a v4 UUID")
        check(isinstance(ver, list) and len(ver) == 3 and all(isinstance(x, int) for x in ver), f"{label}: header.version must be [x,y,z]")
        mev = h.get("min_engine_version")
        if not (isinstance(mev, list) and len(mev) == 3 and all(isinstance(x, int) and not isinstance(x, bool) for x in mev)) or mev == [0, 0, 0]:
            fails.append(f"{label}: header.min_engine_version must be 3 integers (the looked-up game version), not a placeholder")
        mevs[label] = mev
        t = TITLE.match(name) if isinstance(name, str) else None
        check(bool(t), f"{label}: header.name must end with one (vX.Y.Z)")
        if t and isinstance(ver, list):
            check([int(x) for x in t.groups()] == ver, f"{label}: title version does not match header.version")
            check(name.count("(v") == 1, f"{label}: duplicated version suffix in name")
        check(isinstance(desc, str) and desc.count(ATTRIBUTION) == 1, f"{label}: description needs '{ATTRIBUTION}' exactly once")
        check(ATTRIBUTION not in str(name), f"{label}: keep attribution out of the name")
        uuids.append(uuid)
        mods = m.get("modules") or []
        check(bool(mods), f"{label}: no modules")
        for mod in mods:
            uuids.append(mod.get("uuid"))
            check(isinstance(mod.get("uuid"), str) and bool(UUID_V4.match(mod["uuid"])), f"{label}: module uuid missing or invalid")
            if mod.get("type") == "script":
                entry = mod.get("entry")
                check(isinstance(entry, str) and (root / label / entry).is_file(), f"{label}: script module entry file missing")
                check(mod.get("language") == "javascript", f"{label}: script module language must be javascript")
        versions[label], huuid[label] = ver, uuid
        if label == "resource_pack":
            check(any(x.get("type") == "resources" for x in mods), "resource_pack needs a 'resources' module")
    check(len(uuids) == len(set(uuids)), "UUIDs must all be unique (headers and modules)")
    if repo_root:
        mine = {u.lower() for u in uuids if isinstance(u, str)}
        for mp in sorted((Path(repo_root) / "MCPEDL").glob("*/*/*/manifest.json")):
            proj = mp.parent.parent
            if folder and proj.name == folder and proj.parent.name == DEST_DIR.get(ptype):
                continue
            try:
                other = json.loads(mp.read_text(encoding="utf-8-sig"))
            except Exception:
                continue
            ids = [(other.get("header") or {}).get("uuid")] + [m.get("uuid") for m in other.get("modules") or []]
            for u in ids:
                if isinstance(u, str) and u.lower() in mine:
                    fails.append(f"UUID {u} is already used by MCPEDL/{proj.parent.name}/{proj.name}: generate fresh UUIDs")
    if "resource_pack" in versions:
        check(versions.get("behavior_pack") == versions.get("resource_pack"), "behavior and resource pack versions must match")
        dep = [d for d in bpm.get("dependencies") or [] if d.get("uuid") == huuid.get("resource_pack")]
        check(bool(dep), "behavior pack must depend on the resource pack header.uuid")
        if dep:
            check(dep[0].get("version") == versions.get("resource_pack"), "resource pack dependency version must match")
    modnames = {}
    for d in bpm.get("dependencies") or []:
        if "module_name" in d:
            modnames[d["module_name"]] = d.get("version")
            check(isinstance(d.get("version"), str) and bool(d.get("version")), f"dependency {d['module_name']} needs a version string")
            check(d["module_name"] in ALLOWED_MODULES, f"dependency {d['module_name']} is not allowed (allowed: {', '.join(ALLOWED_MODULES)})")
        if any(w in str(d.get("version", "")).lower() for w in ("beta", "rc", "preview")):
            warns.append(f"preview/beta dependency {d.get('module_name')}: allowed only if a requested feature needs it; say so in the README")
    has_script = any(x.get("type") == "script" for x in bpm.get("modules") or [])
    if ptype in ("script", "minigame"):
        check(has_script, f"{ptype} project: behavior_pack manifest needs a script module")
    if has_script:
        check("@minecraft/server" in modnames, "script module needs a @minecraft/server dependency")
    if controller:
        check("@minecraft/server-ui" in modnames, "controller needs a @minecraft/server-ui dependency")
    mev_bp = mevs.get("behavior_pack")
    if isinstance(mev_bp, list) and len(mev_bp) == 3 and all(isinstance(x, int) for x in mev_bp) and mev_bp != [0, 0, 0]:
        for item_rel in ("behavior_pack/items/trader_ledger.json", "behavior_pack/items/controller.json"):
            ip = root / item_rel
            if not ip.is_file():
                continue
            try:
                fv = [int(x) for x in str(json.loads(ip.read_text(encoding="utf-8-sig")).get("format_version", "")).split(".")]
            except Exception:
                continue
            if len(fv) == 3 and fv > mev_bp:
                warns.append(f"{item_rel}: format_version {'.'.join(map(str, fv))} is newer than min_engine_version {'.'.join(map(str, mev_bp))}: raise min_engine_version")

    rec = root / "behavior_pack/recipes"
    for p in rec.glob("**/*.json") if rec.is_dir() else []:
        rrel = p.relative_to(root).as_posix()
        try:
            rdata = json.loads(p.read_text(encoding="utf-8-sig"))
        except Exception as exc:
            fails.append(f"{rrel}: invalid JSON ({exc})")
            continue
        rkey = next((k for k in CRAFTING_RECIPES if isinstance(rdata, dict) and k in rdata), None)
        if rkey is None:
            continue
        body = rdata[rkey] if isinstance(rdata[rkey], dict) else {}
        unlock = body.get("unlock")
        entries = unlock if isinstance(unlock, list) else [unlock]
        check(any(isinstance(e, dict) and e.get("context") == "AlwaysUnlocked" for e in entries),
              f"{rrel}: missing the unlock context AlwaysUnlocked inside {rkey}")

    def icon_checks(key, ident, item_rel, want_sha, size):
        png = root / f"resource_pack/textures/items/{key}.png"
        if png.is_file():
            d = png.read_bytes()
            ref = (Path(repo_root) / ASSET_DIR / f"{key}.png") if repo_root else None
            if ref is not None:
                check(ref.is_file(), f"repo asset {ASSET_DIR}/{key}.png is missing")
                if ref.is_file():
                    check(d == ref.read_bytes(), f"{key}.png is not a byte-for-byte copy of the repo asset")
            else:
                check(hashlib.sha256(d).hexdigest() == want_sha, f"{key}.png is not a byte-for-byte copy (SHA-256 mismatch)")
            check(_png_size(d) == size, f"{key}.png must be a {size[0]}x{size[1]} PNG")
        its = jload("resource_pack/textures/item_texture.json") or {}
        check(its.get("texture_name") == "atlas.items", "item_texture.json: texture_name must be atlas.items")
        check(((its.get("texture_data") or {}).get(key) or {}).get("textures") == f"textures/items/{key}",
              f"item_texture.json: {key} entry missing or wrong path")
        item = (jload(item_rel) or {}).get("minecraft:item") or {}
        check((item.get("description") or {}).get("identifier") == ident, f"{item_rel}: identifier must be {ident}")
        icon = ((item.get("components") or {}).get("minecraft:icon") or {}).get("textures", {}).get("default")
        check(icon == key, f"{item_rel}: icon default must be {key}")

    scripts_dir = root / "behavior_pack/scripts"
    for jp in sorted(scripts_dir.rglob("*.js")) if scripts_dir.is_dir() else []:
        jrel = jp.relative_to(root).as_posix()
        jtxt = jp.read_text(encoding="utf-8")
        for spec in IMPORT_RE.findall(jtxt):
            check(spec in ALLOWED_MODULES or spec.startswith(("./", "../")),
                  f"{jrel}: import of '{spec}' is not allowed (only {', '.join(ALLOWED_MODULES)} and relative files)")
        check(not re.search(r"\beval\s*\(|\bFunction\s*\(", jtxt), f"{jrel}: eval and the Function constructor are not allowed")
        check(not re.search(r"\bimport\s*\(\s*(?![\"'])", jtxt), f"{jrel}: dynamic import with a computed specifier is not allowed (it bypasses the module allowlist)")
        if re.search(r"\b(?:globalThis|window|self)\s*\[", jtxt):
            warns.append(f"{jrel}: indexed global access can bypass the eval and import checks: a reviewer must read this file")
        if _has_comment(jtxt):
            warns.append(jrel + ": remove comments (no comments in scripts)")
        ran, err = _node_syntax(jp)
        if ran and err:
            fails.append(f"{jrel}: syntax error: {err}")
        elif not ran:
            warns.append(f"{jrel}: syntax not checked (node not found or not working)")

    if ptype == "map":
        icon_checks("trader_ledger", "allenscreations:trader_ledger", "behavior_pack/items/trader_ledger.json", LEDGER_SHA256, (32, 32))
    if controller:
        icon_checks("controller", "allenscreations:controller", "behavior_pack/items/controller.json", CONTROLLER_SHA256, (16, 16))
        mj = root / "behavior_pack/scripts/main.js"
        cj = root / CONTROLLER_JS_REL
        if mj.is_file():
            js = mj.read_text(encoding="utf-8")
            if "registerController(" in js:
                check(cj.is_file(), "missing required file: " + CONTROLLER_JS_REL)
                if cj.is_file():
                    check(cj.read_text(encoding="utf-8").replace("\r\n", "\n") == CONTROLLER_JS,

def pack(root, out):
    root = Path(root)
    buf = io.BytesIO()
    with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as z:
        for p in sorted(root.rglob("*")):
            if p.is_file():
                info = zipfile.ZipInfo(p.relative_to(root).as_posix(), date_time=(2020, 1, 1, 0, 0, 0))
                info.compress_type = zipfile.ZIP_DEFLATED
                info.external_attr = 0o644 << 16
                z.writestr(info, p.read_bytes(), compresslevel=9)
    data = buf.getvalue()
    out.write_text(base64.b64encode(data).decode("ascii"), encoding="ascii")  # always one line
    back = zipfile.ZipFile(io.BytesIO(base64.b64decode(out.read_text(), validate=True)))
    names = set(back.namelist())
    want = {
        p.relative_to(root).as_posix()
        for p in root.rglob("*")
        if p.is_file()
    }
    if back.testzip() is not None or names != want:
        print(f"ROUND-TRIP FAILED: {back.testzip()} / diff {sorted(names ^ want)}")
        sys.exit(2)
    print(f"ZIP bytes: {len(data)}   payload.b64 chars: {out.stat().st_size}   round-trip OK ({len(names)} files)")
    if len(data) > ZIP_LIMIT:
        print(f"ZIP is over {ZIP_LIMIT} bytes: do NOT paste it into the workflow. Deliver {out.name} as a file "
              "(commit to MCPEDL/incoming/<n>.b64 and use payload_file).")
    return len(data)

"""Import a base64 ZIP produced by an AI agent into the repository.

Supports four project types, read from project.json inside the payload:
  addon    -> MCPEDL/Addons/<Name>/    (LOCK markers are written here)

  create    new project only; fails if the name already exists anywhere under MCPEDL/ (default)

def payload_from_text(text: str, name: str = "") -> str:
    """A pack script (.sh) carries its base64 between two marker lines, each payload line starting with '#'.
    The file is only read, never executed. Any other text is returned unchanged."""

#!/usr/bin/env python3
"""Canonical renderer for the sole crafting-board source: scripts/CRAFTING_TABLE.svg.

The board image, every slot position, the output slot, and every board color come
from that SVG. No raster board, screenshot, or alternate GUI is ever accepted as
input. Generated PNGs are outputs only; they are not design references.

Expected SVG ids: slot-0 .. slot-8 (row-major 3x3 groups), slot-N-fill (item area),
output (group), output-fill (item area).
"""
from __future__ import annotations

from dataclasses import dataclass
from functools import lru_cache
from io import BytesIO
from pathlib import Path
import re
import xml.etree.ElementTree as ET

import cairosvg
from PIL import Image

CRAFTING_TABLE_SVG = Path(__file__).resolve().parent / "CRAFTING_TABLE.svg"
NATIVE_SIZE = (400, 222)

SLOT_ICON_SIZE = 32
OUTPUT_ICON_SIZE = 38

Rect = tuple[float, float, float, float]  # x, y, width, height in SVG pixels

@dataclass(frozen=True)
class BoardLayout:
    slots: tuple[Rect, ...]  # nine item areas, row-major
    output: Rect  # output item area

def _read_svg() -> bytes:
    if not CRAFTING_TABLE_SVG.is_file():
        raise FileNotFoundError(
            f"Required sole crafting-board reference is missing: {CRAFTING_TABLE_SVG}. "
            "No alternate image or raster fallback is permitted."
        )
    return CRAFTING_TABLE_SVG.read_bytes()

def _parse_svg(source: bytes) -> ET.Element:
    try:
        return ET.fromstring(source)
    except ET.ParseError as exc:
        raise ValueError(f"Unable to parse the sole crafting-board SVG: {CRAFTING_TABLE_SVG}") from exc

def _translate(element: ET.Element) -> tuple[float, float]:
    match = re.search(
        r"translate\(\s*(-?[0-9.]+)[\s,]+(-?[0-9.]+)\s*\)", element.attrib.get("transform", "")
    )
    return (float(match.group(1)), float(match.group(2))) if match else (0.0, 0.0)

def _fill_rect(group: ET.Element, fill_id: str) -> Rect:
    """Absolute rectangle of the item area `fill_id` inside `group`."""
    gx, gy = _translate(group)
    for child in group.iter():
        if child.attrib.get("id") == fill_id:
            try:
                return (
                    gx + float(child.attrib.get("x", 0)),
                    gy + float(child.attrib.get("y", 0)),
                    float(child.attrib["width"]),
                    float(child.attrib["height"]),
                )
            except (KeyError, ValueError) as exc:
                raise ValueError(f"Element {fill_id!r} in CRAFTING_TABLE.svg needs numeric width and height") from exc
    raise ValueError(f"CRAFTING_TABLE.svg is missing required element id {fill_id!r}")

@lru_cache(maxsize=1)
def load_board_layout() -> BoardLayout:
    """Read slot and output positions directly from the SVG's ids."""
    root = _parse_svg(_read_svg())
    groups = {el.attrib["id"]: el for el in root.iter() if "id" in el.attrib}

    slots = []
    for index in range(9):
        group = groups.get(f"slot-{index}")
        if group is None:
            raise ValueError(f"CRAFTING_TABLE.svg is missing required element id 'slot-{index}'")
        slots.append(_fill_rect(group, f"slot-{index}-fill"))

    output_group = groups.get("output")
    if output_group is None:
        raise ValueError("CRAFTING_TABLE.svg is missing required element id 'output'")
    return BoardLayout(slots=tuple(slots), output=_fill_rect(output_group, "output-fill"))

def render_crafting_table_svg() -> Image.Image:
    """Render the exact source SVG at its native dimensions and return RGBA pixels."""
    source = _read_svg()
    root = _parse_svg(source)

    def dimension(value: str) -> float:
        match = re.fullmatch(r"\s*([0-9]+(?:\.[0-9]+)?)(?:px)?\s*", value)
        if not match:
            raise ValueError(
                "CRAFTING_TABLE.svg must declare numeric width and height so its native "
                "proportions can be preserved."
            )
        return float(match.group(1))

    width, height = dimension(root.attrib.get("width", "")), dimension(root.attrib.get("height", ""))
    if width <= 0 or height <= 0:
        raise ValueError(f"Invalid CRAFTING_TABLE.svg dimensions: {width} x {height}")

    expected = (round(width), round(height))
    png = cairosvg.svg2png(
        bytestring=source,
        output_width=expected[0],
        output_height=expected[1],
    )
    image = Image.open(BytesIO(png)).convert("RGBA")
    if image.size != expected:
        raise ValueError(
            f"Rendered CRAFTING_TABLE.svg dimensions {image.size} do not match "
            f"its declared native dimensions {expected}."
        )
    return image

def _centered(rect: Rect, size: int) -> tuple[int, int]:
    x, y, w, h = rect
    return round(x + (w - size) / 2), round(y + (h - size) / 2)

def compose_recipe_diagram(recipe_type: str, recipe_data: dict, ingredient_icon, result_icon: Image.Image, output_count: int = 1, scale: int = 1) -> Image.Image:
    """Overlay recipe icons onto the real 3x3 grid in CRAFTING_TABLE.svg.

    Shaped recipes retain their exact pattern positions. Shapeless ingredients
    fill grid cells in row-major order. The SVG-rendered board is never rebuilt,
    and all positions and colors come from the SVG.
    """
    board = render_crafting_table_svg()
    if board.size != NATIVE_SIZE:
        raise ValueError(f"Expected native {NATIVE_SIZE[0]}x{NATIVE_SIZE[1]} crafting SVG, got {board.size}")
    layout = load_board_layout()

    grid = [[None for _ in range(3)] for _ in range(3)]
    if recipe_type == "minecraft:recipe_shaped":
        pattern = recipe_data.get("pattern", [])
        key = recipe_data.get("key", {})
        if len(pattern) > 3 or any(len(row) > 3 for row in pattern):
            raise ValueError("Shaped recipe pattern exceeds the 3x3 crafting grid")
        for row, pattern_row in enumerate(pattern):
            for col, symbol in enumerate(pattern_row):
                if symbol == " ":
                    continue
                if symbol not in key or not key[symbol].get("item"):
                    raise ValueError(f"Missing item mapping for shaped recipe symbol {symbol!r}")
                grid[row][col] = key[symbol]["item"]
    elif recipe_type == "minecraft:recipe_shapeless":
        ingredients = recipe_data.get("ingredients", [])
        if len(ingredients) > 9:
            raise ValueError(f"Shapeless recipe has {len(ingredients)} ingredients; crafting grid holds 9")
        for index, entry in enumerate(ingredients):
            item_id = entry.get("item") if isinstance(entry, dict) else None
            if not item_id:
                raise ValueError(f"Unsupported shapeless ingredient entry: {entry!r}")
            grid[index // 3][index % 3] = item_id
    else:
        raise ValueError(f"Unsupported recipe type: {recipe_type}")

    for row in range(3):
        for col in range(3):
            item_id = grid[row][col]
            if item_id is None:
                continue
            icon = ingredient_icon(item_id).convert("RGBA").resize(
                (SLOT_ICON_SIZE, SLOT_ICON_SIZE), Image.Resampling.NEAREST
            )
            board.alpha_composite(icon, _centered(layout.slots[row * 3 + col], SLOT_ICON_SIZE))

    result = result_icon.convert("RGBA").resize((OUTPUT_ICON_SIZE, OUTPUT_ICON_SIZE), Image.Resampling.NEAREST)
    board.alpha_composite(result, _centered(layout.output, OUTPUT_ICON_SIZE))

    if output_count > 1:
        try:
            from PIL import ImageDraw, ImageFont

            ox, oy, ow, oh = layout.output
            right, bottom = round(ox + ow) - 2, round(oy + oh) - 4
            left, top = right - 14, bottom - 13
            # Badge background uses the slot color sampled from the SVG render.
            badge_fill = board.getpixel((round(ox) + 1, round(oy) + 1))

            draw = ImageDraw.Draw(board)
            font_path = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
            font = ImageFont.truetype(str(font_path), 10) if font_path.is_file() else ImageFont.load_default()
            draw.rectangle((left, top, right, bottom), fill=badge_fill)
            draw.text((left + 2, top), str(output_count), font=font, fill=(255, 255, 255, 255), stroke_width=1, stroke_fill=(35, 35, 35, 255))
        except Exception as exc:
            raise RuntimeError("Unable to render recipe output count") from exc

    if scale != 1:
        board = board.resize((board.width * scale, board.height * scale), Image.Resampling.NEAREST)
    return board

def load_json(path: Path):
    text = path.read_text(encoding="utf-8-sig")
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        stripped = "\n".join(l for l in text.splitlines() if not l.lstrip().startswith("//"))
        return json.loads(stripped)

def short(item_id: str) -> str:
    return item_id.split(":")[-1] if item_id.count(":") <= 1 else item_id.split(":")[1]

def _samples_rp(samples_root: Path) -> Path:
    return samples_root / "resource_pack"

def _shade(img: Image.Image, factor: float) -> Image.Image:
    img = img.copy()
    px = img.load()
    for y in range(img.height):
        for x in range(img.width):
            r, g, b, a = px[x, y]
            px[x, y] = (min(255, int(r * factor)), min(255, int(g * factor)), min(255, int(b * factor)), a)
    return img

def make_block_item_icon(face_path: Path) -> Image.Image:
    face = Image.open(face_path).convert("RGBA").resize((16, 16), Image.Resampling.NEAREST)
    out = Image.new("RGBA", (16, 16), (0, 0, 0, 0))
    out.paste(face.resize((12, 12), Image.Resampling.NEAREST), (2, 2))
    for x in range(2, 14):
        r, g, b, a = out.getpixel((x, 2))
        if a > 0:
            out.putpixel((x, 2), (min(255, int(r * 1.25)), min(255, int(g * 1.25)), min(255, int(b * 1.25)), a))
        r, g, b, a = out.getpixel((x, 13))
        if a > 0:
            out.putpixel((x, 13), (int(r * 0.65), int(g * 0.65), int(b * 0.65), a))
    for y in range(2, 14):
        r, g, b, a = out.getpixel((2, y))
        if a > 0:
            out.putpixel((2, y), (min(255, int(r * 1.15)), min(255, int(g * 1.15)), min(255, int(b * 1.15)), a))
        r, g, b, a = out.getpixel((13, y))
        if a > 0:
            out.putpixel((13, y), (int(r * 0.75), int(g * 0.75), int(b * 0.75), a))
    return out

def isometric_block_icon(side_path: Path, top_path: Path, right_shade: float = 0.72) -> Image.Image:
    icon = Image.new("RGBA", (32, 32), (0, 0, 0, 0))

    def face(texture_path, box, points, shade=1.0):
        x0, y0, x1, y1 = box
        tex = Image.open(texture_path).convert("RGBA").resize((x1 - x0, y1 - y0), Image.Resampling.NEAREST)
        if shade != 1.0:
            tex = _shade(tex, shade)
        layer = Image.new("RGBA", (32, 32), (0, 0, 0, 0))
        layer.alpha_composite(tex, (x0, y0))
        mask = Image.new("L", (32, 32), 0)
        ImageDraw.Draw(mask).polygon(points, fill=255)
        layer.putalpha(ImageChops.multiply(layer.getchannel("A"), mask))
        icon.alpha_composite(layer)

    face(top_path, (4, 2, 28, 16), [(16, 2), (28, 9), (16, 16), (4, 9)])
    face(side_path, (4, 9, 17, 30), [(4, 9), (16, 16), (16, 30), (4, 23)])
    face(side_path, (16, 9, 29, 30), [(16, 16), (28, 9), (28, 23), (16, 30)], right_shade)
    return icon

def _texture_from_json(rp: Path, name: str) -> Path | None:
    """item_texture.json: texture_data[name].textures -> <rp>/<path>.png"""
    index = rp / "textures" / "item_texture.json"
    if not index.is_file():
        return None
    entry = load_json(index).get("texture_data", {}).get(name)
    if not entry:
        return None
    tex = entry.get("textures")
    if isinstance(tex, list):
        tex = tex[0] if tex else None
    if isinstance(tex, dict):
        tex = tex.get("path")
    if not isinstance(tex, str):
        return None
    path = rp / (tex if tex.endswith(".png") else tex + ".png")
    return path if path.is_file() else None

def vanilla_icon(item_id: str, samples_root: Path = None) -> Image.Image:
    rp = _samples_rp(samples_root or SAMPLES_ROOT)
    name = short(item_id)
    entry = VANILLA_OVERRIDES.get(f"minecraft:{name}")
    if entry:
        path, kind = rp / entry[0], entry[1]
        if not path.is_file():
            raise Unresolved(f"{item_id}: mapped Samples texture is missing: {path}")
        if kind == "block_item":
            return make_block_item_icon(path)
        if kind == "iso_block":
            top = rp / entry[2]
            if not top.is_file():
                raise Unresolved(f"{item_id}: Samples top texture is missing: {top}")
            return isometric_block_icon(path, top)
        if kind == "head_item":
            head = Image.open(path).convert("RGBA").crop((0, 0, 16, 16))
            out = Image.new("RGBA", (16, 16), (0, 0, 0, 0))
            out.paste(head.resize((12, 12), Image.Resampling.NEAREST), (2, 2))
            return out
        im = Image.open(path).convert("RGBA")  # item, plant_item, flat_block
        return im if im.size == (16, 16) else im.resize((16, 16), Image.Resampling.NEAREST)
    direct = rp / "textures" / "items" / f"{name}.png"
    if direct.is_file():
        return Image.open(direct).convert("RGBA")
    indexed = _texture_from_json(rp, name)
    if indexed:
        return Image.open(indexed).convert("RGBA")
    raise Unresolved(f"{item_id}: no verified vanilla texture. Add it to VANILLA_OVERRIDES after checking Samples "
                     "(never guess a block texture).")

def custom_texture_path(pack: Path, item_id: str) -> Path | None:
    rp = pack / "resource_pack"
    name = short(item_id)
    icon_name = None
    items_dir = pack / "behavior_pack" / "items"
    if items_dir.is_dir():
        for f in sorted(items_dir.rglob("*.json")):
            try:
                data = load_json(f).get("minecraft:item", {})
            except (json.JSONDecodeError, OSError):
                continue
            if data.get("description", {}).get("identifier") != item_id:
                continue
            icon = data.get("components", {}).get("minecraft:icon")
            if isinstance(icon, str):
                icon_name = icon
            elif isinstance(icon, dict):
                tex = icon.get("texture") or (icon.get("textures") or {}).get("default")
                icon_name = tex if isinstance(tex, str) else None
            break
    for candidate in (icon_name, name):
        if candidate:
            found = _texture_from_json(rp, candidate)
            if found:
                return found
    hits = sorted((rp / "textures" / "items").rglob(f"{name}.png")) if (rp / "textures" / "items").is_dir() else []
    return hits[0] if hits else None

def icon_for(pack: Path, item_id: str, samples_root: Path = None) -> Image.Image:
    if item_id.startswith("minecraft:") or ":" not in item_id:
        return vanilla_icon(item_id, samples_root)
    path = custom_texture_path(pack, item_id)
    if not path:
        raise Unresolved(f"{item_id}: no texture found in the pack's resource_pack (checked the BP item icon, "
                         "item_texture.json and textures/items/**).")
    return Image.open(path).convert("RGBA")

# ---------------------------------------------------------------- recipes
def recipe_files(pack: Path) -> list[Path]:
    d = pack / "behavior_pack" / "recipes"
    return sorted(d.rglob("*.json")) if d.is_dir() else []

def parse_recipe(path: Path):
    """-> (type, data) for a crafting-table recipe, or None when it is another kind of recipe."""
    doc = load_json(path)
    for kind in SUPPORTED:
        if kind in doc:
            data = doc[kind]
            tags = data.get("tags")
            if tags is not None and "crafting_table" not in tags:
                return None
            return kind, data
    return None

def result_of(data: dict) -> tuple[str, int]:
    res = data.get("result")
    if isinstance(res, str):
        return res, 1
    if isinstance(res, dict) and res.get("item"):
        return res["item"], int(res.get("count", 1))
    raise Unresolved("recipe has no usable result item")

def ingredient_entries(kind: str, data: dict):
    if kind == "minecraft:recipe_shaped":
        used = {c for row in data.get("pattern", []) for c in row if c != " "}
        return [data.get("key", {}).get(c) for c in sorted(used)]
    return list(data.get("ingredients", []))

def plan_pack(pack: Path, samples_root: Path = None):
    """Resolve every recipe of a pack. -> (jobs, skipped, problems). Nothing is written here."""
    jobs, skipped, problems = [], [], []
    names_used = {}
    for f in recipe_files(pack):
        label = f.relative_to(pack).as_posix()
        try:
            parsed = parse_recipe(f)
        except (json.JSONDecodeError, OSError) as exc:
            problems.append(f"{label}: cannot read recipe ({exc})")
            continue
        if parsed is None:
            skipped.append(label)
            continue
        kind, data = parsed
        try:
            result_id, count = result_of(data)
            for entry in ingredient_entries(kind, data):
                if not isinstance(entry, dict) or not entry.get("item"):
                    raise Unresolved(f"ingredient {entry!r} has no single item (tag or missing key); add an explicit item")
                if entry.get("data") not in (None, 0):
                    raise Unresolved(f"{entry['item']} uses data value {entry['data']}; one icon cannot show it")
            ingredients = {e["item"] for e in ingredient_entries(kind, data)}
            icons = {i: icon_for(pack, i, samples_root) for i in sorted(ingredients)}
            result_icon = icon_for(pack, result_id, samples_root)
        except Unresolved as exc:
            problems.append(f"{label}: {exc}")
            continue
        slug = short(result_id)
        out_name = f"{slug}_recipe.png"
        if out_name in names_used:
            out_name = f"{slug}_{f.stem}_recipe.png"
        names_used[out_name] = f
        jobs.append(dict(file=f, kind=kind, data=data, count=count, icons=icons, result=result_icon, result_id=result_id, out=out_name))
    return jobs, skipped, problems

def write_pack(pack: Path, jobs: list, with_sheet: bool):
    from crafting_template import compose_recipe_diagram  # needs CairoSVG; imported only when writing
    out_dir = pack / "assets" / "recipes"
    out_dir.mkdir(parents=True, exist_ok=True)
    written = []
    for job in jobs:
        icons = job["icons"]
        diagram = compose_recipe_diagram(job["kind"], job["data"], lambda i, _m=icons: _m[i], job["result"], job["count"], scale=1)
        target = out_dir / job["out"]
        diagram.save(target)
        written.append(target)
    sheet = pack / "assets" / "all_recipes_sheet.png"
    if written and (with_sheet or sheet.is_file()):
        pad, w, h = 8, 400, 222
        canvas = Image.new("RGBA", (w + 2 * pad, len(written) * h + (len(written) + 1) * pad), (35, 35, 38, 255))
        for i, p in enumerate(sorted(written)):
            with Image.open(p) as frame:
                canvas.paste(frame, (pad, pad + i * (h + pad)))
        canvas.save(sheet)
        written.append(sheet)
    return written

def export_items(pack: Path, item_ids: list[str], samples_root: Path = None):
    """Copy item sprites to <pack>/assets/items/<name>.png, nearest-neighbor ITEM_SPRITE_SIZE square."""
    out_dir = pack / "assets" / "items"
    problems, written = [], []
    for item_id in item_ids:
        try:
            icon = icon_for(pack, item_id, samples_root)
        except Unresolved as exc:
            problems.append(str(exc))
            continue
        out_dir.mkdir(parents=True, exist_ok=True)
        target = out_dir / f"{short(item_id)}.png"
        icon.resize((ITEM_SPRITE_SIZE, ITEM_SPRITE_SIZE), Image.Resampling.NEAREST).save(target)
        written.append(target)
    return written, problems

def custom_result_ids(jobs: list) -> list[str]:
    """Custom (non-vanilla) result items of the planned recipes, in order, without duplicates."""
    seen, out = set(), []
    for job in jobs:
        rid = job["result_id"]
        if ":" in rid and not rid.startswith("minecraft:") and rid not in seen:
            seen.add(rid)
            out.append(rid)
    return out

def find_packs(root: Path, project: str | None):
    if project:

"""Export SVG assets to PNG. Crafting-table PNG is a derivative of CRAFTING_TABLE.svg, never a separate reference."""
from pathlib import Path
from urllib.request import Request, urlopen
import base64
import re

import cairosvg

ROOT = Path(__file__).resolve().parents[1]
ASSETS = ROOT / "assets"

def embed_remote_images(svg: str) -> str:
    pattern = re.compile(r'(<image\b[^>]*?\b(?:href|xlink:href)=["\'])(https?://[^"\']+)(["\'][^>]*>)', re.I)
    def replace(match):
        url = match.group(2)
        request = Request(url, headers={"User-Agent": "ALNStudios-RecipeImageBuilder/1.0"})
        try:
            with urlopen(request, timeout=30) as response:
                data = response.read()
                mime = response.headers.get_content_type()
        except Exception as exc:
            raise RuntimeError(f"Unable to fetch linked recipe texture {url}: {exc}") from exc
        encoded = base64.b64encode(data).decode("ascii")
        return match.group(1) + f"data:{mime};base64,{encoded}" + match.group(3)
    return pattern.sub(replace, svg)

def main():
    candidates = sorted(ASSETS.glob("CRAFTING_RECIPE*.svg"))
    # Always export the crafting board template to PNG as well.
    template_svg = ROOT / "scripts" / "CRAFTING_TABLE.svg"
    if template_svg.exists():
        candidates.insert(0, template_svg)
    if not candidates:
        print("No crafting recipe SVGs found; nothing to convert.")
        return
    for source in candidates:
        output = source.with_suffix(".png")
        svg = source.read_text(encoding="utf-8")
        svg = embed_remote_images(svg)
        cairosvg.svg2png(bytestring=svg.encode("utf-8"), write_to=str(output), output_width=400 if source == template_svg else 800)
        print(f"Generated {output.relative_to(ROOT)}")
if __name__ == "__main__":
    main()

            skipped += 1
            continue

        # Bedrock uses a single unlock object for shaped recipes and an array for shapeless recipes.

            already_correct += 1
            continue
        recipe["unlock"] = desired
        path.write_text(json.dumps(data, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
        changed += 1

#!/usr/bin/env python3
"""Guarantee every pack has a valid pack_icon.png (256x256 PNG, pack root).

Used two ways:
  * CLI (refresh workflow):  python3 scripts/make_pack_icon.py [--strict]
  * Imported by auto_bump_and_build.py so EVERY build gets the icon, not only
    the "Refresh Add-ons and Rebuild Packs" workflow.

Source lookup inside each behavior_pack/ and resource_pack/ (top level only,
case-insensitive): pack_icon.*, icon.*, logo.* in png/jpg/jpeg/webp/bmp/gif.
Fallbacks when a pack has none: the sibling pack's pack_icon.png, then an
icon image at the project root. Leftover source files are deleted after
conversion. Packs that still have no icon are reported loudly.
"""
import glob, os, sys
from PIL import Image, ImageOps

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
SIZE = 256
EXTS = {".png", ".jpg", ".jpeg", ".webp", ".bmp", ".gif"}
STEMS = ["pack_icon", "packicon", "icon", "logo"]   # priority order
PACKS = ("behavior_pack", "resource_pack")

def _norm(stem):
    return stem.lower().replace(" ", "_").replace("-", "_")

def _find_sources(folder):
    found = []
    for name in os.listdir(folder):
        stem, ext = os.path.splitext(name)
        if ext.lower() in EXTS and _norm(stem) in STEMS and name != "pack_icon.png":
            found.append(name)
    return sorted(found, key=lambda n: (STEMS.index(_norm(os.path.splitext(n)[0])), n))

def _convert(src, dest):
    img = ImageOps.exif_transpose(Image.open(src)).convert("RGBA")
    side = min(img.size)
    left, top = (img.width - side) // 2, (img.height - side) // 2
    img = img.crop((left, top, left + side, top + side))
    img = img.resize((SIZE, SIZE), Image.Resampling.LANCZOS)
    tmp = dest + ".tmp"
    img.save(tmp, "PNG", optimize=True)
    os.replace(tmp, dest)

def _rel(p):
    return os.path.relpath(p, ROOT)

def ensure_pack_icon(pack_dir, fallbacks=(), warn=True):
    """Return True when pack_dir ends up with a usable pack_icon.png."""
    dest = os.path.join(pack_dir, "pack_icon.png")
    sources = _find_sources(pack_dir)
    try:
        if sources:
            _convert(os.path.join(pack_dir, sources[0]), dest)
            for s in sources:
                os.remove(os.path.join(pack_dir, s))
            print(f"[Pack Icon] {_rel(pack_dir)}/{sources[0]} -> pack_icon.png (sources removed: {len(sources)})")
            return True
        if os.path.isfile(dest):
            with Image.open(dest) as im:
                w, h = im.size
            if w != h or w > SIZE:
                _convert(dest, dest)
                print(f"[Pack Icon] {_rel(dest)} normalized to {SIZE}x{SIZE}")
            return True
        for fb in fallbacks:
            if os.path.isfile(fb):
                _convert(fb, dest)
                print(f"[Pack Icon] {_rel(dest)} created from {_rel(fb)}")
                return True
    except Exception as exc:  # corrupt image etc.
        print(f"[Pack Icon] ERROR in {_rel(pack_dir)}: {exc}", file=sys.stderr)
        return False
    if warn:
        print(f"[Pack Icon] WARNING: {_rel(pack_dir)} has NO pack_icon.png "
              f"(add icon.png next to manifest.json)", file=sys.stderr)
    return False

def ensure_project_icons(project_path):
    """Handle both packs of one project. Returns list of pack dirs still missing an icon."""
    dirs = {p: os.path.join(project_path, p) for p in PACKS}
    present = [p for p in PACKS if os.path.isdir(dirs[p])]
    project_srcs = []
    if os.path.isdir(project_path):
        project_srcs = [os.path.join(project_path, n) for n in _find_sources(project_path)]
        pi = os.path.join(project_path, "pack_icon.png")
        if os.path.isfile(pi):
            project_srcs.insert(0, pi)
    missing = []
    # first pass: packs that have their own icon; second pass: fill gaps from siblings
    done = {p: ensure_pack_icon(dirs[p], fallbacks=project_srcs, warn=False) for p in present}
    for p in present:
        if not done[p]:
            sib = [os.path.join(dirs[q], "pack_icon.png") for q in present if q != p]
            if not ensure_pack_icon(dirs[p], fallbacks=sib):
                missing.append(dirs[p])
    return missing

def main():
    strict = "--strict" in sys.argv
    missing, n = [], 0

from PIL import Image

MAX_BYTES = 200 * 1024

def optimize(path, check):
    original = path.read_bytes()
    img = Image.open(io.BytesIO(original))
    needs = img.width > MAX_WIDTH or len(original) > MAX_BYTES
    if check:
        return needs, len(original)
    if not needs:
        return False, len(original)
    img = img.convert("RGB")
    if img.width > MAX_WIDTH:
        img = img.resize((MAX_WIDTH, round(img.height * MAX_WIDTH / img.width)), Image.LANCZOS)
    best = None
    for quality in range(85, 34, -5):
        buf = io.BytesIO()
        img.save(buf, "JPEG", quality=quality, optimize=True, progressive=True)
        best = buf.getvalue()
        if len(best) <= MAX_BYTES:
            break
    path.write_bytes(best)
    return True, len(best)

def main():
    p = argparse.ArgumentParser(description="Auto resize (max 800 px wide) and compress (max 200 KB) JPEG files.")
    p.add_argument("folder", nargs="?", default=str(ROOT / "assets" / "Images"))

        return 2

        bad += changed

What lives here
  * align_project_manifests - make every source manifest of a project share ONE version

  * ensure_metadata - fill metadata.authors (and metadata.license when MANIFEST_LICENSE is set).

def _normalized_manifest_bytes(raw, own_uuids):
    """Manifest content minus everything the version machinery rewrites."""
    try:
        data = json.loads(raw.decode("utf-8"))
        header = data.get("header", {})
        header.pop("version", None)
        if isinstance(data.get("metadata"), dict):
            data["metadata"].pop("version", None)
        if isinstance(header.get("name"), str):
            header["name"] = TITLE_SUFFIX_RE.sub("", header["name"]).rstrip()
        for module in data.get("modules", []) or []:
            if isinstance(module, dict):
                module.pop("version", None)
        for dep in data.get("dependencies", []) or []:
            if isinstance(dep, dict) and dep.get("uuid") in own_uuids:
                dep.pop("version", None)
        return json.dumps(data, sort_keys=True, ensure_ascii=False).encode("utf-8")
    except (ValueError, AttributeError, UnicodeDecodeError):
        return raw

    own_uuids = set()

        except (OSError, ValueError, AttributeError):
            pass

    digest = hashlib.sha256()
    entries = []

            raw = fh.read()

            raw = _normalized_manifest_bytes(raw, own_uuids)
        digest.update(rel.encode("utf-8") + b"\0")

KEYWORD_ANCHORS = ("Minecraft Bedrock add-on", "MCPEDL add-on", "CurseForge Bedrock", "ALNStudios")
CHANGELOG_GROUPS = ("Added", "Improved & Balanced", "Fixed")
INSTALL_STEPS = ("1. Download the package using the link above.",
                 "2. Open the downloaded file with Minecraft to import it.",
                 "3. If the package contains a pack that must be enabled separately, activate it in your world's settings.")
INFO_LINE = re.compile(r"^For gameplay features, usage details, compatibility notes, and other project information, read the \[full description\]\(([^)\s]+)\)\.$")
EMOJI = re.compile("[\U0001F300-\U0001FAFF\u2600-\u27BF\uFE0F]")
BANNED = (
    ("a Markdown table", re.compile(r"(?m)^\s*\|.*\|\s*$")),
    ("a code block", re.compile(r"```")),
    ("a badge", re.compile(r"img\.shields\.io|badge", re.I)),
    ("'BP' or 'RP' (write Behavior Pack / Resource Pack)", re.compile(r"\b(?:BP|RP)\b")),
    ("'Script API' or an @minecraft module name", re.compile(r"Script API|@minecraft/", re.I)),
    ("a raw identifier such as minecraft:diamond", re.compile(r"\bminecraft:[a-z0-9_]+")),
    ("an emoji", EMOJI),
    ("a duplicate version suffix such as (v1.0.0) (v1.0.1)", re.compile(r"\(v\d+\.\d+\.\d+(?:\.\d+)?\)\s*\(v\d+")),
)

def norm(text):
    return "\n".join(line.rstrip() for line in text.replace("\r\n", "\n").split("\n"))

def manifest_version(root):
    try:
        data = json.loads((Path(root) / "behavior_pack/manifest.json").read_text(encoding="utf-8-sig"))
        v = data["header"]["version"]
        if isinstance(v, list) and len(v) == 3 and all(type(x) is int for x in v):
            return ".".join(map(str, v))
    except Exception:
        pass
    return None

def vtuple(text):
    return tuple(int(x) for x in text.split("."))

def banned_content(name, text, errors, images_allowed=False):
    if images_allowed:
        for badge in DESCRIPTION_BADGES + LEGACY_BADGES:
            text = text.replace(badge, "")
    for label, rx in BANNED:
        if label == "a badge" and images_allowed:
            rx = re.compile(r"img\.shields\.io", re.I)
        if rx.search(text):
            errors.append(f"DOC: {name} must not contain {label} (Part 2).")
    if not images_allowed and (re.search(r"!\[[^\]]*\]\(", text) or re.search(r"<img\b", text, re.I)):
        errors.append(f"DOC: {name} must not contain images (Part 2).")

def check_description_badges(text, errors):
    lines = text.splitlines()
    for badge in DESCRIPTION_BADGES:
        if lines.count(badge) != 1:
            errors.append(f"DOC: DESCRIPTION.md must include this badge exactly once: {badge}")
    for badge in LEGACY_BADGES:
        if badge in lines:
            errors.append("DOC: DESCRIPTION.md uses the old hardcoded-version Minecraft Bedrock badge; run apply_description_policy.")

def check_listing_images(creators, errors, images_dir=None):
    n = "DESCRIPTION.md"
    for label, (prefix, file, default) in LISTINGS.items():
        rx = re.compile(r'<a href="(https://[^"\s]+)"><img src="' + re.escape(IMG_BASE + file) + r'" alt="' + label + r'" width="45%"></a>')
        found = rx.findall(creators)
        if len(found) != 1:
            errors.append(f"DOC: {n} '## Creator and Permissions' must contain the clickable {label} image exactly once, e.g. {listing_html(label)}")
        elif not found[0].startswith(prefix):
            errors.append(f"DOC: {n} the {label} image link must point to {prefix}")
    if len(re.findall(r"<img\b", creators, re.I)) != 2 or re.search(r"\bheight=", creators, re.I):
        errors.append(f"DOC: {n} the Creator section may hold only the two listing images, with relative width and no fixed height.")
    folder = Path(images_dir) if images_dir else ROOT / "assets" / "Images"
    for label, (prefix, file, default) in LISTINGS.items():
        f = folder / file
        if f.is_file() and f.stat().st_size > IMG_MAX_BYTES:
            errors.append(f"DOC: {file} is {f.stat().st_size // 1024} KB; run scripts/optimize_images.py (limit {IMG_MAX_BYTES // 1024} KB).")

def apply_description_policy(text, addon=False):
    drop = DESCRIPTION_BADGES + LEGACY_BADGES
    lines = [line for line in norm(text).split("\n") if line not in drop]
    insert_at = next((i + 1 for i, line in enumerate(lines) if line.strip() == "**By ALNStudios**"), None)
    if insert_at is None:
        insert_at = next((i + 1 for i, line in enumerate(lines) if line.startswith("# ")), 0)
    while insert_at < len(lines) and not lines[insert_at].strip():
        lines.pop(insert_at)
    lines[insert_at:insert_at] = ["", *DESCRIPTION_BADGES, ""]
    if addon:
        found = False
        for i, line in enumerate(lines):
            if re.match(r"^- (?:\*\*)?Achievements(?:\*\*)?:", line):
                lines[i] = "- Achievements: Likely achievement-friendly."
                found = True
        if not found:
            compatibility = next((i for i, line in enumerate(lines) if line.strip().lower() == "## compatibility"), None)
            if compatibility is not None:
                insert_at = next((i for i in range(compatibility + 1, len(lines)) if lines[i].startswith("## ")), len(lines))
                lines.insert(insert_at, "- Achievements: Likely achievement-friendly.")
    return apply_listing_images("\n".join(lines))

def apply_listing_images(text):
    text = norm(text)
    if "<img" in text and LISTING_LINE in text:
        return text
    lines = text.split("\n")
    start = next((i for i, l in enumerate(lines) if l.strip() == "## Creator and Permissions"), None)
    if start is None:
        return text
    end = next((i for i in range(start + 1, len(lines)) if lines[i].startswith("## ")), len(lines))
    while end > start + 1 and not lines[end - 1].strip():
        end -= 1
    lines[end:end] = ["", LISTING_LINE]
    return "\n".join(lines)

def check_project_readme(text, errors):
    lines = [l for l in norm(text).split("\n") if l.strip()]
    n = "README.md"
    if len(lines) != 11:
        errors.append(f"DOC: {n} must have exactly 11 non-empty lines in the version-free download template (found {len(lines)}).")
    title = re.match(r"^# (.+)$", lines[0]) if lines else None
    if not title:
        errors.append(f"DOC: {n} line 1 must be '# [Pack Name]' with no release version.")
        return
    name = title.group(1)
    if re.search(r"\(v\d+\.\d+\.\d+(?:\.\d+)?\)$", name, re.I):
        errors.append(f"DOC: {n} title must not contain a release version.")
    expect = [
        (1, f"Download the current release of **{name}** for Minecraft Bedrock Edition.", "the intro sentence"),
        (2, "## Download", "the '## Download' heading"),
        (5, "## Installation", "the '## Installation' heading"),
        (6, INSTALL_STEPS[0], "installation step 1"),
        (7, INSTALL_STEPS[1], "installation step 2"),
        (9, "## Pack Information", "the '## Pack Information' heading"),
    ]
    for index, wanted, label in expect:
        if index >= len(lines) or lines[index] != wanted:
            errors.append(f"DOC: {n} {label} must read exactly: {wanted}")
    fmt = re.match(r"^- \*\*Package format:\*\* `\.(mcaddon|mcpack|mcworld)`$", lines[3]) if len(lines) > 3 else None
    if not fmt:
        errors.append(f"DOC: {n} line 4 must name a supported package format: `.mcaddon`, `.mcpack`, or `.mcworld`.")
    dl = re.match(r"^- \*\*Direct download:\*\* \[Download (.+) \(\.(mcaddon|mcpack|mcworld)\)\]\(([^)\s]+)\)", lines[4]) if len(lines) > 4 else None
    if not dl:
        errors.append(f"DOC: {n} line 5 must be a direct download link for {name} and its package format.")
    else:
        if dl.group(1) != name and not dl.group(1).startswith(name + " — "):
            errors.append(f"DOC: {n} the download link text must name '{name}'.")
        if fmt and dl.group(2) != fmt.group(1):
            errors.append(f"DOC: {n} the package format and the download link extension must match.")
    if fmt and fmt.group(1) == "mcworld" and len(lines) > 8:
        expected_install = "3. Find the imported world in your Worlds list and open it."
    else:
        expected_install = INSTALL_STEPS[2]
    if len(lines) > 8 and lines[8] != expected_install:
        errors.append(f"DOC: {n} installation step 3 must read exactly: {expected_install}")
    info = INFO_LINE.match(lines[10]) if len(lines) > 10 else None
    if not info:
        errors.append(f"DOC: {n} the last line must link the full DESCRIPTION.md.")
    elif "DESCRIPTION.md" not in info.group(1):
        errors.append(f"DOC: {n} the full description link must point to DESCRIPTION.md.")
    banned_content(n, text, errors)

def section_map(text):
    parts = re.split(r"(?m)^(?=## )", text)
    return parts[0], [(p.split("\n", 1)[0][3:].strip(), p.split("\n", 1)[1] if "\n" in p else "") for p in parts[1:]]

def check_description(text, version, errors, ptype="addon", images_dir=None):
    n = "DESCRIPTION.md"
    text = norm(text)
    pre, sections = section_map(text)
    head = re.search(r"(?m)^# (.+?) — (\S.*)$", pre)
    if not head or pre.lstrip().split("\n", 1)[0] != head.group(0):
        errors.append(f"DOC: {n} line 1 must be '# [Pack Name] — [Clear, factual subtitle]' with no release version.")
    elif re.search(r"\(v\d+\.\d+\.\d+(?:\.\d+)?\)$", head.group(1), re.I):
        errors.append(f"DOC: {n} title must not contain a release version.")
    after = [l for l in pre.split("\n")[1:] if l.strip()]
    if not after or after[0] != "**By ALNStudios**":
        errors.append(f"DOC: {n} the line after the title must be exactly: **By ALNStudios**")
    elif len(after) < 2 or after[1].startswith(("#", "-", "|", "*")):
        errors.append(f"DOC: {n} needs a short summary paragraph after the By line and before ## Overview.")
    names = [s[0] for s in sections]
    for opt in ("Commands", "Common Questions"):
        at = names.index("How to Use") + 1 if "How to Use" in names else None
        if at is None or at >= len(names) or names[at] != opt:
            continue
        opt_body = sections[at][1]
        sections = sections[:at] + sections[at + 1:]
        names = [s[0] for s in sections]
        opt_lines = [l for l in opt_body.split("\n") if l.strip()]
        if not opt_lines:
            errors.append(f"DOC: {n} '## {opt}' must not be empty; remove it or add verified lines.")
        pat = r"^- \*\*[^*]+\*\*: \S" if opt == "Common Questions" else r"^- `?/\S.*`?: \S"
        for l in opt_lines:
            if not re.match(pat, l):
                want = "- **Question**: answer" if opt == "Common Questions" else "- `/command options`: what it does"
                errors.append(f"DOC: {n} '## {opt}' lines must read '{want}'.")
                break
    if names != list(DESCRIPTION_SECTIONS):
        errors.append(f"DOC: {n} the ## sections must be exactly, in this order: {', '.join(DESCRIPTION_SECTIONS)}. Found: {', '.join(names) or 'none'}.")
        return banned_content(n, text, errors, images_allowed=True)
    body = dict(sections)
    key = body["Key Features"].strip("\n")
    cards = re.split(r"(?m)^(?=### )", key)
    if not key.strip() or not key.lstrip().startswith("### "):
        errors.append(f"DOC: {n} '## Key Features' must contain only '### Feature Name' cards.")
    else:
        for card in cards:
            if not card.strip():
                continue
            if not re.search(r"(?m)^- \*\*What Happens\*\*: \S", card):
                errors.append(f"DOC: {n} the feature card '{card.splitlines()[0][4:].strip()}' needs a '- **What Happens**: ...' bullet.")
    for heading in ("Overview", "How to Use"):
        if not body[heading].strip():
            errors.append(f"DOC: {n} '## {heading}' must not be empty.")
    compat = [l for l in body["Compatibility"].split("\n") if l.strip()]
    found = []
    for line in compat:
        m = re.match(r"^- (Minecraft Bedrock Edition|Experimental features|Achievements|Verification): (\S.*)$", line)
        found.append((m.group(1), m.group(2)) if m else None)
    if [f[0] if f else None for f in found] != list(COMPAT_LABELS):
        errors.append(f"DOC: {n} '## Compatibility' must be exactly four bullets, in this order: " + ", ".join(f"'- {l}: ...'" for l in COMPAT_LABELS) + ".")
    else:
        ach, ver = found[2][1].lower(), found[3][1].lower()
        if not any(a in ach for a in ACHIEVEMENT):
            errors.append(f"DOC: {n} the Achievements line must use one of: Likely achievement-friendly, Not achievement-friendly, Unknown / needs confirmation.")
        if ptype == "addon" and not ach.startswith("likely achievement-friendly"):
            errors.append(f"DOC: {n} add-on Achievements must start with 'Likely achievement-friendly'.")
        if not any(ver.startswith(v) for v in VERIFIED):
            errors.append(f"DOC: {n} the Verification line must start with 'Source created', 'Package built', or 'In-game tested'.")
    creators = body["Creator and Permissions"]
    for label, url in (("CurseForge", CF_URL), ("MCPEDL", MP_URL)):
        if url not in creators:
            errors.append(f"DOC: {n} '## Creator and Permissions' must link the {label} profile ({url}).")
    check_listing_images(creators, errors, images_dir)
    if GUIDELINES + "## Search & Discovery Keywords" not in text:
        errors.append(f"DOC: {n} the attribution block must be the exact Part 2 text, directly before '## Search & Discovery Keywords'.")
    kw = body["Search & Discovery Keywords"]
    for anchor in KEYWORD_ANCHORS:
        if anchor.lower() not in kw.lower():
            errors.append(f"DOC: {n} the keyword section must include '{anchor}'.")
    banned_content(n, text, errors, images_allowed=True)

def check_changelog(text, version, errors):
    n = "CHANGELOG.md"
    text = norm(text)
    lines = text.split("\n")
    ne = [(i, l) for i, l in enumerate(lines) if l.strip()]
    if not ne or not re.match(r"^# Changelog — \S.*$", ne[0][1]):
        errors.append(f"DOC: {n} line 1 must be '# Changelog — [Add-on Name]'.")
        return
    if len(ne) < 2 or ne[1][1].strip() != "---":
        errors.append(f"DOC: {n} the title must be followed directly by a '---' line, with no boilerplate.")
    versions, group_order, current, bullets_in_group = [], -1, None, 0
    in_entries = False
    for i, line in ne[2:]:
        entry = re.match(r"^## \[(\d+\.\d+\.\d+(?:\.\d+)?)\] - (\d{4}-\d{2}-\d{2})(?: - \S.*)?$", line)
        group = re.match(r"^### (.+)$", line)
        bullet = re.match(r"^- \*\*[^*]+\*\*: \S", line)
        if entry:
            if current and bullets_in_group == 0:
                errors.append(f"DOC: {n} entry {current} has an empty group.")
            try:
                datetime.date.fromisoformat(entry.group(2))
            except ValueError:
                errors.append(f"DOC: {n} entry {entry.group(1)} has an invalid date {entry.group(2)}.")
            versions.append(entry.group(1))
            current, group_order, bullets_in_group, in_entries = entry.group(1), -1, 0, True
        elif group:
            if not in_entries:
                errors.append(f"DOC: {n} a '### {group.group(1)}' group appears before any release entry.")
                continue
            if group.group(1) not in CHANGELOG_GROUPS:
                errors.append(f"DOC: {n} group '### {group.group(1)}' is not allowed; use Added, Improved & Balanced, or Fixed.")
                continue
            idx = CHANGELOG_GROUPS.index(group.group(1))
            if idx <= group_order:
                errors.append(f"DOC: {n} entry {current}: groups must appear once, in the order Added, Improved & Balanced, Fixed.")
            if group_order >= 0 and bullets_in_group == 0:
                errors.append(f"DOC: {n} entry {current} has an empty group.")
            group_order, bullets_in_group = idx, 0
        elif bullet:
            if group_order < 0:
                errors.append(f"DOC: {n} entry {current}: bullets must be under a ### group.")
            bullets_in_group += 1
        else:
            errors.append(f"DOC: {n} line {i + 1} is not allowed: only release headings, ### groups, and '- **Title**: text' bullets.")
    if current and bullets_in_group == 0:
        errors.append(f"DOC: {n} entry {current} has an empty group.")
    if not versions:
        errors.append(f"DOC: {n} needs at least one entry like '## [1.0.0] - YYYY-MM-DD'.")
    else:
        if [vtuple(v) for v in versions] != sorted((vtuple(v) for v in versions), reverse=True) or len(set(versions)) != len(versions):
            errors.append(f"DOC: {n} entries must be newest first with no duplicate versions.")
        if version and versions[0] != version:
            errors.append(f"DOC: {n} the newest entry is {versions[0]} but the manifest version is {version}.")
    if re.search(r"https?://|\]\(|www\.", text):
        errors.append(f"DOC: {n} must not contain links or URLs (Part 2).")
    if re.search(r"keep a changelog|semantic versioning", text, re.I):
        errors.append(f"DOC: {n} must not mention Keep a Changelog or Semantic Versioning.")
    if EMOJI.search(text):
        errors.append(f"DOC: {n} must not contain emojis.")
    if re.search(r"\.(?:js|json|mcfunction)\b|\bminecraft:[a-z_]+", text):
        errors.append(f"DOC: {n} must not name files, functions, or identifiers; describe player-visible changes.")

def check_docs(root, ptype, images_dir=None):
    root = Path(root)
    errors = []
    version = manifest_version(root)
    texts = {}
    for name in ("README.md", "DESCRIPTION.md", "CHANGELOG.md"):
        p = root / name
        texts[name] = p.read_text(encoding="utf-8") if p.is_file() else None
    if texts["README.md"] is not None:
        check_project_readme(texts["README.md"], errors)
    if texts["DESCRIPTION.md"] is not None:
        check_description_badges(texts["DESCRIPTION.md"], errors)
        check_description(texts["DESCRIPTION.md"], version, errors, ptype, images_dir)
    if texts["CHANGELOG.md"] is not None:
        check_changelog(texts["CHANGELOG.md"], version, errors)
    seen, unique = set(), []
    for e in errors:
        if e not in seen:
            seen.add(e)
            unique.append(e)
    return unique

def style_warnings(text, name="DESCRIPTION.md"):
    hits = []
    lines = norm(text).split("\n")
    for i, line in enumerate(lines, 1):
        low = line.lower()
        for phrase in STYLE_PHRASES:
            if re.search(r"\b" + re.escape(phrase) + r"\b", low):
                hits.append(f"STYLE: {name} line {i} uses '{phrase}'. Replace it with a specific fact about what players can do.")
        if "!" in line.replace("![", ""):
            hits.append(f"STYLE: {name} line {i} has an exclamation mark. State the fact plainly.")
    dashes = sum(l.count("\u2014") for l in lines[1:] if not l.startswith("#"))
    if dashes > 3:
        hits.append(f"STYLE: {name} has {dashes} em dashes outside headings. Use plain sentences instead.")
    for block in re.split(r"\n\s*\n", norm(text)):
        b = block.strip()
        if not b or b[0] in "#->*<[|" or b[0].isdigit():
            continue
        if len(re.findall(r"[.!?](?:\s|$)", b)) > 4:
            hits.append(f"STYLE: {name} has a paragraph of more than four sentences: '{b[:50]}...'. Split or shorten it.")
    tags = [l for l in lines if re.match(r"^\*\*[^*].*\*\*$", l.strip()) and "By ALNStudios" not in l]
    if not tags:
        hits.append(f"STYLE: {name} has no bold tagline. Add one concrete sentence of 25 words or fewer.")
    elif len(tags[0].strip("*").split()) > 25:
        hits.append(f"STYLE: {name} tagline is longer than 25 words. It should work as the CurseForge Summary.")
    elif not re.search(r"\d|[A-Z][a-z]+ [A-Z]|[A-Z][a-z]+, [A-Z]", tags[0].strip("*")[1:]):
        hits.append(f"STYLE: {name} tagline names no number or specific item. Add one verified detail.")
    howto = re.search(r"(?ms)^## How to Use\n(.*?)(?=^## |\Z)", norm(text))
    if howto and not re.search(r"(?m)^\d+\. ", howto.group(1)):
        hits.append(f"STYLE: {name} How to Use has no numbered steps.")
    for card in re.findall(r"(?m)^### (.+)$", norm(text)):
        if card.strip().lower() in ("features", "overview", "benefits", "gameplay", "items", "improvements", "extras"):

#!/usr/bin/env python3
"""Turn pixel JSON into PNG textures. Standard library only (no Pillow), so it runs in Termux, GitHub and any AI sandbox.

Source: <project>/texture_sources/*.json (never inside resource_pack/). Each file holds an array of textures
(a single object, or {"textures": [...]}, also works):

  {
    "name": "shepherds_pie",              lowercase letters, digits, underscore; becomes <name>.png
    "size": 16,                           16, 32, 64 or 128; pixels must be exactly size x size
    "folder": "items",                    optional: items (default), blocks, or items/<sub> e.g. items/banquet
    "palette": {"0": "#00000000", "1": "#3d2817"},   index -> #RRGGBB or #RRGGBBAA
    "pixels": [[0, 0, 1, ...], ...]       size rows, each with size palette indexes
  }

Output: <project>/resource_pack/textures/<folder>/<name>.png (nearest-neighbor exact pixels, same JSON = same PNG).
Every problem in every file is reported together; nothing is written unless all textures are valid.

Usage
  python3 scripts/pixel_json_to_png.py MCPEDL/Addons/<pack>     convert (also works on a build/<Name> scratch folder)
  python3 scripts/pixel_json_to_png.py <project> --check        validate only
  python3 scripts/pixel_json_to_png.py <project> --preview [FILE]   also write a contact sheet PNG to FILE (default
                                                                 ./pixel_preview.png, OUTSIDE the project so it never ships)
  python3 scripts/pixel_json_to_png.py <project> --input FILE   use this JSON file instead of texture_sources/*.json
Edit the JSON, never the PNG: the PNG is regenerated from it.
"""
from __future__ import annotations

import argparse
import json
import re
import struct
import sys
import zlib
from pathlib import Path

SIZES = (16, 32, 64, 128)
NAME_RE = re.compile(r"^[a-z0-9_]+$")
FOLDER_RE = re.compile(r"^(items|blocks)(/[a-z0-9_]+)*$")
HEX_RE = re.compile(r"^#([0-9a-fA-F]{6})([0-9a-fA-F]{2})?$")
SOURCE_DIR = "texture_sources"

# ---------------------------------------------------------------- PNG writer (RGBA, 8 bit, filter 0)
def _chunk(tag: bytes, data: bytes) -> bytes:
    return struct.pack(">I", len(data)) + tag + data + struct.pack(">I", zlib.crc32(tag + data) & 0xFFFFFFFF)

def encode_png(width: int, height: int, rgba_rows: list[bytes]) -> bytes:
    raw = b"".join(b"\x00" + row for row in rgba_rows)
    return (b"\x89PNG\r\n\x1a\n"
            + _chunk(b"IHDR", struct.pack(">IIBBBBB", width, height, 8, 6, 0, 0, 0))
            + _chunk(b"IDAT", zlib.compress(raw, 9))
            + _chunk(b"IEND", b""))

# ---------------------------------------------------------------- validation
def parse_color(value) -> tuple[int, int, int, int] | None:
    m = HEX_RE.match(value) if isinstance(value, str) else None
    if not m:
        return None
    rgb = m.group(1)
    alpha = int(m.group(2), 16) if m.group(2) else 255
    return int(rgb[0:2], 16), int(rgb[2:4], 16), int(rgb[4:6], 16), alpha

def textures_in(doc):
    if isinstance(doc, dict) and isinstance(doc.get("textures"), list):
        return doc["textures"]
    if isinstance(doc, dict):
        return [doc]
    return doc if isinstance(doc, list) else None

def validate(tex, where: str) -> tuple[list[str], dict | None]:
    """-> (problems, normalized texture). normalized = name, folder, size, rows (list of list of RGBA tuples), notes."""
    errs: list[str] = []
    if not isinstance(tex, dict):
        return [f"{where}: each texture must be an object"], None
    name = tex.get("name")
    where = f"{where} '{name}'" if isinstance(name, str) else where
    if not isinstance(name, str) or not NAME_RE.match(name):
        errs.append(f"{where}: name must be lowercase letters, digits and underscore only")
    size = tex.get("size")
    if size not in SIZES or isinstance(size, bool):
        errs.append(f"{where}: size must be one of {', '.join(map(str, SIZES))}")
    folder = tex.get("folder", "items")
    if not isinstance(folder, str) or not FOLDER_RE.match(folder):
        errs.append(f"{where}: folder must be items, blocks or items/<name> (lowercase)")
    palette = tex.get("palette")
    colors: dict[int, tuple] = {}
    if not isinstance(palette, dict) or not palette:
        errs.append(f"{where}: palette must be an object like {{\"0\": \"#00000000\"}}")
    else:
        for key, value in palette.items():
            if not (isinstance(key, str) and key.isdigit()):
                errs.append(f"{where}: palette key {key!r} must be a whole number in quotes")
                continue
            rgba = parse_color(value)
            if rgba is None:
                errs.append(f"{where}: palette {key} = {value!r} is not #RRGGBB or #RRGGBBAA")
            else:
                colors[int(key)] = rgba
    pixels = tex.get("pixels")
    rows: list[list[tuple]] = []
    used: set[int] = set()
    if not isinstance(pixels, list):
        errs.append(f"{where}: pixels must be a list of rows")
    elif size in SIZES:
        if len(pixels) != size:
            errs.append(f"{where}: needs {size} rows, found {len(pixels)}")
        for y, row in enumerate(pixels):
            if not isinstance(row, list) or len(row) != size:
                errs.append(f"{where}: row {y} needs {size} values, found {len(row) if isinstance(row, list) else 'a non-list'}")
                continue
            out = []
            for x, idx in enumerate(row):
                if isinstance(idx, str) and idx.isdigit():
                    idx = int(idx)
                if isinstance(idx, bool) or not isinstance(idx, int) or idx not in colors:
                    errs.append(f"{where}: pixel ({x},{y}) uses {row[x]!r}, which is not in the palette")
                    out.append((0, 0, 0, 0))
                else:
                    used.add(idx)
                    out.append(colors[idx])
            rows.append(out)
    if errs:
        return errs, None
    if all(px[3] == 0 for r in rows for px in r):
        return [f"{where}: every pixel is transparent"], None
    unused = sorted(set(colors) - used)
    return [], {"name": name, "folder": folder, "size": size, "rows": rows,
                "notes": [f"{where}: unused palette entries {unused}"] if unused else []}

def load_all(sources: list[Path]):
    """-> (textures, problems, notes) across every source file."""
    textures, problems, notes, seen = [], [], [], {}
    for src in sources:
        label = src.name
        try:
            doc = json.loads(src.read_text(encoding="utf-8-sig"))
        except (OSError, json.JSONDecodeError) as exc:
            problems.append(f"{label}: cannot read JSON ({exc})")
            continue
        items = textures_in(doc)
        if items is None:
            problems.append(f"{label}: expected an array of textures")
            continue
        for i, tex in enumerate(items):
            errs, norm = validate(tex, f"{label}[{i}]")
            problems += errs
            if norm is None:
                continue
            key = (norm["folder"], norm["name"])
            if key in seen:
                problems.append(f"{label}[{i}] '{norm['name']}': duplicate of {seen[key]} (same folder and name)")
                continue
            seen[key] = f"{label}[{i}]"
            notes += norm["notes"]
            textures.append(norm)
    return textures, problems, notes

# ---------------------------------------------------------------- output
def to_png(tex: dict) -> bytes:
    rows = [b"".join(bytes(px) for px in row) for row in tex["rows"]]
    return encode_png(tex["size"], tex["size"], rows)

def preview_png(textures: list[dict], per_row: int = 6, tile: int = 128, gap: int = 12) -> bytes:
    """Contact sheet: every texture scaled to ~tile px over a light/dark checkerboard, for quick human review."""
    cols = min(per_row, max(1, len(textures)))
    rows_n = -(-len(textures) // cols)
    w, h = gap + cols * (tile + gap), gap + rows_n * (tile + gap)
    canvas = [[(38, 38, 42, 255)] * w for _ in range(h)]
    for n, tex in enumerate(textures):
        ox, oy = gap + (n % cols) * (tile + gap), gap + (n // cols) * (tile + gap)
        scale = max(1, tile // tex["size"])
        side = tex["size"] * scale
        for y in range(side):
            for x in range(side):
                r, g, b, a = tex["rows"][y // scale][x // scale]
                bg = 200 if ((x // 8) + (y // 8)) % 2 == 0 else 150
                canvas[oy + y][ox + x] = ((r * a + bg * (255 - a)) // 255, (g * a + bg * (255 - a)) // 255,
                                          (b * a + bg * (255 - a)) // 255, 255)
    return encode_png(w, h, [b"".join(bytes(px) for px in row) for row in canvas])

def find_sources(project: Path, explicit: str | None) -> list[Path]:
    if explicit:
        return [Path(explicit)]
    d = project / SOURCE_DIR
    return sorted(p for p in d.glob("*.json")) if d.is_dir() else []

def run(project: Path, *, check: bool = False, preview: Path | None = None, explicit: str | None = None, log=print) -> int:
    sources = find_sources(project, explicit)
    if not sources:
        log(f"No pixel JSON found (expected {project / SOURCE_DIR}/*.json).")
        return 0
    textures, problems, notes = load_all(sources)
    for n in notes:
        log(f"note   {n}")
    for p in problems:
        log(f"FAIL   {p}")
    if problems:
        log(f"Nothing written: {len(problems)} problem(s) above.")
        return 1
    if check:
        log(f"ok     {len(textures)} texture(s) valid")
        return 0
    base = project / "resource_pack" / "textures"
    for tex in textures:
        target = base / tex["folder"] / f"{tex['name']}.png"
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(to_png(tex))
        log(f"wrote  {target.relative_to(project).as_posix()}  ({tex['size']}x{tex['size']})")
    if preview:
        Path(preview).parent.mkdir(parents=True, exist_ok=True)
        Path(preview).write_bytes(preview_png(textures))
        log(f"wrote  {preview}  (review sheet, not part of the pack)")
    log(f"PIXEL TEXTURES OK: {len(textures)} PNG(s)")
    return 0

def main(argv=None):
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("project", help="pack folder (repo pack or build/<Name>)")
    ap.add_argument("--check", action="store_true", help="validate only, write nothing")

    ap.add_argument("--input", help="use this JSON file instead of texture_sources/*.json")
    a = ap.parse_args(argv)
    project = Path(a.project)
    if not project.is_dir():
        print(f"Project folder not found: {project}")
        return 1
    return run(project, check=a.check, preview=Path(a.preview) if a.preview else None, explicit=a.input)

if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""Search-and-replace updater for ALNStudios/MCPE-Curseforges (runs inside every update .sh).

Applies a small text spec to a project instead of rewriting whole files. All-or-nothing: every edit is
checked in memory first, nothing is written if any edit fails. Safe to run twice (applied edits are skipped).

Spec format (lines starting with "@@ " are directives; a content line that must start with "@@" is written
"@@@"; the first "@" is removed):
  @@ version 1.2.4                      new X.Y.Z (must be the current highest + 0.0.1, or already applied)
                                        repo level (@@ project .): just the number, e.g. @@ version 80 (= REPO_VERSION + 1)
  @@ project MCPEDL/Addons/Mining-Helmet   folder every @@ file path is relative to ("." = repo root, no version step)
  @@ file behavior_pack/scripts/main.js
  @@ replace                            exact text that must occur ONCE (partial lines are fine)
  old text
  @@ with
  new text
  @@ replace-all ... @@ with ...        every occurrence
  @@ insert-after / @@ insert-before    anchor line(s) occurring once, then @@ with + the lines to insert
  @@ delete                             exact text occurring once (whole lines remove their newline too)
  @@ append                             text added to the end (skipped when already there)
  @@ create                             new file (error when it exists with other content)
  @@ overwrite                          whole-file replacement (only for new content over 60% of the file)
  @@ changelog Update Name              project CHANGELOG.md entry "## [X.Y.Z] - date - Update Name" + body
  @@ root-changelog Update Name         optional; default = same name and body in the root CHANGELOG.md
Block text is the lines between directives, joined with newlines (no trimming). Put no blank line before the
next directive unless the text needs it.

Usage: python3 scripts/apply_update.py SPEC --date YYYY-MM-DD [--root DIR] [--dry-run] [--touched FILE]
Exit: 0 ok, 1 spec or match error, 3 version is stale, 4 validation failed.
"""
import argparse
import json
import re
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

NAMES = {"version", "project", "file", "replace", "replace-all", "insert-after", "insert-before", "delete", "with",
         "append", "create", "overwrite", "changelog", "root-changelog"}
NEEDS_WITH = {"replace", "replace-all", "insert-after", "insert-before"}
VER = re.compile(r"^(?:v\d+|\d+\.\d+\.\d+)$", re.IGNORECASE)
REPO_VER = re.compile(r"^(?:\d+\.\d+\.)?(\d+)$")

class Fail(Exception):
    def __init__(self, msg, code=1):
        super().__init__(msg)
        self.code = code

def parse(text):
    toks = []
    for n, line in enumerate(text.replace("\r\n", "\n").split("\n"), 1):
        if line.startswith("@@@"):
            toks[-1][2].append(line[1:]) if toks else None
        elif line == "@@" or line.startswith("@@ "):
            name, _, arg = line[3:].strip().partition(" ")
            if name not in NAMES:
                raise Fail(f"spec line {n}: unknown directive '@@ {name}'")
            toks.append([name, arg.strip(), []])
        elif toks:
            toks[-1][2].append(line)
        elif line.strip():
            raise Fail(f"spec line {n}: text before the first directive")
    if toks and toks[-1][2] and toks[-1][2][-1] == "":
        toks[-1][2].pop()
    spec = {"version": None, "project": None, "ops": [], "changelog": None, "root": None}
    cur, i = None, 0
    while i < len(toks):
        name, arg, lines = toks[i]
        body = "\n".join(lines)
        i += 1
        if name == "version":
            spec["version"] = arg
        elif name == "project":
            spec["project"] = arg
        elif name == "file":
            cur = arg
        elif name in ("changelog", "root-changelog"):
            spec["changelog" if name == "changelog" else "root"] = (arg or "Update", body)
        elif name == "with":
            raise Fail("'@@ with' without a matching @@ replace, @@ replace-all, @@ insert-after or @@ insert-before")
        else:
            if cur is None:
                raise Fail(f"'@@ {name}' needs a '@@ file <path>' before it")
            new = None
            if name in NEEDS_WITH:
                if i >= len(toks) or toks[i][0] != "with":
                    raise Fail(f"'@@ {name}' on {cur} needs '@@ with'")
                new = "\n".join(toks[i][2])
                i += 1
            spec["ops"].append({"op": name, "file": cur, "old": body, "new": new})
    repo = (spec["project"] or "").strip().rstrip("/") == "."
    if not spec["version"] or not (REPO_VER if repo else VER).match(spec["version"]):
        raise Fail("spec needs '@@ version vX or X.Y.Z' (a repo-level spec uses just the number, '@@ version 80')")
    if not spec["project"]:
        raise Fail("spec needs '@@ project <folder>'")
    return spec

def apply_op(t, o):
    op, old, new = o["op"], o["old"], o["new"]
    if op in ("replace", "replace-all"):
        c = t.count(old) if old else 0
        if not old:
            raise Fail(f"{o['file']}: empty search text")
        if c == 0:
            if new in t:
                return t, False
            raise Fail(f"{o['file']}: text not found: {old.splitlines()[0][:80]!r}")
        if new in t and old in new:
            return t, False
        if op == "replace" and c > 1:
            raise Fail(f"{o['file']}: text found {c} times, add more context or use replace-all: {old.splitlines()[0][:80]!r}")
        return t.replace(old, new), True
    if op in ("insert-after", "insert-before"):
        if new in t and old in t:
            return t, False
        c = t.count(old)
        if c != 1:
            raise Fail(f"{o['file']}: anchor found {c} times (needs exactly 1): {old.splitlines()[0][:80]!r}")
        a = t.index(old)
        if op == "insert-after":
            e = t.find("\n", a + len(old))
            e = len(t) if e < 0 else e
            return t[:e] + "\n" + new + t[e:], True
        s = t.rfind("\n", 0, a) + 1
        return t[:s] + new + "\n" + t[s:], True
    if op == "delete":
        c = t.count(old)
        if c == 0:
            return t, False
        if c > 1:
            raise Fail(f"{o['file']}: delete text found {c} times: {old.splitlines()[0][:80]!r}")
        a = t.index(old)
        e = a + len(old)
        if (a == 0 or t[a - 1] == "\n") and t[e:e + 1] == "\n":
            e += 1
        return t[:a] + t[e:], True
    if op == "append":
        if old.strip("\n") and old.strip("\n") in t:
            return t, False
        return (t if t.endswith("\n") or not t else t + "\n") + old.rstrip("\n") + "\n", True
    raise Fail(f"unknown op {op}")

def vtuple(v):
    cleaned = str(v or "").strip().lstrip("vV")
    parts = [int(p) for p in cleaned.split(".")]
    if len(parts) == 1:
        return (parts[0], 0, 0)
    if len(parts) == 3:
        return (parts[0], parts[1], parts[2])
    return tuple(parts)

def current_highest(proj):
    best = None
    for m in proj.rglob("manifest.json"):
        try:
            v = json.loads(m.read_text(encoding="utf-8-sig"))["header"]["version"]
            if isinstance(v, list) and len(v) == 3:
                best = max(best or (0, 0, 0), tuple(v))
        except (OSError, ValueError, KeyError, TypeError):
            pass
    c = proj / "CHANGELOG.md"
    if c.is_file():
        m = re.search(r"(?m)^## \[(?:v)?(\d+(?:\.\d+\.\d+)?)\]", c.read_text(encoding="utf-8"))
        if m:
            best = max(best or (0, 0, 0), vtuple(m.group(1)))
    return best

def add_entry(t, heading, body):
    if heading in t:
        return t, False
    i = t.index("---\n") + 4 if "---\n" in t else len(t)
    return t[:i] + "\n" + heading + "\n\n" + body.strip("\n") + "\n" + t[i:], True

def js_syntax(path):
    """(ran, error). A missing or broken node (for example a Termux library error) is 'not checked', never a syntax error."""
    node = shutil.which("node")
    if not node:
        return False, ""
    with tempfile.TemporaryDirectory() as t:
        tmp = Path(t) / "check.mjs"
        shutil.copyfile(path, tmp)
        r = subprocess.run([node, "--check", str(tmp)], capture_output=True, text=True)
    if r.returncode == 0:
        return True, ""
    if "SyntaxError" not in r.stderr:
        return False, ""
    errs = [ln for ln in r.stderr.splitlines() if "Error" in ln]
    return True, (errs[-1] if errs else "syntax error").strip()

def run(spec, root, date, dry=False):
    root = Path(root).resolve()
    ver = spec["version"]
    project = spec["project"].strip().rstrip("/")
    repo_level = project == "."
    proj = (root / project).resolve()
    if not repo_level and root not in proj.parents:
        raise Fail(f"project path leaves the repo: {project}")
    if not proj.is_dir():
        raise Fail(f"project folder not found: {project}")
    if not repo_level:
        hi = current_highest(proj)
        want = vtuple(ver)
        allowed = [hi] if hi is not None else []
        if hi is not None:
            allowed.append((hi[0], hi[1], hi[2] + 1))
            allowed.append((hi[0] + 1, 0, 0))
        if hi is not None and want not in allowed:
            raise Fail(f"stale script: {project} is at {'.'.join(map(str, hi))}, this script makes {ver}. "

            raise Fail("edit CHANGELOG.md with @@ changelog, not with a file edit")

                continue
            if t is not None and o["op"] == "create":

            continue

                    raise Fail(f"{p.relative_to(root)}: JavaScript syntax error: {err}", 4)

```
ALN UI Redesign - Pixelify Sans font (menu-only glyph page F0)

For any pack with a menu, form, HUD or screen that follows the ALN UI Redesign skin:
1. Copy glyph_F0.png into <pack>/resource_pack/font/ (copy, never link).
2. Copy pixelify.js into <pack>/behavior_pack/scripts/lib/.
3. Wrap menu strings: form.title(pixelify("...")), form.button(pixelify("...")).
   Spaces stay normal. Color codes (section sign + letter) stay outside pixelify().
4. Bump the Resource Pack version with manifest_tools.py --set.

Covers plain ASCII letters, digits, punctuation only. Vanilla font is untouched elsewhere.
Page F0 (U+F000-U+F0FF) is assumed unused by Minecraft; not tested in game.
Font: Pixelify Sans, SIL OFL 1.1 (see PixelifySans-OFL.txt). Keep that file with any pack that ships the glyphs.
```

## FILE: assets/aln-ui-redesign/font/pixelify.js (11 lines, 319 bytes)

// Pixelify Sans glyph mapper (page F0). Spaces and section-sign color codes stay vanilla.
const PAGE = 0x00F0;
export function pixelify(text) {
  let out = "";
  for (const ch of String(text)) {
    const c = ch.codePointAt(0);
    out += c > 0x20 && c < 0x7F ? String.fromCharCode(PAGE + c) : ch;
  }
  return out;
}
